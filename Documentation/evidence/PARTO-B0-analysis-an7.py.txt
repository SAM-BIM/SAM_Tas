import pandas as pd, numpy as np
W='C:/TasOut/b0w/'
D=pd.read_csv(W+'out/hourly-dataset.csv')
A=pd.read_csv(W+'A.csv')
DU=pd.read_csv('C:/TasOut/b0r4/ducts.csv')
BOUND=['Bathroom_2','Studio 1_0','Bedroom 2_3','Ensuite_5','Kitchen_4','Bedroom 2_6','Ensuite_8','Kitchen_7']
odb=A['BLD:externalTemperature'].values
MV={'MVHR-01':'{10ECD9E6-FBBB-478D-BE3F-ECF00808F335}','MVHR-02':'{31313469-2008-46C2-9006-3535F83DEC3B}','MVHR-03':'{D301D74D-D9CD-4080-88C5-D374A8CD5A02}'}
UNIT={'Studio 1_0':'MVHR-01','Bathroom_2':'MVHR-01','Bedroom 2_3':'MVHR-02','Kitchen_4':'MVHR-02','Ensuite_5':'MVHR-02','Bedroom 2_6':'MVHR-03','Kitchen_7':'MVHR-03','Ensuite_8':'MVHR-03'}
days=[31,28,31,30,31,30,31,31,30,31,30,31]; mon=np.concatenate([np.full(d*24,i) for i,d in enumerate(days)])
MON=['Jan','Feb','Mar','Apr','May','Jun','Jul','Aug','Sep','Oct','Nov','Dec']
def dcol(sub,var=3):
    m=[c for c in DU.columns if sub in c and c.endswith('|var%d'%var)]
    return DU[m[0]].values if len(m)==1 else None
ENTER={'Studio 1_0':'MVHR-01|Damper:1.29903811034702->SystemZone:Studio 1_0','Bathroom_2':'MVHR-01|Damper:918.960311873252->SystemZone:Bathroom_2',
 'Bedroom 2_3':'MVHR-02|Damper:3.15233098071178->SystemZone:Bedroom 2_3','Kitchen_4':'MVHR-02|Damper:3.15233131035354->SystemZone:Kitchen_4',
 'Ensuite_5':'MVHR-02|Damper:7.35272125882989->SystemZone:Ensuite_5','Bedroom 2_6':'MVHR-03|Damper:3.15444375498756->SystemZone:Bedroom 2_6',
 'Kitchen_7':'MVHR-03|Damper:3.15444370415334->SystemZone:Kitchen_7','Ensuite_8':'MVHR-03|Damper:171.427909252452->SystemZone:Ensuite_8'}
SRC={'Studio 1_0':None,'Bathroom_2':'Studio 1_0','Bedroom 2_3':None,'Kitchen_4':'Bedroom 2_3','Ensuite_5':'Kitchen_4','Bedroom 2_6':None,'Kitchen_7':'Bedroom 2_6','Ensuite_8':'Kitchen_7'}

print('=== free-running no-IZAM SOURCE offset (the temperature the zone loads were evaluated at) ===')
print('%-12s %10s %10s %10s %10s'%('room','S-B mean','S-A mean','B-A mean','corr(S-B, B-A)'))
for r in BOUND:
    d=D[D.room==r]
    sb=(d.S_air-d.B_air).values; sa=(d.S_air-d.A_air).values; ba=d.d_air.values
    print('%-12s %10.3f %10.3f %10.3f %10.4f'%(r,sb.mean(),sa.mean(),ba.mean(),np.corrcoef(sb,ba)[0,1]))
print('')
print('monthly: pooled (S-B) vs pooled (B-A) air')
sb=np.concatenate([(D[D.room==r].S_air-D[D.room==r].B_air).values for r in BOUND]).reshape(8,8760)
ba=np.concatenate([D[D.room==r].d_air.values for r in BOUND]).reshape(8,8760)
print('%-10s'%'month'+''.join('%8s'%m for m in MON))
print('%-10s'%'S-B'+''.join('%8.3f'%sb[:,mon==i].mean() for i in range(12)))
print('%-10s'%'B-A'+''.join('%8.3f'%ba[:,mon==i].mean() for i in range(12)))
print('monthly correlation of the two series across the 12 months: r=%.4f'%np.corrcoef([sb[:,mon==i].mean() for i in range(12)],[ba[:,mon==i].mean() for i in range(12)])[0,1])

print('')
print('=== ANALYSIS 8b: BATHROOM_2 - the 12 extra >26 C hours ===')
d=D[D.room=='Bathroom_2'].reset_index(drop=True)
src=D[D.room=='Studio 1_0'].reset_index(drop=True)
bin_=dcol(ENTER['Bathroom_2'])
a26=d.A_res.values>26; b26=d.B_res.values>26
print('A>26: %d hours %s'%(a26.sum(),list(d.hour.values[a26])))
print('B>26: %d hours %s'%(b26.sum(),list(d.hour.values[b26])))
extra=np.where(b26&~a26)[0]
print('B-only hours: %d'%len(extra))
print('%6s %5s %8s %8s %8s %8s %8s %8s %8s %8s'%('hour','mon','A_res','B_res','delta','A_over','B_over','B_inlet','A_inlet(Studio)','ODB'))
rows=[]
for i in extra:
    h=int(d.hour.values[i])
    print('%6d %5s %8.3f %8.3f %+8.3f %8.3f %8.3f %8.3f %8.3f %8.2f'%(h,MON[mon[h]],d.A_res.values[i],d.B_res.values[i],d.d_res.values[i],
        d.A_res.values[i]-26,d.B_res.values[i]-26,bin_[h],src.A_air.values[i],odb[h]))
    rows.append(dict(hour=h,month=MON[mon[h]],A_res=d.A_res.values[i],B_res=d.B_res.values[i],delta=d.d_res.values[i],
        B_margin_over_26=d.B_res.values[i]-26,A_margin_over_26=d.A_res.values[i]-26,B_inlet_C=bin_[h],A_inlet_C=src.A_air.values[i],odb=odb[h]))
pd.DataFrame(rows).to_csv(W+'out/bathroom2-extra-hours.csv',index=False,float_format='%.6f')
print('B-only hours: B margin over 26 C: mean=%.3f max=%.3f min=%.3f'%(np.mean([r['B_margin_over_26'] for r in rows]),np.max([r['B_margin_over_26'] for r in rows]),np.min([r['B_margin_over_26'] for r in rows])))
print('B-only hours: A already at    : mean=%.3f max=%.3f'%(np.mean([r['A_res'] for r in rows]),np.max([r['A_res'] for r in rows])))
print('hours where A>26 and B<=26   : %d'%int((a26&~b26).sum()))
print('Bathroom_2 hours within 0.5 K below 26 in A: %d ; in B: %d'%(int(((d.A_res>25.5)&(d.A_res<=26)).sum()),int(((d.B_res>25.5)&(d.B_res<=26)).sum())))

print('')
print('=== top-10 |B-A| resultant hours per room ===')
top=[]
for r in BOUND:
    dd=D[D.room==r].reset_index(drop=True)
    o=np.argsort(-np.abs(dd.d_res.values))[:10]

    for i in o: top.append(dict(room=r,hour=int(dd.hour.values[i]),month=MON[mon[dd.hour.values[i]]],A_res=dd.A_res.values[i],B_res=dd.B_res.values[i],d_res=dd.d_res.values[i],odb=odb[dd.hour.values[i]]))
pd.DataFrame(top).to_csv(W+'out/top-hours.csv',index=False,float_format='%.6f')
