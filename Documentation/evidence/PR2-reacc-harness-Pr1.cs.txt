// PHASE B - the frozen PR1 materialisation, given the real Iteration 1a design and nothing else.
//
// This mode CONSTRUCTS NOTHING. It reads the prepared model Phase A produced and hands it to
// SAM.Analytical.Systems.Create.MechanicalVentilation, then reports what came back against the
// Phase A intent table.

using System;
using System.Collections.Generic;
using System.Globalization;
using System.IO;
using System.Text;

namespace Inv
{
    internal static class Pr1
    {
        private static readonly StringBuilder log = new StringBuilder();

        private static void W(string format, params object[] args)
        {
            string line = args == null || args.Length == 0 ? format : string.Format(CultureInfo.InvariantCulture, format, args);
            Console.WriteLine(line);
            log.AppendLine(line);
        }

        internal static string TemplatePath()
        {
            return @"C:\Users\michal.dengusiak\Documents\GitHub\SAM-BIM\SAM_Systems\files\resources\Analytical\Systems\SystemEnergyCentre\MV.json";
        }

        /// <summary>
        /// The scope PR1 is asked to materialise: the spaces the model's own MVHR ventilation systems
        /// serve, plus every space their dwelling transfer air crosses.
        /// <para>
        /// <b>Read, never authored.</b> A space is in scope because a <c>VentilationTerminal</c> the design
        /// carries connects it to a <c>VentilationSystem</c>, or because a <c>SpaceAirMovement</c> the
        /// design carries reaches it from one that is. No name is matched, no guid is ordered, no role is
        /// assigned and no duty is read to decide this.
        /// </para>
        /// </summary>
        internal static List<SAM.Analytical.Space> Scope(SAM.Analytical.AdjacencyCluster adjacencyCluster, out List<string> notes)
        {
            notes = new List<string>();

            HashSet<Guid> guids = new HashSet<Guid>();

            Dictionary<Guid, SAM.Analytical.Space> dictionary_Space = new Dictionary<Guid, SAM.Analytical.Space>();
            foreach (SAM.Analytical.Space space in adjacencyCluster.GetSpaces() ?? new List<SAM.Analytical.Space>())
            {
                dictionary_Space[space.Guid] = space;
            }

            foreach (SAM.Analytical.VentilationSystem ventilationSystem in adjacencyCluster.GetObjects<SAM.Analytical.VentilationSystem>() ?? new List<SAM.Analytical.VentilationSystem>())
            {
                List<SAM.Analytical.VentilationTerminal> ventilationTerminals = adjacencyCluster.GetRelatedObjects<SAM.Analytical.VentilationTerminal>(ventilationSystem) ?? new List<SAM.Analytical.VentilationTerminal>();

                if (ventilationTerminals.Count == 0)
                {
                    continue;
                }

                foreach (SAM.Analytical.VentilationTerminal ventilationTerminal in ventilationTerminals)
                {
                    foreach (SAM.Analytical.Space space in adjacencyCluster.GetRelatedObjects<SAM.Analytical.Space>(ventilationTerminal) ?? new List<SAM.Analytical.Space>())
                    {
                        guids.Add(space.Guid);
                    }
                }
            }

            notes.Add(string.Format("{0} space(s) carry a design ventilation terminal.", guids.Count));

            //The transitive closure over the design's own transfer legs, so a zero-terminal room the
            //design's air genuinely crosses is in scope and a room it never reaches is not.
            Dictionary<(Guid, Guid), SAM.Analytical.Query.DesignTransferAirMovement> dictionary_Transfer =
                SAM.Analytical.Query.DesignTransferSpaceAirMovements(adjacencyCluster) ?? new Dictionary<(Guid, Guid), SAM.Analytical.Query.DesignTransferAirMovement>();

            bool changed = true;

            while (changed)
            {
                changed = false;

                foreach (KeyValuePair<(Guid, Guid), SAM.Analytical.Query.DesignTransferAirMovement> keyValuePair in dictionary_Transfer)
                {
                    Guid guid_From = keyValuePair.Value.FromGuid;
                    Guid guid_To = keyValuePair.Value.ToGuid;

                    if (guids.Contains(guid_From) && !guids.Contains(guid_To))
                    {
                        guids.Add(guid_To);
                        changed = true;
                    }
                    else if (guids.Contains(guid_To) && !guids.Contains(guid_From))
                    {
                        guids.Add(guid_From);
                        changed = true;
                    }
                }
            }

            List<SAM.Analytical.Space> result = new List<SAM.Analytical.Space>();

            foreach (Guid guid in guids)
            {
                SAM.Analytical.Space space;
                if (dictionary_Space.TryGetValue(guid, out space))
                {
                    result.Add(space);
                }
            }

            result.Sort((x, y) => string.Compare(x.Name, y.Name, StringComparison.Ordinal));

            List<string> names = new List<string>();
            foreach (SAM.Analytical.Space space in result)
            {
                names.Add(space.Name);
            }

            notes.Add(string.Format("{0} space(s) in scope after closing over the design's own transfer legs: {1}.", result.Count, string.Join(", ", names)));

            return result;
        }

        /// <summary>
        /// Removes from a working copy of the cluster the ventilation systems that carry NO design
        /// ventilation terminal, so PR1 is handed the ventilation design and nothing beside it.
        /// <para>
        /// <b>What this is, and what it is emphatically not.</b> It is a SCOPING step, stated in terms of
        /// the design itself: a <c>VentilationSystem</c> with no design terminal materialises no duty and
        /// is not part of the mechanical ventilation being materialised. No name, type, guid order or
        /// enumeration index decides it. Nothing is added. No terminal, air movement, air handling unit,
        /// space, duty or transfer direction is touched, so the intent table is provably identical before
        /// and after - which the caller asserts rather than assumes.
        /// </para>
        /// <para>
        /// It exists because <c>Create.MechanicalVentilation</c> walks EVERY <c>VentilationSystem</c> the
        /// cluster carries and requires each to name a resolvable air handling unit, while production
        /// <c>Modify.PreparePartOIteration</c> deliberately PRESERVES the model's authored NV / UV / MV
        /// systems ("that relation is left as the model authored it"). The two are in genuine
        /// contradiction on any real model, and PR1 offers the caller no system scope to say so.
        /// </para>
        /// </summary>
        internal static int ScopeSystems(SAM.Analytical.AdjacencyCluster adjacencyCluster, string rule, List<string> notes)
        {
            int result = 0;

            foreach (SAM.Analytical.VentilationSystem ventilationSystem in adjacencyCluster.GetObjects<SAM.Analytical.VentilationSystem>() ?? new List<SAM.Analytical.VentilationSystem>())
            {
                int count_Terminal = (adjacencyCluster.GetRelatedObjects<SAM.Analytical.VentilationTerminal>(ventilationSystem) ?? new List<SAM.Analytical.VentilationTerminal>()).Count;

                bool namesUnit = !string.IsNullOrWhiteSpace(ventilationSystem.GetValue<string>(SAM.Analytical.VentilationSystemParameter.SupplyUnitName));

                bool remove;

                if (string.Equals(rule, "nounit", StringComparison.OrdinalIgnoreCase))
                {
                    remove = count_Terminal == 0 && !namesUnit;
                }
                else if (string.Equals(rule, "noterminal", StringComparison.OrdinalIgnoreCase))
                {
                    remove = count_Terminal == 0;
                }
                else
                {
                    remove = false;
                }

                if (!remove)
                {
                    continue;
                }

                notes.Add(string.Format("out of scope: ventilation system '{0}' of type '{1}' carries {2} design terminal(s) and names unit '{3}'.",
                    ventilationSystem.FullName,
                    ventilationSystem.Type == null ? "<none>" : ventilationSystem.Type.Name,
                    count_Terminal,
                    ventilationSystem.GetValue<string>(SAM.Analytical.VentilationSystemParameter.SupplyUnitName) ?? "<none>"));

                if (adjacencyCluster.RemoveObject(ventilationSystem.GetType(), ventilationSystem.Guid))
                {
                    result++;
                }
            }

            return result;
        }

        public static int Main(string path_SAM, string directory, string rule)
        {
            log.Clear();

            Directory.CreateDirectory(directory);

            W("PHASE B - PR1 materialisation of the real design");
            W("  model     {0}", path_SAM);
            W("  template  {0}", TemplatePath());
            W("");

            SAM.Analytical.AnalyticalModel analyticalModel = SAM.Core.Convert.ToSAM<SAM.Analytical.AnalyticalModel>(path_SAM)[0];

            SAM.Analytical.AdjacencyCluster adjacencyCluster = analyticalModel.AdjacencyCluster;

            List<string> notes_System = new List<string>();

            int count_OutOfScope = ScopeSystems(adjacencyCluster, rule, notes_System);

            W("  system scope rule: {0}, {1} system(s) taken out of scope", rule ?? "none", count_OutOfScope);

            foreach (string note in notes_System)
            {
                W("    {0}", note);
            }

            // ---- the scoping step is proved to change nothing about the design -----------------------
            //
            // The intent table is recomputed off the SCOPED cluster and compared, row for row, with the
            // one Phase A wrote off the unscoped prepared model. Identical means no duty, role, direction,
            // membership or room moved - which is the whole claim the scoping step makes.

            string path_Intent = Path.Combine(directory, "intent-scoped.tsv");

            Prep.Oracle(Prep.MeasureModel(new SAM.Analytical.AnalyticalModel(analyticalModel, adjacencyCluster)), path_Intent);

            List<SAM.Analytical.Space> spaces_Scope = Scope(adjacencyCluster, out List<string> notes_Scope);

            foreach (string note in notes_Scope)
            {
                W("  scope: {0}", note);
            }

            SAM.Core.Systems.SystemEnergyCentre systemEnergyCentre_Template =
                SAM.Analytical.Systems.Query.SystemEnergyCentre(TemplatePath());

            W("");
            W("== PRODUCTION SAM_Systems Create.MechanicalVentilation ==");

            SAM.Analytical.Systems.MechanicalVentilationMaterialisation mechanicalVentilationMaterialisation =
                SAM.Analytical.Systems.Create.MechanicalVentilation(adjacencyCluster, systemEnergyCentre_Template, null, spaces_Scope);

            foreach (string note in mechanicalVentilationMaterialisation.Notes)
            {
                W("  note: {0}", note);
            }

            foreach (string refusal in mechanicalVentilationMaterialisation.Refusals)
            {
                W("  REFUSAL: {0}", refusal);
            }

            W("");
            W("  IsMaterialised: {0}", mechanicalVentilationMaterialisation.IsMaterialised);

            if (mechanicalVentilationMaterialisation.IsMaterialised)
            {
                Dump(mechanicalVentilationMaterialisation, adjacencyCluster);
            }

            File.WriteAllText(Path.Combine(directory, "pr1.log"), log.ToString());

            return mechanicalVentilationMaterialisation.IsMaterialised ? 0 : 7;
        }

        /// <summary>What PR1 produced, read back off the materialised graph rather than off its notes.</summary>
        private static void Dump(SAM.Analytical.Systems.MechanicalVentilationMaterialisation mechanicalVentilationMaterialisation, SAM.Analytical.AdjacencyCluster adjacencyCluster)
        {
            Dictionary<Guid, string> dictionary_SpaceName = new Dictionary<Guid, string>();
            foreach (SAM.Analytical.Space space in adjacencyCluster.GetSpaces() ?? new List<SAM.Analytical.Space>())
            {
                dictionary_SpaceName[space.Guid] = space.Name;
            }

            W("");
            W("== the materialised graph ==");

            foreach (SAM.Core.Systems.SystemPlantRoom systemPlantRoom in mechanicalVentilationMaterialisation.SystemEnergyCentre.GetSystemPlantRooms() ?? new List<SAM.Core.Systems.SystemPlantRoom>())
            {
                W("  plant room '{0}'", systemPlantRoom.Name);

                foreach (SAM.Analytical.Systems.AirSystem airSystem in systemPlantRoom.GetSystems<SAM.Analytical.Systems.AirSystem>() ?? new List<SAM.Analytical.Systems.AirSystem>())
                {
                    List<SAM.Analytical.Systems.SystemSpace> systemSpaces = systemPlantRoom.GetSystemComponents<SAM.Analytical.Systems.SystemSpace>(airSystem) ?? new List<SAM.Analytical.Systems.SystemSpace>();

                    List<string> names = new List<string>();
                    foreach (SAM.Analytical.Systems.SystemSpace systemSpace in systemSpaces)
                    {
                        names.Add(systemSpace.Name);
                    }

                    names.Sort(StringComparer.Ordinal);

                    W("    air system '{0}'  {1} system space(s): {2}", airSystem.Name, systemSpaces.Count, string.Join(", ", names));
                }
            }

            W("");
            W("== the lineage bindings ==");

            List<string> lines = new List<string>();

            foreach (SAM.Analytical.Systems.MechanicalVentilationBinding mechanicalVentilationBinding in mechanicalVentilationMaterialisation.Bindings)
            {
                string name;
                lines.Add(string.Format("  {0,-22}  analytical {1}  {2}",
                    mechanicalVentilationBinding.BindingType,
                    mechanicalVentilationBinding.Guid_Analytical.ToString("D"),
                    dictionary_SpaceName.TryGetValue(mechanicalVentilationBinding.Guid_Analytical, out name) ? name : string.Empty));
            }

            lines.Sort(StringComparer.Ordinal);

            foreach (string line in lines)
            {
                W(line);
            }
        }
    }
