// Controls for the fan-operation evidence. Each works on its OWN COPY of a produced TPD, changes one
// thing about the fans' operation carrier, re-simulates every air system with a raw late-bound call and
// then runs the same per-fan hourly read as FanProbe. The route's output is never touched.
//
//   gap          attach a yearly schedule that is OFF for hours 0..23 and ON for 24..8759 to every fan.
//                Proves the native hourly fan Load (variable 9) tracks operation rather than restating a
//                design value.
//   funcload N   keep the fans' existing function schedule but set its FunctionLoads to N (4 = heating,
//                8 = cooling, 1024 = occupant sensible). Proves whether a demand-driven function
//                schedule switches the fan off when the attached zones' demand is absent.
//
// Nothing here is production code.

using System;
using System.Collections.Generic;
using System.Globalization;
using System.IO;

namespace Inv
{
    internal static class FanControl
    {
        public static int Run(string path_TPD, string mode, int argument, string pathOut)
        {
            string path_Copy = Path.ChangeExtension(path_TPD, "." + mode + (mode == "funcload" ? argument.ToString(CultureInfo.InvariantCulture) : string.Empty) + ".tpd");
            File.Copy(path_TPD, path_Copy, true);

            List<string> lines = new List<string>();
            lines.Add(string.Format(CultureInfo.InvariantCulture, "FANCONTROL {0} mode={1} argument={2}", path_TPD, mode, argument));
            lines.Add("  edited copy " + path_Copy);

            global::TPD.TPDDoc tPDDoc = new global::TPD.TPDDoc();

            try
            {
                tPDDoc.Open(path_Copy);

                object energyCentre = tPDDoc.EnergyCentre;

                object plantSchedule_Gap = null;
                if (mode == "gap")
                {
                    plantSchedule_Gap = Dump.Call(energyCentre, "AddSchedule", (int)global::TPD.tpdScheduleType.tpdScheduleYearly);
                    int[] values = new int[8760];
                    for (int i = 0; i < values.Length; i++)
                    {
                        values[i] = i < 24 ? 0 : 1;
                    }

                    ((dynamic)plantSchedule_Gap).SetYearlyValues(values);
                    ((dynamic)plantSchedule_Gap).Name = "Control Gap Hours 0-23";

                    lines.Add(string.Format(CultureInfo.InvariantCulture,
                        "  added yearly schedule, GetNumOperableHours={0}, GetYearlyValue(1)={1} (24)={2} (25)={3} (8760)={4}",
                        Dump.CallText(plantSchedule_Gap, "GetNumOperableHours"),
                        Dump.CallText(plantSchedule_Gap, "GetYearlyValue", 1),
                        Dump.CallText(plantSchedule_Gap, "GetYearlyValue", 24),
                        Dump.CallText(plantSchedule_Gap, "GetYearlyValue", 25),
                        Dump.CallText(plantSchedule_Gap, "GetYearlyValue", 8760)));
                }

                object plantSchedule_New = null;
                if (mode == "hourly1")
                {
                    plantSchedule_New = Dump.Call(energyCentre, "AddSchedule", (int)global::TPD.tpdScheduleType.tpdScheduleHourly);
                    ((dynamic)plantSchedule_New).Name = "Control Hourly All On";
                    int count_Day = (int)Convert.ToDouble(Dump.Call(plantSchedule_New, "GetScheduleDayCount"), CultureInfo.InvariantCulture);
                    List<string> dayTypes = new List<string>();
                    for (int day = 1; day <= count_Day; day++)
                    {
                        object plantScheduleDay = Dump.Call(plantSchedule_New, "GetScheduleDay", day);
                        for (int hour = 1; hour <= 24; hour++)
                        {
                            ((dynamic)plantScheduleDay).SetHourlyValue(hour, 1);
                        }

                        dayTypes.Add(Dump.Get(Dump.Call(plantScheduleDay, "GetDayType"), "Name"));
                    }

                    lines.Add(string.Format(CultureInfo.InvariantCulture, "  added hourly schedule, {0} day type(s) all on: {1}; GetNumOperableHours={2}",
                        count_Day, string.Join(", ", dayTypes), Dump.CallText(plantSchedule_New, "GetNumOperableHours")));
                }
                else if (mode == "funcnone")
                {
                    plantSchedule_New = Dump.Call(energyCentre, "AddSchedule", (int)global::TPD.tpdScheduleType.tpdScheduleFunction);
                    ((dynamic)plantSchedule_New).Name = "Control Function None";
                    ((dynamic)plantSchedule_New).FunctionType = global::TPD.tpdScheduleFunctionType.tpdScheduleFunctionNone;
                    ((dynamic)plantSchedule_New).FunctionLoads = 0;
                    lines.Add("  added function schedule, FunctionType=" + Dump.Get(plantSchedule_New, "FunctionType") + " FunctionLoads=" + Dump.Get(plantSchedule_New, "FunctionLoads"));
                }
                else if (mode == "occupancy")
                {
                    for (int index = 1; index <= 1000; index++)
                    {
                        object plantSchedule = Dump.Call(energyCentre, "GetSchedule", index);
                        if (plantSchedule == null)
                        {
                            break;
                        }

                        if (Dump.Get(plantSchedule, "Name") == "\"Occupancy Schedule\"")
                        {
                            plantSchedule_New = plantSchedule;
                            break;
                        }
                    }

                    lines.Add("  found existing schedule: " + (plantSchedule_New == null ? "<none>" : Dump.Get(plantSchedule_New, "Name")));
                }

                int count_PlantRoom = (int)Convert.ToDouble(Dump.Call(energyCentre, "GetPlantRoomCount"), CultureInfo.InvariantCulture);
                HashSet<string> edited = new HashSet<string>(StringComparer.Ordinal);

                for (int p = 1; p <= count_PlantRoom; p++)
                {
                    object plantRoom = Dump.Call(energyCentre, "GetPlantRoom", p);
                    int count_System = (int)Convert.ToDouble(Dump.Call(plantRoom, "GetSystemCount"), CultureInfo.InvariantCulture);

                    for (int s = 1; s <= count_System; s++)
                    {
                        object system = Dump.Call(plantRoom, "GetSystem", s);
                        object firstFanSchedule = null;

                        for (int index = 1; index <= 20000; index++)
                        {
                            object component = Dump.Call(system, "GetComponent", index);
                            if (component == null)
                            {
                                break;
                            }

                            if (!(component is global::TPD.Fan))
                            {
                                continue;
                            }

                            if (mode == "gap")
                            {
                                ((dynamic)component).SetSchedule(plantSchedule_Gap);
                                lines.Add("  fan " + Dump.Get(component, "Name") + " -> " + Dump.Get(((dynamic)component).GetSchedule(), "Name"));
                            }
                            else if (mode == "hourly1" || mode == "funcnone" || mode == "occupancy")
                            {
                                ((dynamic)component).SetSchedule(plantSchedule_New);
                                lines.Add("  fan " + Dump.Get(component, "Name") + " -> " + Dump.Get(((dynamic)component).GetSchedule(), "Name"));
                            }
                            else if (mode == "none")
                            {
                                ((dynamic)component).RemoveSchedule();
                                object after = ((dynamic)component).GetSchedule();
                                lines.Add("  fan " + Dump.Get(component, "Name") + " schedule removed; GetSchedule() now " + (after == null ? "null" : Dump.Get(after, "Name")));
                            }
                            else if (mode == "sysyearly")
                            {
                                if (firstFanSchedule == null)
                                {
                                    firstFanSchedule = ((dynamic)component).GetSchedule();
                                }
                            }
                            else if (mode == "funcload")
                            {
                                object plantSchedule = ((dynamic)component).GetSchedule();
                                string name = Dump.Get(plantSchedule, "Name");
                                if (edited.Add(name))
                                {
                                    string before = Dump.Get(plantSchedule, "FunctionLoads");
                                    ((dynamic)plantSchedule).FunctionLoads = argument;
                                    lines.Add(string.Format(CultureInfo.InvariantCulture, "  schedule {0}: FunctionLoads {1} -> {2}", name, before, Dump.Get(plantSchedule, "FunctionLoads")));
                                }
                            }
                        }

                        if (mode == "sysyearly" && firstFanSchedule != null)
                        {
                            ((dynamic)system).SetSystemSchedule(firstFanSchedule);
                            lines.Add("  system " + Dump.Get(system, "Name") + " SetSystemSchedule(" + Dump.Get(firstFanSchedule, "Name") + ")");
                        }
                    }
                }

                Dump.Call(tPDDoc, "Save");
            }
            finally
            {
                try { tPDDoc.Close(); } catch { }
            }

            File.WriteAllLines(pathOut + ".edit.txt", lines);
            foreach (string line in lines)
            {
                Console.WriteLine(line);
            }

            return 0;
        }
    }
}
