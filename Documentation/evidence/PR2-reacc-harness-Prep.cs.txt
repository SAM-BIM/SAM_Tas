// PR2 corrective re-acceptance - PHASE A.
//
// Reproduces the SAME production route SAM_UI's Iteration 1a runs, on a real .sam, and reports what
// the analytical design actually says. NOTHING here authors a ventilation design, assigns a room
// role, invents an airflow or reorders anything by guid. Every engineering decision is one of these
// production calls:
//
//   SAM.Analytical.Query.DefaultPartFCalculator / PartFCalculator.Calculate   the Part F sizing
//   SAM.Analytical.Query.PartFDwellingZones                                   the dwelling scope
//   SAM.Analytical.Modify.PreparePartOIteration                               Iteration 1a / Base MVHR
//
// The dwelling scope, the room roles, the duties and the transfer directions are all READ back off
// what those calls produced.

using System;
using System.Collections.Generic;
using System.Globalization;
using System.IO;
using System.Text;

namespace Inv
{
    internal static class Prep
    {
        private static readonly StringBuilder log = new StringBuilder();

        private static void W(string format, params object[] args)
        {
            string line = args == null || args.Length == 0 ? format : string.Format(CultureInfo.InvariantCulture, format, args);
            Console.WriteLine(line);
            log.AppendLine(line);
        }

        private static void Flush(string path)
        {
            try
            {
                File.WriteAllText(path, log.ToString());
            }
            catch
            {
            }
        }

        /// <summary>One room of one materialised dwelling system, as the analytical design states it.</summary>
        internal sealed class Room
        {
            public string System;
            public Guid SystemGuid;
            public string Dwelling;
            public Guid DwellingGuid;
            public Guid SpaceGuid;
            public string SpaceName;
            public double Supply_Lps;
            public double Extract_Lps;
        }

        /// <summary>One transfer leg, with the direction the design states and no other.</summary>
        internal sealed class Leg
        {
            public Guid FromGuid;
            public string FromName;
            public Guid ToGuid;
            public string ToName;
            public double Flow_Lps;
            public string Dwelling;
        }

        /// <summary>
        /// Everything Phase A measured, so Phase B's oracle and the native comparison read one object
        /// rather than re-deriving the design a second time.
        /// </summary>
        internal sealed class Design
        {
            public SAM.Analytical.AnalyticalModel AnalyticalModel;
            public List<Room> Rooms = new List<Room>();
            public List<Leg> Legs = new List<Leg>();
            public List<string> AirHandlingUnitNames = new List<string>();
            public bool Successful;
            public string Refusal;
        }

        // =========================================================================== the production route

        /// <summary>
        /// The production Iteration 1a preparation, on the supplied model, exactly as SAM_UI drives it.
        /// </summary>
        public static Design Prepare(SAM.Analytical.AnalyticalModel analyticalModel, string zoneCategoryName, bool sizePartF)
        {
            Design result = new Design();

            // ---- the Part F sizing -----------------------------------------------------------------
            //
            // WPF Modify.AddVentilationByPartF, without its two dialogs: the shipped default rule set, the
            // shipped default setback factor, and PartFCalculator.Calculate(zoneCategoryName) - which is
            // the ONLY thing that applies the explicit dwelling filter and the duplicate-zone check.

            if (sizePartF)
            {
                SAM.Analytical.PartFCalculator partFCalculator = SAM.Analytical.Query.DefaultPartFCalculator();

                if (partFCalculator == null)
                {
                    result.Refusal = "The shipped Part F rule set could not be read, so the model cannot be sized.";

                    return result;
                }

                partFCalculator.AdjacencyCluster = analyticalModel.AdjacencyCluster;

                double setbackFlowRateFactor = SAM.Analytical.Query.DefaultPartFData() != null
                    ? SAM.Analytical.Query.DefaultPartFData().SetbackFlowRateFactor
                    : SAM.Analytical.PartFData.DefaultSetbackFlowRateFactor;

                partFCalculator.SetbackFlowRateFactor = setbackFlowRateFactor;

                W("== PRODUCTION PartFCalculator.Calculate ==");
                W("  rule set setback factor {0}", setbackFlowRateFactor.ToString("0.###", CultureInfo.InvariantCulture));
                W("  zone category: {0}", string.IsNullOrWhiteSpace(zoneCategoryName) ? "<none - single house mode>" : zoneCategoryName);

                bool calculated = partFCalculator.Calculate(zoneCategoryName);

                W("  Calculate returned {0}, {1} dwelling result(s)", calculated, partFCalculator.DwellingResults == null ? 0 : partFCalculator.DwellingResults.Count);

                analyticalModel = new SAM.Analytical.AnalyticalModel(analyticalModel, partFCalculator.AdjacencyCluster);
            }
            else
            {
                W("== Part F sizing NOT re-run - the model's own PartFSpaceData is used ==");
            }

            // ---- the dwelling scope ----------------------------------------------------------------

            List<SAM.Analytical.Zone> zones = analyticalModel.GetZones() ?? new List<SAM.Analytical.Zone>();

            List<SAM.Analytical.Zone> zones_Dwelling = SAM.Analytical.Query.PartFDwellingZones(zones) ?? new List<SAM.Analytical.Zone>();

            W("");
            W("== PRODUCTION Query.PartFDwellingZones ==");
            W("  {0} zone(s) on the model, {1} of them dwelling zone(s)", zones.Count, zones_Dwelling.Count);

            foreach (SAM.Analytical.Zone zone in zones_Dwelling)
            {
                W("    dwelling zone  {0}  {1}", zone.Guid.ToString("D"), zone.Name);
            }

            if (zones_Dwelling.Count == 0)
            {
                result.Refusal = "The model carries no dwelling zone Approved Document F would size, so there is no Iteration 1a to prepare.";

                return result;
            }

            // ---- the route each assessed zone states -----------------------------------------------
            //
            // The canonical word SAM_UI's picker hands over for BasePassive, and the only word it offers.
            // See SAM.Analytical.UI.PartOVentilationStrategyOption.

            Dictionary<Guid, string> dictionary_VentilationStrategy = new Dictionary<Guid, string>();

            foreach (SAM.Analytical.Zone zone in zones_Dwelling)
            {
                dictionary_VentilationStrategy[zone.Guid] = "MVHR";
            }

            // ---- Iteration 1a ----------------------------------------------------------------------
            //
            // ventilationUnitCapacityDescriptors null: no catalogue offered, which IS Iteration 1a.
            // isolate false: the whole building's thermal model, as the acceptance is defined over.

            W("");
            W("== PRODUCTION Modify.PreparePartOIteration  (BasePassive / Base MVHR) ==");

            SAM.Analytical.PartOIterationPreparation partOIterationPreparation = SAM.Analytical.Modify.PreparePartOIteration(
                analyticalModel,
                SAM.Analytical.Enums.PartOIteration.BasePassive,
                zones_Dwelling,
                dictionary_VentilationStrategy,
                null,
                false);

            W("  VentilationMode      {0}", partOIterationPreparation.VentilationMode);
            W("  AirflowApplication   {0}", partOIterationPreparation.AirflowApplication);
            W("  Successful           {0}", partOIterationPreparation.Successful);
            W("  DesignSupplyDuty     {0} l/s", partOIterationPreparation.DesignSupplyDuty_Lps.ToString("0.###", CultureInfo.InvariantCulture));
            W("  DesignExtractDuty    {0} l/s", partOIterationPreparation.DesignExtractDuty_Lps.ToString("0.###", CultureInfo.InvariantCulture));
            W("  systems              {0}", partOIterationPreparation.VentilationSystems.Count);
            W("  terminals            {0}", partOIterationPreparation.VentilationTerminals.Count);

            W("");
            foreach (string note in partOIterationPreparation.Notes)
            {
                W("  note: {0}", note);
            }

            foreach (string warning in partOIterationPreparation.Warnings)
            {
                W("  warning: {0}", warning);
            }

            foreach (string refusal in partOIterationPreparation.Refusals)
            {
                W("  REFUSAL: {0}", refusal);
            }

            if (partOIterationPreparation.Refusal != null)
            {
                W("");
                W("  FATAL REFUSAL: {0}", partOIterationPreparation.Refusal);

                result.Refusal = partOIterationPreparation.Refusal;

                return result;
            }

            if (partOIterationPreparation.AnalyticalModel == null)
            {
                result.Refusal = "The preparation returned no model.";

                return result;
            }

            result.AnalyticalModel = partOIterationPreparation.AnalyticalModel;
            result.Successful = partOIterationPreparation.Successful;

            Measure(result);

            return result;
        }

        // ================================================================== reading the design back off it

        /// <summary>
        /// Reads the prepared model's own ventilation objects back. This is a READ: the roles are whatever
        /// the design's terminals say, the duties are the terminals' own values, and a transfer direction is
        /// the direction the model's <c>SpaceAirMovement</c> states.
        /// </summary>
        internal static void Measure(Design design)
        {
            SAM.Analytical.AdjacencyCluster adjacencyCluster = design.AnalyticalModel.AdjacencyCluster;

            List<SAM.Analytical.Space> spaces = adjacencyCluster.GetSpaces() ?? new List<SAM.Analytical.Space>();

            Dictionary<Guid, SAM.Analytical.Space> dictionary_Space = new Dictionary<Guid, SAM.Analytical.Space>();
            foreach (SAM.Analytical.Space space in spaces)
            {
                dictionary_Space[space.Guid] = space;
            }

            //Which dwelling zone each space belongs to, so a leg can be checked against a dwelling boundary
            //without asking the model twice per leg.
            Dictionary<Guid, SAM.Analytical.Zone> dictionary_Dwelling = new Dictionary<Guid, SAM.Analytical.Zone>();

            foreach (SAM.Analytical.Zone zone in SAM.Analytical.Query.PartFDwellingZones(adjacencyCluster.GetObjects<SAM.Analytical.Zone>() ?? new List<SAM.Analytical.Zone>()) ?? new List<SAM.Analytical.Zone>())
            {
                foreach (SAM.Analytical.Space space in adjacencyCluster.GetRelatedObjects<SAM.Analytical.Space>(zone) ?? new List<SAM.Analytical.Space>())
                {
                    dictionary_Dwelling[space.Guid] = zone;
                }
            }

            List<SAM.Analytical.AirHandlingUnit> airHandlingUnits = adjacencyCluster.GetObjects<SAM.Analytical.AirHandlingUnit>() ?? new List<SAM.Analytical.AirHandlingUnit>();

            foreach (SAM.Analytical.AirHandlingUnit airHandlingUnit in airHandlingUnits)
            {
                design.AirHandlingUnitNames.Add(airHandlingUnit.Name);
            }

            design.AirHandlingUnitNames.Sort(StringComparer.Ordinal);

            // ---- one row per (system, room) --------------------------------------------------------

            foreach (SAM.Analytical.VentilationSystem ventilationSystem in adjacencyCluster.GetObjects<SAM.Analytical.VentilationSystem>() ?? new List<SAM.Analytical.VentilationSystem>())
            {
                Dictionary<Guid, Room> dictionary_Room = new Dictionary<Guid, Room>();

                foreach (SAM.Analytical.VentilationTerminal ventilationTerminal in adjacencyCluster.GetRelatedObjects<SAM.Analytical.VentilationTerminal>(ventilationSystem) ?? new List<SAM.Analytical.VentilationTerminal>())
                {
                    double designFlowRate_Lps = ventilationTerminal.DesignFlowRate_Lps ?? 0;

                    foreach (SAM.Analytical.Space space in adjacencyCluster.GetRelatedObjects<SAM.Analytical.Space>(ventilationTerminal) ?? new List<SAM.Analytical.Space>())
                    {
                        Room room;
                        if (!dictionary_Room.TryGetValue(space.Guid, out room))
                        {
                            SAM.Analytical.Zone zone;
                            dictionary_Dwelling.TryGetValue(space.Guid, out zone);

                            room = new Room
                            {
                                System = ventilationSystem.FullName,
                                SystemGuid = ventilationSystem.Guid,
                                Dwelling = zone == null ? "<none>" : zone.Name,
                                DwellingGuid = zone == null ? Guid.Empty : zone.Guid,
                                SpaceGuid = space.Guid,
                                SpaceName = space.Name,
                            };

                            dictionary_Room[space.Guid] = room;
                        }

                        //SUMMED per direction. A room subdivided into several terminals has one duty per
                        //direction, which is what the design states and what PR1 materialises.
                        if (ventilationTerminal.FlowClassification == SAM.Analytical.FlowClassification.Supply)
                        {
                            room.Supply_Lps += designFlowRate_Lps;
                        }
                        else if (ventilationTerminal.FlowClassification == SAM.Analytical.FlowClassification.Extract)
                        {
                            room.Extract_Lps += designFlowRate_Lps;
                        }
                    }
                }

                List<Room> rooms = new List<Room>(dictionary_Room.Values);
                rooms.Sort((x, y) => string.Compare(x.SpaceName, y.SpaceName, StringComparison.Ordinal));

                design.Rooms.AddRange(rooms);
            }

            design.Rooms.Sort((x, y) =>
            {
                int comparison = string.Compare(x.System, y.System, StringComparison.Ordinal);
                return comparison != 0 ? comparison : string.Compare(x.SpaceName, y.SpaceName, StringComparison.Ordinal);
            });

            // ---- the transfer legs, in the direction the design states -----------------------------
            //
            // Query.DesignTransferSpaceAirMovements is the SAME authority PR1 reads. Its key is
            // order-normalised, so the direction is taken from FromGuid/ToGuid, never from the key.

            Dictionary<(Guid, Guid), SAM.Analytical.Query.DesignTransferAirMovement> dictionary_Transfer =
                SAM.Analytical.Query.DesignTransferSpaceAirMovements(adjacencyCluster) ?? new Dictionary<(Guid, Guid), SAM.Analytical.Query.DesignTransferAirMovement>();

            foreach (KeyValuePair<(Guid, Guid), SAM.Analytical.Query.DesignTransferAirMovement> keyValuePair in dictionary_Transfer)
            {
                Guid guid_From = keyValuePair.Value.FromGuid;
                Guid guid_To = keyValuePair.Value.ToGuid;

                SAM.Analytical.Space space_From;
                SAM.Analytical.Space space_To;

                dictionary_Space.TryGetValue(guid_From, out space_From);
                dictionary_Space.TryGetValue(guid_To, out space_To);

                SAM.Analytical.Zone zone_From;
                dictionary_Dwelling.TryGetValue(guid_From, out zone_From);

                design.Legs.Add(new Leg
                {
                    FromGuid = guid_From,
                    FromName = space_From == null ? "<unknown>" : space_From.Name,
                    ToGuid = guid_To,
                    ToName = space_To == null ? "<unknown>" : space_To.Name,
                    //m3/s on the model; l/s is the unit every table and TAS itself uses.
                    Flow_Lps = keyValuePair.Value.SpaceAirMovement.AirFlow * 1000.0,
                    Dwelling = zone_From == null ? "<none>" : zone_From.Name,
                });
            }

            design.Legs.Sort((x, y) =>
            {
                int comparison = string.Compare(x.FromName, y.FromName, StringComparison.Ordinal);
                return comparison != 0 ? comparison : string.Compare(x.ToName, y.ToName, StringComparison.Ordinal);
            });
        }

        // ============================================================================ the Phase A report

        /// <summary>
        /// The Phase A transcript: the per-dwelling table, the dwelling-boundary check, and the per-room
        /// airflow continuity TAS actually gates on.
        /// </summary>
        public static void Transcript(Design design, string path)
        {
            W("");
            W("=================================== THE ANALYTICAL DESIGN ===================================");
            W("");
            W("air handling unit(s): {0}", design.AirHandlingUnitNames.Count == 0 ? "<none>" : string.Join(", ", design.AirHandlingUnitNames));
            W("");
            W("{0,-26} | {1,-22} | {2,-9} | {3,-9}", "system", "space", "supply l/s", "extract l/s");
            W(new string('-', 78));

            foreach (Room room in design.Rooms)
            {
                W("{0,-26} | {1,-22} | {2,9} | {3,9}",
                    Trim(room.System, 26),
                    Trim(room.SpaceName, 22),
                    room.Supply_Lps.ToString("0.###", CultureInfo.InvariantCulture),
                    room.Extract_Lps.ToString("0.###", CultureInfo.InvariantCulture));
            }

            W("");
            W("{0,-22} | {1,-22} | {2,-9} | {3}", "transfer from", "transfer to", "l/s", "dwelling");
            W(new string('-', 78));

            foreach (Leg leg in design.Legs)
            {
                W("{0,-22} | {1,-22} | {2,9} | {3}",
                    Trim(leg.FromName, 22),
                    Trim(leg.ToName, 22),
                    leg.Flow_Lps.ToString("0.###", CultureInfo.InvariantCulture),
                    leg.Dwelling);
            }

            // ---- the checks Phase A has to answer --------------------------------------------------

            Dictionary<Guid, Room> dictionary_Room = new Dictionary<Guid, Room>();
            Dictionary<Guid, string> dictionary_Dwelling = new Dictionary<Guid, string>();

            foreach (Room room in design.Rooms)
            {
                dictionary_Room[room.SpaceGuid] = room;
                dictionary_Dwelling[room.SpaceGuid] = string.Concat(room.DwellingGuid.ToString("D"), "|", room.System);
            }

            //Per-room continuity: inflow (supply + transfer in) against outflow (extract + transfer out).
            //This is what TAS gates on - unit-level balance is a consequence of it, never the condition.
            Dictionary<Guid, double> dictionary_In = new Dictionary<Guid, double>();
            Dictionary<Guid, double> dictionary_Out = new Dictionary<Guid, double>();
            Dictionary<Guid, string> dictionary_Name = new Dictionary<Guid, string>();

            foreach (Room room in design.Rooms)
            {
                Add(dictionary_In, room.SpaceGuid, room.Supply_Lps);
                Add(dictionary_Out, room.SpaceGuid, room.Extract_Lps);
                dictionary_Name[room.SpaceGuid] = room.SpaceName;
            }

            foreach (Leg leg in design.Legs)
            {
                Add(dictionary_Out, leg.FromGuid, leg.Flow_Lps);
                Add(dictionary_In, leg.ToGuid, leg.Flow_Lps);

                if (!dictionary_Name.ContainsKey(leg.FromGuid))
                {
                    dictionary_Name[leg.FromGuid] = leg.FromName;
                }

                if (!dictionary_Name.ContainsKey(leg.ToGuid))
                {
                    dictionary_Name[leg.ToGuid] = leg.ToName;
                }
            }

            List<Guid> guids = new List<Guid>(dictionary_Name.Keys);
            guids.Sort((x, y) => string.Compare(dictionary_Name[x], dictionary_Name[y], StringComparison.Ordinal));

            int count_Discontinuous = 0;

            W("");
            W("{0,-22} | {1,-9} | {2,-9} | {3}", "room", "in l/s", "out l/s", "continuity");
            W(new string('-', 68));

            foreach (Guid guid in guids)
            {
                double flow_In;
                double flow_Out;

                dictionary_In.TryGetValue(guid, out flow_In);
                dictionary_Out.TryGetValue(guid, out flow_Out);

                bool continuous = Math.Abs(flow_In - flow_Out) <= 1e-6;

                if (!continuous)
                {
                    count_Discontinuous++;
                }

                W("{0,-22} | {1,9} | {2,9} | {3}",
                    Trim(dictionary_Name[guid], 22),
                    flow_In.ToString("0.###", CultureInfo.InvariantCulture),
                    flow_Out.ToString("0.###", CultureInfo.InvariantCulture),
                    continuous ? "ok" : "*** DISCONTINUOUS ***");
            }

            //Cross-dwelling transfer: a leg whose two ends are in different dwellings, or in different
            //systems, is a shortcut between two flats and must not exist.
            int count_CrossDwelling = 0;

            foreach (Leg leg in design.Legs)
            {
                string key_From;
                string key_To;

                dictionary_Dwelling.TryGetValue(leg.FromGuid, out key_From);
                dictionary_Dwelling.TryGetValue(leg.ToGuid, out key_To);

                //A zero-terminal hall carries no system relation, so a null end is not a mismatch on its
                //own - only two NON-null ends that disagree are.
                if (key_From != null && key_To != null && !string.Equals(key_From, key_To, StringComparison.Ordinal))
                {
                    count_CrossDwelling++;
                    W("*** CROSS-DWELLING TRANSFER: {0} -> {1}", leg.FromName, leg.ToName);
                }
            }

            W("");
            W("PHASE A: {0} room(s), {1} transfer leg(s), {2} discontinuous room(s), {3} cross-dwelling leg(s)",
                guids.Count, design.Legs.Count, count_Discontinuous, count_CrossDwelling);

            W("PHASE A VERDICT: {0}",
                design.Successful && count_Discontinuous == 0 && count_CrossDwelling == 0
                    ? "the fixture is a valid real-design reference"
                    : "FIXTURE UNSUITABLE FOR PR2 RE-ACCEPTANCE");

            Flush(path);
        }

        private static void Add(Dictionary<Guid, double> dictionary, Guid guid, double value)
        {
            double existing;
            dictionary.TryGetValue(guid, out existing);
            dictionary[guid] = existing + value;
        }

        private static string Trim(string value, int length)
        {
            if (value == null)
            {
                return string.Empty;
            }

            return value.Length <= length ? value : value.Substring(0, length);
        }

        // ================================================================================ the entry point

        public static int Main(string path_SAM, string directory, string zoneCategoryName, bool sizePartF)
        {
            log.Clear();

            Directory.CreateDirectory(directory);

            W("PHASE A - prove the fixture");
            W("  model  {0}", path_SAM);
            W("  bytes  {0}", new FileInfo(path_SAM).Length);
            W("");

            SAM.Analytical.AnalyticalModel analyticalModel = SAM.Core.Convert.ToSAM<SAM.Analytical.AnalyticalModel>(path_SAM)[0];

            SAM.Analytical.AdjacencyCluster adjacencyCluster = analyticalModel.AdjacencyCluster;

            List<SAM.Analytical.Space> spaces = adjacencyCluster.GetSpaces() ?? new List<SAM.Analytical.Space>();
            List<SAM.Analytical.Zone> zones = adjacencyCluster.GetObjects<SAM.Analytical.Zone>() ?? new List<SAM.Analytical.Zone>();

            W("== the model as it arrives ==");
            W("  spaces                {0}", spaces.Count);
            W("  zones                 {0}", zones.Count);

            HashSet<string> zoneCategories = adjacencyCluster.GetZoneCategories() ?? new HashSet<string>();
            W("  zone categories       {0}", zoneCategories.Count == 0 ? "<none>" : string.Join(", ", zoneCategories));

            int count_PartFSpaceData = 0;
            foreach (SAM.Analytical.Space space in spaces)
            {
                if (space.GetValue<SAM.Core.IJSAMObject>(SAM.Analytical.SpaceParameter.PartFSpaceData) != null)
                {
                    count_PartFSpaceData++;
                }
            }

            W("  spaces with PartFSpaceData    {0} of {1}", count_PartFSpaceData, spaces.Count);
            W("  AirHandlingUnit               {0}", (adjacencyCluster.GetObjects<SAM.Analytical.AirHandlingUnit>() ?? new List<SAM.Analytical.AirHandlingUnit>()).Count);
            W("  VentilationSystem             {0}", (adjacencyCluster.GetObjects<SAM.Analytical.VentilationSystem>() ?? new List<SAM.Analytical.VentilationSystem>()).Count);
            W("  VentilationTerminal           {0}", (adjacencyCluster.GetObjects<SAM.Analytical.VentilationTerminal>() ?? new List<SAM.Analytical.VentilationTerminal>()).Count);
            W("  SpaceAirMovement              {0}", (adjacencyCluster.GetObjects<SAM.Analytical.SpaceAirMovement>() ?? new List<SAM.Analytical.SpaceAirMovement>()).Count);

            foreach (SAM.Analytical.Zone zone in zones)
            {
                W("  zone  {0}  {1,-28}  category={2}  spaces={3}",
                    zone.Guid.ToString("D"),
                    Trim(zone.Name, 28),
                    zone.GetValue<string>(SAM.Analytical.ZoneParameter.ZoneCategory) ?? "<none>",
                    (adjacencyCluster.GetRelatedObjects<SAM.Analytical.Space>(zone) ?? new List<SAM.Analytical.Space>()).Count);
            }

            W("");

            Design design = Prepare(analyticalModel, zoneCategoryName, sizePartF);

            if (design.Refusal != null)
            {
                W("");
                W("PHASE A VERDICT: FIXTURE UNSUITABLE FOR PR2 RE-ACCEPTANCE");
                W("  production refusal: {0}", design.Refusal);

                Flush(Path.Combine(directory, "phase-a.log"));

                return 5;
            }

            //The PREPARED model, saved so Phase B materialises exactly what Phase A proved.
            string path_Prepared = Path.Combine(directory, "prepared.sam");

            SAM.Core.Convert.ToFile(new SAM.Core.IJSAMObject[] { design.AnalyticalModel }, path_Prepared);

            W("prepared model saved: {0}  ({1} bytes)", path_Prepared, new FileInfo(path_Prepared).Length);

            Transcript(design, Path.Combine(directory, "phase-a.log"));

            //The machine-readable oracle, written from the SAME measurement the transcript printed.
            Oracle(design, Path.Combine(directory, "intent.tsv"));

            return design.Successful ? 0 : 5;
        }

        /// <summary>
        /// The engineering oracle the native TAS comparison is made against - one tab-separated row per
        /// materialised duty, written before any TAS conversion happens.
        /// </summary>
        /// <summary>The same measurement, over a model a caller already has - so a scoped cluster can be
        /// compared with the Phase A design row for row rather than by argument.</summary>
        internal static Design MeasureModel(SAM.Analytical.AnalyticalModel analyticalModel)
        {
            Design result = new Design { AnalyticalModel = analyticalModel, Successful = true };

            Measure(result);

            return result;
        }

        public static void Oracle(Design design, string path)
        {
            List<string> lines = new List<string>();

            lines.Add(string.Join("\t", new string[] { "kind", "system", "dwelling", "room_guid", "room_name", "supply_lps", "extract_lps", "from_guid", "to_guid", "transfer_lps" }));

            foreach (Room room in design.Rooms)
            {
                lines.Add(string.Join("\t", new string[]
                {
                    "room",
                    room.System,
                    room.Dwelling,
                    room.SpaceGuid.ToString("D"),
                    room.SpaceName,
                    room.Supply_Lps.ToString("R", CultureInfo.InvariantCulture),
                    room.Extract_Lps.ToString("R", CultureInfo.InvariantCulture),
                    string.Empty,
                    string.Empty,
                    string.Empty,
                }));
            }

            foreach (Leg leg in design.Legs)
            {
                lines.Add(string.Join("\t", new string[]
                {
                    "transfer",
                    string.Empty,
                    leg.Dwelling,
                    string.Empty,
                    string.Concat(leg.FromName, " -> ", leg.ToName),
                    string.Empty,
                    string.Empty,
                    leg.FromGuid.ToString("D"),
                    leg.ToGuid.ToString("D"),
                    leg.Flow_Lps.ToString("R", CultureInfo.InvariantCulture),
                }));
            }

            File.WriteAllLines(path, lines);

            W("intent table written: {0}  ({1} row(s))", path, lines.Count - 1);
        }
    }
