// Native hourly evidence for every fan of a produced TPD: what operation carrier TAS holds for it, and
// what it actually delivered hour by hour.
//
// Works on a COPY of the document (the route's own output is left as produced). Each air system is
// simulated with a raw late-bound ISystem.Simulate and its fans are read straight afterwards, because
// TAS replaces a system's in-memory result surface when the next system in the document is simulated.
//
// Every read is late-bound: the typed ISystemComponent accessors throw on this interop.
//
// Nothing here is production code.

using System;
using System.Collections.Generic;
using System.Globalization;
using System.IO;
using System.Reflection;
using System.Text;

namespace Inv
{
    internal static class FanProbe
    {
        private static readonly StringBuilder log = new StringBuilder();

        private static void W(string format, params object[] args)
        {
            string line = args == null || args.Length == 0 ? format : string.Format(CultureInfo.InvariantCulture, format, args);
            Console.WriteLine(line);
            log.AppendLine(line);
        }

        private static double Dbl(object value)
        {
            try
            {
                return value == null ? double.NaN : Convert.ToDouble(value, CultureInfo.InvariantCulture);
            }
            catch
            {
                return double.NaN;
            }
        }

        private static List<object> Components(object container, int depth = 0)
        {
            List<object> result = new List<object>();
            if (container == null || depth > 8)
            {
                return result;
            }

            for (int index = 1; index <= 20000; index++)
            {
                object component = Dump.Call(container, "GetComponent", index);
                if (component == null)
                {
                    break;
                }

                result.Add(component);

                if (component is global::TPD.ComponentGroup)
                {
                    result.AddRange(Components(component, depth + 1));
                }
            }

            return result;
        }

        /// <summary>One results variable of one component, hourly, walked with foreach.</summary>
        private static bool TryResults(object component, int variable, int start, int count, out List<double> values, out string failure)
        {
            values = null;
            failure = null;

            object data;

            try
            {
                data = component.GetType().InvokeMember(
                    "GetResultsData",
                    BindingFlags.InvokeMethod,
                    null,
                    component,
                    new object[]
                    {
                        (int)global::TPD.tpdResultsPeriod.tpdResultsPeriodHourly,
                        (int)global::TPD.tpdCombinerType.tpdCombinerTypeMax,
                        variable,
                        start,
                        count,
                    },
                    CultureInfo.InvariantCulture);
            }
            catch (Exception exception)
            {
                failure = "threw " + (exception.InnerException ?? exception).Message.Trim();
                return false;
            }

            Array array = data as Array;
            if (array == null)
            {
                failure = data == null ? "<null>" : "<not an array: " + data.GetType().Name + ">";
                return false;
            }

            values = new List<double>();
            foreach (object element in array)
            {
                values.Add(Dbl(element));
            }

            return true;
        }

        private static string Schedule(object fan)
        {
            object plantSchedule = Dump.Call(fan, "GetSchedule");
            if (plantSchedule == null)
            {
                return "schedule=<none attached>";
            }

            int ones = 0;
            int zeros = 0;
            int other = 0;
            int read = 0;

            for (int hour = 1; hour <= 8760; hour++)
            {
                object value = Dump.Call(plantSchedule, "GetYearlyValue", hour);
                if (value == null)
                {
                    continue;
                }

                read++;
                double v = Dbl(value);
                if (v == 1) { ones++; } else if (v == 0) { zeros++; } else { other++; }
            }

            return string.Format(
                CultureInfo.InvariantCulture,
                "schedule name={0} Type={1} FunctionType={2} FunctionLoads={3} GetNumOperableHours={4} GetYearlyValue(1..8760): read={5} ones={6} zeros={7} other={8}; GetYearlyValue(0)={9} GetYearlyValue(8761)={10}",
                Dump.Get(plantSchedule, "Name"),
                Dump.Get(plantSchedule, "Type"),
                Dump.Get(plantSchedule, "FunctionType"),
                Dump.Get(plantSchedule, "FunctionLoads"),
                Dump.CallText(plantSchedule, "GetNumOperableHours"),
                read, ones, zeros, other,
                Dump.CallText(plantSchedule, "GetYearlyValue", 0),
                Dump.CallText(plantSchedule, "GetYearlyValue", 8761));
        }

        public static int Run(string path_TPD, int startHour, int endHour, string pathOut)
        {
            log.Clear();

            int expected = endHour - startHour + 1;
            string path_Copy = Path.ChangeExtension(path_TPD, ".fanprobe.tpd");
            File.Copy(path_TPD, path_Copy, true);

            W("FANPROBE {0}", path_TPD);
            W("  on copy {0}; expected period hours {1}..{2} = {3} hour(s)", path_Copy, startHour, endHour, expected);
            W("  results read: ISystemComponent.GetResultsData(tpdResultsPeriodHourly, Max, variable, {0}, {1}), late-bound", startHour + 1, expected);
            W("  SAM's FanDataType maps variable 1 = FlowRate, 2 = Pressure, 3 = Temperature ... 9 = Load");

            int count_Fans = 0;
            int count_Continuous = 0;

            global::TPD.TPDDoc tPDDoc = new global::TPD.TPDDoc();

            try
            {
                tPDDoc.Open(path_Copy);

                object energyCentre = tPDDoc.EnergyCentre;
                int count_PlantRoom = (int)Dbl(Dump.Call(energyCentre, "GetPlantRoomCount"));

                for (int p = 1; p <= count_PlantRoom; p++)
                {
                    object plantRoom = Dump.Call(energyCentre, "GetPlantRoom", p);
                    int count_System = (int)Dbl(Dump.Call(plantRoom, "GetSystemCount"));

                    for (int s = 1; s <= count_System; s++)
                    {
                        object system = Dump.Call(plantRoom, "GetSystem", s);

                        W("");
                        W("== SYSTEM {0} {1}", Dump.Get(system, "Name"), Dump.Get(system, "GUID"));

                        string returned = Dump.CallText(system, "Simulate", startHour + 1, endHour + 1, 0);
                        W("  raw late-bound Simulate({0}, {1}, 0) answered {2}", startHour + 1, endHour + 1, returned);

                        foreach (object component in Components(system))
                        {
                            if (!(component is global::TPD.Fan))
                            {
                                continue;
                            }

                            count_Fans++;

                            object designFlowRate = Dump.Obj(component, "DesignFlowRate");
                            double design = Dbl(Dump.Obj(designFlowRate, "Value"));

                            W("");
                            W("  FAN {0} {1}", Dump.Get(component, "GUID"), Dump.Get(component, "Name"));
                            W("    DesignFlowType={0} DesignFlowRate.Value={1} (derived duty, l/s) HeatGainFactor={2} ControlType={3} MinimumFlowFraction={4}",
                                Dump.Get(component, "DesignFlowType"), design.ToString("R", CultureInfo.InvariantCulture),
                                Dump.Get(component, "HeatGainFactor"), Dump.Get(component, "ControlType"), Dump.Get(component, "MinimumFlowFraction"));
                            W("    {0}", Schedule(component));

                            bool continuous = false;
                            int atDutyHours = 0;

                            //Variable 9 is the only hourly series a fan answers. It is the fan's Load in W,
                            //Q x dp / eta: 0.044 m3/s x 1000 Pa / 1.0 = 44 W on the fresh air fan and
                            //x 600 Pa = 26.4 W on the return fan. So the delivered flow in l/s is
                            //Load x eta / dp x 1000, and that is what is held against the derived duty.
                            double pressure = Dbl(Dump.Obj(component, "Pressure"));
                            double efficiency = Dbl(Dump.Obj(Dump.Obj(component, "OverallEfficiency"), "Value"));
                            W("    Pressure={0:R} Pa OverallEfficiency.Value={1:R} -> delivered flow (l/s) = var 9 Load (W) x {1:R} / {0:R} x 1000",
                                pressure, efficiency);

                            for (int variable = 0; variable <= 24; variable++)
                            {
                                if (!TryResults(component, variable, startHour + 1, expected, out List<double> values, out string failure))
                                {
                                    W("    var {0,2}: {1}", variable, failure);
                                    continue;
                                }

                                double min = double.MaxValue;
                                double max = double.MinValue;
                                int running = 0;
                                int atDuty = 0;
                                int nonFinite = 0;
                                int firstOff = -1;

                                for (int i = 0; i < values.Count; i++)
                                {
                                    double v = values[i];
                                    if (double.IsNaN(v) || double.IsInfinity(v))
                                    {
                                        nonFinite++;
                                        continue;
                                    }

                                    if (v < min) { min = v; }
                                    if (v > max) { max = v; }

                                    if (Math.Abs(v) > 1e-9)
                                    {
                                        running++;
                                    }
                                    else if (firstOff < 0)
                                    {
                                        firstOff = startHour + i;
                                    }

                                    if (!double.IsNaN(design) && Math.Abs(v - design) <= Math.Max(1e-6, Math.Abs(design) * 1e-4))
                                    {
                                        atDuty++;
                                    }
                                }

                                W("    var {0,2}: n={1} min={2:R} max={3:R} nonZeroHours={4} hoursAtDesignDuty={5} nonFinite={6} firstZeroHour={7}",
                                    variable, values.Count, min, max, running, atDuty, nonFinite, firstOff < 0 ? "none" : firstOff.ToString(CultureInfo.InvariantCulture));

                                if (variable == 9)
                                {
                                    double flowMin = double.MaxValue;
                                    double flowMax = double.MinValue;
                                    int hoursRunning = 0;
                                    int hoursAtDuty = 0;
                                    int firstOffHour = -1;

                                    for (int i = 0; i < values.Count; i++)
                                    {
                                        double flow = values[i] * efficiency / pressure * 1000.0;
                                        if (double.IsNaN(flow) || double.IsInfinity(flow))
                                        {
                                            continue;
                                        }

                                        if (flow < flowMin) { flowMin = flow; }
                                        if (flow > flowMax) { flowMax = flow; }

                                        if (flow > 1e-6)
                                        {
                                            hoursRunning++;
                                        }
                                        else if (firstOffHour < 0)
                                        {
                                            firstOffHour = startHour + i;
                                        }

                                        if (Math.Abs(flow - design) <= Math.Max(1e-3, Math.Abs(design) * 1e-4))
                                        {
                                            hoursAtDuty++;
                                        }
                                    }

                                    W("    delivered flow from var 9: hours available={0} running={1} atDerivedDuty={2} min={3:R} l/s max={4:R} l/s firstOffHour={5}",
                                        values.Count, hoursRunning, hoursAtDuty, flowMin, flowMax, firstOffHour < 0 ? "none" : firstOffHour.ToString(CultureInfo.InvariantCulture));

                                    continuous = values.Count == expected && nonFinite == 0 && hoursAtDuty == expected;
                                    atDutyHours = hoursAtDuty;
                                }
                            }

                            W("    VERDICT {0}: delivered its derived design duty in {1} of {2} hour(s)",
                                continuous ? "CONTINUOUS" : "NOT CONTINUOUS", atDutyHours, expected);

                            if (continuous)
                            {
                                count_Continuous++;
                            }
                        }
                    }
                }
            }
            finally
            {
                try { tPDDoc.Close(); } catch { }
            }

            W("");
            W("SUMMARY {0} of {1} fan(s) delivered their derived design duty in every hour of {2}..{3}", count_Continuous, count_Fans, startHour, endHour);

            if (!string.IsNullOrWhiteSpace(pathOut))
            {
                File.WriteAllText(pathOut, log.ToString());
            }

            return count_Fans > 0 && count_Continuous == count_Fans ? 0 : 4;
        }
    }
}
