// Exhaustive, late-bound observation of a TPD's native COM graph.
//
// Every read goes through IDispatch (Type.InvokeMember) rather than the typed interop accessors,
// because checkpoint 1 measured that the typed ISystemComponent.GUID/.Name accessors THROW on a
// SystemZone. Reading late-bound is therefore the only way to observe the graph honestly, and it also
// lets a property be reported as "threw" instead of aborting the walk.
//
// The property NAMES come from the interop type library (reflection over the interface types), so this
// dumps what TAS actually declares, not a hand-written guess at what might be there.

using System;
using System.Collections.Generic;
using System.Globalization;
using System.Linq;
using System.Reflection;
using System.Text;

namespace Inv
{
    internal static class Dump
    {
        /// <summary>Property names declared by an interop interface, in a stable order.</summary>
        private static readonly Dictionary<Type, string[]> propertyNames = new Dictionary<Type, string[]>();

        private static string[] PropertyNames(Type type)
        {
            if (!propertyNames.TryGetValue(type, out string[] result))
            {
                List<string> names = new List<string>();

                foreach (Type current in new[] { type }.Concat(type.GetInterfaces()))
                {
                    foreach (PropertyInfo propertyInfo in current.GetProperties())
                    {
                        if (propertyInfo.GetIndexParameters().Length == 0 && !names.Contains(propertyInfo.Name))
                        {
                            names.Add(propertyInfo.Name);
                        }
                    }
                }

                names.Sort(StringComparer.Ordinal);
                result = names.ToArray();
                propertyNames[type] = result;
            }

            return result;
        }

        /// <summary>Reads one property late-bound, reporting a refusal rather than throwing.</summary>
        public static string Get(object comObject, string name)
        {
            if (comObject == null)
            {
                return "<no object>";
            }

            try
            {
                object value = comObject.GetType().InvokeMember(
                    name,
                    BindingFlags.GetProperty,
                    null,
                    comObject,
                    null,
                    CultureInfo.InvariantCulture);

                return Format(value);
            }
            catch (Exception exception)
            {
                Exception inner = exception is TargetInvocationException && exception.InnerException != null
                    ? exception.InnerException
                    : exception;

                return string.Concat("<threw ", inner.GetType().Name, ": ", Compact(inner.Message), ">");
            }
        }


        /// <summary>
        /// Reads an OBJECT-valued property late-bound (FlowRate, FreshAir, DesignFlowRate ...).
        /// These must go through BindingFlags.GetProperty on the property NAME; invoking "get_Xxx" as a
        /// method is not what IDispatch exposes and silently answers null, which reads in a dump as
        /// "TAS has no such carrier" when in fact it does.
        /// </summary>
        public static object Obj(object comObject, string name)
        {
            if (comObject == null)
            {
                return null;
            }

            try
            {
                return comObject.GetType().InvokeMember(
                    name,
                    BindingFlags.GetProperty,
                    null,
                    comObject,
                    null,
                    CultureInfo.InvariantCulture);
            }
            catch
            {
                return null;
            }
        }

        /// <summary>Calls one method late-bound, reporting a refusal rather than throwing.</summary>
        public static object Call(object comObject, string name, params object[] args)
        {
            if (comObject == null)
            {
                return null;
            }

            try
            {
                return comObject.GetType().InvokeMember(
                    name,
                    BindingFlags.InvokeMethod,
                    null,
                    comObject,
                    args,
                    CultureInfo.InvariantCulture);
            }
            catch
            {
                return null;
            }
        }

        public static string CallText(object comObject, string name, params object[] args)
        {
            if (comObject == null)
            {
                return "<no object>";
            }

            try
            {
                return Format(comObject.GetType().InvokeMember(
                    name,
                    BindingFlags.InvokeMethod,
                    null,
                    comObject,
                    args,
                    CultureInfo.InvariantCulture));
            }
            catch (Exception exception)
            {
                Exception inner = exception is TargetInvocationException && exception.InnerException != null
                    ? exception.InnerException
                    : exception;

                return string.Concat("<threw ", inner.GetType().Name, ": ", Compact(inner.Message), ">");
            }
        }

        private static string Compact(string message)
        {
            return (message ?? string.Empty).Replace("\r", " ").Replace("\n", " ").Trim();
        }

        private static string Format(object value)
        {
            if (value == null)
            {
                return "<null>";
            }

            if (value is string)
            {
                return string.Concat("\"", value, "\"");
            }

            if (value is double || value is float)
            {
                return Convert.ToDouble(value, CultureInfo.InvariantCulture).ToString("R", CultureInfo.InvariantCulture);
            }

            if (value.GetType().IsEnum || value is int || value is short || value is bool)
            {
                return value.ToString();
            }

            if (System.Runtime.InteropServices.Marshal.IsComObject(value))
            {
                return string.Concat("<com ", value.GetType().Name, ">");
            }

            return value.ToString();
        }

        /// <summary>Every declared property of an object, one per line, late-bound.</summary>
        public static void Properties(StringBuilder stringBuilder, string indent, object comObject, Type declaringType)
        {
            foreach (string name in PropertyNames(declaringType))
            {
                stringBuilder.AppendLine(string.Concat(indent, name, " = ", Get(comObject, name)));
            }
        }

        /// <summary>A SizedFlowVariable is where an absolute duty actually lives, so it gets its own block.</summary>
        public static void SizedFlow(StringBuilder stringBuilder, string indent, string label, object sizedFlowVariable)
        {
            if (sizedFlowVariable == null)
            {
                stringBuilder.AppendLine(string.Concat(indent, label, " = <null>"));
                return;
            }

            stringBuilder.AppendLine(string.Concat(
                indent, label,
                " Value=", Get(sizedFlowVariable, "Value"),
                " Type=", Get(sizedFlowVariable, "Type"),
                " Method=", Get(sizedFlowVariable, "Method"),
                " SizeFraction=", Get(sizedFlowVariable, "SizeFraction"),
                " SizeValue1=", Get(sizedFlowVariable, "SizeValue1"),
                " SizeValue2=", Get(sizedFlowVariable, "SizeValue2"),
                " DesignConditions=", CallText(sizedFlowVariable, "GetDesignConditionCount")));
        }
    }
}
