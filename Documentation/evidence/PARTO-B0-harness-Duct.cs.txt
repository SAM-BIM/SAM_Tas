using System;
using System.Collections;
using System.Collections.Generic;
using System.Globalization;
using System.IO;
using System.Linq;
using TPD;

namespace B0P
{
    public static partial class Steps
    {
        // ducts <file.tpd> <tsd> <tbd> <outCsv>
        public static void Ducts(string[] args)
        {
            string path = args[1], tsd = args[2], tbd = args[3], outCsv = args[4];
            TPDDoc doc = new TPDDoc();
            doc.Open(path);
            var series = new List<Tuple<string, double[]>>();
            try
            {
                EnergyCentre ec = doc.EnergyCentre;
                for (int i = 1; i <= ec.GetTSDDataCount(); i++) { dynamic td = ec.GetTSDData(i); td.TSDPath = tsd; td.TBDPath = tbd; }
                if (args.Length > 5 && args[5] == "dv0")
                {
                    for (int q = 1; q <= ec.GetPlantRoomCount(); q++)
                        for (int s2 = 1; s2 <= ec.GetPlantRoom(q).GetSystemCount(); s2++)
                            foreach (SystemComponent sc2 in Components(ec.GetPlantRoom(q).GetSystem(s2)))
                                if (sc2 is SystemZone) ((SystemZone)sc2).DisplacementVent = 0;
                    W("DisplacementVent cleared");
                }
                int simdata = (int)tpdSimulationData.tpdSimulationDataLoad + (int)tpdSimulationData.tpdSimulationDataPipe
                    + (int)tpdSimulationData.tpdSimulationDataDuct + (int)tpdSimulationData.tpdSimulationDataSimEvents
                    + (int)tpdSimulationData.tpdSimulationDataCont;
                for (int p = 1; p <= ec.GetPlantRoomCount(); p++)
                {
                    PlantRoom pr = ec.GetPlantRoom(p);
                    DateTime t0 = DateTime.UtcNow;
                    string answer = pr.SimulateEx(1, 8760, 15, ec.ExternalPollutant.Value, 10.0, simdata, 1, 0);
                    W("SIMULATE-EX|{0}|{1}|{2:F1}s", CName(pr), answer, (DateTime.UtcNow - t0).TotalSeconds);
                    for (int s = 1; s <= pr.GetSystemCount(); s++)
                    {
                        global::TPD.System sys = pr.GetSystem(s);
                        string sname = CName(sys);
                        int dc = sys.GetDuctCount();
                        for (int d = 1; d <= dc; d++)
                        {
                            Duct duct = sys.GetDuct(d);
                            if (duct == null) continue;
                            object up = duct.GetUpstreamComponent(), dn = duct.GetDownstreamComponent();
                            string tag = string.Format("{0}|{1}:{2}->{3}:{4}", sname, CKind(up), CName(up), CKind(dn), CName(dn));
                            for (int v = 1; v <= 8; v++)
                            {
                                double[] r = Results(duct, v, 1, 8760);
                                if (r == null) { if (v <= 2) W("  DUCT {0,-64} var{1} NULL {2}", tag, v, LastDiag); continue; }
                                W("  DUCT {0,-64} var{1,-2} n={2} min={3:F3} max={4:F3} mean={5:F3}", tag, v, r.Length, r.Min(), r.Max(), r.Average());
                                series.Add(Tuple.Create(tag + "|var" + v, r));
                            }
                        }
                        foreach (SystemComponent sc in Components(sys))
                        {
                            if (!(sc is SystemZone)) continue;
                            double[] r9 = Results(sc, 9, 1, 8760);
                            if (r9 != null) series.Add(Tuple.Create(sname + "|ZONE:" + CName(sc) + "|var9", r9));
                        }
                    }
                }
            }
            finally { doc.Close(); }
            using (StreamWriter sw = new StreamWriter(outCsv, false))
            {
                sw.WriteLine("hour," + string.Join(",", series.Select(x => x.Item1)));
                int n = series.Count == 0 ? 0 : series.Min(x => x.Item2.Length);
                for (int h = 0; h < n; h++)
                {
                    sw.Write(h);
                    foreach (var s in series) { sw.Write(','); sw.Write(s.Item2[h].ToString("R", CultureInfo.InvariantCulture)); }
                    sw.WriteLine();
                }
            }
            W("WROTE {0} ({1} series)", outCsv, series.Count);
        }
    }
}
