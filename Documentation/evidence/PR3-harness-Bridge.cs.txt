// PR3 - the ResultantTemperature thermostat bridge on the real Iteration 1a chain (SAM-BIM/SAM#111).
//
//   bridge      the PR2 production route, then the PR3 production Create.ThermostatBridge on the route it
//               returned, in one process. Persists the route's ZoneTemperature, bindings and the bridge's
//               ResultantTemperature as TSV so the read-backs below are independent of the code under test.
//   bridgecheck raw late-bound read-back through TAS's own accessors: every thermostat slot of every internal
//               condition of every bridged zone against the Systems series; the second TSD's resultant and
//               dry bulb; the hour alignment by +-1 h shift tests.
//   rename      the identity challenge: every zone in a copy of the source renamed to ONE shared string, the
//               bridge re-run on the route's own persisted output, and each room's resultant compared.
//
// Nothing here authors, balances or corrects the design. The only cluster write is Pr1.ScopeSystems, whose
// invariance is asserted exactly as Run does.

using System;
using System.Collections.Generic;
using System.Globalization;
using System.IO;
using System.Linq;

namespace Inv
{
    internal static class Bridge
    {
        private const int Hours = 8760;

        private static void W(string format, params object[] args)
        {
            Route.W(format, args);
        }

        private static string R(double value)
        {
            return value.ToString("R", CultureInfo.InvariantCulture);
        }

        private static string Key(string reference)
        {
            return Guid.TryParse((reference ?? string.Empty).Trim(), out Guid guid) ? guid.ToString("D") : (reference ?? string.Empty).Trim().ToUpperInvariant();
        }

        // =========================================================================================== bridge
        public static int Main(string directory_Design, string directory_Source, string directory, double tolerance)
        {
            Directory.CreateDirectory(directory);

            string path_Design = Path.Combine(directory_Design, "prepared.sam");
            string path_TBD = Path.Combine(directory, "src.tbd");
            string path_TSD = Path.Combine(directory, "src.tsd");
            string path_TPD = Path.Combine(directory, "acc.tpd");
            string path_Bridge = Path.Combine(directory, "bridge.tbd");

            File.Copy(Path.Combine(directory_Source, "src.tbd"), path_TBD, true);
            File.Copy(Path.Combine(directory_Source, "src.tsd"), path_TSD, true);

            string hash_TBD_Before = Src.Hash(path_TBD);
            string hash_TSD_Before = Src.Hash(path_TSD);

            W("PR3 - the ResultantTemperature thermostat bridge on the real chain");
            W("  design    {0}  sha256 {1}", path_Design, Src.Hash(path_Design));
            W("  src.tbd   sha256 {0}  (copied from {1})", hash_TBD_Before, directory_Source);
            W("  src.tsd   sha256 {0}", hash_TSD_Before);
            W("  tolerance {0} K (achieved air vs imposed)", R(tolerance));
            W("");

            SAM.Analytical.AnalyticalModel analyticalModel = SAM.Core.Convert.ToSAM<SAM.Analytical.AnalyticalModel>(path_Design)[0];
            SAM.Analytical.AdjacencyCluster adjacencyCluster = analyticalModel.AdjacencyCluster;

            Dictionary<Guid, string> name_By_Space = new Dictionary<Guid, string>();
            foreach (SAM.Analytical.Space space in adjacencyCluster.GetSpaces())
            {
                name_By_Space[space.Guid] = space.Name;
            }

            // ------------------------------------------------------------ the design scope, exactly as Run
            Prep.Design design_Before = Prep.MeasureModel(new SAM.Analytical.AnalyticalModel(analyticalModel, adjacencyCluster));
            string path_Intent_Before = Path.Combine(directory, "intent-before-scope.tsv");
            string path_Intent_After = Path.Combine(directory, "intent.tsv");
            Prep.Oracle(design_Before, path_Intent_Before);

            List<string> notes_System = new List<string>();
            int count_OutOfScope = Pr1.ScopeSystems(adjacencyCluster, "noterminal", notes_System);
            W("{0} ventilation system(s) out of scope (no design terminal).", count_OutOfScope);

            Prep.Design design = Prep.MeasureModel(new SAM.Analytical.AnalyticalModel(analyticalModel, adjacencyCluster));
            Prep.Oracle(design, path_Intent_After);

            bool identical = string.Equals(File.ReadAllText(path_Intent_Before), File.ReadAllText(path_Intent_After), StringComparison.Ordinal);
            W("intent table {0} before and after the scoping step.", identical ? "BYTE-IDENTICAL" : "*** DIFFERENT ***");
            if (!identical)
            {
                Route.Report(directory, "bridge.log");
                return 9;
            }

            // ------------------------------------------------------------ the thermal source
            Dictionary<Guid, string> zoneReferences = new Dictionary<Guid, string>();
            foreach (string line in File.ReadAllLines(Path.Combine(directory_Source, "src-zones.txt")))
            {
                string[] parts = line.Split('\t');
                if (parts.Length == 2)
                {
                    zoneReferences[Guid.Parse(parts[0])] = parts[1];
                }
            }

            SAM.Core.Tas.SimulationEvidence simulationEvidence_Source = new SAM.Core.Tas.SimulationEvidence(SAM.Core.Tas.SimulationOutputShape.SeparateOutputFile, path_TBD, path_TSD);
            simulationEvidence_Source.RecordCallReturned();
            simulationEvidence_Source.Conclude();

            SAM.Core.Tas.NoIzamThermalSource noIzamThermalSource = new SAM.Core.Tas.NoIzamThermalSource(path_TBD, path_TSD, true, true, simulationEvidence_Source, zoneReferences, null, null);
            W("thermal source IsComplete: {0}, {1} room identities", noIzamThermalSource.IsComplete, noIzamThermalSource.Count_ZoneReferences);

            // ------------------------------------------------------------ PR1, the frozen parity settings
            SAM.Core.Systems.SystemEnergyCentre systemEnergyCentre_Template = SAM.Analytical.Systems.Query.SystemEnergyCentre(Route.TemplatePath());

            SAM.Analytical.Systems.YearlySchedule yearlySchedule = new SAM.Analytical.Systems.YearlySchedule("Part O continuous operation");
            double[] ones = new double[Hours];
            for (int i = 0; i < ones.Length; i++)
            {
                ones[i] = 1.0;
            }

            yearlySchedule.Values = ones;

            SAM.Analytical.Systems.MechanicalVentilationSettings mechanicalVentilationSettings = new SAM.Analytical.Systems.MechanicalVentilationSettings { Schedule = yearlySchedule };

            List<SAM.Analytical.Space> spaces_Scope = Pr1.Scope(adjacencyCluster, out List<string> notes_Scope);

            SAM.Analytical.Systems.MechanicalVentilationMaterialisation mechanicalVentilationMaterialisation =
                SAM.Analytical.Systems.Create.MechanicalVentilation(adjacencyCluster, systemEnergyCentre_Template, mechanicalVentilationSettings, spaces_Scope);

            W("PR1 IsMaterialised: {0}", mechanicalVentilationMaterialisation.IsMaterialised);
            if (!mechanicalVentilationMaterialisation.IsMaterialised)
            {
                foreach (string refusal in mechanicalVentilationMaterialisation.Refusals)
                {
                    W("  refusal: {0}", refusal);
                }

                Route.Report(directory, "bridge.log");
                return 7;
            }

            // ------------------------------------------------------------ PR2, full year
            W("");
            W("== PRODUCTION TPD.Create.SystemVentilationRoute, hours 0..8759 ==");

            SAM.Analytical.Tas.TPD.SystemVentilationRoute systemVentilationRoute =
                SAM.Analytical.Tas.TPD.Create.SystemVentilationRoute(noIzamThermalSource, mechanicalVentilationMaterialisation, path_TPD, 0, Hours - 1);

            foreach (string refusal in systemVentilationRoute.Refusals)
            {
                W("  refusal: {0}", refusal);
            }

            W("  IsComplete: {0}  NativeDiagnostic [{1}]", systemVentilationRoute.IsComplete, systemVentilationRoute.SimulationEvidence?.NativeDiagnostic ?? "<null>");

            if (!systemVentilationRoute.IsComplete)
            {
                Route.Report(directory, "bridge.log");
                return 8;
            }

            // The route's own output, persisted BEFORE the bridge runs: the read-backs compare against this.
            List<string> lines_ZT = new List<string>();
            List<string> lines_Binding = new List<string>();

            foreach (SAM.Analytical.Tas.TPD.SystemVentilationBinding binding in systemVentilationRoute.Bindings)
            {
                SAM.Analytical.Tas.TPD.SystemZoneTemperatureResult zt = systemVentilationRoute.SystemZoneTemperatureResults.Result(binding.Guid_Space);

                List<string> row = new List<string> { binding.Guid_Space.ToString("D"), noIzamThermalSource.ZoneReference(binding.Guid_Space) };
                for (int h = 0; h < Hours; h++)
                {
                    zt.TryGetValue(h, out double value);
                    row.Add(R(value));
                }

                lines_ZT.Add(string.Join("\t", row));

                lines_Binding.Add(string.Join("\t", new string[]
                {
                    binding.Guid_Space.ToString("D"),
                    binding.Guid_SystemSpace.ToString("D"),
                    binding.Guid_AirSystem.ToString("D"),
                    binding.Reference_SystemZone,
                    binding.Reference_ZoneLoad,
                    binding.Reference_System,
                    binding.DesignFlowRate_Supply_Lps.HasValue ? R(binding.DesignFlowRate_Supply_Lps.Value) : string.Empty,
                    binding.DesignFlowRate_Extract_Lps.HasValue ? R(binding.DesignFlowRate_Extract_Lps.Value) : string.Empty,
                    name_By_Space.TryGetValue(binding.Guid_Space, out string name) ? name : "<unknown>",
                }));
            }

            File.WriteAllLines(Path.Combine(directory, "zt.tsv"), lines_ZT);
            File.WriteAllLines(Path.Combine(directory, "bindings.tsv"), lines_Binding);
            File.WriteAllText(Path.Combine(directory, "tolerance.txt"), R(tolerance));

            W("  route output persisted: zt.tsv ({0} room(s)), bindings.tsv", lines_ZT.Count);

            // ------------------------------------------------------------ PR3
            int result = RunBridge(systemVentilationRoute, path_Bridge, tolerance, name_By_Space, directory, "resultant.tsv", "bridge-rooms.tsv");

            W("");
            W("== provenance, after the run ==");
            W("  src.tbd  sha256 {0}  {1}", Src.Hash(path_TBD), string.Equals(hash_TBD_Before, Src.Hash(path_TBD), StringComparison.OrdinalIgnoreCase) ? "UNCHANGED" : "*** CHANGED ***");
            W("  src.tsd  sha256 {0}  {1}", Src.Hash(path_TSD), string.Equals(hash_TSD_Before, Src.Hash(path_TSD), StringComparison.OrdinalIgnoreCase) ? "UNCHANGED" : "*** CHANGED ***");
            W("  acc.tpd  sha256 {0}", Src.Hash(path_TPD));

            File.WriteAllLines(Path.Combine(directory, "provenance.txt"), new string[]
            {
                string.Concat("prepared.sam\t", Src.Hash(path_Design)),
                string.Concat("src.tbd.before\t", hash_TBD_Before),
                string.Concat("src.tbd.after\t", Src.Hash(path_TBD)),
                string.Concat("src.tsd.before\t", hash_TSD_Before),
                string.Concat("src.tsd.after\t", Src.Hash(path_TSD)),
                string.Concat("acc.tpd\t", Src.Hash(path_TPD)),
                string.Concat("bridge.tbd\t", Src.Hash(path_Bridge)),
                string.Concat("bridge.tsd\t", Src.Hash(Path.ChangeExtension(path_Bridge, ".tsd"))),
            });

            Route.Report(directory, "bridge.log");

            return result;
        }

        /// <summary>The PR3 production call, and its evidence written out.</summary>
        private static int RunBridge(
            SAM.Analytical.Tas.TPD.SystemVentilationRoute systemVentilationRoute,
            string path_Bridge,
            double tolerance,
            Dictionary<Guid, string> name_By_Space,
            string directory,
            string file_Resultant,
            string file_Rooms)
        {
            W("");
            W("== PRODUCTION TPD.Create.ThermostatBridge ==");

            DateTime start = DateTime.UtcNow;

            SAM.Analytical.Tas.TPD.ThermostatBridge thermostatBridge = SAM.Analytical.Tas.TPD.Create.ThermostatBridge(systemVentilationRoute, path_Bridge, tolerance);

            W("  elapsed {0:0.0} s", (DateTime.UtcNow - start).TotalSeconds);

            SAM.Analytical.Tas.TPD.ResultantTemperatureResults results = thermostatBridge.ResultantTemperatureResults;

            foreach (string note in results.Notes)
            {
                W("  note: {0}", note);
            }

            foreach (string refusal in results.Refusals)
            {
                W("  REFUSAL: {0}", refusal);
            }

            if (thermostatBridge.SimulationEvidence != null)
            {
                foreach (string note in thermostatBridge.SimulationEvidence.Notes)
                {
                    W("  evidence: {0}", note);
                }

                W("  evidence Completed: {0}  ResultsReconciled: {1}  CallReturned: {2}", thermostatBridge.SimulationEvidence.Completed, thermostatBridge.SimulationEvidence.ResultsReconciled, thermostatBridge.SimulationEvidence.CallReturned);
                W("  evidence result: {0} bytes, written {1:o}, run started {2:o}", thermostatBridge.SimulationEvidence.ResultLengthAfter, thermostatBridge.SimulationEvidence.UtcResultAfter, thermostatBridge.SimulationEvidence.UtcStarted);
            }

            W("  source TBD  {0}  before {1}  after {2}", thermostatBridge.Path_TBD_Source, thermostatBridge.Hash_TBD_Source_Before, thermostatBridge.Hash_TBD_Source_After);
            W("  source TSD  {0}  before {1}  after {2}", thermostatBridge.Path_TSD_Source, thermostatBridge.Hash_TSD_Source_Before, thermostatBridge.Hash_TSD_Source_After);
            W("  bridge TBD  {0}  copied {1}  final {2}", thermostatBridge.Path_TBD, thermostatBridge.Hash_TBD_Copied, thermostatBridge.Hash_TBD);
            W("  bridge TSD  {0}  {1}", thermostatBridge.Path_TSD, thermostatBridge.Hash_TSD);
            W("  IsComplete  {0}", thermostatBridge.IsComplete);

            List<string> lines_Room = new List<string>
            {
                string.Join("\t", new string[] { "space_guid", "space_name", "zone_guid", "zt_count", "ic_count", "heating_count", "cooling_count", "max_transfer_delta", "rt_count", "rt_finite", "max_air_dev", "mean_air_dev" }),
            };

            W("");
            W("  {0,-36}  {1,-12}  {2,5} {3,3} {4,5} {5,5} {6,10} {7,5} {8,5} {9,8} {10,8}", "space guid", "name", "ZT", "IC", "heat", "cool", "maxXfer", "RT", "fin", "maxDev", "meanDev");

            foreach (SAM.Analytical.Tas.TPD.ThermostatBridgeRoom room in thermostatBridge.Rooms)
            {
                string name = name_By_Space.TryGetValue(room.Guid_Space, out string value) ? value : "<unknown>";

                W("  {0,-36}  {1,-12}  {2,5} {3,3} {4,5} {5,5} {6,10} {7,5} {8,5} {9,8:0.0000} {10,8:0.0000}",
                    room.Guid_Space, name, room.Count_ZoneTemperature, room.Count_InternalConditions, room.Count_Heating, room.Count_Cooling,
                    R(room.MaxTransferDelta), room.Count_ResultantTemperature, room.Count_ResultantTemperature_Finite,
                    room.MaxAchievedAirTemperatureDeviation, room.MeanAchievedAirTemperatureDeviation);

                lines_Room.Add(string.Join("\t", new string[]
                {
                    room.Guid_Space.ToString("D"), name, room.Reference_Zone,
                    room.Count_ZoneTemperature.ToString(CultureInfo.InvariantCulture), room.Count_InternalConditions.ToString(CultureInfo.InvariantCulture),
                    room.Count_Heating.ToString(CultureInfo.InvariantCulture), room.Count_Cooling.ToString(CultureInfo.InvariantCulture),
                    R(room.MaxTransferDelta), room.Count_ResultantTemperature.ToString(CultureInfo.InvariantCulture),
                    room.Count_ResultantTemperature_Finite.ToString(CultureInfo.InvariantCulture),
                    R(room.MaxAchievedAirTemperatureDeviation), R(room.MeanAchievedAirTemperatureDeviation),
                }));
            }

            File.WriteAllLines(Path.Combine(directory, file_Rooms), lines_Room);

            List<string> lines_Resultant = new List<string>();
            foreach (SAM.Analytical.Tas.TPD.ResultantTemperatureResult resultantTemperatureResult in results.Results)
            {
                List<string> row = new List<string> { resultantTemperatureResult.Guid_Space.ToString("D"), resultantTemperatureResult.Reference_Zone };
                for (int h = 0; h < Hours; h++)
                {
                    resultantTemperatureResult.TryGetValue(h, out double v);
                    row.Add(R(v));
                }

                lines_Resultant.Add(string.Join("\t", row));
            }

            File.WriteAllLines(Path.Combine(directory, file_Resultant), lines_Resultant);

            W("  {0} complete resultant series written to {1}", lines_Resultant.Count, file_Resultant);

            return thermostatBridge.IsComplete ? 0 : 10;
        }

        // ====================================================================================== bridgecheck
        public static int Check(string directory)
        {
            Dictionary<string, double[]> zt = ReadSeries(Path.Combine(directory, "zt.tsv"), out Dictionary<string, string> zone_By_Space);
            Dictionary<string, string> name_By_Space = File.ReadAllLines(Path.Combine(directory, "bindings.tsv")).Select(x => x.Split('\t')).ToDictionary(x => x[0], x => x[8]);
            double tolerance = double.Parse(File.ReadAllText(Path.Combine(directory, "tolerance.txt")), CultureInfo.InvariantCulture);

            W("PR3 bridgecheck - raw read-back through TAS's own accessors, independent of the code under test");
            W("  {0} room(s) in zt.tsv", zt.Count);

            // ---------------------------------------------------------------- A. thermostats in the bridge TBD
            double max_Transfer = 0;
            int count_Rows_Pass = 0;
            List<string> lines = new List<string>();

            global::TBD.TBDDocument tBDDocument = new global::TBD.TBDDocument();
            tBDDocument.openReadOnly(Path.Combine(directory, "bridge.tbd"));
            try
            {
                global::TBD.Building building = tBDDocument.Building;

                Dictionary<string, global::TBD.zone> zones = new Dictionary<string, global::TBD.zone>();
                int index = 0;
                global::TBD.zone zone;
                while ((zone = building.GetZone(index)) != null)
                {
                    zones[Key(zone.GUID)] = zone;
                    index++;
                }

                W("");
                W("A. INPUT TRANSFER - every slot of ticLL (heating) and ticUL (cooling) on every IC of each zone");
                W("  {0,-36} {1,-12} {2,5} {3,3} {4,6} {5,6} {6,12}", "space guid", "name", "ZT", "IC", "heat", "cool", "max|d|");

                foreach (KeyValuePair<string, double[]> keyValuePair in zt.OrderBy(x => x.Key, StringComparer.Ordinal))
                {
                    global::TBD.zone zone_Room = zones[Key(zone_By_Space[keyValuePair.Key])];
                    double[] series = keyValuePair.Value;

                    int count_IC = 0;
                    int heat = int.MaxValue;
                    int cool = int.MaxValue;
                    double max = 0;

                    global::TBD.InternalCondition ic;
                    while ((ic = zone_Room.GetIC(count_IC)) != null)
                    {
                        count_IC++;
                        global::TBD.Thermostat thermostat = ic.GetThermostat();

                        foreach (global::TBD.Profiles slot in new[] { global::TBD.Profiles.ticLL, global::TBD.Profiles.ticUL })
                        {
                            global::TBD.profile profile = thermostat.GetProfile((int)slot);
                            int equal = 0;
                            for (int h = 1; h <= Hours; h++)
                            {
                                float stored = profile.get_yearlyValues(h);
                                double d = Math.Abs(stored - series[h - 1]);
                                max = Math.Max(max, d);
                                if (stored == (float)series[h - 1])
                                {
                                    equal++;
                                }
                            }

                            if (slot == global::TBD.Profiles.ticLL) heat = Math.Min(heat, equal); else cool = Math.Min(cool, equal);

                            if (profile.type != global::TBD.ProfileTypes.ticYearlyProfile || profile.factor != 1)
                            {
                                W("  !! {0} {1} on IC '{2}' type={3} factor={4}", keyValuePair.Key, slot, ic.name, profile.type, profile.factor);
                            }
                        }
                    }

                    max_Transfer = Math.Max(max_Transfer, max);
                    bool pass = heat == Hours && cool == Hours && series.Length == Hours;
                    if (pass) count_Rows_Pass++;

                    W("  {0,-36} {1,-12} {2,5} {3,3} {4,6} {5,6} {6,12}", keyValuePair.Key, name_By_Space[keyValuePair.Key], series.Length, count_IC, heat, cool, R(max));
                    lines.Add(string.Join("\t", keyValuePair.Key, name_By_Space[keyValuePair.Key], series.Length, count_IC, heat, cool, R(max)));
                }
            }
            finally
            {
                try { tBDDocument.close(); } catch { }
            }

            W("  MAX ABSOLUTE TRANSFER DELTA (read-back single vs Systems double) = {0}", R(max_Transfer));
            W("  rooms with 8760/8760 heating and cooling slots exactly as written: {0} of {1}", count_Rows_Pass, zt.Count);

            // ---------------------------------------------------------------- B. the second TSD
            Dictionary<string, float[][]> bridge = ReadTsd(Path.Combine(directory, "bridge.tsd"));
            Dictionary<string, float[][]> source = ReadTsd(Path.Combine(directory, "src.tsd"));
            Dictionary<string, double[]> resultant = ReadSeries(Path.Combine(directory, "resultant.tsv"), out _);

            W("");
            W("B. COMPLETENESS and ACHIEVED AIR - second TSD read by zoneGUID");
            W("  {0,-36} {1,-12} {2,5} {3,5} {4,9} {5,9} {6,9} {7,9} {8,9}", "space guid", "name", "RT", "fin", "air-1h", "air0", "air+1h", "air0max", "prod=raw");

            int count_Complete = 0;
            double max_ProductionVsRaw = 0;
            foreach (KeyValuePair<string, double[]> keyValuePair in zt.OrderBy(x => x.Key, StringComparer.Ordinal))
            {
                float[][] b = bridge[Key(zone_By_Space[keyValuePair.Key])];
                int finite = b[1].Count(x => !float.IsNaN(x) && !float.IsInfinity(x));
                double[] mean = new double[3];
                double max0 = 0;
                for (int s = -1; s <= 1; s++)
                {
                    double sum = 0; int n = 0;
                    for (int h = 0; h < Hours; h++)
                    {
                        int k = h + s;
                        if (k < 0 || k >= Hours) continue;
                        double d = Math.Abs(b[0][h] - keyValuePair.Value[k]);
                        sum += d; n++;
                        if (s == 0) max0 = Math.Max(max0, d);
                    }

                    mean[s + 1] = sum / n;
                }

                double maxPR = 0;
                double[] prod = resultant[keyValuePair.Key];
                for (int h = 0; h < Hours; h++)
                {
                    maxPR = Math.Max(maxPR, Math.Abs(prod[h] - b[1][h]));
                }

                max_ProductionVsRaw = Math.Max(max_ProductionVsRaw, maxPR);
                if (b[1].Length == Hours && finite == Hours) count_Complete++;

                W("  {0,-36} {1,-12} {2,5} {3,5} {4,9:0.0000} {5,9:0.0000} {6,9:0.0000} {7,9:0.0000} {8,9}", keyValuePair.Key, name_By_Space[keyValuePair.Key], b[1].Length, finite, mean[0], mean[1], mean[2], max0, R(maxPR));
            }

            W("  rooms with 8760 finite ResultantTemperature: {0} of {1}; production series vs raw read max|d| = {2}", count_Complete, zt.Count, R(max_ProductionVsRaw));
            W("  zones in the bridge TSD: {0}; not bridged (free-running, not a ventilation-system member): {1}", bridge.Count, string.Join(", ", bridge.Keys.Where(x => !zone_By_Space.Values.Select(Key).Contains(x))));

            // ---------------------------------------------------------------- C. hour alignment Systems vs TBD
            W("");
            W("C. HOUR ALIGNMENT - Systems ZoneTemperature vs the no-IZAM source's own free-running dry bulb (mean|d| K)");
            foreach (KeyValuePair<string, double[]> keyValuePair in zt.OrderBy(x => x.Key, StringComparer.Ordinal))
            {
                float[][] s = source[Key(zone_By_Space[keyValuePair.Key])];
                double[] mean = new double[3];
                for (int sh = -1; sh <= 1; sh++)
                {
                    double sum = 0; int n = 0;
                    for (int h = 0; h < Hours; h++)
                    {
                        int k = h + sh;
                        if (k < 0 || k >= Hours) continue;
                        sum += Math.Abs(keyValuePair.Value[h] - s[0][k]); n++;
                    }

                    mean[sh + 1] = sum / n;
                }

                double[] diffDiurnal = new double[3];
                W("  {0,-12} ZT[h] vs srcDB[h-1] {1:0.0000}  [h] {2:0.0000}  [h+1] {3:0.0000}", name_By_Space[keyValuePair.Key], mean[0], mean[1], mean[2]);
            }

            File.WriteAllLines(Path.Combine(directory, "bridgecheck-transfer.tsv"), lines);
            Route.Report(directory, "bridgecheck.log");
            return count_Rows_Pass == zt.Count && count_Complete == zt.Count && max_ProductionVsRaw == 0 ? 0 : 11;
        }

        // =========================================================================================== rename
        public static int Rename(string directory, string directory_Out)
        {
            Directory.CreateDirectory(directory_Out);

            string path_TBD = Path.Combine(directory_Out, "src.tbd");
            string path_TSD = Path.Combine(directory_Out, "src.tsd");
            File.Copy(Path.Combine(directory, "src.tbd"), path_TBD, true);
            File.Copy(Path.Combine(directory, "src.tsd"), path_TSD, true);

            double tolerance = double.Parse(File.ReadAllText(Path.Combine(directory, "tolerance.txt")), CultureInfo.InvariantCulture);

            W("PR3 rename - identity challenge: every zone in a copy of the source renamed to ONE shared string");

            // Rename every zone in the copy.
            global::TBD.TBDDocument tBDDocument = new global::TBD.TBDDocument();
            tBDDocument.open(path_TBD);
            int count_Renamed = 0;
            try
            {
                int index = 0;
                global::TBD.zone zone;
                while ((zone = tBDDocument.Building.GetZone(index)) != null)
                {
                    W("  zone {0} '{1}' -> 'Zone'", zone.GUID, zone.name);
                    zone.name = "Zone";
                    index++;
                    count_Renamed++;
                }

                tBDDocument.save();
            }
            finally
            {
                try { tBDDocument.close(); } catch { }
            }

            W("  {0} zone(s) renamed; renamed source sha256 {1}", count_Renamed, Src.Hash(path_TBD));

            // The route, rebuilt from its OWN persisted output (bindings.tsv, zt.tsv): nothing is authored.
            Dictionary<string, double[]> zt = ReadSeries(Path.Combine(directory, "zt.tsv"), out Dictionary<string, string> zone_By_Space);
            List<string[]> rows = File.ReadAllLines(Path.Combine(directory, "bindings.tsv")).Select(x => x.Split('\t')).ToList();

            Dictionary<Guid, string> zoneReferences = new Dictionary<Guid, string>();
            foreach (string line in File.ReadAllLines(Path.Combine(directory, "src-zones.txt")))
            {
                string[] parts = line.Split('\t');
                zoneReferences[Guid.Parse(parts[0])] = parts[1];
            }

            List<SAM.Analytical.Tas.TPD.SystemVentilationBinding> bindings = new List<SAM.Analytical.Tas.TPD.SystemVentilationBinding>();
            SAM.Analytical.Tas.TPD.SystemZoneTemperatureResults systemZoneTemperatureResults = new SAM.Analytical.Tas.TPD.SystemZoneTemperatureResults(0, Hours - 1);

            Dictionary<Guid, string> name_By_Space = new Dictionary<Guid, string>();

            foreach (string[] r in rows)
            {
                SAM.Analytical.Tas.TPD.SystemVentilationBinding binding = new SAM.Analytical.Tas.TPD.SystemVentilationBinding(
                    Guid.Parse(r[0]), Guid.Parse(r[1]), Guid.Parse(r[2]), r[3], r[4], r[5],
                    r[6].Length == 0 ? (double?)null : double.Parse(r[6], CultureInfo.InvariantCulture),
                    r[7].Length == 0 ? (double?)null : double.Parse(r[7], CultureInfo.InvariantCulture));

                bindings.Add(binding);
                name_By_Space[binding.Guid_Space] = r[8];

                SAM.Core.IndexedDoubles indexedDoubles = new SAM.Core.IndexedDoubles();
                double[] series = zt[r[0]];
                for (int h = 0; h < Hours; h++) indexedDoubles[h] = series[h];

                systemZoneTemperatureResults.Add(new SAM.Analytical.Tas.TPD.SystemZoneTemperatureResult(binding.Guid_Space, binding.Guid_SystemSpace, binding.Reference_SystemZone, binding.Reference_ZoneLoad, 0, Hours - 1, indexedDoubles, null));
            }

            systemZoneTemperatureResults.Validate(bindings);

            SAM.Core.Tas.SimulationEvidence simulationEvidence_Source = new SAM.Core.Tas.SimulationEvidence(SAM.Core.Tas.SimulationOutputShape.SeparateOutputFile, path_TBD, path_TSD);
            simulationEvidence_Source.RecordCallReturned();
            simulationEvidence_Source.Conclude();

            SAM.Core.Tas.NoIzamThermalSource noIzamThermalSource = new SAM.Core.Tas.NoIzamThermalSource(path_TBD, path_TSD, true, true, simulationEvidence_Source, zoneReferences, null, null);

            SAM.Core.Tas.SimulationEvidence simulationEvidence_Route = new SAM.Core.Tas.SimulationEvidence(SAM.Core.Tas.SimulationOutputShape.InPlaceDocument, Path.Combine(directory, "acc.tpd"), null);
            simulationEvidence_Route.RecordCallReturned("Done");
            simulationEvidence_Route.RecordResultsReconciled(bindings.Count, 0, Hours - 1);

            SAM.Analytical.Tas.TPD.SystemVentilationRoute systemVentilationRoute = new SAM.Analytical.Tas.TPD.SystemVentilationRoute(
                noIzamThermalSource, Path.Combine(directory, "acc.tpd"), simulationEvidence_Route, bindings, null, systemZoneTemperatureResults, null, null);

            W("  rebuilt route IsComplete {0} ({1} room(s))", systemVentilationRoute.IsComplete, bindings.Count);

            int result = RunBridge(systemVentilationRoute, Path.Combine(directory_Out, "bridge.tbd"), tolerance, name_By_Space, directory_Out, "resultant.tsv", "bridge-rooms.tsv");

            // Compare with the canonical bridge, room by room, by guid.
            Dictionary<string, double[]> canonical = ReadSeries(Path.Combine(directory, "resultant.tsv"), out _);
            Dictionary<string, double[]> renamed = File.Exists(Path.Combine(directory_Out, "resultant.tsv")) ? ReadSeries(Path.Combine(directory_Out, "resultant.tsv"), out _) : new Dictionary<string, double[]>();

            W("");
            W("C. IDENTITY - renamed-source bridge vs canonical bridge, ResultantTemperature per room guid");
            int identical = 0;
            foreach (KeyValuePair<string, double[]> keyValuePair in canonical.OrderBy(x => x.Key, StringComparer.Ordinal))
            {
                if (!renamed.TryGetValue(keyValuePair.Key, out double[] other))
                {
                    W("  {0,-36} {1,-12} MISSING in renamed run", keyValuePair.Key, name_By_Space[Guid.Parse(keyValuePair.Key)]);
                    continue;
                }

                int equal = 0; double max = 0;
                for (int h = 0; h < Hours; h++)
                {
                    if (other[h] == keyValuePair.Value[h]) equal++;
                    max = Math.Max(max, Math.Abs(other[h] - keyValuePair.Value[h]));
                }

                if (equal == Hours) identical++;
                W("  {0,-36} {1,-12} identical hours {2}/{3}  max|d| {4}", keyValuePair.Key, name_By_Space[Guid.Parse(keyValuePair.Key)], equal, Hours, R(max));
            }

            W("  rooms bit-identical: {0} of {1}", identical, canonical.Count);
            Route.Report(directory_Out, "rename.log");
            return result == 0 && identical == canonical.Count ? 0 : 12;
        }

        // ============================================================================================ helpers
        private static Dictionary<string, double[]> ReadSeries(string path, out Dictionary<string, string> zone_By_Space)
        {
            Dictionary<string, double[]> result = new Dictionary<string, double[]>(StringComparer.Ordinal);
            zone_By_Space = new Dictionary<string, string>(StringComparer.Ordinal);

            foreach (string line in File.ReadAllLines(path))
            {
                string[] parts = line.Split('\t');
                double[] values = new double[parts.Length - 2];
                for (int i = 2; i < parts.Length; i++)
                {
                    values[i - 2] = double.Parse(parts[i], CultureInfo.InvariantCulture);
                }

                result[parts[0]] = values;
                zone_By_Space[parts[0]] = parts[1];
            }

            return result;
        }

        private static Dictionary<string, float[][]> ReadTsd(string path_TSD)
        {
            Dictionary<string, float[][]> result = new Dictionary<string, float[][]>(StringComparer.Ordinal);
            global::TSD.TSDDocument tSDDocument = new global::TSD.TSDDocument();
            tSDDocument.openReadOnly(path_TSD);
            try
            {
                global::TSD.BuildingData buildingData = tSDDocument.SimulationData.GetBuildingData();
                int i = 1;
                global::TSD.ZoneData zoneData;
                while ((zoneData = buildingData.GetZoneData(i)) != null)
                {
                    float[][] series = new float[2][];
                    int k = 0;
                    foreach (global::TSD.tsdZoneArray q in new[] { global::TSD.tsdZoneArray.dryBulbTemp, global::TSD.tsdZoneArray.resultantTemp })
                    {
                        List<float> values = new List<float>();
                        foreach (object o in (System.Collections.IEnumerable)zoneData.GetAnnualZoneResult((int)q))
                        {
                            values.Add(Convert.ToSingle(o, CultureInfo.InvariantCulture));
                        }

                        series[k++] = values.ToArray();
                    }

                    result[Key(zoneData.zoneGUID)] = series;
                    i++;
                }
            }
            finally
            {
                try { tSDDocument.close(); } catch { }
            }

            return result;
        }
