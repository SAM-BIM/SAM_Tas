using System;
using System.Collections;
using System.Collections.Generic;
using System.Globalization;
using System.IO;
using System.Linq;
using SAM.Analytical;

namespace B0P
{
    public static partial class Steps
    {
        static void W(string f, params object[] a) { Console.WriteLine(a.Length == 0 ? f : string.Format(CultureInfo.InvariantCulture, f, a)); }

        // map <model.sam>
        public static void MapModel(string[] args)
        {
            string path = args[1];
            AnalyticalModel model = SAM.Core.Convert.ToSAM<AnalyticalModel>(path).FirstOrDefault();
            W("model '{0}' spaces={1}", path, model.GetSpaces().Count);
            foreach (Space space in model.GetSpaces().OrderBy(x => x.Name))
            {
                string zoneGuid = null;
                space.TryGetValue(SAM.Analytical.Tas.SpaceParameter.ZoneGuid, out zoneGuid);
                W("SPACE|{0}|{1}|{2}", space.Guid, space.Name, zoneGuid ?? "-");
            }
        }

        // tsdzones <file.tsd>
        public static void TsdZones(string[] args)
        {
            string path = args[1];
            TSD.TSDDocument doc = new TSD.TSDDocument();
            doc.openReadOnly(path);
            try
            {
                TSD.SimulationData sd = doc.SimulationData;
                W("TSD '{0}' firstDay={1} lastDay={2}", path, sd.firstDay, sd.lastDay);
                TSD.BuildingData bd = sd.GetBuildingData();
                int i = 1;
                TSD.ZoneData zd = bd.GetZoneData(i);
                while (zd != null)
                {
                    W("ZONE|{0}|{1}|{2}", i, zd.name, zd.zoneGUID);
                    i++;
                    zd = bd.GetZoneData(i);
                }
                foreach (TSD.tsdBuildingArray ba in Enum.GetValues(typeof(TSD.tsdBuildingArray)))
                {
                    try
                    {
                        float[] v = (bd.GetDailyBuildingResult(180, (short)ba) as IEnumerable).Cast<float>().ToArray();
                        W("BLD|{0}|{1}|{2:F3},{3:F3},{4:F3},{5:F3}", ba, v.Length, v[0], v[6], v[12], v[18]);
                    }
                    catch (Exception ex) { W("BLD|{0}|ERR|{1}", ba, ex.Message.Split('\n')[0]); }
                }
            }
            finally { doc.close(); }
        }
    }
}
