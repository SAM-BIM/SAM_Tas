def stats(d):
    return dict(n=len(d),bias=float(np.mean(d)),rmse=float(np.sqrt(np.mean(d**2))),mae=float(np.mean(np.abs(d))),
                maxabs=float(np.max(np.abs(d))),sd=float(np.std(d)))
