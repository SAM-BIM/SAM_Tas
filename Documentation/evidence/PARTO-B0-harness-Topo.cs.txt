using System;
using System.Collections;
using System.Collections.Generic;
using System.Globalization;
using System.IO;
using System.Linq;
using TPD;

namespace B0P
{
    public static partial class Steps
    {
        static string CName(object o) { if (o == null) return "-"; try { return ((SystemComponent)o).Name; } catch { } try { return (string)((dynamic)o).Name; } catch { return "?"; } }
        static string CGuid(object o) { if (o == null) return "-"; try { return (string)((dynamic)o).GUID; } catch { } try { return ((SystemComponent)o).GUID; } catch { return "?"; } }
        static string CKind(object o)
        {
            if (o == null) return "null";
            if (o is SystemZone) return "SystemZone";
            if (o is Damper) return "Damper";
            if (o is Fan) return "Fan";
            if (o is Junction) return "Junction";
            if (o is Exchanger) return "Exchanger";
            if (o is HeatingCoil) return "HeatingCoil";
            if (o is CoolingCoil) return "CoolingCoil";
            if (o is ComponentGroup) return "ComponentGroup";
            return o.GetType().Name;
        }
        static string PD(object o) { if (o == null) return "null"; try { return ((dynamic)o).Value.ToString(); } catch { } try { return ((ProfileData)o).Value.ToString(); } catch { return "?"; } }

        static double SV(object o) { if (o == null) return double.NaN; try { return ((dynamic)o).Value; } catch { return double.NaN; } }

        // topo <file.tpd>
        public static void Topo(string[] args)
        {
            TPDDoc doc = new TPDDoc();
            doc.OpenReadOnly(args[1]);
            try
            {
                EnergyCentre ec = doc.EnergyCentre;
                for (int p = 1; p <= ec.GetPlantRoomCount(); p++)
                {
                    PlantRoom pr = ec.GetPlantRoom(p);
                    W("PLANTROOM|{0}", CName(pr));
                    for (int s = 1; s <= pr.GetSystemCount(); s++)
                    {
                        global::TPD.System sys = pr.GetSystem(s);
                        W("SYSTEM|{0}|{1}", CName(sys), CGuid(sys));
                        foreach (SystemComponent sc in Components(sys))
                        {
                            string kind = CKind(sc);
                            string extra = "";
                            if (sc is SystemZone)
                            {
                                SystemZone sz = (SystemZone)sc;
                                extra = " ZONE";
                                try { extra += " DisplacementVent=" + sz.DisplacementVent; } catch (Exception e) { extra += " DisplacementVent!" + e.GetType().Name; }
                                try { extra += " FlowRate=" + SV(sz.FlowRate); } catch { }
                                try { extra += " FreshAir=" + SV(sz.FreshAir); } catch { }
                                try { extra += " MinFlowFrac=" + sz.MinimumFlowFraction; } catch { }
                                try { ZoneLoad zl = ((SystemZone)sc).GetSystemZoneZoneLoad(); extra += string.Format(" load='{0}' loadGuid={1}", zl.Name, zl.GUID); } catch { }
                                try { Duct din = ((SystemZone)sc).GetSystemZoneInputDuct(); extra += string.Format(" | in<-{0}:{1}", CKind(din.GetUpstreamComponent()), CName(din.GetUpstreamComponent())); } catch { }
                                try { Duct dout = ((SystemZone)sc).GetSystemZoneOutputDuct(); extra += string.Format(" | out->{0}:{1}", CKind(dout.GetDownstreamComponent()), CName(dout.GetDownstreamComponent())); } catch { }
                            }
                            else if (sc is Damper)
                            {
                                Damper dm = (Damper)sc;
                                extra = " DAMPER";
                                try { extra += " DesignFlow=" + SV(dm.DesignFlowRate); } catch { }
                                try { extra += " DesignFlowType=" + dm.DesignFlowType; } catch { }
                                try { extra += " MinFlow=" + SV(dm.MinimumFlowRate); } catch { }
                                try { extra += " MinFlowType=" + dm.MinimumFlowType; } catch { }
                                try { extra += " Capacity=" + dm.Capacity; } catch { }

                            }
                            else if (sc is Fan)
                            {
                                Fan fn = (Fan)sc;
                                extra = " FAN";
                                try { extra += " DesignFlowRate=" + SV(fn.DesignFlowRate); } catch (Exception e) { extra += " DesignFlowRate!" + e.GetType().Name; }
                                try { extra += " DesignFlowType=" + fn.DesignFlowType; } catch { }
                                try { extra += " ControlType=" + fn.ControlType; } catch { }
                                try { extra += " Pressure=" + SV(fn.Pressure); } catch { }
                                try { extra += " OverallEfficiency=" + PD(fn.OverallEfficiency); } catch { }
                                try { extra += " HeatGainFactor=" + fn.HeatGainFactor; } catch { }

                            }
                            else if (sc is Exchanger)
                            {
                                Exchanger ex = (Exchanger)sc;
                                extra = " EXCHANGER";
                                try { extra += " SensibleEfficiency=" + PD(ex.SensibleEfficiency); } catch { }
                                try { extra += " LatentEfficiency=" + PD(ex.LatentEfficiency); } catch { }
                                try { extra += " ExchangerType=" + ex.ExchangerType; } catch { }
                                try { extra += " SetpointMethod=" + ex.SetpointMethod; } catch { }
                                try { extra += " BypassFactor=" + PD(ex.BypassFactor); } catch { }
                                try { extra += " ExchCalcType=" + ex.ExchCalcType; } catch { }
                            }
                            W("  COMP|{0}|{1}|{2}|{3}", kind, CName(sc), CGuid(sc), extra);
                        }
                        // ducts
                        int dc = sys.GetDuctCount();
                        for (int d = 1; d <= dc; d++)
                        {
                            Duct duct = sys.GetDuct(d);
                            if (duct == null) continue;
                            object up = duct.GetUpstreamComponent(), dn = duct.GetDownstreamComponent();
                            W("  DUCT|{0}:{1}:p{2} -> {3}:{4}:p{5}", CKind(up), CName(up), duct.GetUpstreamComponentPort(), CKind(dn), CName(dn), duct.GetDownstreamComponentPort());
                        }
                    }
                }
            }
            finally { doc.Close(); }
        }
    }
}
