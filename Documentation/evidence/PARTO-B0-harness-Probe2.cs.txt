using System;
using System.Collections;
using System.Linq;
using TPD;

namespace B0P
{
    public static partial class Steps
    {
        // tpdinfo <file.tpd> [rw]
        public static void TpdInfo(string[] args)
        {
            TPDDoc doc = new TPDDoc();
            if (args.Length > 2 && args[2] == "rw") doc.Open(args[1]); else doc.OpenReadOnly(args[1]);
            try
            {
                EnergyCentre ec = doc.EnergyCentre;
                W("start={0} end={1} simStart={2} simEnd={3} precond={4} simPrecond={5}",
                    ec.GetStartHour(), ec.GetEndHour(), ec.GetSimStartHour(), ec.GetSimEndHour(), ec.GetPrecondHours(), ec.GetSimPrecondHours());
                W("tsdDataCount={0} plantRooms={1}", ec.GetTSDDataCount(), ec.GetPlantRoomCount());
                for (int i = 1; i <= ec.GetTSDDataCount(); i++)
                {
                    dynamic td = ec.GetTSDData(i);
                    W("TSDData[{0}] file='{1}' tsd='{2}' tbd='{3}' bld='{4}' days {5}..{6}", i, td.FileName, td.TSDPath, td.TBDPath, td.BuildingName, td.StartDay, td.EndDay);
                }
                // first plant room, first system, first zone -> try result read
                PlantRoom pr = ec.GetPlantRoom(1);
                W("plantRoom='{0}' systemCount={1}", ((dynamic)pr).Name, pr.GetSystemCount());
                global::TPD.System sys = pr.GetSystem(1);
                var comps = Components(sys);
                var zone = comps.FirstOrDefault(x => x is SystemZone);
                W("zone='{0}'", ((dynamic)zone).Name);
                foreach (int v in new[] { 1, 3, 9 })
                {
                    try
                    {
                        object o = ((dynamic)zone).GetResultsData(tpdResultsPeriod.tpdResultsPeriodHourly, tpdCombinerType.tpdCombinerTypeMax, v, 1, 24);
                        W("var{0} -> {1}", v, o == null ? "null" : (o as IEnumerable).Cast<float>().Take(6).Select(x => x.ToString("F3")).Aggregate((a, b) => a + "," + b));
                    }
                    catch (Exception ex) { W("var{0} -> {1}: {2}", v, ex.GetType().Name, Compact(ex.Message)); }
                }
                try
                {
                    object rs = ec.GetResultSet(tpdResultsPeriod.tpdResultsPeriodHourly, 1, 1, 24, null);
                    W("GetResultSet -> {0}", rs == null ? "null" : rs.GetType().Name);
                }
                catch (Exception ex) { W("GetResultSet -> {0}: {1}", ex.GetType().Name, Compact(ex.Message)); }
            }
            finally { doc.Close(); }
        }
    }
}
