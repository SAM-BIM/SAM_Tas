// Diagnosing "Sizing Flow Failed" by ONE-CHANGE experiments on the production-converted TPD.
//
// Each run applies exactly the switches named on the command line, saves, reopens, simulates, and
// reports the raw ITPD.Simulate diagnostic plus what TAS did or did not size. Nothing is changed in
// production code until a switch is shown to change the diagnostic.
//
// Switches (comma separated, order irrelevant):
//   loads        bind every unbound SystemZone to a TSD ZoneLoad by explicit AddZoneLoad
//   zonevalue    zone FlowRate/FreshAir -> tpdSizedVariableValue, keeping the authored number
//   dampervalue  every Damper -> DesignFlowType=tpdFlowRateValue with an explicit DesignFlowRate
//   fanvalue     every Fan    -> DesignFlowType=tpdFlowRateValue with an explicit DesignFlowRate
//   flags        every SystemZone -> ModelVentFlow | ModelInterzoneFlow
//   nogroup      set the component group multiplicity to 1
//   period       set the energy centre hours to the requested window

using System;
using System.Collections.Generic;
using System.Globalization;
using System.IO;
using System.Reflection;
using System.Text;

namespace Inv
{
    internal static class Experiment
    {
        private static readonly StringBuilder log = new StringBuilder();

        private static void W(string format, params object[] args)
        {
            string line = args == null || args.Length == 0 ? format : string.Format(CultureInfo.InvariantCulture, format, args);
            Console.WriteLine(line);
            log.AppendLine(line);
        }

        private static void SetProperty(object target, string name, object value)
        {
            target.GetType().InvokeMember(
                name, BindingFlags.SetProperty, null, target, new object[] { value }, CultureInfo.InvariantCulture);
        }

        private static double Dbl(object value)
        {
            try
            {
                return value == null ? double.NaN : Convert.ToDouble(value, CultureInfo.InvariantCulture);
            }
            catch
            {
                return double.NaN;
            }
        }

        private static void Collect(object container, List<object> all, int depth)
        {
            if (container == null || depth > 8)
            {
                return;
            }

            for (int index = 1; index <= 4000; index++)
            {
                object systemComponent = Dump.Call(container, "GetComponent", index);
                if (systemComponent == null)
                {
                    break;
                }

                all.Add(systemComponent);

                if (systemComponent is global::TPD.ComponentGroup)
                {
                    Collect(systemComponent, all, depth + 1);
                }
            }
        }

        public static int Run(string path_TPD, string switches, int startHour, int endHour)
        {
            log.Clear();

            HashSet<string> on = new HashSet<string>(StringComparer.OrdinalIgnoreCase);
            foreach (string part in (switches ?? string.Empty).Split(','))
            {
                if (!string.IsNullOrWhiteSpace(part))
                {
                    on.Add(part.Trim());
                }
            }

            W("experiment {0}", path_TPD);
            W("switches: [{0}]  hours {1}..{2}", string.Join(",", new List<string>(on).ToArray()), startHour, endHour);

            if (!File.Exists(path_TPD))
            {
                W("REFUSED: TPD missing");
                Flush(path_TPD);
                return 5;
            }

            global::TPD.TPDDoc tPDDoc = new global::TPD.TPDDoc();
            try
            {
                tPDDoc.Open(path_TPD);

                object energyCentre = tPDDoc.EnergyCentre;
                if (energyCentre == null)
                {
                    W("REFUSED: no energy centre");
                    return 3;
                }

                W("energy centre hours before: {0}..{1}",
                    Dump.CallText(energyCentre, "GetStartHour"), Dump.CallText(energyCentre, "GetEndHour"));

                object tSDData = Dump.Call(energyCentre, "GetTSDData", 1);
                int count_ZoneLoad = (int)Dbl(Dump.Call(tSDData, "GetZoneLoadCount"));
                W("TSD zone loads: {0}", count_ZoneLoad);

                List<object> all = new List<object>();
                List<object> zones = new List<object>();
                List<object> fans = new List<object>();
                List<object> dampers = new List<object>();
                List<object> groups = new List<object>();

                int count_PlantRoom = (int)Dbl(Dump.Call(energyCentre, "GetPlantRoomCount"));
                for (int p = 1; p <= count_PlantRoom; p++)
                {
                    object plantRoom = Dump.Call(energyCentre, "GetPlantRoom", p);
                    W("PlantRoom[{0}] plantComponents={1} systems={2}",
                        p, Dump.CallText(plantRoom, "GetComponentCount"), Dump.CallText(plantRoom, "GetSystemCount"));

                    int count_System = (int)Dbl(Dump.Call(plantRoom, "GetSystemCount"));
                    for (int s = 1; s <= count_System; s++)
                    {
                        Collect(Dump.Call(plantRoom, "GetSystem", s), all, 0);
                    }
                }

                foreach (object systemComponent in all)
                {
                    if (systemComponent is global::TPD.SystemZone) { zones.Add(systemComponent); }
                    else if (systemComponent is global::TPD.Fan) { fans.Add(systemComponent); }
                    else if (systemComponent is global::TPD.Damper) { dampers.Add(systemComponent); }
                    else if (systemComponent is global::TPD.ComponentGroup) { groups.Add(systemComponent); }
                }

                W("components={0} zones={1} fans={2} dampers={3} groups={4}",
                    all.Count, zones.Count, fans.Count, dampers.Count, groups.Count);

                // ---------------------------------------------------------------- switches

                if (on.Contains("period"))
                {
                    Try("set energy centre hours", delegate
                    {
                        SetProperty(energyCentre, "StartHour", startHour + 1);
                        SetProperty(energyCentre, "EndHour", endHour + 1);
                    });
                    W("  hours now {0}..{1}",
                        Dump.CallText(energyCentre, "GetStartHour"), Dump.CallText(energyCentre, "GetEndHour"));
                }

                if (on.Contains("nogroup"))
                {
                    foreach (object group in groups)
                    {
                        Try("group multiplicity -> 1", delegate
                        {
                            group.GetType().InvokeMember(
                                "SetMultiplicity", BindingFlags.InvokeMethod, null, group,
                                new object[] { 1 }, CultureInfo.InvariantCulture);
                        });
                        W("  group {0} multiplicity now {1}",
                            Dump.Get(group, "GUID"), Dump.CallText(group, "GetMultiplicity"));
                    }
                }

                if (on.Contains("dhw"))
                {
                    // The plant walk showed "DHW Junction In" with pipesIn=0 and "DHW Junction Out" with
                    // pipesOut=0 - a liquid loop that is not closed. Close it with one pipe and see whether
                    // the diagnostic moves.
                    for (int p2 = 1; p2 <= count_PlantRoom; p2++)
                    {
                        object plantRoom = Dump.Call(energyCentre, "GetPlantRoom", p2);
                        int count_PlantComponent = (int)Dbl(Dump.Call(plantRoom, "GetComponentCount"));

                        object junction_In = null;
                        object junction_Out = null;

                        for (int pc = 1; pc <= count_PlantComponent; pc++)
                        {
                            object plantComponent = Dump.Call(plantRoom, "GetComponent", pc);
                            if (plantComponent == null)
                            {
                                continue;
                            }

                            string type = Dump.CallText(plantComponent, "GetComponentType");
                            if (type == null || type.IndexOf("PlantJunction", StringComparison.OrdinalIgnoreCase) < 0)
                            {
                                continue;
                            }

                            int pipesIn = 0;
                            int pipesOut = 0;
                            int inPorts = (int)Dbl(Dump.Call(plantComponent, "GetInputPortCount"));
                            int outPorts = (int)Dbl(Dump.Call(plantComponent, "GetOutputPortCount"));

                            for (int port = 1; port <= inPorts; port++)
                            {
                                pipesIn += (int)Dbl(Dump.Call(plantComponent, "GetInputPipeCount", port));
                            }

                            for (int port = 1; port <= outPorts; port++)
                            {
                                pipesOut += (int)Dbl(Dump.Call(plantComponent, "GetOutputPipeCount", port));
                            }

                            W("  plant junction {0} pipesIn={1} pipesOut={2}", Dump.Get(plantComponent, "Name"), pipesIn, pipesOut);

                            if (pipesIn == 0 && junction_In == null) { junction_In = plantComponent; }
                            if (pipesOut == 0 && junction_Out == null) { junction_Out = plantComponent; }
                        }

                        if (junction_In != null && junction_Out != null)
                        {
                            Try("AddPipe to close the dangling loop", delegate
                            {
                                plantRoom.GetType().InvokeMember(
                                    "AddPipe", BindingFlags.InvokeMethod, null, plantRoom,
                                    new object[] { junction_Out, 1, junction_In, 1 }, CultureInfo.InvariantCulture);
                            });

                            W("  closed {0} -> {1}", Dump.Get(junction_Out, "Name"), Dump.Get(junction_In, "Name"));
                        }
                        else
                        {
                            W("  no dangling plant junction pair found");
                        }
                    }
                }

                if (on.Contains("loads"))
                {
                    int next = 1;
                    foreach (object zone in zones)
                    {
                        object existing = Dump.Call(zone, "GetSystemZoneZoneLoad");
                        if (existing != null)
                        {
                            W("  zone {0} already bound to {1}", Dump.Get(zone, "GUID"), Dump.Get(existing, "Name"));
                            continue;
                        }

                        if (next > count_ZoneLoad)
                        {
                            W("  zone {0} has no load left to bind", Dump.Get(zone, "GUID"));
                            continue;
                        }

                        object zoneLoad = Dump.Call(tSDData, "GetZoneLoad", next);
                        next++;

                        Try("AddZoneLoad", delegate
                        {
                            zone.GetType().InvokeMember(
                                "AddZoneLoad", BindingFlags.InvokeMethod, null, zone,
                                new object[] { zoneLoad }, CultureInfo.InvariantCulture);
                        });

                        W("  zone {0} bound -> {1} volume {2}",
                            Dump.Get(zone, "GUID"),
                            Dump.Get(Dump.Call(zone, "GetSystemZoneZoneLoad"), "Name"),
                            Dump.Get(zoneLoad, "Volume"));
                    }
                }

                if (on.Contains("zonevalue"))
                {
                    foreach (object zone in zones)
                    {
                        Absolute("zone FlowRate", Dump.Obj(zone, "FlowRate"), 17.0);
                        Absolute("zone FreshAir", Dump.Obj(zone, "FreshAir"), 17.0);
                    }
                }

                if (on.Contains("flags"))
                {
                    foreach (object zone in zones)
                    {
                        Try("zone flags", delegate
                        {
                            int flags = Convert.ToInt32(Dump.Obj(zone, "Flags"), CultureInfo.InvariantCulture);
                            SetProperty(
                                zone,
                                "Flags",
                                flags
                                    | (int)global::TPD.tpdSystemZoneFlags.tpdSystemZoneFlagModelVentFlow
                                    | (int)global::TPD.tpdSystemZoneFlags.tpdSystemZoneFlagModelInterzoneFlow);
                        });
                        W("  zone {0} flags now {1}", Dump.Get(zone, "GUID"), Dump.Get(zone, "Flags"));
                    }
                }

                if (on.Contains("dampervalue"))
                {
                    foreach (object damper in dampers)
                    {
                        W("  damper {0} DesignFlowType before {1} DesignFlowRate {2}",
                            Dump.Get(damper, "GUID"),
                            Dump.Get(damper, "DesignFlowType"),
                            Dump.Get(Dump.Obj(damper, "DesignFlowRate"), "Value"));

                        Try("damper DesignFlowType -> Value", delegate
                        {
                            SetProperty(damper, "DesignFlowType", (int)global::TPD.tpdFlowRateType.tpdFlowRateValue);
                        });

                        Absolute("damper DesignFlowRate", Dump.Obj(damper, "DesignFlowRate"), 17.0);

                        W("  damper {0} DesignFlowType after {1}", Dump.Get(damper, "GUID"), Dump.Get(damper, "DesignFlowType"));
                    }
                }

                if (on.Contains("fanvalue"))
                {
                    foreach (object fan in fans)
                    {
                        W("  fan {0} DesignFlowType before {1} DesignFlowRate {2}",
                            Dump.Get(fan, "GUID"),
                            Dump.Get(fan, "DesignFlowType"),
                            Dump.Get(Dump.Obj(fan, "DesignFlowRate"), "Value"));

                        Try("fan DesignFlowType -> Value", delegate
                        {
                            SetProperty(fan, "DesignFlowType", (int)global::TPD.tpdFlowRateType.tpdFlowRateValue);
                        });

                        Absolute("fan DesignFlowRate", Dump.Obj(fan, "DesignFlowRate"), 34.0);

                        W("  fan {0} DesignFlowType after {1}", Dump.Get(fan, "GUID"), Dump.Get(fan, "DesignFlowType"));
                    }
                }

                tPDDoc.Save();

                // ---------------------------------------------------------------- simulate

                // IPlantRoom.SimulateExx exposes resetSizes / simSystem, and ISystem.Simulate runs the
                // AIR SYSTEM on its own - which is what a ventilation-only route actually needs. Both
                // return strings, so both diagnostics are captured.
                if (on.Contains("exx") || on.Contains("syssim"))
                {
                    for (int p2 = 1; p2 <= count_PlantRoom; p2++)
                    {
                        object plantRoom = Dump.Call(energyCentre, "GetPlantRoom", p2);

                        if (on.Contains("exx"))
                        {
                            foreach (int resetSizes in new[] { 1, 0 })
                            {
                                try
                                {
                                    object answer = plantRoom.GetType().InvokeMember(
                                        "SimulateExx",
                                        BindingFlags.InvokeMethod,
                                        null,
                                        plantRoom,
                                        new object[] { startHour + 1, endHour + 1, 0, resetSizes, 1, 0, null, 1, 1, 0 },
                                        CultureInfo.InvariantCulture);

                                    W("  SimulateExx(resetSizes={0}) -> isNull={1} value=[{2}]",
                                        resetSizes, answer == null, answer == null ? "<null>" : answer.ToString());
                                }
                                catch (Exception exception)
                                {
                                    Exception inner = exception.InnerException ?? exception;
                                    W("  SimulateExx(resetSizes={0}) THREW {1}: {2}",
                                        resetSizes, inner.GetType().Name, inner.Message.Replace(Environment.NewLine, " ").Trim());
                                }
                            }
                        }

                        if (on.Contains("syssim"))
                        {
                            int count_System2 = (int)Dbl(Dump.Call(plantRoom, "GetSystemCount"));
                            for (int s2 = 1; s2 <= count_System2; s2++)
                            {
                                object system2 = Dump.Call(plantRoom, "GetSystem", s2);
                                try
                                {
                                    object answer = system2.GetType().InvokeMember(
                                        "Simulate",
                                        BindingFlags.InvokeMethod,
                                        null,
                                        system2,
                                        new object[] { startHour + 1, endHour + 1, 0 },
                                        CultureInfo.InvariantCulture);

                                    W("  ISystem.Simulate -> isNull={0} value=[{1}]",
                                        answer == null, answer == null ? "<null>" : answer.ToString());
                                }
                                catch (Exception exception)
                                {
                                    Exception inner = exception.InnerException ?? exception;
                                    W("  ISystem.Simulate THREW {0}: {1}",
                                        inner.GetType().Name, inner.Message.Replace(Environment.NewLine, " ").Trim());
                                }
                            }

                            // IPlantRoom.Simulate returns a string too.
                            try
                            {
                                object answer = plantRoom.GetType().InvokeMember(
                                    "Simulate", BindingFlags.InvokeMethod, null, plantRoom,
                                    new object[] { startHour + 1, endHour + 1, 0 }, CultureInfo.InvariantCulture);

                                W("  IPlantRoom.Simulate -> isNull={0} value=[{1}]",
                                    answer == null, answer == null ? "<null>" : answer.ToString());
                            }
                            catch (Exception exception)
                            {
                                Exception inner = exception.InnerException ?? exception;
                                W("  IPlantRoom.Simulate THREW {0}: {1}",
                                    inner.GetType().Name, inner.Message.Replace(Environment.NewLine, " ").Trim());
                            }
                        }
                    }
                }

                W("");
                W("== SIMULATE ==");
                string returned = null;
                try
                {
                    returned = tPDDoc.Simulate(startHour + 1, endHour + 1, 0);
                    W("RESULT: isNull={0} length={1} value=[{2}]",
                        returned == null, returned == null ? -1 : returned.Length, returned ?? "<null>");
                }
                catch (Exception exception)
                {
                    Exception inner = exception.InnerException ?? exception;
                    W("RESULT: THREW {0}: {1}", inner.GetType().Name, inner.Message.Replace(Environment.NewLine, " ").Trim());
                }

                tPDDoc.Save();

                // ---------------------------------------------------------------- what came back

                W("");
                W("== AFTER ==");
                foreach (object zone in zones)
                {
                    object flowRate = Dump.Obj(zone, "FlowRate");
                    W("  zone {0} FlowRate Value={1} Type={2} | FreshAir Value={3} Type={4}",
                        Dump.Get(zone, "GUID"),
                        Dump.Get(flowRate, "Value"), Dump.Get(flowRate, "Type"),
                        Dump.Get(Dump.Obj(zone, "FreshAir"), "Value"), Dump.Get(Dump.Obj(zone, "FreshAir"), "Type"));
                }

                foreach (object fan in fans)
                {
                    W("  fan {0} DesignFlowType={1} DesignFlowRate Value={2} Type={3}",
                        Dump.Get(fan, "Name"),
                        Dump.Get(fan, "DesignFlowType"),
                        Dump.Get(Dump.Obj(fan, "DesignFlowRate"), "Value"),
                        Dump.Get(Dump.Obj(fan, "DesignFlowRate"), "Type"));
                }

                foreach (object damper in dampers)
                {
                    W("  damper {0} DesignFlowType={1} DesignFlowRate Value={2} Type={3}",
                        Dump.Get(damper, "GUID"),
                        Dump.Get(damper, "DesignFlowType"),
                        Dump.Get(Dump.Obj(damper, "DesignFlowRate"), "Value"),
                        Dump.Get(Dump.Obj(damper, "DesignFlowRate"), "Type"));
                }

                // Did any result actually appear?
                W("");
                W("== RESULT PROBE: ZoneTemperature on each zone ==");
                foreach (object zone in zones)
                {
                    object data = null;
                    try
                    {
                        data = zone.GetType().InvokeMember(
                            "GetResultsData",
                            BindingFlags.InvokeMethod,
                            null,
                            zone,
                            new object[]
                            {
                                (int)global::TPD.tpdResultsPeriod.tpdResultsPeriodHourly,
                                (int)global::TPD.tpdCombinerType.tpdCombinerTypeMax,
                                9,
                                startHour + 1,
                                endHour - startHour + 1,
                            },
                            CultureInfo.InvariantCulture);
                    }
                    catch (Exception exception)
                    {
                        Exception inner = exception.InnerException ?? exception;
                        W("  zone {0}: GetResultsData THREW {1}: {2}",
                            Dump.Get(zone, "GUID"), inner.GetType().Name, inner.Message.Replace(Environment.NewLine, " ").Trim());
                        continue;
                    }

                    if (data == null)
                    {
                        W("  zone {0}: GetResultsData -> <null>", Dump.Get(zone, "GUID"));
                        continue;
                    }

                    Array array = data as Array;
                    if (array == null)
                    {
                        W("  zone {0}: GetResultsData -> {1} (not an array)", Dump.Get(zone, "GUID"), data.GetType().Name);
                        continue;
                    }

                    // TAS hands back a MULTI-DIMENSIONAL array here, so GetValue(int) is out of bounds.
                    // foreach walks any rank, which is what the production IndexedDoubles path relies on.
                    int finite = 0;
                    int total = 0;
                    double min = double.MaxValue;
                    double max = double.MinValue;
                    StringBuilder sample = new StringBuilder();

                    foreach (object element in array)
                    {
                        double value = Dbl(element);
                        total++;

                        if (!double.IsNaN(value) && !double.IsInfinity(value))
                        {
                            finite++;
                            if (value < min) { min = value; }
                            if (value > max) { max = value; }
                        }

                        if (total <= 8)
                        {
                            sample.Append(value.ToString("F3", CultureInfo.InvariantCulture)).Append(' ');
                        }
                    }

                    W("  zone {0}: rank={1} length={2} counted={3} FINITE={4} min={5} max={6}",
                        Dump.Get(zone, "GUID"),
                        array.Rank,
                        array.Length,
                        total,
                        finite,
                        min == double.MaxValue ? double.NaN : min,
                        max == double.MinValue ? double.NaN : max);
                    W("      first values: {0}", sample.ToString().Trim());
                }
            }
            finally
            {
                try { tPDDoc.Close(); } catch { }
            }

            Flush(path_TPD);
            return 0;
        }

        private static void Absolute(string label, object sizedFlowVariable, double fallback)
        {
            if (sizedFlowVariable == null)
            {
                W("  {0}: no carrier", label);
                return;
            }

            Try(label, delegate
            {
                double authored = Dbl(Dump.Obj(sizedFlowVariable, "Value"));
                if (double.IsNaN(authored) || authored <= 0.0)
                {
                    SetProperty(sizedFlowVariable, "Value", fallback);
                }

                SetProperty(sizedFlowVariable, "Type", (int)global::TPD.tpdSizedVariable.tpdSizedVariableValue);
            });

            W("  {0} -> Value={1} Type={2}",
                label, Dump.Get(sizedFlowVariable, "Value"), Dump.Get(sizedFlowVariable, "Type"));
        }

        private static void Try(string what, Action action)
        {
            try
            {
                action();
            }
            catch (Exception exception)
            {
                Exception inner = exception.InnerException ?? exception;
                W("  !! {0} FAILED: {1}: {2}", what, inner.GetType().Name, inner.Message.Replace(Environment.NewLine, " ").Trim());
            }
        }

        private static void Flush(string path_TPD)
        {
            try
            {
                File.WriteAllText(Path.ChangeExtension(path_TPD, ".exp.log"), log.ToString());
            }
            catch
            {
            }
        }
    }
}
