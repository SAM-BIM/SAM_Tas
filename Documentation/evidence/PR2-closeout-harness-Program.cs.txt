// PR2 closeout harness. One operation per process; every path absolute and under C:\TasOut.
//
// route-source / route-acceptance are the archived PR2 harness (Route.cs) unchanged except for this
// machine's template path and the "const1" mode, which supplies the frozen #111 parity schedule to PR1.
// source-check and fanprobe are new and read TAS late-bound, independently of the code under test.

using System;
using System.IO;

namespace Inv
{
    internal static class Program
    {
        private static void Usage()
        {
            Console.WriteLine("modes:");
            Console.WriteLine("  route-source     <in.sam> <dir> <weather.twd> <sizing|full>");
            Console.WriteLine("  route-acceptance <in.sam> <dir> <weather.twd> <full[ balanced][ continuity][ const1]>");
            Console.WriteLine("  source-check     <in.tbd> <out.txt>                 IZAM / ticV / ticI on this exact TBD");
            Console.WriteLine("  fanprobe         <in.tpd> <startHour> <endHour> <out.txt>   native hourly fan evidence, on a copy");
        }

        private static int Main(string[] args)
        {
            if (args == null || args.Length == 0)
            {
                Usage();
                return 1;
            }

            try
            {
                switch (args[0].ToLowerInvariant())
                {
                    case "route-source":
                        return Route.Source(args[1], args[2], args[3], args.Length > 4 ? args[4] : "sizing");
                    case "route-acceptance":
                        return Route.Run(args[1], args[2], args[3], args.Length > 4 ? args[4] : "sizing");
                    case "source-check":
                        return Check.Run(args[1], args.Length > 2 ? args[2] : null);
                    case "spread":
                        return Spread.Move(args[1], args[2]);
                    case "zt":
                        return Spread.ZoneTemperatures(args[1], args[2]);
                    case "layout":
                        return Layout.Run(args[1], args.Length > 2 ? args[2] : null);
                    case "graph":
                        Graph.Walk(args[1], args[2]);
                        return 0;
                    case "net":
                        Net.Run(args[1], args[2]);
                        return 0;
                    case "fancontrol":
                        return FanControl.Run(args[1], args[2], args.Length > 3 ? int.Parse(args[3]) : 0, args.Length > 4 ? args[4] : @"C:\TasOut\inv3\fancontrol");
                    case "fanprobe":
                        return FanProbe.Run(args[1], int.Parse(args[2]), int.Parse(args[3]), args.Length > 4 ? args[4] : null);
                    default:
                        Usage();
                        return 1;
                }
            }
            catch (Exception exception)
            {
                string text = string.Concat("EXCEPTION ", exception.GetType().Name, ": ", exception.Message, Environment.NewLine, exception.StackTrace);
                Console.WriteLine(text);

                try
                {
                    File.WriteAllText(@"C:\TasOut\inv3\last-exception.log", text);
                }
                catch
                {
                }

                return 2;
            }
        }
    }
}
