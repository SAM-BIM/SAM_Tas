// Which call actually reads a fan's schedule? The typed and the dynamic reads both answered
// DISP_E_MEMBERNOTFOUND from production while a raw late-bound InvokeMember answered a schedule from
// this harness, so every path is tried here on the same object rather than reasoned about.
//
// Nothing here is production code.

using System;
using System.Globalization;
using System.Reflection;
using SAM.Core.Tas;

namespace Inv
{
    internal static class SchedProbe
    {
        private static string Try(string label, Func<object> read)
        {
            try
            {
                object result = read();
                return string.Format(
                    CultureInfo.InvariantCulture,
                    "{0,-52} -> {1}",
                    label,
                    result == null ? "<null>" : "OK " + Dump.Get(result, "Name") + " Type=" + Dump.Get(result, "Type"));
            }
            catch (Exception exception)
            {
                Exception inner = exception.InnerException ?? exception;
                return string.Format(
                    CultureInfo.InvariantCulture,
                    "{0,-52} -> {1}: {2}",
                    label,
                    inner.GetType().Name,
                    inner.Message.Trim());
            }
        }

        public static void Run(string path)
        {
            using (SAMTPDDocument sAMTPDDocument = new SAMTPDDocument(path))
            {
                object energyCentre = sAMTPDDocument.TPDDocument.EnergyCentre;
                object system = Graph.Systems(energyCentre)[0];

                // (a) raw late-bound enumeration - what this harness used
                object raw = null;
                for (int i = 1; i <= 200; i++)
                {
                    object component = Dump.Call(system, "GetComponent", i);
                    if (component == null)
                    {
                        break;
                    }

                    if (Graph.Kind(component).Contains("Fan"))
                    {
                        raw = component;
                        break;
                    }
                }

                Console.WriteLine("raw object runtime type   : " + (raw == null ? "<none>" : raw.GetType().FullName));
                Console.WriteLine("raw is TPD.Fan            : " + (raw is global::TPD.Fan));
                Console.WriteLine("raw is TPD.SystemComponent: " + (raw is global::TPD.SystemComponent));
                Console.WriteLine();

                Console.WriteLine(Try("(a) raw InvokeMember GetSchedule", () => raw.GetType().InvokeMember(
                    "GetSchedule", BindingFlags.InvokeMethod, null, raw, null, CultureInfo.InvariantCulture)));

                Console.WriteLine(Try("(b) raw dynamic .GetSchedule()", () => ((dynamic)raw).GetSchedule()));

                // (c..f) the typed enumeration - what production uses
                global::TPD.System systemTyped = (global::TPD.System)system;
                global::TPD.Fan fan = null;
                int countTyped = systemTyped.GetComponentCount();

                for (int i = 1; i <= countTyped; i++)
                {
                    global::TPD.SystemComponent systemComponent = systemTyped.GetComponent(i);
                    if (systemComponent is global::TPD.Fan)
                    {
                        fan = (global::TPD.Fan)systemComponent;
                        break;
                    }
                }

                Console.WriteLine();
                Console.WriteLine("typed object runtime type : " + (fan == null ? "<none>" : fan.GetType().FullName));

                Console.WriteLine(Try("(c) typed ((SystemComponent)fan).GetSchedule()",
                    () => ((global::TPD.SystemComponent)fan).GetSchedule()));

                Console.WriteLine(Try("(d) typed ((dynamic)fan).GetSchedule()",
                    () => ((dynamic)fan).GetSchedule()));

                Console.WriteLine(Try("(e) typed InvokeMember GetSchedule",
                    () => fan.GetType().InvokeMember("GetSchedule", BindingFlags.InvokeMethod, null, fan, null, CultureInfo.InvariantCulture)));

                Console.WriteLine(Try("(f) typed cast to ISystemComponent GetSchedule",
                    () => ((global::TPD.ISystemComponent)fan).GetSchedule()));

                Console.WriteLine();
                Console.WriteLine("-- what the raw enumeration reports for every component --");
                for (int i = 1; i <= 200; i++)
                {
                    object component = Dump.Call(system, "GetComponent", i);
                    if (component == null)
                    {
                        break;
                    }

                    Console.WriteLine(string.Format(
                        CultureInfo.InvariantCulture,
                        "  {0,-14} {1,-30} raw={2}",
                        Graph.Kind(component),
                        Dump.Get(component, "Name"),
                        Dump.Call(component, "GetSchedule") == null ? "<null>" : "schedule"));
                }
            }
        }
    }
}
