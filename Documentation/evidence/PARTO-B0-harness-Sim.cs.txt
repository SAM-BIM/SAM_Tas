using System;
using System.Collections;
using System.Collections.Generic;
using System.Globalization;
using System.IO;
using System.Linq;
using TPD;

namespace B0P
{
    public static partial class Steps
    {
        // simsys <file.tpd> <tsdPath> <tbdPath> <outCsv> [dv0|dvkeep]
        public static void SimSys(string[] args)
        {
            string path = args[1], tsd = args[2], tbd = args[3], outCsv = args[4];
            string mode = args.Length > 5 ? args[5] : "dvkeep";
            TPDDoc doc = new TPDDoc();
            doc.Open(path);
            var series = new List<Tuple<string, double[]>>();
            try
            {
                EnergyCentre ec = doc.EnergyCentre;
                for (int i = 1; i <= ec.GetTSDDataCount(); i++)
                {
                    dynamic td = ec.GetTSDData(i);
                    W("TSDData[{0}] before tsd='{1}' tbd='{2}'", i, td.TSDPath, td.TBDPath);
                    td.TSDPath = tsd; td.TBDPath = tbd;
                    W("TSDData[{0}] after  tsd='{1}' tbd='{2}'", i, td.TSDPath, td.TBDPath);
                }
                if (mode == "dv0")
                {
                    int changed = 0;
                    for (int p = 1; p <= ec.GetPlantRoomCount(); p++)
                    {
                        PlantRoom pr = ec.GetPlantRoom(p);
                        for (int s = 1; s <= pr.GetSystemCount(); s++)
                            foreach (SystemComponent sc in Components(pr.GetSystem(s)))
                                if (sc is SystemZone)
                                {
                                    SystemZone sz = (SystemZone)sc;
                                    int before = sz.DisplacementVent;
                                    sz.DisplacementVent = 0;
                                    W("DV|{0}|{1} -> {2}", CName(sc), before, sz.DisplacementVent);
                                    changed++;
                                }
                    }
                    W("DisplacementVent cleared on {0} zone(s)", changed);
                }
                for (int p = 1; p <= ec.GetPlantRoomCount(); p++)
                {
                    PlantRoom pr = ec.GetPlantRoom(p);
                    for (int s = 1; s <= pr.GetSystemCount(); s++)
                    {
                        global::TPD.System sys = pr.GetSystem(s);
                        string sname = CName(sys);
                        DateTime t0 = DateTime.UtcNow;
                        string answer = sys.Simulate(1, 8760, 0);
                        W("SIMULATE|{0}|{1}|{2:F1}s", sname, answer, (DateTime.UtcNow - t0).TotalSeconds);
                        int dc = sys.GetDuctCount();
                        for (int d = 1; d <= dc; d++)
                        {
                            Duct duct = sys.GetDuct(d);
                            if (duct == null) continue;
                            object up = duct.GetUpstreamComponent(), dn = duct.GetDownstreamComponent();
                            string tag = string.Format("{0}:{1}->{2}:{3}", CKind(up), CName(up), CKind(dn), CName(dn));
                            for (int v = 1; v <= 6; v++)
                            {
                                double[] r = Results(duct, v, 1, 8760);
                                if (r == null) { if (v == 1) W("   DUCT {0,-58} var{1} NULL {2}", tag, v, LastDiag); continue; }
                                W("   DUCT {0,-58} var{1,-2} n={2} min={3:F3} max={4:F3} mean={5:F3}", tag, v, r.Length, r.Min(), r.Max(), r.Average());
                                series.Add(Tuple.Create("DUCT|" + tag + "|var" + v, r));
                            }
                        }
                        foreach (SystemComponent sc in Components(sys))
                        {
                            if (!(sc is SystemZone)) continue;
                            string zn = CName(sc);
                            foreach (int v in new[] { 1, 3, 9 })
                            {
                                double[] r = Results(sc, v, 1, 8760);
                                if (r == null) { W("   {0} var{1} NULL {2}", zn, v, LastDiag); continue; }
                                W("   {0,-12} var{1,-2} n={2} min={3:F4} max={4:F4} mean={5:F4}", zn, v, r.Length, r.Min(), r.Max(), r.Average());
                                series.Add(Tuple.Create(zn + "|var" + v, r));
                            }
                        }
                    }
                }
                doc.Save();
            }
            finally { doc.Close(); }
            using (StreamWriter sw = new StreamWriter(outCsv, false))
            {
                sw.WriteLine("hour," + string.Join(",", series.Select(x => x.Item1)));
                int n = series.Count == 0 ? 0 : series.Min(x => x.Item2.Length);
                for (int h = 0; h < n; h++)
                {
                    sw.Write(h);
                    foreach (var s in series) { sw.Write(','); sw.Write(s.Item2[h].ToString("R", CultureInfo.InvariantCulture)); }
                    sw.WriteLine();
                }
            }
            W("WROTE {0} ({1} series)", outCsv, series.Count);
        }
    }
}
