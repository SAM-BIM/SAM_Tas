using System;
using System.Collections;
using System.Linq;
using TPD;

namespace B0P
{
    public static partial class Steps
    {
        // zonecomp <file.tpd>
        public static void ZoneComp(string[] args)
        {
            TPDDoc doc = new TPDDoc();
            doc.OpenReadOnly(args[1]);
            try
            {
                EnergyCentre ec = doc.EnergyCentre;
                for (int p = 1; p <= ec.GetPlantRoomCount(); p++)
                {
                    PlantRoom pr = ec.GetPlantRoom(p);
                    W("PLANTROOM|{0}|plantComponentCount={1}|controllerCount={2}|heatingGroups={3}|coolingGroups={4}|dhwGroups={5}",
                        CName(pr), pr.GetComponentCount(), pr.GetControllerCount(), pr.GetHeatingGroupCount(), pr.GetCoolingGroupCount(), pr.GetDHWGroupCount());
                    for (int s = 1; s <= pr.GetSystemCount(); s++)
                    {
                        global::TPD.System sys = pr.GetSystem(s);
                        foreach (SystemComponent sc in Components(sys))
                        {
                            if (!(sc is SystemZone)) continue;
                            SystemZone sz = (SystemZone)sc;
                            int zc = 0;
                            try { zc = sz.GetZoneComponentCount(); } catch (Exception e) { W("  zonecompcount threw {0}", e.GetType().Name); }
                            string names = "";
                            for (int i = 1; i <= zc; i++) { try { names += CKind(sz.GetZoneComponent(i)) + ":" + CName(sz.GetZoneComponent(i)) + " "; } catch { names += "? "; } }
                            string tsp = "-", rsp = "-", psp = "-";
                            try { tsp = PD(sz.TemperatureSetpoint); } catch (Exception e) { tsp = "!" + e.GetType().Name; }
                            try { rsp = PD(sz.RHSetpoint); } catch (Exception e) { rsp = "!" + e.GetType().Name; }
                            try { psp = PD(sz.PollutantSetpoint); } catch (Exception e) { psp = "!" + e.GetType().Name; }
                            W("SYSZONE|{0}|{1}|zoneComponents={2} [{3}]|TemperatureSetpoint={4}|RHSetpoint={5}|PollutantSetpoint={6}|DisplacementVent={7}",
                                CName(sys), CName(sc), zc, names.Trim(), tsp, rsp, psp, sz.DisplacementVent);
                        }
                    }
                    // plant components attached to each zone load
                    for (int i = 1; i <= ec.GetTSDDataCount(); i++)
                    {
                        dynamic td = ec.GetTSDData(i);
                        int zlc = td.GetZoneLoadCount();
                        for (int z = 1; z <= zlc; z++)
                        {
                            dynamic zl = td.GetZoneLoad(z);
                            int pc = 0, scn = 0;
                            try { pc = zl.GetPlantComponentCount(); } catch { }
                            try { scn = zl.GetSystemComponentCount(); } catch { }
                            if (pc != 0) W("ZONELOAD|{0}|plantComponents={1}|systemComponents={2}", zl.Name, pc, scn);
                        }
                        W("ZONELOADS|{0} total, all with 0 plant components unless listed above", zlc);
                    }
                }
            }
            finally { doc.Close(); }
        }
    }
}
