using System;
using System.Collections;
using System.Collections.Generic;
using System.Globalization;
using System.IO;
using System.Linq;
using TBD;

namespace B0P
{
    public static partial class Steps
    {
        // tbdizam <file.tbd>
        public static void TbdIzam(string[] args)
        {
            TBDDocument doc = new TBDDocument();
            doc.open(args[1]);
            try
            {
                Building b = doc.Building;
                W("TBD {0}", args[1]);
                int zi = 0;
                dynamic z = b.GetZone(zi);
                while (z != null)
                {
                    double ticV = 0, ticIC = 0;
                    try { ticV = z.ticV; } catch { }
                    W("ZONE|{0}|{1}|{2}|vol={3}|floorArea={4}|ticV={5}", zi, z.name, z.GUID, z.volume, z.floorArea, ticV);
                    int ici = 0;
                    try
                    {
                        dynamic ic = z.GetIC(ici);
                        while (ic != null)
                        {
                            W("   IC|{0}|{1}", ici, ic.name);
                            ici++;
                            ic = z.GetIC(ici);
                        }
                    }
                    catch { }
                    zi++;
                    z = b.GetZone(zi);
                }
                int i = 0;
                dynamic izam = b.GetIZAM(i);
                while (izam != null)
                {
                    dynamic src = null;
                    try { src = izam.GetSourceZone(); } catch { }
                    string srcName = src == null ? "(outside/none)" : (string)src.name;
                    List<string> tgts = new List<string>();
                    int t = 0;
                    while (true)
                    {
                        dynamic tz = null;
                        try { tz = izam.GetTargetZone(t); } catch { }
                        if (tz == null) break;
                        tgts.Add((string)tz.name);
                        t++;
                    }
                    dynamic prof = null;
                    try { prof = izam.GetProfile(); } catch { }
                    string profInfo = "-";
                    if (prof != null)
                    {
                        try { profInfo = string.Format("name='{0}' type={1}", prof.name, prof.type); } catch { }
                        try
                        {
                            object v = prof.GetHourlyValues();
                            if (v is IEnumerable en)
                            {
                                var arr = en.Cast<object>().Select(x => Convert.ToDouble(x)).ToArray();
                                profInfo += string.Format(" hourly n={0} min={1:F3} max={2:F3} mean={3:F3}", arr.Length, arr.Min(), arr.Max(), arr.Average());
                            }
                        }
                        catch (Exception ex) { profInfo += " (hourly: " + Compact(ex.Message) + ")"; }
                    }
                    W("IZAM|{0}|name='{1}'|desc='{2}'|fromOutside={3}|source={4}|targets=[{5}]|profile: {6}",
                        i, izam.name, izam.description, izam.fromOutside, srcName, string.Join(";", tgts), profInfo);
                    i++;
                    izam = b.GetIZAM(i);
                }
                W("IZAM count = {0}", i);
            }
            finally { doc.close(); }
        }
    }
}
