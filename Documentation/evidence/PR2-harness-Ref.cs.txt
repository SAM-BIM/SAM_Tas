// Checkpoint 1B: discover how TAS Systems actually represents supply, extract, transfer and branching
// transfer airflow, by observation rather than assumption.
//
// Two tools:
//   dump-graph  - an exhaustive late-bound walk of every native object in a TPD
//   build-ref   - a minimal reference system with deliberately asymmetric, non-round flows
//
// The differential method: build-ref twice with ONE value changed, dump-graph both, diff. Whatever
// moved is the carrier. Nothing here forces the approved design onto TAS.

using System;
using System.Collections.Generic;
using System.Globalization;
using System.IO;
using System.Text;

namespace Inv
{
    internal static class Ref
    {
        // Deliberately asymmetric, non-round, mutually distinct - so a carrier is unmistakable in a dump.
        public const double Supply_Only_Lps = 17.0;
        public const double Extract_Only_Lps = 23.0;
        public const double Both_Supply_Lps = 31.0;
        public const double Both_Extract_Lps = 19.0;
        public const double Transfer_Lps = 11.0;
        public const double Transfer_Second_Lps = 7.0;

        // ------------------------------------------------------------------ walking helpers

        /// <summary>
        /// Every component reachable from a system, INCLUDING the ones nested inside a ComponentGroup.
        /// A real TAS-authored system puts its fans, coils and zones inside a group, so a walk that stops
        /// at the group sees nothing at all.
        /// </summary>
        public static List<object> Components(object system)
        {
            List<object> result = new List<object>();
            Collect(system, result, 0);
            return result;
        }

        private static void Collect(object container, List<object> result, int depth)
        {
            if (container == null || depth > 8)
            {
                return;
            }

            for (int index = 1; index <= 2000; index++)
            {
                object systemComponent = Dump.Call(container, "GetComponent", index);
                if (systemComponent == null)
                {
                    break;
                }

                result.Add(systemComponent);

                // A ComponentGroup holds the real components; recurse into it.
                if (systemComponent is global::TPD.ComponentGroup)
                {
                    Collect(systemComponent, result, depth + 1);
                }
            }
        }

        /// <summary>Every duct a group holds, which is where a grouped system's topology actually lives.</summary>
        public static List<object> GroupDucts(object componentGroup)
        {
            List<object> result = new List<object>();

            int count = 0;
            try
            {
                object value = Dump.Call(componentGroup, "GetDuctCount");
                count = value == null ? 0 : Convert.ToInt32(value, CultureInfo.InvariantCulture);
            }
            catch
            {
                return result;
            }

            for (int i = 1; i <= count; i++)
            {
                object duct = Dump.Call(componentGroup, "GetDuct", i);
                if (duct != null)
                {
                    result.Add(duct);
                }
            }

            return result;
        }

        /// <summary>Every duct reachable from a component's input and output ports.</summary>
        private static void Ducts(StringBuilder stringBuilder, string indent, object systemComponent)
        {
            int inputPorts = ToInt(Dump.Call(systemComponent, "GetInputPortCount"));
            int outputPorts = ToInt(Dump.Call(systemComponent, "GetOutputPortCount"));

            stringBuilder.AppendLine(string.Concat(indent, "ports in=", inputPorts, " out=", outputPorts));

            for (int port = 1; port <= inputPorts; port++)
            {
                int count = ToInt(Dump.Call(systemComponent, "GetInputDuctCount", port));
                for (int i = 1; i <= count; i++)
                {
                    object duct = Dump.Call(systemComponent, "GetInputDuct", port, i);
                    DescribeDuct(stringBuilder, string.Concat(indent, "  IN  port ", port, " duct ", i, ": "), duct);
                }
            }

            for (int port = 1; port <= outputPorts; port++)
            {
                int count = ToInt(Dump.Call(systemComponent, "GetOutputDuctCount", port));
                for (int i = 1; i <= count; i++)
                {
                    object duct = Dump.Call(systemComponent, "GetOutputDuct", port, i);
                    DescribeDuct(stringBuilder, string.Concat(indent, "  OUT port ", port, " duct ", i, ": "), duct);
                }
            }
        }

        private static void DescribeDuct(StringBuilder stringBuilder, string prefix, object duct)
        {
            if (duct == null)
            {
                stringBuilder.AppendLine(string.Concat(prefix, "<null>"));
                return;
            }

            object upstream = Dump.Call(duct, "GetUpstreamComponent");
            object downstream = Dump.Call(duct, "GetDownstreamComponent");

            stringBuilder.AppendLine(string.Concat(
                prefix,
                "up=", Dump.Get(upstream, "Name"), "/", Dump.Get(upstream, "GUID"),
                " port=", Dump.CallText(duct, "GetUpstreamComponentPort"),
                " -> down=", Dump.Get(downstream, "Name"), "/", Dump.Get(downstream, "GUID"),
                " port=", Dump.CallText(duct, "GetDownstreamComponentPort")));

            // Every declared property of a duct, so a design-flow carrier cannot hide.
            Dump.Properties(stringBuilder, string.Concat(prefix, "    prop "), duct, typeof(global::TPD.IDuct));
        }

        private static int ToInt(object value)
        {
            try
            {
                return value == null ? 0 : Convert.ToInt32(value, CultureInfo.InvariantCulture);
            }
            catch
            {
                return 0;
            }
        }

        // ------------------------------------------------------------------ dump-graph

        public static string Graph(string path_TPD)
        {
            StringBuilder stringBuilder = new StringBuilder();

            global::TPD.TPDDoc tPDDoc = new global::TPD.TPDDoc();
            try
            {
                tPDDoc.Open(path_TPD);

                object energyCentre = tPDDoc.EnergyCentre;
                stringBuilder.AppendLine(string.Concat("FILE ", Path.GetFileName(path_TPD)));
                stringBuilder.AppendLine(string.Concat("EnergyCentre ", energyCentre == null ? "<null>" : "present"));

                if (energyCentre == null)
                {
                    return stringBuilder.ToString();
                }

                stringBuilder.AppendLine(string.Concat("StartHour ", Dump.CallText(energyCentre, "GetStartHour")));
                stringBuilder.AppendLine(string.Concat("EndHour ", Dump.CallText(energyCentre, "GetEndHour")));

                int count_TSDData = ToInt(Dump.Call(energyCentre, "GetTSDDataCount"));
                stringBuilder.AppendLine(string.Concat("TSDDataCount ", count_TSDData));

                for (int t = 1; t <= count_TSDData; t++)
                {
                    object tSDData = Dump.Call(energyCentre, "GetTSDData", t);
                    if (tSDData == null)
                    {
                        continue;
                    }

                    stringBuilder.AppendLine(string.Concat("  TSDData[", t, "]"));
                    Dump.Properties(stringBuilder, "    ", tSDData, typeof(global::TPD.ITSDData));

                    int count_ZoneLoad = ToInt(Dump.Call(tSDData, "GetZoneLoadCount"));
                    stringBuilder.AppendLine(string.Concat("    ZoneLoadCount ", count_ZoneLoad));

                    for (int z = 1; z <= count_ZoneLoad; z++)
                    {
                        object zoneLoad = Dump.Call(tSDData, "GetZoneLoad", z);
                        if (zoneLoad == null)
                        {
                            continue;
                        }

                        string guid = Dump.Get(zoneLoad, "GUID");

                        stringBuilder.AppendLine(string.Concat(
                            "      ZoneLoad[", z, "] Name=", Dump.Get(zoneLoad, "Name"),
                            " GUID=", guid,
                            " FloorArea=", Dump.Get(zoneLoad, "FloorArea"),
                            " Volume=", Dump.Get(zoneLoad, "Volume")));

                        // Does the load-side bridge resolve this exact load back?
                        string trimmed = guid.Trim('"');
                        object roundTripped = Dump.Call(tSDData, "GetZoneLoadForGuid", trimmed);
                        stringBuilder.AppendLine(string.Concat(
                            "        GetZoneLoadForGuid -> ",
                            roundTripped == null ? "<null>" : string.Concat("Name=", Dump.Get(roundTripped, "Name"), " GUID=", Dump.Get(roundTripped, "GUID"))));

                        stringBuilder.AppendLine(string.Concat(
                            "        SystemComponentCount=", Dump.CallText(zoneLoad, "GetSystemComponentCount")));

                        int count_SystemComponent = ToInt(Dump.Call(zoneLoad, "GetSystemComponentCount"));
                        for (int c = 1; c <= count_SystemComponent; c++)
                        {
                            object systemComponent = Dump.Call(zoneLoad, "GetSystemComponent", c);
                            stringBuilder.AppendLine(string.Concat(
                                "          -> component Name=", Dump.Get(systemComponent, "Name"),
                                " GUID=", Dump.Get(systemComponent, "GUID"),
                                " Type=", Dump.CallText(systemComponent, "GetComponentType")));
                        }
                    }
                }

                int count_PlantRoom = ToInt(Dump.Call(energyCentre, "GetPlantRoomCount"));
                stringBuilder.AppendLine(string.Concat("PlantRoomCount ", count_PlantRoom));

                for (int p = 1; p <= count_PlantRoom; p++)
                {
                    object plantRoom = Dump.Call(energyCentre, "GetPlantRoom", p);
                    if (plantRoom == null)
                    {
                        continue;
                    }

                    stringBuilder.AppendLine(string.Concat("PlantRoom[", p, "] Name=", Dump.Get(plantRoom, "Name")));

                    int count_System = ToInt(Dump.Call(plantRoom, "GetSystemCount"));
                    stringBuilder.AppendLine(string.Concat("  SystemCount ", count_System));

                    for (int s = 1; s <= count_System; s++)
                    {
                        object system = Dump.Call(plantRoom, "GetSystem", s);
                        if (system == null)
                        {
                            continue;
                        }

                        stringBuilder.AppendLine(string.Concat(
                            "  System[", s, "] Name=", Dump.Get(system, "Name"),
                            " GUID=", Dump.Get(system, "GUID"),
                            " Multiplicity=", Dump.Get(system, "Multiplicity")));

                        Dump.Properties(stringBuilder, "    sysprop ", system, typeof(global::TPD.ISystem));

                        List<object> components = Components(system);
                        stringBuilder.AppendLine(string.Concat("    ComponentCount ", components.Count));

                        for (int c = 0; c < components.Count; c++)
                        {
                            object systemComponent = components[c];

                            string componentType = Dump.CallText(systemComponent, "GetComponentType");

                            stringBuilder.AppendLine(string.Concat(
                                "    Component[", c + 1, "] Type=", componentType,
                                " Name=", Dump.Get(systemComponent, "Name"),
                                " GUID=", Dump.Get(systemComponent, "GUID"),
                                " ZoneLoadCount=", Dump.CallText(systemComponent, "GetZoneLoadCount")));

                            // Round trip the identity through the system, exactly as a binding would.
                            string guid_Component = Dump.Get(systemComponent, "GUID").Trim('"');
                            if (!string.IsNullOrWhiteSpace(guid_Component) && !guid_Component.StartsWith("<"))
                            {
                                object back = Dump.Call(system, "GetComponentByGUID", guid_Component);
                                stringBuilder.AppendLine(string.Concat(
                                    "      GetComponentByGUID -> ",
                                    back == null ? "<null>" : string.Concat("GUID=", Dump.Get(back, "GUID"), " Type=", Dump.CallText(back, "GetComponentType"))));
                            }

                            global::TPD.SystemZone systemZone = systemComponent as global::TPD.SystemZone;
                            if (systemZone != null)
                            {
                                stringBuilder.AppendLine("      -- SystemZone --");
                                stringBuilder.AppendLine(string.Concat("      Flags=", Dump.Get(systemComponent, "Flags")));
                                stringBuilder.AppendLine(string.Concat("      DisplacementVent=", Dump.Get(systemComponent, "DisplacementVent")));
                                stringBuilder.AppendLine(string.Concat("      MinimumFlowFraction=", Dump.Get(systemComponent, "MinimumFlowFraction")));

                                Dump.SizedFlow(stringBuilder, "      ", "FlowRate", Dump.Obj(systemComponent, "FlowRate"));
                                Dump.SizedFlow(stringBuilder, "      ", "FreshAir", Dump.Obj(systemComponent, "FreshAir"));

                                object zoneLoad = Dump.Call(systemComponent, "GetSystemZoneZoneLoad");
                                stringBuilder.AppendLine(string.Concat(
                                    "      GetSystemZoneZoneLoad -> ",
                                    zoneLoad == null ? "<null>" : string.Concat("Name=", Dump.Get(zoneLoad, "Name"), " GUID=", Dump.Get(zoneLoad, "GUID"))));

                                stringBuilder.AppendLine(string.Concat("      ZoneComponentCount=", Dump.CallText(systemComponent, "GetZoneComponentCount")));

                                object duct_In = Dump.Call(systemComponent, "GetSystemZoneInputDuct");
                                object duct_Out = Dump.Call(systemComponent, "GetSystemZoneOutputDuct");
                                stringBuilder.AppendLine(string.Concat("      SystemZoneInputDuct  = ", duct_In == null ? "<null>" : "present"));
                                stringBuilder.AppendLine(string.Concat("      SystemZoneOutputDuct = ", duct_Out == null ? "<null>" : "present"));
                            }

                            global::TPD.Damper damper = systemComponent as global::TPD.Damper;
                            if (damper != null)
                            {
                                stringBuilder.AppendLine("      -- Damper --");
                                stringBuilder.AppendLine(string.Concat("      DesignFlowType=", Dump.Get(systemComponent, "DesignFlowType")));
                                stringBuilder.AppendLine(string.Concat("      MinimumFlowType=", Dump.Get(systemComponent, "MinimumFlowType")));
                                stringBuilder.AppendLine(string.Concat("      MinimumFlowFraction=", Dump.Get(systemComponent, "MinimumFlowFraction")));
                                stringBuilder.AppendLine(string.Concat("      DesignPressureDrop=", Dump.Get(systemComponent, "DesignPressureDrop")));
                                Dump.SizedFlow(stringBuilder, "      ", "DesignFlowRate", Dump.Obj(systemComponent, "DesignFlowRate"));
                                Dump.SizedFlow(stringBuilder, "      ", "MinimumFlowRate", Dump.Obj(systemComponent, "MinimumFlowRate"));
                            }

                            global::TPD.Fan fan = systemComponent as global::TPD.Fan;
                            if (fan != null)
                            {
                                stringBuilder.AppendLine("      -- Fan --");
                                stringBuilder.AppendLine(string.Concat("      DesignFlowType=", Dump.Get(systemComponent, "DesignFlowType")));
                                stringBuilder.AppendLine(string.Concat("      MinimumFlowType=", Dump.Get(systemComponent, "MinimumFlowType")));
                                stringBuilder.AppendLine(string.Concat("      HeatGainFactor=", Dump.Get(systemComponent, "HeatGainFactor")));
                                stringBuilder.AppendLine(string.Concat("      Pressure=", Dump.Get(systemComponent, "Pressure")));
                                Dump.SizedFlow(stringBuilder, "      ", "DesignFlowRate", Dump.Obj(systemComponent, "DesignFlowRate"));
                                Dump.SizedFlow(stringBuilder, "      ", "MinimumFlowRate", Dump.Obj(systemComponent, "MinimumFlowRate"));
                            }

                            for (int z = 1; z <= ToInt(Dump.Call(systemComponent, "GetZoneLoadCount")); z++)
                            {
                                object zoneLoad = Dump.Call(systemComponent, "GetZoneLoad", z);
                                stringBuilder.AppendLine(string.Concat(
                                    "      ZoneLoad[", z, "] Name=", Dump.Get(zoneLoad, "Name"),
                                    " GUID=", Dump.Get(zoneLoad, "GUID")));
                            }

                            if (systemComponent is global::TPD.ComponentGroup)
                            {
                                stringBuilder.AppendLine(string.Concat(
                                    "      -- ComponentGroup -- Multiplicity=", Dump.CallText(systemComponent, "GetMultiplicity"),
                                    " ComponentCount=", Dump.CallText(systemComponent, "GetComponentCount"),
                                    " BaseComponentCount=", Dump.CallText(systemComponent, "GetBaseComponentCount"),
                                    " DuctCount=", Dump.CallText(systemComponent, "GetDuctCount"),
                                    " BaseDuctCount=", Dump.CallText(systemComponent, "GetBaseDuctCount")));

                                List<object> ducts_Group = GroupDucts(systemComponent);
                                for (int d = 0; d < ducts_Group.Count; d++)
                                {
                                    DescribeDuct(stringBuilder, string.Concat("      groupduct[", d + 1, "]: "), ducts_Group[d]);
                                }
                            }

                            Ducts(stringBuilder, "      ", systemComponent);
                        }
                    }
                }
            }
            finally
            {
                try { tPDDoc.Close(); } catch { }
            }

            return stringBuilder.ToString();
        }
    }
}
