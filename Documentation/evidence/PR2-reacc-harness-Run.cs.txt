// PHASE D - the PR2 route on the real design, and the licensed acceptance items 4 to 15.
//
// The design comes from Phase A's prepared model. The thermal source comes from Phase C. Nothing here
// authors, balances, completes or corrects any part of the design: the only cluster write is
// Pr1.ScopeSystems, whose invariance is asserted here rather than assumed.
//
// Every engineering step is a production call:
//
//   SAM.Analytical.Systems.Create.MechanicalVentilation      the PR1 graph
//   SAM.Analytical.Tas.TPD.Create.SystemVentilationRoute     the PR2 route and the TAS simulation

using System;
using System.Collections.Generic;
using System.Globalization;
using System.IO;
using System.Text;

namespace Inv
{
    internal static class Run
    {
        private static void W(string format, params object[] args)
        {
            Route.W(format, args);
        }

        /// <summary>
        /// The design's shape, taken from the SAME measurement Phase A published - so items 4 to 6 are
        /// asserted against the analytical design and never against a literal.
        /// </summary>
        internal static Route.Shape ShapeOf(Prep.Design design)
        {
            Route.Shape result = new Route.Shape();

            HashSet<Guid> rooms = new HashSet<Guid>();
            HashSet<Guid> supply = new HashSet<Guid>();
            HashSet<Guid> extract = new HashSet<Guid>();
            HashSet<Guid> systems = new HashSet<Guid>();

            foreach (Prep.Room room in design.Rooms)
            {
                rooms.Add(room.SpaceGuid);
                systems.Add(room.SystemGuid);

                if (room.Supply_Lps > 0)
                {
                    supply.Add(room.SpaceGuid);
                }

                if (room.Extract_Lps > 0)
                {
                    extract.Add(room.SpaceGuid);
                }
            }

            foreach (Prep.Leg leg in design.Legs)
            {
                rooms.Add(leg.FromGuid);
                rooms.Add(leg.ToGuid);
            }

            result.AirHandlingUnits = systems.Count;
            result.Rooms = rooms.Count;
            result.Supply = supply.Count;
            result.Extract = extract.Count;
            result.Transfer = design.Legs.Count;

            foreach (Guid guid in rooms)
            {
                bool hasSupply = supply.Contains(guid);
                bool hasExtract = extract.Contains(guid);

                if (hasSupply && hasExtract)
                {
                    result.Both++;
                }
                else if (hasSupply)
                {
                    result.SupplyOnly++;
                }
                else if (hasExtract)
                {
                    result.ExtractOnly++;
                }
                else
                {
                    result.TransferOnly++;
                }
            }

            return result;
        }

        /// <summary>
        /// Writes what the PR2 route CLAIMS - room identity and per-leg topology - so the directed
        /// topology acceptance can compare three independent statements: the analytical intent, this
        /// claim, and the native connector ports read straight off the document.
        /// </summary>
        private static void Claim(
            SAM.Analytical.Tas.TPD.SystemVentilationRoute systemVentilationRoute,
            SAM.Analytical.Systems.MechanicalVentilationMaterialisation mechanicalVentilationMaterialisation,
            Prep.Design design,
            string directory)
        {
            //SystemSpace guid -> analytical space guid, off PR1's own lineage.
            Dictionary<Guid, Guid> space_By_SystemSpace = new Dictionary<Guid, Guid>();

            foreach (SAM.Analytical.Systems.MechanicalVentilationBinding mechanicalVentilationBinding in mechanicalVentilationMaterialisation.Bindings)
            {
                if (mechanicalVentilationBinding.BindingType == SAM.Analytical.Systems.MechanicalVentilationBindingType.SystemSpace)
                {
                    space_By_SystemSpace[mechanicalVentilationBinding.Guid_Systems] = mechanicalVentilationBinding.Guid_Analytical;
                }
            }

            Dictionary<Guid, string> name_By_Space = new Dictionary<Guid, string>();
            Dictionary<Guid, string> system_By_Space = new Dictionary<Guid, string>();

            foreach (Prep.Room room in design.Rooms)
            {
                name_By_Space[room.SpaceGuid] = room.SpaceName;
                system_By_Space[room.SpaceGuid] = room.System;
            }

            foreach (Prep.Leg leg in design.Legs)
            {
                if (!name_By_Space.ContainsKey(leg.FromGuid))
                {
                    name_By_Space[leg.FromGuid] = leg.FromName;
                }

                if (!name_By_Space.ContainsKey(leg.ToGuid))
                {
                    name_By_Space[leg.ToGuid] = leg.ToName;
                }
            }

            List<string> lines_Room = new List<string>();
            lines_Room.Add(string.Join("\t", new string[] { "space_guid", "space_name", "system", "reference_system", "reference_systemzone", "reference_zoneload", "supply_lps", "extract_lps" }));

            foreach (SAM.Analytical.Tas.TPD.SystemVentilationBinding binding in systemVentilationRoute.Bindings)
            {
                string name;
                string system;

                lines_Room.Add(string.Join("\t", new string[]
                {
                    binding.Guid_Space.ToString("D"),
                    name_By_Space.TryGetValue(binding.Guid_Space, out name) ? name : "<unknown>",
                    system_By_Space.TryGetValue(binding.Guid_Space, out system) ? system : "<none>",
                    binding.Reference_System,
                    binding.Reference_SystemZone,
                    binding.Reference_ZoneLoad,
                    binding.DesignFlowRate_Supply_Lps.HasValue ? binding.DesignFlowRate_Supply_Lps.Value.ToString("R", CultureInfo.InvariantCulture) : string.Empty,
                    binding.DesignFlowRate_Extract_Lps.HasValue ? binding.DesignFlowRate_Extract_Lps.Value.ToString("R", CultureInfo.InvariantCulture) : string.Empty,
                }));
            }

            File.WriteAllLines(Path.Combine(directory, "claim-rooms.tsv"), lines_Room);

            List<string> lines_Leg = new List<string>();
            lines_Leg.Add(string.Join("\t", new string[] { "type", "from_space_guid", "from_name", "to_space_guid", "to_name", "duty_lps", "reference_flowcontroller", "air_system" }));

            foreach (SAM.Analytical.Tas.TPD.SystemVentilationConnectionBinding connectionBinding in systemVentilationRoute.ConnectionBindings)
            {
                Guid guid_From;
                Guid guid_To;

                space_By_SystemSpace.TryGetValue(connectionBinding.Guid_SystemSpace_From, out guid_From);
                space_By_SystemSpace.TryGetValue(connectionBinding.Guid_SystemSpace_To, out guid_To);

                string name_From;
                string name_To;

                lines_Leg.Add(string.Join("\t", new string[]
                {
                    connectionBinding.ConnectionType.ToString(),
                    guid_From.ToString("D"),
                    name_By_Space.TryGetValue(guid_From, out name_From) ? name_From : "<the air handling unit>",
                    guid_To.ToString("D"),
                    name_By_Space.TryGetValue(guid_To, out name_To) ? name_To : "<the air handling unit>",
                    connectionBinding.DesignFlowRate_Lps.ToString("R", CultureInfo.InvariantCulture),
                    connectionBinding.Reference_FlowController ?? string.Empty,
                    connectionBinding.Guid_AirSystem.ToString("D"),
                }));
            }

            File.WriteAllLines(Path.Combine(directory, "claim-legs.tsv"), lines_Leg);

            W("");
            W("the route's claim written: claim-rooms.tsv ({0} room(s)), claim-legs.tsv ({1} leg(s))", lines_Room.Count - 1, lines_Leg.Count - 1);
        }

        public static int Main(string directory_Design, string directory_Source, string directory, string mode)
        {
            bool full = mode != null && mode.IndexOf("full", StringComparison.OrdinalIgnoreCase) >= 0;
            bool const1 = mode == null || mode.IndexOf("noconst", StringComparison.OrdinalIgnoreCase) < 0;

            int startHour = 0;
            int endHour = full ? 8759 : 23;

            Directory.CreateDirectory(directory);

            string path_Design = Path.Combine(directory_Design, "prepared.sam");
            string path_TBD = Path.Combine(directory, "src.tbd");
            string path_TSD = Path.Combine(directory, "src.tsd");
            string path_TPD = Path.Combine(directory, "acc.tpd");

            //A COPY of the one thermal source, hashed on the way in and again at the end, so the run can
            //be shown to have read it and never written it.
            File.Copy(Path.Combine(directory_Source, "src.tbd"), path_TBD, true);
            File.Copy(Path.Combine(directory_Source, "src.tsd"), path_TSD, true);

            string hash_TBD_Before = Src.Hash(path_TBD);
            string hash_TSD_Before = Src.Hash(path_TSD);

            W("PHASE D - the PR2 explicit Systems / TPD route");
            W("  design    {0}  sha256 {1}", path_Design, Src.Hash(path_Design));
            W("  src.tbd   sha256 {0}", hash_TBD_Before);
            W("  src.tsd   sha256 {0}", hash_TSD_Before);
            W("  hours     {0}..{1}", startHour, endHour);
            W("");

            SAM.Analytical.AnalyticalModel analyticalModel = SAM.Core.Convert.ToSAM<SAM.Analytical.AnalyticalModel>(path_Design)[0];

            SAM.Analytical.AdjacencyCluster adjacencyCluster = analyticalModel.AdjacencyCluster;

            // ---------------------------------------------------------------- the design, measured once
            //
            // Measured BEFORE the scoping step and again after it, and the two are required to be
            // identical. That is the whole claim the scoping step makes, asserted rather than argued.

            Prep.Design design_Before = Prep.MeasureModel(new SAM.Analytical.AnalyticalModel(analyticalModel, adjacencyCluster));

            string path_Intent_Before = Path.Combine(directory, "intent-before-scope.tsv");
            string path_Intent_After = Path.Combine(directory, "intent.tsv");

            Prep.Oracle(design_Before, path_Intent_Before);

            List<string> notes_System = new List<string>();

            int count_OutOfScope = Pr1.ScopeSystems(adjacencyCluster, "noterminal", notes_System);

            W("== the system scope handed to PR1 ==");
            W("  {0} ventilation system(s) taken out of scope: they carry no design ventilation terminal.", count_OutOfScope);

            foreach (string note in notes_System)
            {
                W("    {0}", note);
            }

            Prep.Design design = Prep.MeasureModel(new SAM.Analytical.AnalyticalModel(analyticalModel, adjacencyCluster));

            Prep.Oracle(design, path_Intent_After);

            bool identical = string.Equals(File.ReadAllText(path_Intent_Before), File.ReadAllText(path_Intent_After), StringComparison.Ordinal);

            W("  the intent table is {0} before and after the scoping step.", identical ? "BYTE-IDENTICAL" : "*** DIFFERENT ***");

            if (!identical)
            {
                W("REFUSED: the scoping step changed the design, which it must never do.");
                Route.Report(directory, "acceptance.log");
                return 9;
            }

            Route.Shape shape = ShapeOf(design);

            W("");
            W("== the design under test, as the production route states it ==");
            W("  {0} room(s) across {1} air handling unit(s)", shape.Rooms, shape.AirHandlingUnits);
            W("  {0} supply room(s), {1} extract room(s), {2} both, {3} transfer-only", shape.SupplyOnly, shape.ExtractOnly, shape.Both, shape.TransferOnly);
            W("  {0} supply leg(s), {1} extract leg(s), {2} transfer leg(s)", shape.Supply, shape.Extract, shape.Transfer);

            foreach (Prep.Room room in design.Rooms)
            {
                W("    {0,-10} {1,-14} supply {2,7} l/s  extract {3,7} l/s",
                    room.System,
                    room.SpaceName,
                    room.Supply_Lps.ToString("0.###", CultureInfo.InvariantCulture),
                    room.Extract_Lps.ToString("0.###", CultureInfo.InvariantCulture));
            }

            foreach (Prep.Leg leg in design.Legs)
            {
                W("    transfer  {0,-14} -> {1,-14} {2,7} l/s", leg.FromName, leg.ToName, leg.Flow_Lps.ToString("0.###", CultureInfo.InvariantCulture));
            }

            // ---------------------------------------------------------------- the thermal source
            Dictionary<Guid, string> zoneReferences = new Dictionary<Guid, string>();

            foreach (string line in File.ReadAllLines(Path.Combine(directory_Source, "src-zones.txt")))
            {
                string[] parts = line.Split('\t');
                if (parts.Length == 2)
                {
                    zoneReferences[Guid.Parse(parts[0])] = parts[1];
                }
            }

            W("");
            W("room identities read back from Phase C: {0}", zoneReferences.Count);

            //Re-evidenced rather than re-simulated: the files this run stands on already exist, and the
            //workflow would delete the TSD it is being asked to prove.
            SAM.Core.Tas.SimulationEvidence simulationEvidence = new SAM.Core.Tas.SimulationEvidence(
                SAM.Core.Tas.SimulationOutputShape.SeparateOutputFile, path_TBD, path_TSD);

            simulationEvidence.RecordCallReturned();
            simulationEvidence.Conclude();

            SAM.Core.Tas.NoIzamThermalSource noIzamThermalSource = new SAM.Core.Tas.NoIzamThermalSource(
                path_TBD, path_TSD, true, true, simulationEvidence, zoneReferences, null, null);

            W("thermal source IsComplete: {0}", noIzamThermalSource.IsComplete);

            if (!noIzamThermalSource.IsComplete)
            {
                foreach (string refusal in noIzamThermalSource.Refusals)
                {
                    W("  refusal: {0}", refusal);
                }

                Route.Report(directory, "acceptance.log");
                return 6;
            }

            // ---------------------------------------------------------------- PR1
            W("");
            W("== PRODUCTION SAM_Systems Create.MechanicalVentilation ==");

            SAM.Core.Systems.SystemEnergyCentre systemEnergyCentre_Template =
                SAM.Analytical.Systems.Query.SystemEnergyCentre(Route.TemplatePath());

            SAM.Analytical.Systems.MechanicalVentilationSettings mechanicalVentilationSettings = null;

            if (const1)
            {
                //The frozen #111 Part O operating configuration: continuous operation, factor 1.0, stated
                //as an 8760-hour yearly schedule and supplied through PR1's own generic settings.
                SAM.Analytical.Systems.YearlySchedule yearlySchedule = new SAM.Analytical.Systems.YearlySchedule("Part O continuous operation");

                double[] values = new double[8760];
                for (int i = 0; i < values.Length; i++)
                {
                    values[i] = 1.0;
                }

                yearlySchedule.Values = values;

                mechanicalVentilationSettings = new SAM.Analytical.Systems.MechanicalVentilationSettings { Schedule = yearlySchedule };

                W("  PR1 settings: operating schedule \"{0}\", {1} hourly values, all 1.0", yearlySchedule.Name, yearlySchedule.Values.Length);
            }
            else
            {
                W("  PR1 settings: none - the template's own fan schedule assignment is left in place");
            }

            List<SAM.Analytical.Space> spaces_Scope = Pr1.Scope(adjacencyCluster, out List<string> notes_Scope);

            foreach (string note in notes_Scope)
            {
                W("  scope: {0}", note);
            }

            SAM.Analytical.Systems.MechanicalVentilationMaterialisation mechanicalVentilationMaterialisation =
                SAM.Analytical.Systems.Create.MechanicalVentilation(adjacencyCluster, systemEnergyCentre_Template, mechanicalVentilationSettings, spaces_Scope);

            foreach (string note in mechanicalVentilationMaterialisation.Notes)
            {
                W("  note: {0}", note);
            }

            foreach (string refusal in mechanicalVentilationMaterialisation.Refusals)
            {
                W("  refusal: {0}", refusal);
            }

            if (!mechanicalVentilationMaterialisation.IsMaterialised)
            {
                W("REFUSED: PR1 did not materialise.");
                Route.Report(directory, "acceptance.log");
                return 7;
            }

            // ---------------------------------------------------------------- PR2
            W("");
            W("== PRODUCTION Create.SystemVentilationRoute ==");

            SAM.Analytical.Tas.TPD.SystemVentilationRoute systemVentilationRoute =
                SAM.Analytical.Tas.TPD.Create.SystemVentilationRoute(
                    noIzamThermalSource, mechanicalVentilationMaterialisation, path_TPD, startHour, endHour);

            foreach (string note in systemVentilationRoute.Notes)
            {
                W("  note: {0}", note);
            }

            foreach (string refusal in systemVentilationRoute.Refusals)
            {
                W("  refusal: {0}", refusal);
            }

            W("  IsComplete: {0}", systemVentilationRoute.IsComplete);

            if (systemVentilationRoute.SimulationEvidence != null)
            {
                W("  NativeDiagnostic: [{0}]  kind={1}",
                    systemVentilationRoute.SimulationEvidence.NativeDiagnostic ?? "<null>",
                    systemVentilationRoute.SimulationEvidence.NativeDiagnosticKind);
            }

            if (!systemVentilationRoute.IsComplete)
            {
                W("REFUSED: the route did not complete.");
                Route.Report(directory, "acceptance.log");
                return 8;
            }

            //The route's OWN claim about identity and topology, written out before the document is read
            //back - so the directed topology comparison is intent vs native ports vs this claim, three
            //independent statements rather than two.
            Claim(systemVentilationRoute, mechanicalVentilationMaterialisation, design, directory);

            Route.Acceptance(systemVentilationRoute, mechanicalVentilationMaterialisation, adjacencyCluster, shape, path_TPD, startHour, endHour);

            W("");
            W("== provenance, after the run ==");
            W("  src.tbd  sha256 {0}  {1}", Src.Hash(path_TBD), string.Equals(hash_TBD_Before, Src.Hash(path_TBD), StringComparison.OrdinalIgnoreCase) ? "UNCHANGED" : "*** CHANGED ***");
            W("  src.tsd  sha256 {0}  {1}", Src.Hash(path_TSD), string.Equals(hash_TSD_Before, Src.Hash(path_TSD), StringComparison.OrdinalIgnoreCase) ? "UNCHANGED" : "*** CHANGED ***");
            W("  acc.tpd  sha256 {0}  {1} bytes", Src.Hash(path_TPD), File.Exists(path_TPD) ? new FileInfo(path_TPD).Length : 0);

            File.WriteAllLines(Path.Combine(directory, "provenance.txt"), new string[]
            {
                string.Concat("prepared.sam\t", Src.Hash(path_Design)),
                string.Concat("src.tbd.before\t", hash_TBD_Before),
                string.Concat("src.tbd.after\t", Src.Hash(path_TBD)),
                string.Concat("src.tsd.before\t", hash_TSD_Before),
                string.Concat("src.tsd.after\t", Src.Hash(path_TSD)),
                string.Concat("acc.tpd\t", Src.Hash(path_TPD)),
            });

            Route.Report(directory, "acceptance.log");

            return 0;
        }
    }
}
