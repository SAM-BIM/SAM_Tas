using System;
using System.Linq;
using System.Reflection;

namespace B0P
{
    public static partial class Steps
    {
        // refl <TypeName> [...]
        public static void Refl(string[] args)
        {
            Assembly asm = Assembly.Load(args[1] == "TBD" || args[1] == "TSD" ? "Interop." + args[1] : "Interop.TPD");
            int from = (args[1] == "TBD" || args[1] == "TSD") ? 2 : 1;
            for (int i = from; i < args.Length; i++)
            {
                Type t = asm.GetTypes().FirstOrDefault(x => x.Name == args[i] || x.Name == args[i] + "Class" || x.Name == "I" + args[i]);
                if (t == null) { W("no type {0}", args[i]); continue; }
                W("===== {0} =====", t.FullName);
                foreach (Type it in new[] { t }.Concat(t.GetInterfaces()))
                foreach (MemberInfo m in it.GetMembers().OrderBy(x => x.Name))
                    W("  [{0}] {1} {2}", it.Name, m.MemberType, m);
            }
        }
    }
}
