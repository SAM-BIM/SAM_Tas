// Read-only: what ventilation systems, units, terminals and air movements a model actually carries.
// Used to establish what PR1 will be given, before PR1 is given it.

using System;
using System.Collections.Generic;
using System.Globalization;
using System.IO;
using System.Text;

namespace Inv
{
    internal static class Vs
    {
        private static readonly StringBuilder log = new StringBuilder();

        private static void W(string format, params object[] args)
        {
            string line = args == null || args.Length == 0 ? format : string.Format(CultureInfo.InvariantCulture, format, args);
            Console.WriteLine(line);
            log.AppendLine(line);
        }

        public static int Main(string path_SAM, string path_Out)
        {
            SAM.Analytical.AnalyticalModel analyticalModel = SAM.Core.Convert.ToSAM<SAM.Analytical.AnalyticalModel>(path_SAM)[0];

            SAM.Analytical.AdjacencyCluster adjacencyCluster = analyticalModel.AdjacencyCluster;

            W("model {0}", path_SAM);
            W("");

            W("== AirHandlingUnit ==");
            foreach (SAM.Analytical.AirHandlingUnit airHandlingUnit in adjacencyCluster.GetObjects<SAM.Analytical.AirHandlingUnit>() ?? new List<SAM.Analytical.AirHandlingUnit>())
            {
                W("  {0}  name='{1}'", airHandlingUnit.Guid.ToString("D"), airHandlingUnit.Name);
            }

            W("");
            W("== VentilationSystem ==");

            foreach (SAM.Analytical.VentilationSystem ventilationSystem in adjacencyCluster.GetObjects<SAM.Analytical.VentilationSystem>() ?? new List<SAM.Analytical.VentilationSystem>())
            {
                W("  {0}  name='{1}'  full='{2}'  type='{3}'",
                    ventilationSystem.Guid.ToString("D"),
                    ventilationSystem.Name,
                    ventilationSystem.FullName,
                    ventilationSystem.Type == null ? "<none>" : ventilationSystem.Type.Name);

                W("      SupplyUnitName  = '{0}'", ventilationSystem.GetValue<string>(SAM.Analytical.VentilationSystemParameter.SupplyUnitName) ?? "<null>");
                W("      ExhaustUnitName = '{0}'", ventilationSystem.GetValue<string>(SAM.Analytical.VentilationSystemParameter.ExhaustUnitName) ?? "<null>");

                List<SAM.Analytical.Space> spaces = adjacencyCluster.GetRelatedObjects<SAM.Analytical.Space>(ventilationSystem) ?? new List<SAM.Analytical.Space>();
                List<string> names = new List<string>();
                foreach (SAM.Analytical.Space space in spaces)
                {
                    names.Add(space.Name);
                }
                names.Sort(StringComparer.Ordinal);

                W("      spaces    ({0}) {1}", names.Count, string.Join(", ", names));

                List<SAM.Analytical.VentilationTerminal> terminals = adjacencyCluster.GetRelatedObjects<SAM.Analytical.VentilationTerminal>(ventilationSystem) ?? new List<SAM.Analytical.VentilationTerminal>();

                W("      terminals ({0})", terminals.Count);

                foreach (SAM.Analytical.VentilationTerminal ventilationTerminal in terminals)
                {
                    List<string> roomNames = new List<string>();
                    foreach (SAM.Analytical.Space space in adjacencyCluster.GetRelatedObjects<SAM.Analytical.Space>(ventilationTerminal) ?? new List<SAM.Analytical.Space>())
                    {
                        roomNames.Add(space.Name);
                    }

                    W("        {0,-8} {1,8} l/s  '{2}'  -> {3}",
                        ventilationTerminal.FlowClassification,
                        (ventilationTerminal.DesignFlowRate_Lps ?? 0).ToString("0.###", CultureInfo.InvariantCulture),
                        ventilationTerminal.Name,
                        string.Join(", ", roomNames));
                }
            }

            W("");
            W("== SpaceAirMovement, every one, endpoints as written ==");

            Dictionary<Guid, SAM.Analytical.Space> dictionary_Space = new Dictionary<Guid, SAM.Analytical.Space>();
            foreach (SAM.Analytical.Space space in adjacencyCluster.GetSpaces() ?? new List<SAM.Analytical.Space>())
            {
                dictionary_Space[space.Guid] = space;
            }

            foreach (SAM.Analytical.SpaceAirMovement spaceAirMovement in adjacencyCluster.GetObjects<SAM.Analytical.SpaceAirMovement>() ?? new List<SAM.Analytical.SpaceAirMovement>())
            {
                W("  {0,10} l/s  '{1}'", (spaceAirMovement.AirFlow * 1000.0).ToString("0.###", CultureInfo.InvariantCulture), spaceAirMovement.Name);
                W("        From = {0}", spaceAirMovement.From ?? "<null>");
                W("        To   = {0}", spaceAirMovement.To ?? "<null>");
            }

            W("");
            W("== the non-dwelling zones and their spaces ==");

            foreach (SAM.Analytical.Zone zone in adjacencyCluster.GetObjects<SAM.Analytical.Zone>() ?? new List<SAM.Analytical.Zone>())
            {
                List<string> names = new List<string>();
                foreach (SAM.Analytical.Space space in adjacencyCluster.GetRelatedObjects<SAM.Analytical.Space>(zone) ?? new List<SAM.Analytical.Space>())
                {
                    names.Add(space.Name);
                }
                names.Sort(StringComparer.Ordinal);

                W("  zone '{0}'  IsDwelling='{1}'  category='{2}'  spaces: {3}",
                    zone.Name,
                    ValueText(zone, SAM.Analytical.ZoneParameter.IsDwelling),
                    zone.GetValue<string>(SAM.Analytical.ZoneParameter.ZoneCategory) ?? "<none>",
                    string.Join(", ", names));
            }

            File.WriteAllText(path_Out, log.ToString());

            return 0;
        }

        private static string ValueText(SAM.Analytical.Zone zone, SAM.Analytical.ZoneParameter zoneParameter)
        {
            object value = zone.GetValue<object>(zoneParameter);

            return value == null ? "<null>" : value.ToString();
        }
    }
}
