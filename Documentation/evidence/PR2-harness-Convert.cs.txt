// Checkpoint 1B closure, A and B: build a TOPOLOGICALLY VALID current-TAS file through the PRODUCTION
// conversion, then use it to answer the two remaining native questions.
//
//   A. UNIT   - author an absolute flow through the exact COM property PR2 will write, and read it back
//               through TAS's own arithmetic (ACH sizing against a zone load of known volume).
//   B. SUCCESS- run a genuinely successful simulation and record exactly what ITPD.Simulate returns,
//               so the success predicate is measured rather than assumed.
//
// MV.json is the shipped Part O template. It carries a real plant circuit (multi-boiler, multi-chiller,
// pumps, an air-source heat pump) as well as the air system, which is what the bare reference systems
// lacked when TAS answered "Plant room has no components".

using System;
using System.Collections.Generic;
using System.Globalization;
using System.IO;
using System.Reflection;
using System.Text;

namespace Inv
{
    internal static class ConvertRef
    {
        private static readonly StringBuilder log = new StringBuilder();

        private static void W(string format, params object[] args)
        {
            string line = args == null || args.Length == 0 ? format : string.Format(CultureInfo.InvariantCulture, format, args);
            Console.WriteLine(line);
            log.AppendLine(line);
        }

        private static void Flush(string path)
        {
            try
            {
                File.WriteAllText(path, log.ToString());
            }
            catch
            {
            }
        }

        private static void SetProperty(object target, string name, object value)
        {
            target.GetType().InvokeMember(
                name, BindingFlags.SetProperty, null, target, new object[] { value }, CultureInfo.InvariantCulture);
        }

        private static double Dbl(object value)
        {
            try
            {
                return value == null ? double.NaN : Convert.ToDouble(value, CultureInfo.InvariantCulture);
            }
            catch
            {
                return double.NaN;
            }
        }

        private static void CollectZones(object container, List<object> zones, int depth)
        {
            if (container == null || depth > 8)
            {
                return;
            }

            for (int index = 1; index <= 2000; index++)
            {
                object systemComponent = Dump.Call(container, "GetComponent", index);
                if (systemComponent == null)
                {
                    break;
                }

                if (systemComponent is global::TPD.SystemZone)
                {
                    zones.Add(systemComponent);
                }

                if (systemComponent is global::TPD.ComponentGroup)
                {
                    CollectZones(systemComponent, zones, depth + 1);
                }
            }
        }

        /// <summary>Runs the PRODUCTION Convert.ToTPD on the shipped MV template against a real no-IZAM TSD.</summary>
        public static int Build(string path_Template, string path_TPD, string path_TSD)
        {
            log.Clear();
            W("convert-template {0}", path_Template);
            W("  -> TPD {0}", path_TPD);
            W("  <- TSD {0}", path_TSD);

            if (!File.Exists(path_Template) || !File.Exists(path_TSD))
            {
                W("REFUSED: template or TSD missing");
                Flush(Path.ChangeExtension(path_TPD, ".convert.log"));
                return 5;
            }

            List<SAM.Core.Systems.SystemEnergyCentre> systemEnergyCentres =
                SAM.Core.Convert.ToSAM<SAM.Core.Systems.SystemEnergyCentre>(path_Template);

            W("template energy centres read: {0}", systemEnergyCentres == null ? 0 : systemEnergyCentres.Count);

            if (systemEnergyCentres == null || systemEnergyCentres.Count == 0)
            {
                W("REFUSED: the template did not deserialise to a SystemEnergyCentre");
                Flush(Path.ChangeExtension(path_TPD, ".convert.log"));
                return 6;
            }

            SAM.Core.Systems.SystemEnergyCentre systemEnergyCentre = systemEnergyCentres[0];
            W("energy centre name: {0}", systemEnergyCentre.Name);

            if (File.Exists(path_TPD))
            {
                File.Delete(path_TPD);
            }

            // Simulate=false here: this run only needs the CONVERSION, so the simulation can be driven
            // separately and its return string captured.
            SAM.Analytical.Tas.TPD.SystemEnergyCentreConversionSettings settings =
                new SAM.Analytical.Tas.TPD.SystemEnergyCentreConversionSettings
                {
                    Simulate = false,
                    StartHour = 0,
                    EndHour = 8759,
                    IncludeComponentResults = false,
                    IncludeControllerResults = false,
                };

            bool converted = SAM.Analytical.Tas.TPD.Convert.ToTPD(systemEnergyCentre, path_TPD, path_TSD, settings);

            W("PRODUCTION Convert.ToTPD returned: {0}", converted);
            W("TPD exists: {0} ({1} bytes)", File.Exists(path_TPD), File.Exists(path_TPD) ? new FileInfo(path_TPD).Length : 0);

            Flush(Path.ChangeExtension(path_TPD, ".convert.log"));
            return 0;
        }

        /// <summary>
        /// A: prove the unit. B: measure what a SUCCESSFUL Simulate returns.
        /// <para>
        /// Every zone's <c>FlowRate</c> is switched to TAS's own ACH rule at a stated ACH against a zone
        /// load whose volume TAS itself reports. Only one candidate unit can match volume x ACH.
        /// </para>
        /// </summary>
        public static int Prove(string path_TPD, int ach, int startHour, int endHour)
        {
            log.Clear();
            W("prove {0}  ach={1}  hours {2}..{3}", path_TPD, ach, startHour, endHour);

            if (!File.Exists(path_TPD))
            {
                W("REFUSED: TPD missing");
                Flush(Path.ChangeExtension(path_TPD, ".prove.log"));
                return 5;
            }

            global::TPD.TPDDoc tPDDoc = new global::TPD.TPDDoc();
            try
            {
                tPDDoc.Open(path_TPD);

                object energyCentre = tPDDoc.EnergyCentre;
                if (energyCentre == null)
                {
                    W("REFUSED: no energy centre");
                    return 3;
                }

                W("StartHour={0} EndHour={1}", Dump.CallText(energyCentre, "GetStartHour"), Dump.CallText(energyCentre, "GetEndHour"));
                W("TSDDataCount={0} PlantRoomCount={1}",
                    Dump.CallText(energyCentre, "GetTSDDataCount"),
                    Dump.CallText(energyCentre, "GetPlantRoomCount"));

                List<object> zones = new List<object>();
                int count_PlantRoom = (int)Dbl(Dump.Call(energyCentre, "GetPlantRoomCount"));

                for (int p = 1; p <= count_PlantRoom; p++)
                {
                    object plantRoom = Dump.Call(energyCentre, "GetPlantRoom", p);
                    W("PlantRoom[{0}] plantComponents={1} systems={2}",
                        p,
                        Dump.CallText(plantRoom, "GetComponentCount"),
                        Dump.CallText(plantRoom, "GetSystemCount"));

                    int count_System = (int)Dbl(Dump.Call(plantRoom, "GetSystemCount"));
                    for (int s = 1; s <= count_System; s++)
                    {
                        CollectZones(Dump.Call(plantRoom, "GetSystem", s), zones, 0);
                    }
                }

                W("zones found: {0}", zones.Count);

                if (zones.Count == 0)
                {
                    W("REFUSED: no zones to size");
                    return 7;
                }

                // The production conversion left these zones with NO zone load: the template's system
                // spaces are named "System Zone 1" and the TSD's loads are "Cell 1"/"Cell 2", and
                // Query.ZoneLoads(TSDData, systemSpaces) matches by NAME. That is the very defect PR2
                // fixes, and it is why ACH sizing had nothing to compute a volume from. Bind the loads
                // here by explicit pairing - AddZoneLoad - so the unit can be measured.
                W("");
                W("== binding zone loads explicitly (AddZoneLoad), since the name match found none ==");

                object tSDData_ForBinding = Dump.Call(energyCentre, "GetTSDData", 1);
                int count_ZoneLoad_ForBinding = (int)Dbl(Dump.Call(tSDData_ForBinding, "GetZoneLoadCount"));
                W("  TSD zone loads available: {0}", count_ZoneLoad_ForBinding);

                for (int i = 0; i < zones.Count; i++)
                {
                    object existing = Dump.Call(zones[i], "GetSystemZoneZoneLoad");
                    if (existing != null)
                    {
                        W("  zone[{0}] already bound to {1}", i + 1, Dump.Get(existing, "Name"));
                        continue;
                    }

                    if (i >= count_ZoneLoad_ForBinding)
                    {
                        W("  zone[{0}] has no load available to bind", i + 1);
                        continue;
                    }

                    object zoneLoad_ForBinding = Dump.Call(tSDData_ForBinding, "GetZoneLoad", i + 1);

                    try
                    {
                        zones[i].GetType().InvokeMember(
                            "AddZoneLoad", BindingFlags.InvokeMethod, null, zones[i],
                            new object[] { zoneLoad_ForBinding }, CultureInfo.InvariantCulture);

                        object bound = Dump.Call(zones[i], "GetSystemZoneZoneLoad");
                        W("  zone[{0}] bound -> load {1} guid {2} volume {3}",
                            i + 1,
                            Dump.Get(bound, "Name"),
                            Dump.Get(bound, "GUID"),
                            Dump.Get(bound, "Volume"));
                    }
                    catch (Exception exception)
                    {
                        Exception inner = exception.InnerException ?? exception;
                        W("  zone[{0}] AddZoneLoad THREW {1}: {2}", i + 1, inner.GetType().Name, inner.Message.Trim());
                    }
                }

                W("");
                W("== BEFORE: set every zone FlowRate to TAS's own {0} ACH rule ==", ach);

                List<double> volumes = new List<double>();
                List<object> flowRates = new List<object>();

                foreach (object zone in zones)
                {
                    object zoneLoad = Dump.Call(zone, "GetSystemZoneZoneLoad");
                    double volume = Dbl(Dump.Obj(zoneLoad, "Volume"));
                    object flowRate = Dump.Obj(zone, "FlowRate");

                    volumes.Add(volume);
                    flowRates.Add(flowRate);

                    W("  zone {0} load={1} volume={2}",
                        Dump.Get(zone, "GUID"), Dump.Get(zoneLoad, "Name"),
                        volume.ToString("R", CultureInfo.InvariantCulture));

                    if (flowRate == null || ach <= 0)
                    {
                        // ach <= 0 means "leave the authored absolute flows alone" - used to look for a
                        // genuinely SUCCESSFUL simulation, where no sizing is required at all.
                        if (ach < 0)
                        {
                            // ach < 0 means: force the ABSOLUTE value mode PR2 will actually write, so
                            // TAS has nothing left to size. If "Sizing Flow Failed" was about sizing, it
                            // should disappear - and a genuinely successful return becomes observable.
                            try
                            {
                                double authored = Dbl(Dump.Obj(flowRate, "Value"));
                                if (double.IsNaN(authored) || authored <= 0.0)
                                {
                                    authored = 17.0;
                                    SetProperty(flowRate, "Value", authored);
                                }

                                SetProperty(flowRate, "Type", (int)global::TPD.tpdSizedVariable.tpdSizedVariableValue);
                                W("      forced absolute: Value={0} Type={1}",
                                    Dump.Get(flowRate, "Value"), Dump.Get(flowRate, "Type"));
                            }
                            catch (Exception exception)
                            {
                                W("      forcing absolute THREW {0}", exception.GetType().Name);
                            }
                        }
                        else if (ach == 0)
                        {
                            W("      leaving authored flow as-is: Value={0} Type={1}",
                                Dump.Get(flowRate, "Value"), Dump.Get(flowRate, "Type"));
                        }

                        continue;
                    }

                    try
                    {
                        SetProperty(flowRate, "Method", (int)global::TPD.tpdSizeFlowMethod.tpdSizeFlowACH);
                        SetProperty(flowRate, "SizeValue1", (double)ach);
                        SetProperty(flowRate, "SizeValue2", (double)ach);
                        SetProperty(flowRate, "SizeFraction", 1.0);
                        SetProperty(flowRate, "Type", (int)global::TPD.tpdSizedVariable.tpdSizedVariableSize);

                        W("      Method={0} SizeValue1={1} Type={2} (Value before sizing = {3})",
                            Dump.Get(flowRate, "Method"),
                            Dump.Get(flowRate, "SizeValue1"),
                            Dump.Get(flowRate, "Type"),
                            Dump.Get(flowRate, "Value"));
                    }
                    catch (Exception exception)
                    {
                        Exception inner = exception.InnerException ?? exception;
                        W("      sizing setup THREW {0}: {1}", inner.GetType().Name, inner.Message.Trim());
                    }
                }

                tPDDoc.Save();

                W("");
                W("== SIMULATE - and record EXACTLY what it returns ==");

                string returned = null;
                bool threw = false;
                try
                {
                    returned = tPDDoc.Simulate(startHour + 1, endHour + 1, 0);
                }
                catch (Exception exception)
                {
                    threw = true;
                    Exception inner = exception.InnerException ?? exception;
                    W("Simulate THREW {0}: {1}", inner.GetType().Name, inner.Message.Replace(Environment.NewLine, " ").Trim());
                }

                if (!threw)
                {
                    W("Simulate RETURN: isNull={0} isEmpty={1} length={2} value=[{3}]",
                        returned == null,
                        returned != null && returned.Length == 0,
                        returned == null ? -1 : returned.Length,
                        returned ?? "<null>");
                }

                tPDDoc.Save();

                W("");
                W("== AFTER: what did TAS compute? ==");

                bool anySized = false;

                for (int i = 0; i < zones.Count; i++)
                {
                    object flowRate = flowRates[i];
                    double sized = Dbl(Dump.Obj(flowRate, "Value"));
                    double volume = volumes[i];

                    double expected_m3s = volume * ach / 3600.0;
                    double expected_lps = expected_m3s * 1000.0;
                    double expected_m3h = volume * ach;

                    string verdict = "NO MATCH";
                    if (Math.Abs(sized - expected_lps) <= Math.Max(0.05, expected_lps * 0.002)) { verdict = "MATCHES l/s"; }
                    else if (Math.Abs(sized - expected_m3s) <= Math.Max(0.0005, expected_m3s * 0.002)) { verdict = "MATCHES m3/s"; }
                    else if (Math.Abs(sized - expected_m3h) <= Math.Max(0.05, expected_m3h * 0.002)) { verdict = "MATCHES m3/h"; }

                    if (sized != 0.0 && !double.IsNaN(sized))
                    {
                        anySized = true;
                    }

                    W("  zone[{0}] volume={1} sized={2} Type={3}",
                        i + 1,
                        volume.ToString("R", CultureInfo.InvariantCulture),
                        sized.ToString("R", CultureInfo.InvariantCulture),
                        Dump.Get(flowRate, "Type"));
                    W("      candidates  m3/s={0}  l/s={1}  m3/h={2}   -> {3}",
                        expected_m3s.ToString("F5", CultureInfo.InvariantCulture),
                        expected_lps.ToString("F5", CultureInfo.InvariantCulture),
                        expected_m3h.ToString("F5", CultureInfo.InvariantCulture),
                        verdict);
                }

                W("");
                W(anySized
                    ? "TAS produced sized values - the unit verdict above is measured."
                    : "TAS produced NO sized value; the unit remains unmeasured by this route.");
            }
            finally
            {
                try { tPDDoc.Close(); } catch { }
            }

            Flush(Path.ChangeExtension(path_TPD, ".prove.log"));
            return 0;
        }
    }
}
