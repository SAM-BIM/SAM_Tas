// Finds the hour domain and the results variable that actually carry a duct's delivered flow.
// GetFlowRate(0) answers "Hour out of range" on a simulated document, so neither the base nor the
// variable index is assumed here - both are swept and reported.
//
// Nothing here is production code.

using System;
using System.Collections.Generic;
using System.Globalization;
using System.Reflection;
using SAM.Core.Tas;

namespace Inv
{
    internal static class FlowProbe
    {
        public static void Run(string path)
        {
            using (SAMTPDDocument sAMTPDDocument = new SAMTPDDocument(path))
            {
                object energyCentre = sAMTPDDocument.TPDDocument.EnergyCentre;
                List<object> systems = Graph.Systems(energyCentre);

                object system = systems[0];
                Console.WriteLine("simulate answered \"" + (Dump.Call(system, "Simulate", 1, 8760, 0) as string) + "\"");

                List<object> ducts = Graph.Ducts(system);
                object duct = ducts[0];

                Console.WriteLine("duct 1 = " + Graph.Edge(duct));

                Console.WriteLine("-- GetFlowRate(hour) sweep --");
                foreach (int hour in new int[] { -1, 0, 1, 2, 8759, 8760, 8761 })
                {
                    Console.WriteLine(string.Format(
                        CultureInfo.InvariantCulture,
                        "  GetFlowRate({0}) = {1}",
                        hour,
                        Dump.CallText(duct, "GetFlowRate", hour)));
                }

                object probe = duct;
                string label = "duct";

                if (Environment.GetEnvironmentVariable("PROBE") == "zone")
                {
                    foreach (object component in Graph.TopLevel(system))
                    {
                        if (component is global::TPD.SystemZone && (Dump.Get(component, "Name") ?? "").Contains("Ensuite"))
                        {
                            probe = component;
                            label = "zone " + Dump.Get(component, "Name");
                            break;
                        }
                    }
                }

                Console.WriteLine("-- " + label + " GetResultsData variable sweep (Hourly, Max, var, 1, 8760) --");
                for (int variable = 0; variable <= 40; variable++)
                {
                    string note;

                    try
                    {
                        object data = probe.GetType().InvokeMember(
                            "GetResultsData",
                            BindingFlags.InvokeMethod,
                            null,
                            probe,
                            new object[]
                            {
                                (int)global::TPD.tpdResultsPeriod.tpdResultsPeriodHourly,
                                (int)global::TPD.tpdCombinerType.tpdCombinerTypeMax,
                                variable,
                                1,
                                8760,
                            },
                            CultureInfo.InvariantCulture);

                        Array array = data as Array;

                        if (array == null)
                        {
                            note = data == null ? "<null>" : "<not an array: " + data.GetType().Name + ">";
                        }
                        else
                        {
                            double min = double.MaxValue;
                            double max = double.MinValue;
                            int n = 0;

                            foreach (object element in array)
                            {
                                double v = System.Convert.ToDouble(element, CultureInfo.InvariantCulture);
                                n++;
                                if (v < min) { min = v; }
                                if (v > max) { max = v; }
                            }

                            note = string.Format(CultureInfo.InvariantCulture, "n={0} min={1:R} max={2:R}", n, min, max);
                        }
                    }
                    catch (Exception exception)
                    {
                        note = "threw " + (exception.InnerException ?? exception).Message.Trim();
                    }

                    Console.WriteLine(string.Format(CultureInfo.InvariantCulture, "  var {0,2}: {1}", variable, note));
                }
            }
        }
    }
}
