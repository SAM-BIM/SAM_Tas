import pandas as pd, numpy as np
W='C:/TasOut/b0w/'
D=pd.read_csv(W+'out/hourly-dataset.csv')
A=pd.read_csv(W+'A.csv')
DU=pd.read_csv('C:/TasOut/b0r4/ducts.csv')
BOUND=['Bathroom_2','Studio 1_0','Bedroom 2_3','Ensuite_5','Kitchen_4','Bedroom 2_6','Ensuite_8','Kitchen_7']
odb=A['BLD:externalTemperature'].values
MV={'MVHR-01':'{10ECD9E6-FBBB-478D-BE3F-ECF00808F335}','MVHR-02':'{31313469-2008-46C2-9006-3535F83DEC3B}','MVHR-03':'{D301D74D-D9CD-4080-88C5-D374A8CD5A02}'}
# which MVHR feeds which flat
UNIT={'Studio 1_0':'MVHR-01','Bathroom_2':'MVHR-01','Bedroom 2_3':'MVHR-02','Kitchen_4':'MVHR-02','Ensuite_5':'MVHR-02',
      'Bedroom 2_6':'MVHR-03','Kitchen_7':'MVHR-03','Ensuite_8':'MVHR-03'}

def dcol(sub,var=3):
    m=[c for c in DU.columns if sub in c and c.endswith('|var%d'%var)]
    assert len(m)==1,(sub,m)
    return DU[m[0]].values

# B0 duct-side air entering / leaving each room
ENTER={'Studio 1_0':'MVHR-01|Damper:1.29903811034702->SystemZone:Studio 1_0',
 'Bathroom_2':'MVHR-01|Damper:918.960311873252->SystemZone:Bathroom_2',
 'Bedroom 2_3':'MVHR-02|Damper:3.15233098071178->SystemZone:Bedroom 2_3',
 'Kitchen_4':'MVHR-02|Damper:3.15233131035354->SystemZone:Kitchen_4',
 'Ensuite_5':'MVHR-02|Damper:7.35272125882989->SystemZone:Ensuite_5',
 'Bedroom 2_6':'MVHR-03|Damper:3.15444375498756->SystemZone:Bedroom 2_6',
 'Kitchen_7':'MVHR-03|Damper:3.15444370415334->SystemZone:Kitchen_7',
 'Ensuite_8':'MVHR-03|Damper:171.427909252452->SystemZone:Ensuite_8'}
LEAVE={'Studio 1_0':'MVHR-01|SystemZone:Studio 1_0->Junction:Part O Branch Junction',
 'Bathroom_2':'MVHR-01|SystemZone:Bathroom_2->Damper:918.959539445048',
 'Bedroom 2_3':'MVHR-02|SystemZone:Bedroom 2_3->Damper:3.15233131035354',
 'Kitchen_4':'MVHR-02|SystemZone:Kitchen_4->Junction:Part O Branch Junction',
 'Ensuite_5':'MVHR-02|SystemZone:Ensuite_5->Damper:7.35270348394994',
 'Bedroom 2_6':'MVHR-03|SystemZone:Bedroom 2_6->Damper:3.15444370415334',
 'Kitchen_7':'MVHR-03|SystemZone:Kitchen_7->Junction:Part O Branch Junction',
 'Ensuite_8':'MVHR-03|SystemZone:Ensuite_8->Damper:3.77016756905621'}
# the source room whose air enters each room (A-side reference), and the flow
SRC={'Studio 1_0':(None,30),'Bathroom_2':('Studio 1_0',8),'Bedroom 2_3':(None,63),'Kitchen_4':('Bedroom 2_3',63),
     'Ensuite_5':('Kitchen_4',8),'Bedroom 2_6':(None,63),'Kitchen_7':('Bedroom 2_6',63),'Ensuite_8':('Kitchen_7',8)}

print('=== ANALYSIS 6: B0 ACTUAL SUPPLY-AIR CONDITION (native duct results, plant-room SimulateEx) ===')
for u in ['MVHR-01','MVHR-02','MVHR-03']:
    c=[x for x in DU.columns if x.startswith(u+'|Junction:Junction Fresh Air->Fan:Fresh Air Fan|var3')][0]
    t=DU[c].values; d=t-odb
    print('%-8s fresh-air duct: max|T-ODB| = %.9f K   mean=%.6f  (n=%d)'%(u,np.abs(d).max(),t.mean(),len(t)))
print('')
print('=== ANALYSIS 7: air entering and leaving each B0 room vs Reference A equivalent ===')
print('%-12s %5s | %9s %9s | %9s %9s | %9s %9s'%('room','flow','B_in_mean','A_in_mean','B_out_mean','B_zone','A_zone','in_delta'))
for r in BOUND:
    tin=dcol(ENTER[r]); tout=dcol(LEAVE[r])
    src,q=SRC[r]
    ain = odb if src is None else D[D.room==src].A_air.values
    bz=D[D.room==r].B_air.values; az=D[D.room==r].A_air.values
    print('%-12s %5d | %9.3f %9.3f | %9.3f %9.3f | %9.3f %+9.3f'%(r,q,tin.mean(),ain.mean(),tout.mean(),bz.mean(),az.mean(),(tin-ain).mean()))
print('')
print('  (A supply source = MVHR dummy zone, not ODB; recomputed:)')
print('%-12s | %12s %12s %12s'%('room','B_in_mean','A_in_mean','delta'))
for r in BOUND:
    tin=dcol(ENTER[r]); src,q=SRC[r]
    ain = A[MV[UNIT[r]]+'|dryBulbTemp'].values if src is None else D[D.room==src].A_air.values
    print('%-12s | %12.3f %12.3f %+12.3f'%(r,tin.mean(),ain.mean(),(tin-ain).mean()))

print('')
print('=== ANALYSIS 5: sensible ventilation power delivered to each room (rho*cp=1200 J/m3K) ===')
print('%-12s %14s %14s %14s'%('room','A_Qvent_mean_W','B_Qvent_mean_W','delta_W'))
RC=1200.0
for r in BOUND:
    tin=dcol(ENTER[r]); src,q=SRC[r]
    az=D[D.room==r].A_air.values; bz=D[D.room==r].B_air.values
    ain = A[MV[UNIT[r]]+'|dryBulbTemp'].values if src is None else D[D.room==src].A_air.values
    qa=RC*(q/1000.0)*(ain-az); qb=RC*(q/1000.0)*(tin-bz)
    print('%-12s %14.2f %14.2f %14.2f'%(r,qa.mean(),qb.mean(),(qb-qa).mean()))
print('  cross-check against native A airMovementGain (which for supply rooms includes the supply IZAM):')
for r in BOUND:
    src,q=SRC[r]
    az=D[D.room==r].A_air.values
    ain = A[MV[UNIT[r]]+'|dryBulbTemp'].values if src is None else D[D.room==src].A_air.values
    qa=RC*(q/1000.0)*(ain-az)
    nat=D[D.room==r].A_airMoveGain.values
    print('   %-12s hand=%9.2f native=%9.2f  corr=%.5f  implied rho*cp=%.1f'%(r,qa.mean(),nat.mean(),np.corrcoef(qa,nat)[0,1],RC*nat.mean()/qa.mean()))
