// Does the TPD ZoneLoad.GUID equal the TBD zone GUID that SAM stamps onto an analytical Space as
// SpaceParameter.ZoneGuid?
//
// If it does, the whole PR2 room chain is by identity with no display name anywhere:
//
//   Space.Guid -> SpaceParameter.ZoneGuid -> TSDData.GetZoneLoadForGuid -> ZoneLoad -> AddZoneLoad
//
// If it does not, the conversion needs a different explicit key, and this measurement says so rather
// than the code assuming either way.

using System;
using System.Collections.Generic;
using System.Globalization;
using System.IO;
using System.Text;

namespace Inv
{
    internal static class Ident
    {
        private static readonly StringBuilder log = new StringBuilder();

        private static void W(string format, params object[] args)
        {
            string line = args == null || args.Length == 0 ? format : string.Format(CultureInfo.InvariantCulture, format, args);
            Console.WriteLine(line);
            log.AppendLine(line);
        }

        private static double Dbl(object value)
        {
            try
            {
                return value == null ? double.NaN : System.Convert.ToDouble(value, CultureInfo.InvariantCulture);
            }
            catch
            {
                return double.NaN;
            }
        }

        private static void Flush(string path)
        {
            try
            {
                File.WriteAllText(path, log.ToString());
            }
            catch
            {
            }
        }

        /// <summary>
        /// Walks a TBD's zones for (name, GUID), then attaches the TSD to a throwaway TPD energy centre and
        /// walks its zone loads for (name, GUID), and compares the two sets by GUID.
        /// </summary>
        public static int Run(string path_TBD, string path_TSD, string path_TPD_Scratch)
        {
            log.Clear();
            W("zoneload-identity");
            W("  TBD {0}", path_TBD);
            W("  TSD {0}", path_TSD);

            if (!File.Exists(path_TBD) || !File.Exists(path_TSD))
            {
                W("REFUSED: TBD or TSD missing");
                Flush(Path.ChangeExtension(path_TPD_Scratch, ".ident.log"));
                return 5;
            }

            // ---------------------------------------------------------------- the TBD side
            Dictionary<string, string> guid_To_Name_TBD = new Dictionary<string, string>(StringComparer.OrdinalIgnoreCase);

            global::TBD.TBDDocument tBDDocument = new global::TBD.TBDDocument();
            try
            {
                tBDDocument.open(path_TBD);

                global::TBD.Building building = tBDDocument.Building;

                int index = 0;
                while (true)
                {
                    global::TBD.zone zone = building.GetZone(index);
                    if (zone == null)
                    {
                        break;
                    }

                    W("  TBD zone[{0}] name=\"{1}\" GUID={2}", index, zone.name, zone.GUID);
                    guid_To_Name_TBD[zone.GUID ?? string.Empty] = zone.name;

                    index++;
                }

                W("  TBD zones: {0}", guid_To_Name_TBD.Count);
            }
            finally
            {
                try { tBDDocument.close(); } catch { }
            }

            // ---------------------------------------------------------------- the TSD side, through TPD
            Dictionary<string, string> guid_To_Name_Load = new Dictionary<string, string>(StringComparer.OrdinalIgnoreCase);

            if (File.Exists(path_TPD_Scratch))
            {
                File.Delete(path_TPD_Scratch);
            }

            global::TPD.TPDDoc tPDDoc = new global::TPD.TPDDoc();
            try
            {
                tPDDoc.Create(path_TPD_Scratch);

                object energyCentre = tPDDoc.EnergyCentre;

                foreach (int hWnd in new[] { 0, 1 })
                {
                    try
                    {
                        Dump.Call(energyCentre, "AddTSDData", path_TSD, hWnd);
                        W("  AddTSDData(hWnd={0}) returned without throwing", hWnd);
                    }
                    catch (Exception exception)
                    {
                        Exception inner = exception.InnerException ?? exception;
                        W("  AddTSDData(hWnd={0}) THREW {1}: {2}", hWnd, inner.GetType().Name, inner.Message.Replace(Environment.NewLine, " ").Trim());
                    }

                    if (Dbl(Dump.Call(energyCentre, "GetTSDDataCount")) > 0)
                    {
                        W("  attached with hWnd={0}", hWnd);
                        break;
                    }
                }

                object tSDData = Dump.Call(energyCentre, "GetTSDData", 1);
                if (tSDData == null)
                {
                    W("  REFUSED: no TSD data attached");
                    Flush(Path.ChangeExtension(path_TPD_Scratch, ".ident.log"));
                    return 7;
                }

                W("  TSDPath  = {0}", Dump.Get(tSDData, "TSDPath"));

                int count = (int)Dbl(Dump.Call(tSDData, "GetZoneLoadCount"));
                W("  zone loads: {0}", count);

                for (int i = 1; i <= count; i++)
                {
                    object zoneLoad = Dump.Call(tSDData, "GetZoneLoad", i);
                    string guid = Dump.Obj(zoneLoad, "GUID") as string;
                    string name = Dump.Obj(zoneLoad, "Name") as string;

                    W("  load[{0}] name=\"{1}\" GUID={2}", i, name, guid);
                    guid_To_Name_Load[guid ?? string.Empty] = name;

                    // The load-side bridge PR2 uses, driven by the load's own guid.
                    object resolved = null;
                    try
                    {
                        resolved = Dump.Call(tSDData, "GetZoneLoadForGuid", guid);
                    }
                    catch (Exception exception)
                    {
                        W("        GetZoneLoadForGuid THREW {0}: {1}", exception.GetType().Name, exception.Message.Trim());
                    }

                    W("        GetZoneLoadForGuid(own guid) -> {0}", resolved == null ? "<null>" : (Dump.Obj(resolved, "Name") as string));

                    // The same read, with the braces stripped, in case TAS keys on the bare form.
                    object resolved_Bare = Dump.Call(tSDData, "GetZoneLoadForGuid", (guid ?? string.Empty).Trim('{', '}'));
                    W("        GetZoneLoadForGuid(bare guid) -> {0}", resolved_Bare == null ? "<null>" : (Dump.Obj(resolved_Bare, "Name") as string));
                }
            }
            finally
            {
                try { tPDDoc.Close(); } catch { }
            }

            // ---------------------------------------------------------------- the comparison
            W("");
            W("== GUID SET COMPARISON ==");

            int matched = 0;
            foreach (KeyValuePair<string, string> pair in guid_To_Name_Load)
            {
                bool exists = guid_To_Name_TBD.TryGetValue(pair.Key, out string name_TBD);
                if (exists)
                {
                    matched++;
                }

                W("  load \"{0}\" {1}  -> TBD zone {2}",
                    pair.Value,
                    pair.Key,
                    exists ? string.Concat("\"", name_TBD, "\"  MATCH") : "NO MATCH");
            }

            W("");
            W("VERDICT: {0} of {1} ZoneLoad GUIDs are also TBD zone GUIDs (TBD zones: {2})",
                matched, guid_To_Name_Load.Count, guid_To_Name_TBD.Count);

            Flush(Path.ChangeExtension(path_TPD_Scratch, ".ident.log"));
            return 0;
        }
    }
}
