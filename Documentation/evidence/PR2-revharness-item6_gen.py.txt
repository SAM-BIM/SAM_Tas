"""Re-evaluates acceptance item 6 from the native connector graph rather than from counts.

Reads the canonical-network dump produced by `rev.exe net <tpd>` and, for every room, derives its
topological role and its air balance from the EDGES - which duty carrier feeds it, which carries its
extract, which carries each transfer in and out - then checks that every edge in the graph is either
a leg the design states or the unit's own backbone. Nothing is taken on trust from a total.
"""

import collections
import io
import re
import sys


def parse(path):
    systems = collections.OrderedDict()
    system = None
    node = None

    for line in io.open(path, encoding="utf-8"):
        line = line.rstrip("\n")

        match = re.match(r'== SYSTEM \d+ (\{[^}]*\}) name="([^"]*)"', line)
        if match:
            system = match.group(2)
            systems[system] = {"guid": match.group(1), "nodes": {}, "edges": []}
            continue

        if system is None:
            continue

        match = re.match(r'  (\w+) (\{[^}]*\}) "([^"]*)"\s*(.*)$', line)
        if match:
            node = match.group(2)
            systems[system]["nodes"][node] = {
                "kind": match.group(1),
                "name": match.group(3),
                "duty": match.group(4).strip(),
            }
            continue

        match = re.match(r'        -> (\w+) (\{[^}]*\}) "([^"]*)"', line)
        if match:
            systems[system]["edges"].append((node, match.group(2)))

    return systems


def value(text, key):
    match = re.search(key + r"=([-0-9.eE]+)", text)
    return float(match.group(1)) if match else None


def main(path):
    systems = parse(path)

    print("ACCEPTANCE ITEM 6, RE-EVALUATED FROM THE NATIVE CONNECTOR GRAPH")
    print("Source: %s - the canonical network, with Junction / GroupJunction /" % path)
    print("ComponentGroup collapsed as topology-only nodes. Every role and every duty below is")
    print("derived from the edges; no count is used as evidence.")
    print()

    grand = collections.Counter()
    unexplained_total = 0
    discontinuous_total = 0
    node_sets = collections.OrderedDict()

    for name, system in systems.items():
        nodes, edges = system["nodes"], system["edges"]
        zones = {g: n for g, n in nodes.items() if n["kind"] == "SystemZone"}
        dampers = {g: n for g, n in nodes.items() if n["kind"] == "Damper"}
        fans = {g: n for g, n in nodes.items() if n["kind"] == "Fan"}
        node_sets[name] = set(nodes)

        out_of = collections.defaultdict(list)
        into = collections.defaultdict(list)
        for a, b in edges:
            out_of[a].append(b)
            into[b].append(a)

        supply_dampers = [g for g, n in dampers.items() if "DesignFlowType=2" in n["duty"]]
        terminal_fans = [g for g in fans if not out_of[g]]

        print("== %s %s ==  zones=%d dampers=%d fans=%d edges=%d"
              % (name, system["guid"], len(zones), len(dampers), len(fans), len(edges)))
        print("   supply damper, identified by its DERIVED duty switch (DesignFlowType=2): %s"
              % ", ".join('"%s"' % dampers[g]["name"] for g in supply_dampers))
        print("   terminal fan, identified by having no downstream edge: %s"
              % ", ".join('"%s"' % fans[g]["name"] for g in terminal_fans))

        accounted = set()

        # The AHU backbone - fan into the unit's supply damper. Not a room leg, and accounted for as
        # itself rather than left to read as an unexplained cross-connection.
        backbone = 0
        for guid_Fan in fans:
            for downstream in out_of[guid_Fan]:
                if downstream in supply_dampers:
                    accounted.add((guid_Fan, downstream))
                    backbone += 1

        rows = []

        for guid_Zone, zone in zones.items():
            supply = 0.0
            extract, transfer_out, transfer_in = [], [], []

            for upstream in into[guid_Zone]:
                if upstream in supply_dampers:
                    supply = value(zone["duty"], "FlowRate") or 0.0
                    accounted.add((upstream, guid_Zone))
                elif upstream in dampers:
                    transfer_in.append((dampers[upstream]["name"],
                                        value(dampers[upstream]["duty"], "DesignFlowRate")))
                    accounted.add((upstream, guid_Zone))

            for downstream in out_of[guid_Zone]:
                if downstream not in dampers:
                    continue

                accounted.add((guid_Zone, downstream))
                duty = value(dampers[downstream]["duty"], "DesignFlowRate")
                targets = out_of[downstream]

                if any(t in fans for t in targets):
                    extract.append((dampers[downstream]["name"], duty))
                    for t in targets:
                        accounted.add((downstream, t))
                else:
                    for t in targets:
                        transfer_out.append((dampers[downstream]["name"], duty,
                                             zones[t]["name"] if t in zones else "?"))
                        accounted.add((downstream, t))

            role = ("both" if supply and extract else
                    "supply-only" if supply else
                    "extract-only" if extract else "transfer-only")

            grand[role] += 1
            grand["supply"] += 1 if supply else 0
            grand["extract"] += len(extract)
            grand["transfer"] += len(transfer_out)

            in_transfer = sum(v for _, v in transfer_in)
            out_transfer = sum(v for _, v, _ in transfer_out)
            out_extract = sum(v for _, v in extract)
            net = supply + in_transfer - out_extract - out_transfer

            if abs(net) > 1e-9:
                discontinuous_total += 1

            rows.append((zone["name"], role, supply, in_transfer, out_extract, out_transfer, net,
                         transfer_in, extract, transfer_out))

        print()
        print("   %-14s %-13s %7s %9s %8s %9s %5s"
              % ("room", "role", "supply", "transfIn", "extract", "transfOut", "net"))
        for row in sorted(rows):
            print("   %-14s %-13s %7.1f %9.1f %8.1f %9.1f %5.1f%s"
                  % (row[0], row[1], row[2], row[3], row[4], row[5], row[6],
                     "" if abs(row[6]) < 1e-9 else "   <-- DISCONTINUOUS"))

        print()
        for row in sorted(rows):
            for damper, duty in row[7]:
                print('     %-14s <- transfer %5.1f  via "%s"' % (row[0], duty, damper))
            for damper, duty in row[8]:
                print('     %-14s -> extract  %5.1f  via "%s", into the return fan'
                      % (row[0], duty, damper))
            for damper, duty, target in row[9]:
                print('     %-14s -> transfer %5.1f  via "%s", into "%s"'
                      % (row[0], duty, damper, target))

        unexplained = [(a, b) for a, b in edges if (a, b) not in accounted]
        unexplained_total += len(unexplained)

        print()
        print("   edges: %d total = %d design leg(s) + %d backbone (fan -> supply damper) + %d unexplained"
              % (len(edges), len(edges) - backbone - len(unexplained), backbone, len(unexplained)))
        for a, b in unexplained:
            print('     UNEXPLAINED EDGE  %s "%s" -> %s "%s"'
                  % (nodes[a]["kind"], nodes[a]["name"], nodes[b]["kind"], nodes[b]["name"]))

        print("   unit totals: supply %.1f l/s, extract %.1f l/s"
              % (sum(r[2] for r in rows), sum(r[4] for r in rows)))
        print()

    print("-- across every system in the document --")
    print("   rooms: %d supply-only, %d both, %d extract-only, %d transfer-only"
          % (grand["supply-only"], grand["both"], grand["extract-only"], grand["transfer-only"]))
    print("   legs : %d supply, %d extract, %d transfer"
          % (grand["supply"], grand["extract"], grand["transfer"]))

    names = list(node_sets)
    shared = set()
    for i in range(len(names)):
        for j in range(i + 1, len(names)):
            shared |= node_sets[names[i]] & node_sets[names[j]]

    print("   components shared between any two systems: %d" % len(shared))
    print("   unexplained edges anywhere: %d" % unexplained_total)
    print("   rooms failing per-zone air continuity: %d" % discontinuous_total)


if __name__ == "__main__":
    main(sys.argv[1] if len(sys.argv) > 1 else "final_net.txt")
