// The presentation of a produced TPD, measured: every top-level component's box and every duct's
// polyline, per air system, plus the two defects a reader sees - boxes that overlap, and duct segments
// that pass through a box that is neither of the duct's own endpoints.
//
// Read-only: opens, reads, closes; never saves. Every read is late-bound.
//
// Nothing here is production code.

using System;
using System.Collections.Generic;
using System.Globalization;
using System.IO;
using System.Text;

namespace Inv
{
    internal static class Layout
    {
        private sealed class Box
        {
            public object Component;
            public string Guid;
            public string Kind;
            public string Name;
            public int X;
            public int Y;
            public int W;
            public int H;
            public string Direction;
        }

        private static int I(object value)
        {
            try
            {
                return value == null ? 0 : Convert.ToInt32(value, CultureInfo.InvariantCulture);
            }
            catch
            {
                return 0;
            }
        }

        private static string Kind(object component)
        {
            string kind = Dump.CallText(component, "GetComponentType");
            return kind.Trim('"');
        }

        private static bool Overlap(Box a, Box b)
        {
            return a.X < b.X + b.W && b.X < a.X + a.W && a.Y < b.Y + b.H && b.Y < a.Y + a.H;
        }

        //Liang-Barsky: does the segment pass through the INTERIOR of the box (shrunk by one unit, so a
        //duct that only touches a box's edge at a port is not counted).
        private static bool SegmentThroughBox(double x1, double y1, double x2, double y2, Box box)
        {
            double xmin = box.X + 1, xmax = box.X + box.W - 1, ymin = box.Y + 1, ymax = box.Y + box.H - 1;
            if (xmax <= xmin || ymax <= ymin)
            {
                return false;
            }

            double t0 = 0, t1 = 1, dx = x2 - x1, dy = y2 - y1;
            double[] p = { -dx, dx, -dy, dy };
            double[] q = { x1 - xmin, xmax - x1, y1 - ymin, ymax - y1 };

            for (int i = 0; i < 4; i++)
            {
                if (p[i] == 0)
                {
                    if (q[i] < 0)
                    {
                        return false;
                    }
                }
                else
                {
                    double t = q[i] / p[i];
                    if (p[i] < 0) { if (t > t1) return false; if (t > t0) t0 = t; }
                    else { if (t < t0) return false; if (t < t1) t1 = t; }
                }
            }

            return t0 < t1;
        }

        public static int Run(string path_TPD, string pathOut)
        {
            StringBuilder stringBuilder = new StringBuilder();
            stringBuilder.AppendLine("LAYOUT " + path_TPD);

            int total_Overlaps = 0;
            int total_Crossings = 0;

            global::TPD.TPDDoc tPDDoc = new global::TPD.TPDDoc();

            try
            {
                tPDDoc.Open(path_TPD);

                object energyCentre = tPDDoc.EnergyCentre;
                int count_PlantRoom = I(Dump.Call(energyCentre, "GetPlantRoomCount"));

                for (int p = 1; p <= count_PlantRoom; p++)
                {
                    object plantRoom = Dump.Call(energyCentre, "GetPlantRoom", p);
                    int count_System = I(Dump.Call(plantRoom, "GetSystemCount"));

                    for (int s = 1; s <= count_System; s++)
                    {
                        object system = Dump.Call(plantRoom, "GetSystem", s);
                        stringBuilder.AppendLine();
                        stringBuilder.AppendLine("== SYSTEM " + Dump.Get(system, "Name"));

                        List<Box> boxes = new List<Box>();
                        Dictionary<string, Box> box_By_Guid = new Dictionary<string, Box>(StringComparer.OrdinalIgnoreCase);

                        for (int index = 1; index <= 20000; index++)
                        {
                            object component = Dump.Call(system, "GetComponent", index);
                            if (component == null)
                            {
                                break;
                            }

                            object position = Dump.Call(component, "GetPosition");
                            Box box = new Box
                            {
                                Component = component,
                                Guid = Dump.Obj(component, "GUID") as string ?? ("#" + index),
                                Kind = Kind(component),
                                Name = Dump.Get(component, "Name"),
                                X = I(Dump.Obj(position, "x")),
                                Y = I(Dump.Obj(position, "y")),
                                W = I(Dump.Call(component, "GetBaseWidth")),
                                H = I(Dump.Call(component, "GetBaseHeight")),
                                Direction = Dump.CallText(component, "GetDirection"),
                            };

                            boxes.Add(box);
                            box_By_Guid[box.Guid] = box;

                            stringBuilder.AppendLine(string.Format(CultureInfo.InvariantCulture,
                                "  {0,-11} {1,-34} at ({2,5},{3,5}) size {4}x{5} dir={6}",
                                box.Kind, box.Name, box.X, box.Y, box.W, box.H, box.Direction));
                        }

                        int overlaps = 0;
                        for (int i = 0; i < boxes.Count; i++)
                        {
                            for (int j = i + 1; j < boxes.Count; j++)
                            {
                                if (Overlap(boxes[i], boxes[j]))
                                {
                                    overlaps++;
                                    stringBuilder.AppendLine("  OVERLAP " + boxes[i].Kind + " " + boxes[i].Name + "  x  " + boxes[j].Kind + " " + boxes[j].Name);
                                }
                            }
                        }

                        int crossings = 0;
                        int count_Duct = I(Dump.Call(system, "GetDuctCount"));
                        for (int d = 1; d <= count_Duct; d++)
                        {
                            object duct = Dump.Call(system, "GetDuct", d);
                            if (duct == null)
                            {
                                continue;
                            }

                            object up = Dump.Call(duct, "GetUpstreamComponent");
                            object down = Dump.Call(duct, "GetDownstreamComponent");
                            string guid_Up = Dump.Obj(up, "GUID") as string;
                            string guid_Down = Dump.Obj(down, "GUID") as string;

                            int count_Node = I(Dump.Call(duct, "GetNodeCount"));
                            List<int[]> nodes = new List<int[]>();
                            for (int n = 1; n <= count_Node; n++)
                            {
                                nodes.Add(new[] { I(Dump.Call(duct, "GetNodePosX", n)), I(Dump.Call(duct, "GetNodePosY", n)) });
                            }

                            //The drawn polyline: centre of the upstream box, the bend nodes, centre of the
                            //downstream box. Port offsets are not exposed, so box centres stand in for them.
                            List<double[]> polyline = new List<double[]>();
                            if (guid_Up != null && box_By_Guid.TryGetValue(guid_Up, out Box box_Up)) polyline.Add(new double[] { box_Up.X + box_Up.W / 2.0, box_Up.Y + box_Up.H / 2.0 });
                            foreach (int[] node in nodes) polyline.Add(new double[] { node[0], node[1] });
                            if (guid_Down != null && box_By_Guid.TryGetValue(guid_Down, out Box box_Down)) polyline.Add(new double[] { box_Down.X + box_Down.W / 2.0, box_Down.Y + box_Down.H / 2.0 });

                            List<string> through = new List<string>();
                            for (int k = 0; k + 1 < polyline.Count; k++)
                            {
                                foreach (Box box in boxes)
                                {
                                    if (string.Equals(box.Guid, guid_Up, StringComparison.OrdinalIgnoreCase) || string.Equals(box.Guid, guid_Down, StringComparison.OrdinalIgnoreCase))
                                    {
                                        continue;
                                    }

                                    if (SegmentThroughBox(polyline[k][0], polyline[k][1], polyline[k + 1][0], polyline[k + 1][1], box) && !through.Contains(box.Name))
                                    {
                                        through.Add(box.Kind + " " + box.Name);
                                    }
                                }
                            }

                            crossings += through.Count;

                            stringBuilder.AppendLine(string.Format(CultureInfo.InvariantCulture,
                                "  duct {0,2} {1}:{2} -> {3}:{4} nodes={5} [{6}]{7}",
                                d, Dump.Get(up, "Name"), Dump.CallText(duct, "GetUpstreamComponentPort"),
                                Dump.Get(down, "Name"), Dump.CallText(duct, "GetDownstreamComponentPort"),
                                count_Node, string.Join(" ", nodes.ConvertAll(x => x[0] + "," + x[1])),
                                through.Count == 0 ? string.Empty : "  THROUGH " + string.Join("; ", through)));
                        }

                        stringBuilder.AppendLine(string.Format(CultureInfo.InvariantCulture,
                            "  SYSTEM SUMMARY: {0} component box(es), {1} overlapping pair(s), {2} duct pass(es) through a non-endpoint box",
                            boxes.Count, overlaps, crossings));

                        total_Overlaps += overlaps;
                        total_Crossings += crossings;
                    }
                }
            }
            finally
            {
                try { tPDDoc.Close(); } catch { }
            }

            stringBuilder.AppendLine();
            stringBuilder.AppendLine(string.Format(CultureInfo.InvariantCulture, "SUMMARY {0} overlapping pair(s), {1} duct pass(es) through a non-endpoint box", total_Overlaps, total_Crossings));

            string text = stringBuilder.ToString();
            Console.WriteLine(text);
            if (!string.IsNullOrWhiteSpace(pathOut))
            {
                File.WriteAllText(pathOut, text);
            }

            return 0;
        }
    }
}
