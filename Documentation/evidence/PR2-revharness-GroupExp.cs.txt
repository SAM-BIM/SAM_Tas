// The decisive native experiment for the PR2 ComponentGroup finding.
//
// Takes an already-working generated TPD, applies the native ComponentGroup idiom to each room
// terminal - System.AddGroup([SystemZone, terminal Damper], []) - and then measures three things
// against the ungrouped original:
//
//   1. the connector graph, as a set of "upstreamGuid:port -> downstreamGuid:port" edges read from
//      TAS's own IDuct accessors, so a rewiring shows up as an added or removed edge;
//   2. every design duty (zone FlowRate / FreshAir, damper DesignFlowRate) and every component guid;
//   3. ZoneTemperature over 8760 hours per zone guid, so a change in flow authority shows up
//      numerically.
//
// Nothing here is production code.

using System;
using System.Collections.Generic;
using System.Globalization;
using System.IO;
using System.Reflection;
using System.Text;
using SAM.Core.Tas;

namespace Inv
{
    internal static class GroupExp
    {
        private static string Edges(object system, List<string> edges)
        {
            foreach (object duct in Graph.Ducts(system))
            {
                edges.Add(string.Format(
                    CultureInfo.InvariantCulture,
                    "{0}:{1} -> {2}:{3}",
                    Graph.Guid_Of(Dump.Call(duct, "GetUpstreamComponent")),
                    Dump.CallText(duct, "GetUpstreamComponentPort"),
                    Graph.Guid_Of(Dump.Call(duct, "GetDownstreamComponent")),
                    Dump.CallText(duct, "GetDownstreamComponentPort")));
            }

            return null;
        }

        private static void Duties(object container, List<string> duties, int depth)
        {
            if (depth > 6)
            {
                return;
            }

            foreach (object component in Graph.TopLevel(container))
            {
                if (component is global::TPD.SystemZone)
                {
                    duties.Add(string.Format(
                        CultureInfo.InvariantCulture,
                        "ZONE {0} FlowRate={1}/{2} FreshAir={3}/{4} Flags={5}",
                        Graph.Guid_Of(component),
                        Dump.Get(Dump.Obj(component, "FlowRate"), "Value"),
                        Dump.Get(Dump.Obj(component, "FlowRate"), "Type"),
                        Dump.Get(Dump.Obj(component, "FreshAir"), "Value"),
                        Dump.Get(Dump.Obj(component, "FreshAir"), "Type"),
                        Dump.Get(component, "Flags")));
                }
                else if (component is global::TPD.Damper)
                {
                    duties.Add(string.Format(
                        CultureInfo.InvariantCulture,
                        "DAMPER {0} DesignFlowType={1} DesignFlowRate={2}/{3}",
                        Graph.Guid_Of(component),
                        Dump.Get(component, "DesignFlowType"),
                        Dump.Get(Dump.Obj(component, "DesignFlowRate"), "Value"),
                        Dump.Get(Dump.Obj(component, "DesignFlowRate"), "Type")));
                }

                if (component is global::TPD.ComponentGroup)
                {
                    Duties(component, duties, depth + 1);
                }
            }
        }

        /// <summary>
        /// Tries every plausible late-bound marshalling of AddGroup(components, controllers) and
        /// reports which one TAS accepted, rather than assuming one.
        /// </summary>
        private static object AddGroup(object system, List<object> components, out string how)
        {
            global::TPD.SystemComponent[] typed = new global::TPD.SystemComponent[components.Count];
            for (int i = 0; i < components.Count; i++)
            {
                typed[i] = components[i] as global::TPD.SystemComponent;
            }

            List<KeyValuePair<string, object[]>> attempts = new List<KeyValuePair<string, object[]>>
            {
                new KeyValuePair<string, object[]>("object[] + object[0]", new object[] { components.ToArray(), new object[0] }),
                new KeyValuePair<string, object[]>("object[] + null", new object[] { components.ToArray(), null }),
                new KeyValuePair<string, object[]>("object[] + Missing", new object[] { components.ToArray(), Missing.Value }),
                new KeyValuePair<string, object[]>("SystemComponent[] + object[0]", new object[] { typed, new object[0] }),
                new KeyValuePair<string, object[]>("SystemComponent[] + null", new object[] { typed, null }),
                new KeyValuePair<string, object[]>("SystemComponent[] + Controller[0]", new object[] { typed, new global::TPD.Controller[0] }),
                new KeyValuePair<string, object[]>("single component + null", new object[] { components[0], null }),
            };

            List<string> notes = new List<string>();

            foreach (KeyValuePair<string, object[]> attempt in attempts)
            {
                try
                {
                    object result = system.GetType().InvokeMember(
                        "AddGroup",
                        BindingFlags.InvokeMethod,
                        null,
                        system,
                        attempt.Value,
                        CultureInfo.InvariantCulture);

                    if (result != null)
                    {
                        how = attempt.Key + " ACCEPTED";
                        return result;
                    }

                    notes.Add(attempt.Key + " -> null");
                }
                catch (Exception exception)
                {
                    Exception inner = exception.InnerException ?? exception;
                    notes.Add(attempt.Key + " -> " + inner.GetType().Name + ": " + inner.Message.Trim());
                }
            }

            // The typed interop call is a genuinely different marshalling path from IDispatch.
            try
            {
                object result = ((global::TPD.System)system).AddGroup(typed, new global::TPD.Controller[0]);
                if (result != null)
                {
                    how = "typed ISystem.AddGroup ACCEPTED";
                    return result;
                }

                notes.Add("typed ISystem.AddGroup -> null");
            }
            catch (Exception exception)
            {
                Exception inner = exception.InnerException ?? exception;
                notes.Add("typed ISystem.AddGroup -> " + inner.GetType().Name + ": " + inner.Message.Trim());
            }

            how = string.Join(" | ", notes);
            return null;
        }

        public static void Run(string path, string pathCsv, string pathReport)
        {
            StringBuilder stringBuilder = new StringBuilder();
            stringBuilder.AppendLine("GROUPEXP " + path);

            using (SAMTPDDocument sAMTPDDocument = new SAMTPDDocument(path))
            {
                object energyCentre = sAMTPDDocument.TPDDocument.EnergyCentre;
                List<object> systems = Graph.Systems(energyCentre);

                List<string> edgesBefore = new List<string>();
                List<string> dutiesBefore = new List<string>();

                foreach (object system in systems)
                {
                    Edges(system, edgesBefore);
                    Duties(system, dutiesBefore, 0);
                }

                edgesBefore.Sort(StringComparer.Ordinal);
                dutiesBefore.Sort(StringComparer.Ordinal);

                stringBuilder.AppendLine("before: " + edgesBefore.Count + " edge(s), " + dutiesBefore.Count + " duty-carrier(s)");

                int grouped = 0;

                foreach (object system in systems)
                {
                    List<object> zones = new List<object>();

                    foreach (object component in Graph.TopLevel(system))
                    {
                        if (component is global::TPD.SystemZone)
                        {
                            zones.Add(component);
                        }
                    }

                    foreach (object zone in zones)
                    {
                        // The user's stated native shape: the room terminal is the zone plus the
                        // damper immediately downstream of it, grouped, and the group connected into
                        // the main duct.
                        object ductOut = Dump.Call(zone, "GetSystemZoneOutputDuct");
                        object downstream = ductOut == null ? null : Dump.Call(ductOut, "GetDownstreamComponent");

                        List<object> members = new List<object> { zone };

                        if (downstream is global::TPD.Damper)
                        {
                            members.Add(downstream);
                        }

                        object group = AddGroup(system, members, out string how);

                        stringBuilder.AppendLine(string.Format(
                            CultureInfo.InvariantCulture,
                            "  AddGroup zone {0} + {1} member(s) -> {2}   [{3}]",
                            Graph.Guid_Of(zone),
                            members.Count,
                            group == null ? "<null>" : Graph.Guid_Of(group),
                            how));

                        if (group != null)
                        {
                            grouped++;
                            stringBuilder.AppendLine(string.Format(
                                CultureInfo.InvariantCulture,
                                "        multiplicity={0} components={1} baseComponents={2} ducts={3} baseDucts={4} inPorts={5} outPorts={6}",
                                Dump.CallText(group, "GetMultiplicity"),
                                Dump.CallText(group, "GetComponentCount"),
                                Dump.CallText(group, "GetBaseComponentCount"),
                                Dump.CallText(group, "GetDuctCount"),
                                Dump.CallText(group, "GetBaseDuctCount"),
                                Dump.CallText(group, "GetInputPortCount"),
                                Dump.CallText(group, "GetOutputPortCount")));
                        }
                    }
                }

                stringBuilder.AppendLine("groups created: " + grouped);

                List<string> edgesAfter = new List<string>();
                List<string> dutiesAfter = new List<string>();

                foreach (object system in systems)
                {
                    Edges(system, edgesAfter);
                    Duties(system, dutiesAfter, 0);

                    foreach (object component in Graph.TopLevel(system))
                    {
                        if (component is global::TPD.ComponentGroup)
                        {
                            Edges(component, edgesAfter);
                        }
                    }
                }

                edgesAfter.Sort(StringComparer.Ordinal);
                dutiesAfter.Sort(StringComparer.Ordinal);

                stringBuilder.AppendLine("after : " + edgesAfter.Count + " edge(s), " + dutiesAfter.Count + " duty-carrier(s)");

                Diff(stringBuilder, "EDGE", edgesBefore, edgesAfter);
                Diff(stringBuilder, "DUTY", dutiesBefore, dutiesAfter);

                stringBuilder.AppendLine();
                stringBuilder.AppendLine("-- full graph after grouping --");
                Graph.Write(stringBuilder, energyCentre);

                using (StreamWriter streamWriter = new StreamWriter(pathCsv))
                {
                    streamWriter.WriteLine("zone,hour,zonetemperature");

                    foreach (object system in systems)
                    {
                        string returned = Dump.Call(system, "Simulate", 1, 8760, 0) as string;
                        stringBuilder.AppendLine("simulate " + Graph.Guid_Of(system) + " answered \"" + returned + "\"");

                        List<object> zones = new List<object>();
                        Zones(system, zones, 0);

                        foreach (object zone in zones)
                        {
                            string guid = Graph.Guid_Of(zone);
                            Array array = null;

                            try
                            {
                                array = zone.GetType().InvokeMember(
                                    "GetResultsData",
                                    BindingFlags.InvokeMethod,
                                    null,
                                    zone,
                                    new object[]
                                    {
                                        (int)global::TPD.tpdResultsPeriod.tpdResultsPeriodHourly,
                                        (int)global::TPD.tpdCombinerType.tpdCombinerTypeMax,
                                        9,
                                        1,
                                        8760,
                                    },
                                    CultureInfo.InvariantCulture) as Array;
                            }
                            catch (Exception exception)
                            {
                                stringBuilder.AppendLine("  zone " + guid + " results threw: " + (exception.InnerException ?? exception).Message.Trim());
                            }

                            if (array == null)
                            {
                                stringBuilder.AppendLine("  zone " + guid + " NO SERIES");
                                continue;
                            }

                            int hour = 0;
                            foreach (object element in array)
                            {
                                streamWriter.WriteLine(string.Format(CultureInfo.InvariantCulture, "{0},{1},{2:R}", guid, hour, System.Convert.ToDouble(element)));
                                hour++;
                            }

                            stringBuilder.AppendLine("  zone " + guid + " values=" + hour);
                        }
                    }
                }
            }

            string text = stringBuilder.ToString();
            Console.WriteLine(text);
            File.WriteAllText(pathReport, text);
            Console.WriteLine("written " + pathReport + " and " + pathCsv);
        }

        private static void Zones(object container, List<object> zones, int depth)
        {
            if (depth > 6)
            {
                return;
            }

            foreach (object component in Graph.TopLevel(container))
            {
                if (component is global::TPD.SystemZone)
                {
                    zones.Add(component);
                }
                else if (component is global::TPD.ComponentGroup)
                {
                    Zones(component, zones, depth + 1);
                }
            }
        }

        private static void Diff(StringBuilder stringBuilder, string label, List<string> before, List<string> after)
        {
            HashSet<string> setBefore = new HashSet<string>(before, StringComparer.Ordinal);
            HashSet<string> setAfter = new HashSet<string>(after, StringComparer.Ordinal);

            int removed = 0;
            int added = 0;

            foreach (string line in before)
            {
                if (!setAfter.Contains(line))
                {
                    stringBuilder.AppendLine("  " + label + " REMOVED  " + line);
                    removed++;
                }
            }

            foreach (string line in after)
            {
                if (!setBefore.Contains(line))
                {
                    stringBuilder.AppendLine("  " + label + " ADDED    " + line);
                    added++;
                }
            }

            stringBuilder.AppendLine(string.Format(
                CultureInfo.InvariantCulture,
                "  {0}: {1} removed, {2} added, {3} unchanged",
                label,
                removed,
                added,
                before.Count - removed));
        }
    }
}
