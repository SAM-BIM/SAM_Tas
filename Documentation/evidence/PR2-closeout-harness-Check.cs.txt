// The no-IZAM checks (items 1-3), re-run against ONE exact TBD file and nothing else.
//
// Walks TAS's own 0-based, null-terminated accessors - GetIZAM(i), GetIC(i), GetProfile(ticV|ticI) -
// so the result owes nothing to the code under test. Opens, reads, closes; never saves.
//
// Nothing here is production code.

using System;
using System.Collections.Generic;
using System.Globalization;
using System.IO;
using System.Text;

namespace Inv
{
    internal static class Check
    {
        public static int Run(string path_TBD, string pathOut)
        {
            StringBuilder stringBuilder = new StringBuilder();
            stringBuilder.AppendLine("SOURCE-CHECK " + path_TBD);

            int count_IZAM = 0;
            int count_InternalCondition = 0;
            int count_TicVNonZero = 0;
            int count_TicIPresent = 0;
            int count_TicINonZero = 0;
            int count_Zone = 0;
            SortedSet<string> ticI = new SortedSet<string>(StringComparer.Ordinal);

            global::TBD.TBDDocument tBDDocument = new global::TBD.TBDDocument();

            try
            {
                tBDDocument.open(path_TBD);

                global::TBD.Building building = tBDDocument.Building;

                while (building.GetIZAM(count_IZAM) != null && count_IZAM < 100000)
                {
                    count_IZAM++;
                }

                while (building.GetZone(count_Zone) != null && count_Zone < 100000)
                {
                    count_Zone++;
                }

                int index = 0;
                global::TBD.InternalCondition internalCondition = building.GetIC(index);

                while (internalCondition != null && index < 100000)
                {
                    count_InternalCondition++;

                    global::TBD.InternalGain internalGain = internalCondition.GetInternalGain();

                    global::TBD.profile profile_V = internalGain?.GetProfile((int)global::TBD.Profiles.ticV);
                    if (profile_V != null && (profile_V.value != 0 || profile_V.factor != 0 || profile_V.setbackValue != 0 || profile_V.schedule != null))
                    {
                        count_TicVNonZero++;
                    }

                    global::TBD.profile profile_I = internalGain?.GetProfile((int)global::TBD.Profiles.ticI);
                    if (profile_I != null)
                    {
                        count_TicIPresent++;

                        if (profile_I.value != 0 || profile_I.factor != 0)
                        {
                            count_TicINonZero++;
                        }

                        ticI.Add(string.Format(CultureInfo.InvariantCulture, "value={0:R} factor={1:R}", profile_I.value, profile_I.factor));
                    }

                    index++;
                    internalCondition = building.GetIC(index);
                }
            }
            finally
            {
                try { tBDDocument.close(); } catch { }
            }

            stringBuilder.AppendLine(string.Format(CultureInfo.InvariantCulture, "zones                          {0}", count_Zone));
            stringBuilder.AppendLine(string.Format(CultureInfo.InvariantCulture, "IZAM count                     {0}", count_IZAM));
            stringBuilder.AppendLine(string.Format(CultureInfo.InvariantCulture, "internal conditions            {0}", count_InternalCondition));
            stringBuilder.AppendLine(string.Format(CultureInfo.InvariantCulture, "  with a non-zero ticV         {0}", count_TicVNonZero));
            stringBuilder.AppendLine(string.Format(CultureInfo.InvariantCulture, "  with ticI present            {0}", count_TicIPresent));
            stringBuilder.AppendLine(string.Format(CultureInfo.InvariantCulture, "  with a non-zero ticI         {0}", count_TicINonZero));
            stringBuilder.AppendLine("distinct ticI (value, factor): " + string.Join("; ", ticI));

            bool pass = count_IZAM == 0 && count_TicVNonZero == 0 && count_InternalCondition > 0 && count_TicIPresent == count_InternalCondition;
            stringBuilder.AppendLine("VERDICT " + (pass ? "PASS" : "FAIL") + "  no IZAM, zero mechanical ticV, infiltration term present on every internal condition");

            string text = stringBuilder.ToString();
            Console.WriteLine(text);

            if (!string.IsNullOrWhiteSpace(pathOut))
            {
                File.WriteAllText(pathOut, text);
            }

            return pass ? 0 : 3;
        }
    }
}
