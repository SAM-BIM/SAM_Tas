// INDEPENDENT REVIEW harness for SAM Part O Iteration 3 PR2.
// Read-only observation of a produced TPD, plus a zone-flag A/B on copies.
// Nothing here is production code and nothing in the repository is touched.

using System;
using System.Collections.Generic;
using System.Globalization;
using System.IO;
using System.Reflection;
using SAM.Core.Tas;

namespace Inv
{
    internal static class Program
    {
        private static void Main(string[] args)
        {
            if (args.Length == 0)
            {
                Console.WriteLine("modes: inspect <tpd> | zt <tpd> <asis|ventflow|interzone|both> <out.csv>");
                return;
            }

            try
            {
                switch (args[0])
                {
                    case "inspect":
                        Inspect(args[1]);
                        break;

                    case "props":
                        Props(args[1], args.Length > 2 ? args[2] : "");
                        break;

                    case "zt":
                        Zt(args[1], args[2], args[3]);
                        break;

                    case "graph":
                        Graph.Walk(args[1], args.Length > 2 ? args[2] : "");
                        break;

                    case "groupexp":
                        GroupExp.Run(args[1], args[2], args[3]);
                        break;

                    case "net":
                        Net.Run(args[1], args.Length > 2 ? args[2] : "");
                        break;

                    case "oper":
                        Oper.Run(args[1], args.Length > 2 ? args[2] : "");
                        break;

                    case "flow":
                        Flow.Run(args[1], args.Length > 2 ? args[2] : "", args.Length > 3 ? int.Parse(args[3]) : 8760);
                        break;

                    case "flowprobe":
                        FlowProbe.Run(args[1]);
                        break;

                    case "schedprobe":
                        SchedProbe.Run(args[1]);
                        break;

                    case "scan":
                        for (int i = 1; i < args.Length; i++)
                        {
                            Graph.Scan(args[i]);
                        }
                        break;

                    default:
                        Console.WriteLine("unknown mode");
                        break;
                }
            }
            catch (Exception exception)
            {
                Console.WriteLine("HARNESS THREW " + exception.GetType().Name + ": " + exception.Message);
            }
        }

        private static List<object> Systems(object energyCentre)
        {
            List<object> result = new List<object>();

            int countPlantRoom = System.Convert.ToInt32(Dump.Call(energyCentre, "GetPlantRoomCount"));

            for (int i = 1; i <= countPlantRoom; i++)
            {
                object plantRoom = Dump.Call(energyCentre, "GetPlantRoom", i);
                if (plantRoom == null)
                {
                    continue;
                }

                int countSystem = System.Convert.ToInt32(Dump.Call(plantRoom, "GetSystemCount"));

                for (int j = 1; j <= countSystem; j++)
                {
                    object system = Dump.Call(plantRoom, "GetSystem", j);
                    if (system != null)
                    {
                        result.Add(system);
                    }
                }
            }

            return result;
        }

        private static List<object> Components(object container, int depth)
        {
            List<object> result = new List<object>();

            if (container == null || depth > 8)
            {
                return result;
            }

            for (int index = 1; index <= 20000; index++)
            {
                object component = Dump.Call(container, "GetComponent", index);
                if (component == null)
                {
                    break;
                }

                result.Add(component);

                if (component is global::TPD.ComponentGroup)
                {
                    result.AddRange(Components(component, depth + 1));
                }
            }

            return result;
        }

        private static string TypeName(object o)
        {
            foreach (Type type in o.GetType().GetInterfaces())
            {
                if (type.Namespace == "TPD" && type.Name.Length > 1 && type.Name[0] == 'I')
                {
                    return type.Name.Substring(1);
                }
            }

            return o.GetType().Name;
        }

        private static object PropObj(object comObject, string name)
        {
            if (comObject == null)
            {
                return null;
            }

            try
            {
                return comObject.GetType().InvokeMember(name, BindingFlags.GetProperty, null, comObject, null, CultureInfo.InvariantCulture);
            }
            catch
            {
                return null;
            }
        }

        private static List<string> DeclaredProperties(object o)
        {
            List<string> names = new List<string>();

            foreach (Type type in o.GetType().GetInterfaces())
            {
                if (type.Namespace != "TPD")
                {
                    continue;
                }

                foreach (PropertyInfo propertyInfo in type.GetProperties())
                {
                    if (propertyInfo.GetIndexParameters().Length == 0 && !names.Contains(propertyInfo.Name))
                    {
                        names.Add(propertyInfo.Name);
                    }
                }
            }

            names.Sort(StringComparer.Ordinal);

            return names;
        }

        private static void Inspect(string path)
        {
            Console.WriteLine("INSPECT " + path);

            using (SAMTPDDocument sAMTPDDocument = new SAMTPDDocument(path))
            {
                object energyCentre = sAMTPDDocument.TPDDocument.EnergyCentre;

                int s = 0;

                foreach (object system in Systems(energyCentre))
                {
                    s++;
                    Console.WriteLine();
                    Console.WriteLine("== SYSTEM " + s + "  GUID=" + Dump.Get(system, "GUID") + "  Name=" + Dump.Get(system, "Name"));

                    foreach (object component in Components(system, 0))
                    {
                        string typeName = TypeName(component);

                        if (component is global::TPD.SystemZone)
                        {
                            object load = null;
                            try { load = Dump.Call(component, "GetSystemZoneZoneLoad"); } catch { }

                            Console.WriteLine("  ZONE   guid=" + Dump.Get(component, "GUID")
                                + " name=" + Dump.Get(component, "Name")
                                + " Flags=" + Dump.Get(component, "Flags")
                                + " FlowRate=" + Dump.Get(PropObj(component, "FlowRate"), "Value") + "/" + Dump.Get(PropObj(component, "FlowRate"), "Type")
                                + " FreshAir=" + Dump.Get(PropObj(component, "FreshAir"), "Value") + "/" + Dump.Get(PropObj(component, "FreshAir"), "Type")
                                + " load=" + (load == null ? "<none>" : Dump.Get(load, "GUID")));
                        }
                        else if (component is global::TPD.Damper)
                        {
                            Console.WriteLine("  DAMPER guid=" + Dump.Get(component, "GUID")
                                + " name=" + Dump.Get(component, "Name")
                                + " DesignFlowType=" + Dump.Get(component, "DesignFlowType")
                                + " DesignFlowRate=" + Dump.Get(PropObj(component, "DesignFlowRate"), "Value") + "/" + Dump.Get(PropObj(component, "DesignFlowRate"), "Type")
                                + " MinimumFlowType=" + Dump.Get(component, "MinimumFlowType")
                                + " MinimumFlowRate=" + Dump.Get(PropObj(component, "MinimumFlowRate"), "Value"));
                        }
                        else
                        {
                            Console.WriteLine("  " + typeName + " guid=" + Dump.Get(component, "GUID") + " name=" + Dump.Get(component, "Name"));

                            bool interesting = component is global::TPD.Fan
                                || typeName.IndexOf("Exchanger", StringComparison.OrdinalIgnoreCase) >= 0
                                || typeName.IndexOf("Recovery", StringComparison.OrdinalIgnoreCase) >= 0
                                || typeName.IndexOf("Coil", StringComparison.OrdinalIgnoreCase) >= 0
                                || typeName.IndexOf("Heat", StringComparison.OrdinalIgnoreCase) >= 0
                                || typeName.IndexOf("Cool", StringComparison.OrdinalIgnoreCase) >= 0
                                || typeName.IndexOf("Humid", StringComparison.OrdinalIgnoreCase) >= 0;

                            if (interesting)
                            {
                                foreach (string name in DeclaredProperties(component))
                                {
                                    Console.WriteLine("        ." + name + " = " + Dump.Get(component, name));
                                }
                            }
                        }
                    }
                }
            }
        }



        /// <summary>Property names declared by an interop interface type, read from the type library.</summary>
        private static List<string> InterfaceProperties(Type type)
        {
            List<string> names = new List<string>();

            List<Type> types = new List<Type> { type };
            types.AddRange(type.GetInterfaces());

            foreach (Type current in types)
            {
                foreach (PropertyInfo propertyInfo in current.GetProperties())
                {
                    if (propertyInfo.GetIndexParameters().Length == 0 && !names.Contains(propertyInfo.Name))
                    {
                        names.Add(propertyInfo.Name);
                    }
                }
            }

            names.Sort(StringComparer.Ordinal);

            return names;
        }

        private static void Props(string path, string filter)
        {
            Console.WriteLine("PROPS " + path + " filter=" + filter);

            using (SAMTPDDocument sAMTPDDocument = new SAMTPDDocument(path))
            {
                object energyCentre = sAMTPDDocument.TPDDocument.EnergyCentre;

                foreach (object system in Systems(energyCentre))
                {
                    Console.WriteLine();
                    Console.WriteLine("== SYSTEM " + Dump.Get(system, "Name"));

                    if (filter.Length == 0)
                    {
                        foreach (string property in InterfaceProperties(typeof(global::TPD.System)))
                        {
                            Console.WriteLine("      SYS." + property + " = " + Dump.Get(system, property));
                        }
                    }

                    foreach (object component in Components(system, 0))
                    {
                        string name = Dump.Get(component, "Name") ?? "";

                        if (filter.Length != 0 && name.IndexOf(filter, StringComparison.OrdinalIgnoreCase) < 0)
                        {
                            continue;
                        }

                        Console.WriteLine("  COMPONENT " + name + "  guid=" + Dump.Get(component, "GUID"));

                        foreach (string property in InterfaceProperties(typeof(global::TPD.Fan)))
                        {
                            Console.WriteLine("        ." + property + " = " + Dump.Get(component, property));
                        }
                    }
                }
            }
        }

        private static void Zt(string path, string flagMode, string pathCsv)
        {
            Console.WriteLine("ZT " + path + " flagMode=" + flagMode);

            using (SAMTPDDocument sAMTPDDocument = new SAMTPDDocument(path))
            {
                object energyCentre = sAMTPDDocument.TPDDocument.EnergyCentre;

                List<object> systems = Systems(energyCentre);

                if (flagMode == "hgf0")
                {
                    int changedFan = 0;

                    foreach (object system in systems)
                    {
                        foreach (object component in Components(system, 0))
                        {
                            string name = Dump.Get(component, "Name") ?? "";
                            if (name.IndexOf("Fan", StringComparison.OrdinalIgnoreCase) < 0)
                            {
                                continue;
                            }

                            try
                            {
                                component.GetType().InvokeMember("HeatGainFactor", BindingFlags.SetProperty, null, component, new object[] { 0.0 }, CultureInfo.InvariantCulture);
                                changedFan++;
                                Console.WriteLine("fan " + name + " HeatGainFactor now " + Dump.Get(component, "HeatGainFactor"));
                            }
                            catch (Exception exception)
                            {
                                Console.WriteLine("fan " + name + " HeatGainFactor set threw: " + (exception.InnerException ?? exception).Message.Trim());
                            }
                        }
                    }

                    Console.WriteLine("HeatGainFactor zeroed on " + changedFan + " fan(s)");
                }
                else if (flagMode != "asis")
                {
                    int changed = 0;

                    foreach (object system in systems)
                    {
                        foreach (object component in Components(system, 0))
                        {
                            if (!(component is global::TPD.SystemZone))
                            {
                                continue;
                            }

                            int flags = System.Convert.ToInt32(PropObj(component, "Flags"));
                            int wanted = flags;

                            if (flagMode == "ventflow" || flagMode == "both")
                            {
                                wanted |= 4;
                            }

                            if (flagMode == "interzone" || flagMode == "both")
                            {
                                wanted |= 2;
                            }

                            if (wanted != flags)
                            {
                                component.GetType().InvokeMember("Flags", BindingFlags.SetProperty, null, component, new object[] { wanted }, CultureInfo.InvariantCulture);
                                changed++;
                            }
                        }
                    }

                    Console.WriteLine("flags changed on " + changed + " zone(s)");
                }

                using (StreamWriter streamWriter = new StreamWriter(pathCsv))
                {
                    streamWriter.WriteLine("zone,flags,hour,zonetemperature");

                    foreach (object system in systems)
                    {
                        string returned = Dump.Call(system, "Simulate", 1, 8760, 0) as string;
                        Console.WriteLine("system " + Dump.Get(system, "GUID") + " answered \"" + returned + "\"");

                        foreach (object component in Components(system, 0))
                        {
                            if (!(component is global::TPD.SystemZone))
                            {
                                continue;
                            }

                            string guid = Dump.Get(component, "GUID");
                            string flags = Dump.Get(component, "Flags");

                            object data = null;

                            try
                            {
                                data = component.GetType().InvokeMember(
                                    "GetResultsData",
                                    BindingFlags.InvokeMethod,
                                    null,
                                    component,
                                    new object[]
                                    {
                                        (int)global::TPD.tpdResultsPeriod.tpdResultsPeriodHourly,
                                        (int)global::TPD.tpdCombinerType.tpdCombinerTypeMax,
                                        9,
                                        1,
                                        8760,
                                    },
                                    CultureInfo.InvariantCulture);
                            }
                            catch (Exception exception)
                            {
                                Console.WriteLine("  zone " + guid + " results threw: " + (exception.InnerException ?? exception).Message.Trim());
                            }

                            Array array = data as Array;

                            if (array == null)
                            {
                                Console.WriteLine("  zone " + guid + " no series");
                                continue;
                            }

                            int hour = 0;

                            foreach (object element in array)
                            {
                                streamWriter.WriteLine(string.Format(CultureInfo.InvariantCulture, "{0},{1},{2},{3:R}", guid, flags, hour, System.Convert.ToDouble(element)));
                                hour++;
                            }

                            Console.WriteLine("  zone " + guid + " flags=" + flags + " values=" + hour);
                        }
                    }
                }
            }

            Console.WriteLine("written " + pathCsv);
        }
    }
}
