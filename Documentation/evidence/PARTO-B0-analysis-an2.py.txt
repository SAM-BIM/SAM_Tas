import pandas as pd, numpy as np
W='C:/TasOut/b0w/'
D=pd.read_csv(W+'out/hourly-dataset.csv')
BOUND=['Bathroom_2','Studio 1_0','Bedroom 2_3','Ensuite_5','Kitchen_4','Bedroom 2_6','Ensuite_8','Kitchen_7']

def st(d):
    d=d[np.isfinite(d)]
    return dict(n=len(d),bias=float(np.mean(d)),rmse=float(np.sqrt(np.mean(d**2))),mae=float(np.mean(np.abs(d))),maxabs=float(np.max(np.abs(d))))

def shifted(a,b,k):
    # compare A(t) against B(t+k): drop wrapped ends, no circular shift
    if k==0: return a,b
    if k>0: return a[:-k],b[k:]
    return a[-k:],b[:k]

rows=[]
print('=== ANALYSIS 2: TIME SHIFT ===')
for pair,ca,cb in (('A_air vs B_zone(achieved air)','A_air','B_air'),('A_res vs B_res','A_res','B_res')):
    print('')
    print('--- '+pair+' --- (pooled over 8 rooms; B(t+k) against A(t))')
    print('%4s %9s %9s %9s %9s %9s'%('k','bias','RMSE','MAE','max','corr'))
    for k in range(-3,4):
        ds=[];A_=[];B_=[]
        for r in BOUND:
            d=D[D.room==r]
            a,b=shifted(d[ca].values,d[cb].values,k)
            ds.append(b-a); A_.append(a); B_.append(b)
        dd=np.concatenate(ds); aa=np.concatenate(A_); bb=np.concatenate(B_)
        s=st(dd); c=float(np.corrcoef(aa,bb)[0,1])
        print('%+4d %+9.4f %9.4f %9.4f %9.4f %9.6f'%(k,s['bias'],s['rmse'],s['mae'],s['maxabs'],c))
        rows.append(dict(pair=pair,shift=k,scope='pooled',**s,corr=c))
    # per room best shift
    print('  per-room RMSE by shift:')
    print('  %-12s'%'room'+''.join('%9s'%('k=%+d'%k) for k in range(-3,4))+'   best')
    for r in BOUND:
        d=D[D.room==r]; line=[]
        for k in range(-3,4):
            a,b=shifted(d[ca].values,d[cb].values,k); s=st(b-a); line.append(s['rmse'])
            rows.append(dict(pair=pair,shift=k,scope=r,**s,corr=float(np.corrcoef(a,b)[0,1])))
        best=range(-3,4)[int(np.argmin(line))]
        print('  %-12s'%r+''.join('%9.4f'%v for v in line)+'   %+d'%best)
pd.DataFrame(rows).to_csv(W+'out/shift-analysis.csv',index=False,float_format='%.6f')

print('')
print('=== bias-removed (dynamic) residual ===')
for pair,c in (('A_air vs B_zone','A_air,B_air'),('A_res vs B_res','A_res,B_res'),('A_mrt vs B_mrt','A_mrt,B_mrt')):
    ca,cb=c.split(',')
    d=(D[cb]-D[ca]).values
    print('%-20s RMSE=%.4f  bias=%+.4f  RMSE after removing pooled bias=%.4f  after per-room bias=%.4f'%(
        pair,np.sqrt(np.mean(d**2)),np.mean(d),np.sqrt(np.mean((d-np.mean(d))**2)),
        np.sqrt(np.mean(np.concatenate([ (D[D.room==r][cb]-D[D.room==r][ca]).values - (D[D.room==r][cb]-D[D.room==r][ca]).values.mean() for r in BOUND])**2))))
