// The canonical flow network of a TPD air system.
//
// A Junction, a GroupJunction and a ComponentGroup carry topology only - no duty, no duct of their
// own that a design flow travels on. This collapses all three into transparent nodes and reports the
// adjacency of the components that DO carry a duty: fans, dampers, coils, exchangers and system
// zones. Two documents whose canonical networks are equal are the same ventilation system however
// they happen to be drawn or grouped, so this is the evidence that separates a connectivity defect
// from a layout difference.
//
// Group boundaries are traversed explicitly: a ComponentGroup's inlet GroupJunction is the member
// with no incoming internal duct, its outlet GroupJunction the member with no outgoing internal duct,
// and portal edges are added group -> inlet and outlet -> group so the closure runs straight through.
//
// Nothing here is production code.

using System;
using System.Collections.Generic;
using System.Globalization;
using System.IO;
using System.Text;
using SAM.Core.Tas;

namespace Inv
{
    internal static class Net
    {
        private sealed class Node
        {
            public string Guid;
            public string Kind;
            public string Name;
            public bool Transparent;
            public string Duty = "";
            public List<string> Out = new List<string>();
        }

        private static bool IsTransparent(string kind)
        {
            return kind == "Junction" || kind == "GroupJunction" || kind == "ComponentGroup";
        }

        private static string KindOf(object component)
        {
            string kind = Graph.Kind(component);
            int index = kind.IndexOf('=');
            if (index >= 0)
            {
                kind = kind.Substring(index + 1).Trim('"');
            }

            return kind;
        }

        private static void Collect(object container, Dictionary<string, Node> nodes, List<object> groups, int depth)
        {
            if (depth > 6)
            {
                return;
            }

            foreach (object component in Graph.TopLevel(container))
            {
                string guid = Graph.Guid_Of(component);
                string kind = KindOf(component);

                Node node = new Node
                {
                    Guid = guid,
                    Kind = kind,
                    Name = (Dump.Get(component, "Name") ?? "").Trim('"'),
                    Transparent = IsTransparent(kind),
                };

                if (component is global::TPD.SystemZone)
                {
                    node.Duty = string.Format(
                        CultureInfo.InvariantCulture,
                        "FlowRate={0}/{1} FreshAir={2}/{3} Flags={4}",
                        Dump.Get(Dump.Obj(component, "FlowRate"), "Value"),
                        Dump.Get(Dump.Obj(component, "FlowRate"), "Type"),
                        Dump.Get(Dump.Obj(component, "FreshAir"), "Value"),
                        Dump.Get(Dump.Obj(component, "FreshAir"), "Type"),
                        Dump.Get(component, "Flags"));
                }
                else if (component is global::TPD.Damper)
                {
                    node.Duty = string.Format(
                        CultureInfo.InvariantCulture,
                        "DesignFlowType={0} DesignFlowRate={1}/{2}",
                        Dump.Get(component, "DesignFlowType"),
                        Dump.Get(Dump.Obj(component, "DesignFlowRate"), "Value"),
                        Dump.Get(Dump.Obj(component, "DesignFlowRate"), "Type"));
                }
                else if (component is global::TPD.Fan)
                {
                    node.Duty = string.Format(
                        CultureInfo.InvariantCulture,
                        "DesignFlowType={0} DesignFlowRate={1}/{2} HeatGainFactor={3}",
                        Dump.Get(component, "DesignFlowType"),
                        Dump.Get(Dump.Obj(component, "DesignFlowRate"), "Value"),
                        Dump.Get(Dump.Obj(component, "DesignFlowRate"), "Type"),
                        Dump.Get(component, "HeatGainFactor"));
                }

                nodes[guid] = node;

                if (component is global::TPD.ComponentGroup)
                {
                    groups.Add(component);
                    Collect(component, nodes, groups, depth + 1);
                }
            }
        }

        private static void Edges(object container, Dictionary<string, Node> nodes)
        {
            foreach (object duct in Graph.Ducts(container))
            {
                string up = Graph.Guid_Of(Dump.Call(duct, "GetUpstreamComponent"));
                string down = Graph.Guid_Of(Dump.Call(duct, "GetDownstreamComponent"));

                if (nodes.TryGetValue(up, out Node node))
                {
                    node.Out.Add(down);
                }
            }
        }

        /// <summary>Every opaque component reachable downstream of a node through transparent nodes only.</summary>
        private static List<string> Reach(string guid, Dictionary<string, Node> nodes)
        {
            List<string> result = new List<string>();
            HashSet<string> seen = new HashSet<string>(StringComparer.Ordinal);
            Queue<string> queue = new Queue<string>();

            if (nodes.TryGetValue(guid, out Node start))
            {
                foreach (string next in start.Out)
                {
                    queue.Enqueue(next);
                }
            }

            while (queue.Count > 0)
            {
                string current = queue.Dequeue();
                if (!seen.Add(current))
                {
                    continue;
                }

                if (!nodes.TryGetValue(current, out Node node))
                {
                    result.Add(current + " <unknown>");
                    continue;
                }

                if (!node.Transparent)
                {
                    result.Add(current);
                    continue;
                }

                foreach (string next in node.Out)
                {
                    queue.Enqueue(next);
                }
            }

            result.Sort(StringComparer.Ordinal);
            return result;
        }

        public static void Run(string path, string pathOut)
        {
            StringBuilder stringBuilder = new StringBuilder();
            stringBuilder.AppendLine("CANONICAL NETWORK " + path);

            using (SAMTPDDocument sAMTPDDocument = new SAMTPDDocument(path))
            {
                object energyCentre = sAMTPDDocument.TPDDocument.EnergyCentre;

                int s = 0;

                foreach (object system in Graph.Systems(energyCentre))
                {
                    s++;
                    Dictionary<string, Node> nodes = new Dictionary<string, Node>(StringComparer.Ordinal);
                    List<object> groups = new List<object>();

                    Collect(system, nodes, groups, 0);
                    Edges(system, nodes);

                    foreach (object group in groups)
                    {
                        Edges(group, nodes);
                    }

                    // Portal edges across each group boundary.
                    foreach (object group in groups)
                    {
                        string guidGroup = Graph.Guid_Of(group);
                        HashSet<string> hasIncoming = new HashSet<string>(StringComparer.Ordinal);
                        HashSet<string> hasOutgoing = new HashSet<string>(StringComparer.Ordinal);

                        foreach (object duct in Graph.Ducts(group))
                        {
                            hasOutgoing.Add(Graph.Guid_Of(Dump.Call(duct, "GetUpstreamComponent")));
                            hasIncoming.Add(Graph.Guid_Of(Dump.Call(duct, "GetDownstreamComponent")));
                        }

                        foreach (object member in Graph.TopLevel(group))
                        {
                            if (KindOf(member) != "GroupJunction")
                            {
                                continue;
                            }

                            string guidMember = Graph.Guid_Of(member);

                            if (!hasIncoming.Contains(guidMember))
                            {
                                nodes[guidGroup].Out.Add(guidMember);
                            }

                            if (!hasOutgoing.Contains(guidMember))
                            {
                                nodes[guidMember].Out.Add(guidGroup);
                            }
                        }
                    }

                    List<string> opaque = new List<string>();
                    foreach (KeyValuePair<string, Node> keyValuePair in nodes)
                    {
                        if (!keyValuePair.Value.Transparent)
                        {
                            opaque.Add(keyValuePair.Key);
                        }
                    }

                    opaque.Sort(StringComparer.Ordinal);

                    stringBuilder.AppendLine();
                    stringBuilder.AppendLine(string.Format(
                        CultureInfo.InvariantCulture,
                        "== SYSTEM {0} {1} name=\"{2}\"  dutyNodes={3} transparentNodes={4} groups={5}",
                        s,
                        Graph.Guid_Of(system),
                        (Dump.Get(system, "Name") ?? "").Trim('"'),
                        opaque.Count,
                        nodes.Count - opaque.Count,
                        groups.Count));

                    foreach (string guid in opaque)
                    {
                        Node node = nodes[guid];
                        List<string> reach = Reach(guid, nodes);

                        for (int i = 0; i < reach.Count; i++)
                        {
                            string guidTo = reach[i];
                            Node to = nodes.TryGetValue(guidTo, out Node found) ? found : null;
                            reach[i] = string.Format(
                                CultureInfo.InvariantCulture,
                                "{0} {1} \"{2}\"",
                                to == null ? "?" : to.Kind,
                                guidTo,
                                to == null ? "" : to.Name);
                        }

                        stringBuilder.AppendLine(string.Format(
                            CultureInfo.InvariantCulture,
                            "  {0} {1} \"{2}\"  {3}",
                            node.Kind,
                            guid,
                            node.Name,
                            node.Duty));

                        if (reach.Count == 0)
                        {
                            stringBuilder.AppendLine("        -> <terminal>");
                        }

                        foreach (string line in reach)
                        {
                            stringBuilder.AppendLine("        -> " + line);
                        }
                    }
                }
            }

            string text = stringBuilder.ToString();
            Console.WriteLine(text);

            if (!string.IsNullOrWhiteSpace(pathOut))
            {
                File.WriteAllText(pathOut, text);
                Console.WriteLine("written " + pathOut);
            }
        }
    }
}
