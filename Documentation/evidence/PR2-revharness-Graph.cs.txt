// Native connector/group graph observation for the PR2 topology finding.
//
// Nothing here is production code. Every read is late-bound (checkpoint 1 measured that the typed
// ISystemComponent.GUID/.Name accessors throw on a SystemZone), and the graph is read from TAS's own
// duct endpoint accessors - IDuct.GetUpstreamComponent/GetUpstreamComponentPort and their downstream
// pair - so what is reported is the connector graph, never the drawn layout.

using System;
using System.Collections.Generic;
using System.Globalization;
using System.IO;
using System.Text;
using SAM.Core.Tas;

namespace Inv
{
    internal static class Graph
    {
        public static string Guid_Of(object component)
        {
            string result = Dump.Get(component, "GUID");
            return result == null ? "<none>" : result.Trim('"');
        }

        public static string Kind(object component)
        {
            if (component == null)
            {
                return "<null>";
            }

            foreach (Type type in component.GetType().GetInterfaces())
            {
                if (type.Namespace == "TPD" && type.Name.Length > 1 && type.Name[0] == 'I')
                {
                    return type.Name.Substring(1);
                }
            }

            // A late-bound COM object with no useful interop interface still answers GetComponentType().
            return string.Concat("ComponentType=", Dump.CallText(component, "GetComponentType"));
        }

        public static string Short(object component)
        {
            if (component == null)
            {
                return "<null>";
            }

            return string.Concat(Kind(component), " ", Guid_Of(component), " ", Dump.Get(component, "Name"));
        }

        public static List<object> Systems(object energyCentre)
        {
            List<object> result = new List<object>();
            int countPlantRoom = System.Convert.ToInt32(Dump.Call(energyCentre, "GetPlantRoomCount"));

            for (int i = 1; i <= countPlantRoom; i++)
            {
                object plantRoom = Dump.Call(energyCentre, "GetPlantRoom", i);
                if (plantRoom == null)
                {
                    continue;
                }

                int countSystem = System.Convert.ToInt32(Dump.Call(plantRoom, "GetSystemCount"));
                for (int j = 1; j <= countSystem; j++)
                {
                    object system = Dump.Call(plantRoom, "GetSystem", j);
                    if (system != null)
                    {
                        result.Add(system);
                    }
                }
            }

            return result;
        }

        /// <summary>Top-level components of a container, without recursing into groups.</summary>
        public static List<object> TopLevel(object container)
        {
            List<object> result = new List<object>();

            for (int index = 1; index <= 20000; index++)
            {
                object component = Dump.Call(container, "GetComponent", index);
                if (component == null)
                {
                    break;
                }

                result.Add(component);
            }

            return result;
        }

        public static List<object> Ducts(object container)
        {
            List<object> result = new List<object>();
            int count = Int(Dump.Call(container, "GetDuctCount"));

            for (int index = 1; index <= count; index++)
            {
                object duct = Dump.Call(container, "GetDuct", index);
                if (duct != null)
                {
                    result.Add(duct);
                }
            }

            return result;
        }

        public static void Walk(string path, string pathOut)
        {
            StringBuilder stringBuilder = new StringBuilder();
            stringBuilder.AppendLine("GRAPH " + path);

            using (SAMTPDDocument sAMTPDDocument = new SAMTPDDocument(path))
            {
                Write(stringBuilder, sAMTPDDocument.TPDDocument.EnergyCentre);
            }

            string text = stringBuilder.ToString();
            Console.WriteLine(text);

            if (!string.IsNullOrWhiteSpace(pathOut))
            {
                File.WriteAllText(pathOut, text);
                Console.WriteLine("written " + pathOut);
            }
        }

        public static void Write(StringBuilder stringBuilder, object energyCentre)
        {
            int s = 0;

            foreach (object system in Systems(energyCentre))
            {
                s++;
                stringBuilder.AppendLine();
                stringBuilder.AppendLine(string.Format(
                    CultureInfo.InvariantCulture,
                    "== SYSTEM {0} {1} name={2} Multiplicity={3} components={4} ducts={5} zones={6}",
                    s,
                    Guid_Of(system),
                    Dump.Get(system, "Name"),
                    Dump.Get(system, "Multiplicity"),
                    Dump.CallText(system, "GetComponentCount"),
                    Dump.CallText(system, "GetDuctCount"),
                    Dump.CallText(system, "GetSystemZoneCount")));

                List<object> components = TopLevel(system);
                Dictionary<string, object> group_By_Guid = new Dictionary<string, object>(StringComparer.Ordinal);
                Dictionary<string, List<string>> members_By_Group = new Dictionary<string, List<string>>(StringComparer.Ordinal);

                stringBuilder.AppendLine("  -- top-level components (system.GetComponent(i)) --");

                foreach (object component in components)
                {
                    // A ComponentGroup is itself a top-level connectable component: it must be
                    // collected as a group in its own right, not only through a member's GetGroup().
                    if (component is global::TPD.ComponentGroup)
                    {
                        string guidSelf = Guid_Of(component);
                        group_By_Guid[guidSelf] = component;
                        if (!members_By_Group.ContainsKey(guidSelf))
                        {
                            members_By_Group[guidSelf] = new List<string>();
                        }
                    }

                    object group = Dump.Call(component, "GetGroup");
                    string guidGroup = group == null ? null : Guid_Of(group);

                    if (group != null)
                    {
                        group_By_Guid[guidGroup] = group;
                        if (!members_By_Group.TryGetValue(guidGroup, out List<string> members))
                        {
                            members = new List<string>();
                            members_By_Group[guidGroup] = members;
                        }

                        members.Add(Short(component));
                    }

                    stringBuilder.AppendLine(string.Format(
                        CultureInfo.InvariantCulture,
                        "  {0,-22} {1} name={2} type={3} inPorts={4} outPorts={5} group={6}",
                        Kind(component),
                        Guid_Of(component),
                        Dump.Get(component, "Name"),
                        Dump.CallText(component, "GetComponentType"),
                        Dump.CallText(component, "GetInputPortCount"),
                        Dump.CallText(component, "GetOutputPortCount"),
                        guidGroup ?? "<none>"));

                    int inPorts = Int(Dump.Call(component, "GetInputPortCount"));
                    for (int p = 1; p <= inPorts; p++)
                    {
                        int n = Int(Dump.Call(component, "GetInputDuctCount", p));
                        if (n > 0)
                        {
                            stringBuilder.AppendLine(string.Format(CultureInfo.InvariantCulture, "        inPort {0}: {1} duct(s)", p, n));
                        }
                    }

                    int outPorts = Int(Dump.Call(component, "GetOutputPortCount"));
                    for (int p = 1; p <= outPorts; p++)
                    {
                        int n = Int(Dump.Call(component, "GetOutputDuctCount", p));
                        if (n > 0)
                        {
                            stringBuilder.AppendLine(string.Format(CultureInfo.InvariantCulture, "        outPort {0}: {1} duct(s)", p, n));
                        }
                    }

                    if (component is global::TPD.SystemZone)
                    {
                        stringBuilder.AppendLine("        zone inputDuct  : " + Edge(Dump.Call(component, "GetSystemZoneInputDuct")));
                        stringBuilder.AppendLine("        zone outputDuct : " + Edge(Dump.Call(component, "GetSystemZoneOutputDuct")));
                    }
                }

                stringBuilder.AppendLine("  -- ducts (system.GetDuct(i), endpoints from TAS's own accessors) --");

                int d = 0;
                foreach (object duct in Ducts(system))
                {
                    d++;
                    object group = Dump.Call(duct, "GetGroup");
                    stringBuilder.AppendLine(string.Format(
                        CultureInfo.InvariantCulture,
                        "  duct {0,-3} {1}   group={2}",
                        d,
                        Edge(duct),
                        group == null ? "<none>" : Guid_Of(group)));
                }

                if (group_By_Guid.Count == 0)
                {
                    stringBuilder.AppendLine("  -- component groups: NONE --");
                    continue;
                }

                stringBuilder.AppendLine("  -- component groups --");

                foreach (KeyValuePair<string, object> keyValuePair in group_By_Guid)
                {
                    object group = keyValuePair.Value;
                    stringBuilder.AppendLine(string.Format(
                        CultureInfo.InvariantCulture,
                        "  GROUP {0} name={1} multiplicity={2} components={3} baseComponents={4} ducts={5} baseDucts={6}",
                        keyValuePair.Key,
                        Dump.Get(group, "Name"),
                        Dump.CallText(group, "GetMultiplicity"),
                        Dump.CallText(group, "GetComponentCount"),
                        Dump.CallText(group, "GetBaseComponentCount"),
                        Dump.CallText(group, "GetDuctCount"),
                        Dump.CallText(group, "GetBaseDuctCount")));

                    foreach (object component in TopLevel(group))
                    {
                        stringBuilder.AppendLine(string.Format(
                            CultureInfo.InvariantCulture,
                            "        member  {0} inPorts={1} outPorts={2} groupOfMember={3}",
                            Short(component),
                            Dump.CallText(component, "GetInputPortCount"),
                            Dump.CallText(component, "GetOutputPortCount"),
                            Guid_Of(Dump.Call(component, "GetGroup"))));

                        if (component is global::TPD.SystemZone)
                        {
                            stringBuilder.AppendLine("                  Flags=" + Dump.Get(component, "Flags")
                                + " FlowRate=" + Dump.Get(Dump.Obj(component, "FlowRate"), "Value")
                                + "/" + Dump.Get(Dump.Obj(component, "FlowRate"), "Type")
                                + " FreshAir=" + Dump.Get(Dump.Obj(component, "FreshAir"), "Value")
                                + "/" + Dump.Get(Dump.Obj(component, "FreshAir"), "Type"));
                            stringBuilder.AppendLine("                  inputDuct  : " + Edge(Dump.Call(component, "GetSystemZoneInputDuct")));
                            stringBuilder.AppendLine("                  outputDuct : " + Edge(Dump.Call(component, "GetSystemZoneOutputDuct")));
                        }

                        if (component is global::TPD.Damper)
                        {
                            stringBuilder.AppendLine("                  DesignFlowType=" + Dump.Get(component, "DesignFlowType")
                                + " DesignFlowRate=" + Dump.Get(Dump.Obj(component, "DesignFlowRate"), "Value")
                                + "/" + Dump.Get(Dump.Obj(component, "DesignFlowRate"), "Type"));
                        }
                    }

                    foreach (object duct in Ducts(group))
                    {
                        stringBuilder.AppendLine("        ownduct " + Edge(duct));
                    }

                    object zoneLoadGroup = Dump.Call(group, "GetZoneLoadGroup");
                    stringBuilder.AppendLine("        zoneLoadGroup = " + (zoneLoadGroup == null ? "<null or no accessor>" : Short(zoneLoadGroup)));

                    stringBuilder.AppendLine("        (components answering GetGroup() = this group: "
                        + members_By_Group[keyValuePair.Key].Count + ")");
                }
            }
        }

        public static int Int(object value)
        {
            try { return value == null ? 0 : System.Convert.ToInt32(value, CultureInfo.InvariantCulture); }
            catch { return 0; }
        }

        public static string Edge(object duct)
        {
            if (duct == null)
            {
                return "<null>";
            }

            object upstream = Dump.Call(duct, "GetUpstreamComponent");
            object downstream = Dump.Call(duct, "GetDownstreamComponent");

            return string.Format(
                CultureInfo.InvariantCulture,
                "[{0}]:{1}  ->  [{2}]:{3}",
                Short(upstream),
                Dump.CallText(duct, "GetUpstreamComponentPort"),
                Short(downstream),
                Dump.CallText(duct, "GetDownstreamComponentPort"));
        }

        /// <summary>Counts groups in a TPD without printing the whole graph - used to find an authored reference.</summary>
        public static void Scan(string path)
        {
            int countSystem = 0;
            int countGroup = 0;
            int countZone = 0;
            int countJunction = 0;
            int countGroupJunction = 0;
            int countDamper = 0;

            try
            {
                using (SAMTPDDocument sAMTPDDocument = new SAMTPDDocument(path))
                {
                    object energyCentre = sAMTPDDocument.TPDDocument.EnergyCentre;

                    foreach (object system in Systems(energyCentre))
                    {
                        countSystem++;
                        HashSet<string> groups = new HashSet<string>(StringComparer.Ordinal);

                        foreach (object component in TopLevel(system))
                        {
                            string kind = Kind(component);

                            if (component is global::TPD.SystemZone) { countZone++; }
                            else if (component is global::TPD.Damper) { countDamper++; }
                            else if (component is global::TPD.ComponentGroup) { groups.Add(Guid_Of(component)); }
                            else if (kind.Contains("GroupJunction")) { countGroupJunction++; }
                            else if (kind.Contains("Junction")) { countJunction++; }

                            object group = Dump.Call(component, "GetGroup");
                            if (group != null)
                            {
                                groups.Add(Guid_Of(group));
                            }
                        }

                        countGroup += groups.Count;
                    }
                }
            }
            catch (Exception exception)
            {
                Console.WriteLine(string.Format(CultureInfo.InvariantCulture, "{0}\tTHREW {1}", path, exception.GetType().Name));
                return;
            }

            Console.WriteLine(string.Format(
                CultureInfo.InvariantCulture,
                "{0}\tsystems={1}\tgroups={2}\tzones={3}\tdampers={4}\tjunctions={5}\tgroupJunctions={6}",
                path,
                countSystem,
                countGroup,
                countZone,
                countDamper,
                countJunction,
                countGroupJunction));
        }
    }
}
