// Licensed TAS acceptance for the PR2 explicit ventilation route, items 4-15.
//
// One process: build a Part O mechanical ventilation design onto a real SAM model, produce the
// no-IZAM thermal source through the PRODUCTION workflow, materialise the PR1 graph through the
// PRODUCTION SAM_Systems entry point, run the PRODUCTION SystemVentilationRoute, and then read the
// resulting TPD back INDEPENDENTLY - late-bound, through TAS's own accessors - to check that what the
// route says is what TAS holds.
//
// Nothing here reimplements any of it. Every step is the production call a real run makes.

using System;
using System.Collections.Generic;
using System.Globalization;
using System.IO;
using System.Linq;
using System.Reflection;
using System.Text;

namespace Inv
{
    internal static class Route
    {
        private static readonly StringBuilder log = new StringBuilder();
        private static readonly List<string> verdicts = new List<string>();

        private static void W(string format, params object[] args)
        {
            string line = args == null || args.Length == 0 ? format : string.Format(CultureInfo.InvariantCulture, format, args);
            Console.WriteLine(line);
            log.AppendLine(line);
        }

        private static void Item(int number, string title, bool pass, string evidence)
        {
            string line = string.Format(CultureInfo.InvariantCulture, "ITEM {0,2}  {1,-4}  {2}\n            {3}", number, pass ? "PASS" : "FAIL", title, evidence);
            verdicts.Add(line);
            W("");
            W(line);
        }

        private static void Flush(string path)
        {
            try
            {
                File.WriteAllText(path, log.ToString());
            }
            catch
            {
            }
        }

        private static double Dbl(object value)
        {
            try
            {
                return value == null ? double.NaN : Convert.ToDouble(value, CultureInfo.InvariantCulture);
            }
            catch
            {
                return double.NaN;
            }
        }

        // =============================================================================== the design

        /// <summary>
        /// Writes a Part O mechanical ventilation design onto the model: two physical air handling
        /// units, and per unit a supply-only room, a room with unequal supply AND extract, a hall with
        /// no terminals reached only by transfer, and two extract-only rooms fed by a BRANCHING transfer
        /// out of that hall. Spaces are taken in ascending guid so the design is the same on every run.
        /// </summary>
        /// <summary>
        /// Writes a Part O mechanical ventilation design onto the model: <b>two</b> physical air
        /// handling units, and per unit a supply-only room, a room with unequal supply AND extract, a
        /// hall with no terminals reached only by transfer, and one or two extract-only rooms fed by a
        /// transfer out of that hall - branching where the model has the rooms for it.
        /// <para>
        /// Spaces are taken in ascending guid, so the design is the same on every run of the same
        /// model and nothing about it depends on the order the file happened to be written in.
        /// </para>
        /// </summary>
        private static void Design(SAM.Analytical.AdjacencyCluster adjacencyCluster, out Shape shape)
        {
            List<SAM.Analytical.Space> spaces = adjacencyCluster.GetSpaces();
            spaces.Sort((x, y) => x.Guid.CompareTo(y.Guid));

            W("design: {0} spaces available", spaces.Count);

            //Clear whatever ventilation the model already carried, so the acceptance measures the
            //design stated below and nothing inherited. f2.sam ships with a VentilationSystem named
            //"UV" that names no air handling unit, which PR1 rightly refuses.
            int count_Cleared = 0;

            foreach (SAM.Analytical.VentilationSystem ventilationSystem in adjacencyCluster.GetObjects<SAM.Analytical.VentilationSystem>() ?? new List<SAM.Analytical.VentilationSystem>())
            {
                count_Cleared += adjacencyCluster.RemoveObject(ventilationSystem.GetType(), ventilationSystem.Guid) ? 1 : 0;
            }

            foreach (SAM.Analytical.VentilationTerminal ventilationTerminal in adjacencyCluster.GetObjects<SAM.Analytical.VentilationTerminal>() ?? new List<SAM.Analytical.VentilationTerminal>())
            {
                count_Cleared += adjacencyCluster.RemoveObject(ventilationTerminal.GetType(), ventilationTerminal.Guid) ? 1 : 0;
            }

            foreach (SAM.Analytical.SpaceAirMovement spaceAirMovement in adjacencyCluster.GetObjects<SAM.Analytical.SpaceAirMovement>() ?? new List<SAM.Analytical.SpaceAirMovement>())
            {
                count_Cleared += adjacencyCluster.RemoveObject(spaceAirMovement.GetType(), spaceAirMovement.Guid) ? 1 : 0;
            }

            foreach (SAM.Analytical.AirHandlingUnit airHandlingUnit in adjacencyCluster.GetObjects<SAM.Analytical.AirHandlingUnit>() ?? new List<SAM.Analytical.AirHandlingUnit>())
            {
                count_Cleared += adjacencyCluster.RemoveObject(airHandlingUnit.GetType(), airHandlingUnit.Guid) ? 1 : 0;
            }

            W("design: {0} inherited ventilation object(s) cleared", count_Cleared);

            if (spaces.Count < 8)
            {
                throw new InvalidOperationException("The acceptance design needs at least eight spaces: two units of four.");
            }

            int count_Unit_1 = Math.Min(5, spaces.Count / 2);
            int count_Unit_2 = Math.Min(5, spaces.Count - count_Unit_1);

            shape = new Shape();

            Unit(adjacencyCluster, "AHU One", spaces, 0, count_Unit_1, shape);
            Unit(adjacencyCluster, "AHU Two", spaces, count_Unit_1, count_Unit_2, shape);

            W("design: {0} room(s), {1} supply-only, {2} extract-only, {3} both, {4} transfer-only; {5} supply / {6} extract / {7} transfer leg(s)",
                shape.Rooms, shape.SupplyOnly, shape.ExtractOnly, shape.Both, shape.TransferOnly,
                shape.Supply, shape.Extract, shape.Transfer);
        }

        /// <summary>What the design states, so the acceptance asserts against it rather than a literal.</summary>
        internal sealed class Shape
        {
            public int Rooms;
            public int SupplyOnly;
            public int ExtractOnly;
            public int Both;
            public int TransferOnly;
            public int Supply;
            public int Extract;
            public int Transfer;
            public int AirHandlingUnits;
        }

        private static void Unit(SAM.Analytical.AdjacencyCluster adjacencyCluster, string name, List<SAM.Analytical.Space> spaces, int offset, int count, Shape shape)
        {
            SAM.Analytical.AirHandlingUnit airHandlingUnit = SAM.Analytical.Create.AirHandlingUnit(name);
            airHandlingUnit.SummerSupplyTemperature = double.NaN;
            airHandlingUnit.WinterSupplyTemperature = double.NaN;

            adjacencyCluster.AddObject(airHandlingUnit);

            SAM.Analytical.VentilationSystemType ventilationSystemType = SAM.Analytical.Create.VentilationSystemType(Guid.NewGuid(), "MVHR", "Continuous mechanical supply and extract with heat recovery.");

            SAM.Analytical.VentilationSystem ventilationSystem = SAM.Analytical.Create.MechanicalSystem(ventilationSystemType, null, 1) as SAM.Analytical.VentilationSystem;

            ventilationSystem.SetValue(SAM.Analytical.VentilationSystemParameter.SupplyUnitName, name);
            ventilationSystem.SetValue(SAM.Analytical.VentilationSystemParameter.ExhaustUnitName, name);

            adjacencyCluster.AddObject(ventilationSystem);

            SAM.Analytical.Space space_SupplyOnly = spaces[offset + 0];
            SAM.Analytical.Space space_Both = spaces[offset + 1];
            SAM.Analytical.Space space_Hall = spaces[offset + 2];
            SAM.Analytical.Space space_Extract_1 = spaces[offset + 3];
            SAM.Analytical.Space space_Extract_2 = count >= 5 ? spaces[offset + 4] : null;

            List<SAM.Analytical.Space> served = new List<SAM.Analytical.Space> { space_SupplyOnly, space_Both, space_Extract_1 };

            if (space_Extract_2 != null)
            {
                served.Add(space_Extract_2);
            }

            foreach (SAM.Analytical.Space space in served)
            {
                adjacencyCluster.AddRelation(ventilationSystem, space);
            }

            //Deliberately asymmetric and not round, so a value that was substituted, rebalanced or
            //conflated with another shows up as a mismatch rather than as a coincidence.
            Terminal(adjacencyCluster, ventilationSystem, space_SupplyOnly, SAM.Analytical.FlowClassification.Supply, 13.0);
            Terminal(adjacencyCluster, ventilationSystem, space_Both, SAM.Analytical.FlowClassification.Supply, 31.0);
            Terminal(adjacencyCluster, ventilationSystem, space_Both, SAM.Analytical.FlowClassification.Extract, 19.0);
            Terminal(adjacencyCluster, ventilationSystem, space_Extract_1, SAM.Analytical.FlowClassification.Extract, 15.0);

            Transfer(adjacencyCluster, space_SupplyOnly, space_Hall, 0.0065);
            Transfer(adjacencyCluster, space_Both, space_Hall, 0.006);
            Transfer(adjacencyCluster, space_Hall, space_Extract_1, 0.0075);

            shape.AirHandlingUnits++;
            shape.Rooms += 4;
            shape.SupplyOnly++;
            shape.Both++;
            shape.TransferOnly++;
            shape.ExtractOnly++;
            shape.Supply += 2;
            shape.Extract += 2;
            shape.Transfer += 3;

            if (space_Extract_2 != null)
            {
                Terminal(adjacencyCluster, ventilationSystem, space_Extract_2, SAM.Analytical.FlowClassification.Extract, 10.0);
                Transfer(adjacencyCluster, space_Hall, space_Extract_2, 0.005);

                shape.Rooms++;
                shape.ExtractOnly++;
                shape.Extract++;
                shape.Transfer++;
            }

            W("  {0}: supply-only {1}, both {2}, hall {3}, extract {4}{5}",
                name,
                space_SupplyOnly.Name,
                space_Both.Name,
                space_Hall.Name,
                space_Extract_1.Name,
                space_Extract_2 == null ? string.Empty : string.Concat(" / ", space_Extract_2.Name));
        }

        private static void Terminal(SAM.Analytical.AdjacencyCluster adjacencyCluster, SAM.Analytical.VentilationSystem ventilationSystem, SAM.Analytical.Space space, SAM.Analytical.FlowClassification flowClassification, double designFlowRate_Lps)
        {
            SAM.Analytical.VentilationTerminal ventilationTerminal = new SAM.Analytical.VentilationTerminal(
                string.Format("{0} {1}", space.Name, flowClassification),
                flowClassification,
                designFlowRate_Lps);

            adjacencyCluster.AddObject(ventilationTerminal);
            adjacencyCluster.AddRelation(ventilationSystem, ventilationTerminal);
            adjacencyCluster.AddRelation(ventilationTerminal, space);
        }

        private static void Transfer(SAM.Analytical.AdjacencyCluster adjacencyCluster, SAM.Analytical.Space space_From, SAM.Analytical.Space space_To, double airFlow_M3S)
        {
            SAM.Analytical.SpaceAirMovement spaceAirMovement = new SAM.Analytical.SpaceAirMovement(
                string.Format("{0} -> {1}", space_From.Name, space_To.Name),
                airFlow_M3S,
                new SAM.Core.ObjectReference(space_From).ToString(),
                new SAM.Core.ObjectReference(space_To).ToString());

            adjacencyCluster.AddObject(spaceAirMovement);
            adjacencyCluster.AddRelation(spaceAirMovement, space_To);
        }

        // ================================================================================== the run

        /// <summary>
        /// Process one: write the design onto the model, save it, and produce the no-IZAM thermal
        /// source. Nothing here touches TPD - one COM document cycle per process is the rule this
        /// harness lives by, and mixing a TBD/TSD cycle with a TPD one is what leaves orphaned servers
        /// holding the output file.
        /// </summary>
        public static int Source(string path_SAM, string directory, string path_TWD, string mode)
        {
            log.Clear();
            verdicts.Clear();

            bool full = mode != null && mode.StartsWith("full", StringComparison.OrdinalIgnoreCase);

            string path_Design = Path.Combine(directory, "acc-design.sam");
            string path_gbXML = Path.Combine(directory, "acc.xml");
            string path_TBD = Path.Combine(directory, "acc.tbd");

            W("route-source {0} mode={1}", path_SAM, mode);

            Directory.CreateDirectory(directory);

            SAM.Analytical.AnalyticalModel analyticalModel = SAM.Core.Convert.ToSAM<SAM.Analytical.AnalyticalModel>(path_SAM)[0];

            SAM.Analytical.AdjacencyCluster adjacencyCluster = analyticalModel.AdjacencyCluster;

            Design(adjacencyCluster, out Shape shape);

            analyticalModel = new SAM.Analytical.AnalyticalModel(analyticalModel, adjacencyCluster);

            SAM.Core.Convert.ToFile(new SAM.Core.IJSAMObject[] { analyticalModel }, path_Design);
            W("design model saved: {0}", path_Design);

            gbXMLSerializer.gbXML gbXML = SAM.Analytical.gbXML.Convert.TogbXML(analyticalModel, SAM.Core.Tolerance.MacroDistance, SAM.Core.Tolerance.Distance);
            SAM.Core.gbXML.Create.gbXML(gbXML, path_gbXML);

            W("gbXML written: {0} bytes", new FileInfo(path_gbXML).Length);

            List<SAM.Weather.WeatherData> weatherDatas = SAM.Weather.Tas.Convert.ToSAM_WeatherDatas(path_TWD);

            SAM.Analytical.Tas.WorkflowSettings workflowSettings = new SAM.Analytical.Tas.WorkflowSettings
            {
                Path_gbXML = path_gbXML,
                Path_TBD = path_TBD,
                WeatherData = weatherDatas[0],
                Sizing = true,
                Simulate = true,
                SimulateFrom = 1,
                SimulateTo = full ? 365 : 1,
                UpdateZones = true,
                UnmetHours = false,
                RemoveExistingTBD = true,
            };

            W("");
            W("== PRODUCTION Create.NoIzamThermalSource ==");

            SAM.Core.Tas.NoIzamThermalSource noIzamThermalSource = SAM.Analytical.Tas.Create.NoIzamThermalSource(
                analyticalModel, workflowSettings, out SAM.Analytical.AnalyticalModel analyticalModel_Result);

            foreach (string note in noIzamThermalSource.Notes)
            {
                W("  note: {0}", note);
            }

            foreach (string refusal in noIzamThermalSource.Refusals)
            {
                W("  refusal: {0}", refusal);
            }

            if (noIzamThermalSource.SimulationEvidence != null)
            {
                foreach (string note in noIzamThermalSource.SimulationEvidence.Notes)
                {
                    W("  evidence: {0}", note);
                }

                foreach (string refusal in noIzamThermalSource.SimulationEvidence.Refusals)
                {
                    W("  evidence refusal: {0}", refusal);
                }
            }

            W("  TBD {0}  TSD {1}", noIzamThermalSource.Path_TBD, noIzamThermalSource.Path_TSD);
            W("  room identities: {0}", noIzamThermalSource.Count_ZoneReferences);
            W("  IsComplete: {0}", noIzamThermalSource.IsComplete);

            //The zone map, so process two can rebuild the source without re-running the workflow.
            List<string> lines = new List<string>();
            foreach (KeyValuePair<Guid, string> keyValuePair in noIzamThermalSource.ZoneReferences)
            {
                lines.Add(string.Concat(keyValuePair.Key.ToString("D"), "\t", keyValuePair.Value));
            }

            lines.Sort(StringComparer.Ordinal);

            File.WriteAllLines(Path.Combine(directory, "acc-zones.txt"), lines);

            NoIzamEvidence(noIzamThermalSource.Path_TBD);

            Report(directory, "source.log");

            return noIzamThermalSource.IsComplete ? 0 : 6;
        }

        /// <summary>
        /// Process two: the PR1 materialisation, the PR2 route, and the native re-read that checks
        /// what the route says against what TAS holds.
        /// </summary>
        public static int Run(string path_SAM, string directory, string path_TWD, string mode)
        {
            log.Clear();
            verdicts.Clear();

            bool full = mode != null && mode.StartsWith("full", StringComparison.OrdinalIgnoreCase);

            int startHour = 0;
            int endHour = full ? 8759 : 23;

            string path_Design = Path.Combine(directory, "acc-design.sam");
            string path_TBD = Path.Combine(directory, "acc.tbd");
            string path_TSD = Path.Combine(directory, "acc.tsd");
            string path_TPD = Path.Combine(directory, "acc.tpd");

            W("route-convert {0} hours {1}..{2}", directory, startHour, endHour);

            SAM.Analytical.AnalyticalModel analyticalModel = SAM.Core.Convert.ToSAM<SAM.Analytical.AnalyticalModel>(path_Design)[0];

            SAM.Analytical.AdjacencyCluster adjacencyCluster = analyticalModel.AdjacencyCluster;

            if (mode != null && mode.IndexOf("notransfer", StringComparison.OrdinalIgnoreCase) >= 0)
            {
                foreach (SAM.Analytical.SpaceAirMovement movement in adjacencyCluster.GetObjects<SAM.Analytical.SpaceAirMovement>() ?? new List<SAM.Analytical.SpaceAirMovement>())
                {
                    adjacencyCluster.RemoveObject(movement.GetType(), movement.Guid);
                }

                W("diagnostic variant: transfer air movements removed");
            }

            if (mode != null && mode.IndexOf("balanced", StringComparison.OrdinalIgnoreCase) >= 0)
            {
                foreach (SAM.Analytical.VentilationSystem ventilationSystem in adjacencyCluster.GetObjects<SAM.Analytical.VentilationSystem>() ?? new List<SAM.Analytical.VentilationSystem>())
                {
                    List<SAM.Analytical.VentilationTerminal> terminals = adjacencyCluster.GetRelatedObjects<SAM.Analytical.VentilationTerminal>(ventilationSystem) ?? new List<SAM.Analytical.VentilationTerminal>();
                    double supply = 0;
                    double extract = 0;
                    SAM.Analytical.VentilationTerminal extractTerminal = null;

                    foreach (SAM.Analytical.VentilationTerminal terminal in terminals)
                    {
                        double value = terminal.DesignFlowRate_Lps ?? 0;
                        if (terminal.FlowClassification == SAM.Analytical.FlowClassification.Supply)
                        {
                            supply += value;
                        }
                        else if (terminal.FlowClassification == SAM.Analytical.FlowClassification.Extract)
                        {
                            extract += value;
                            extractTerminal = terminal;
                        }
                    }

                    if (extractTerminal != null && Math.Abs(supply - extract) > 1e-9)
                    {
                        extractTerminal.DesignFlowRate_Lps += supply - extract;
                        W("diagnostic variant: balanced {0} from supply {1} / extract {2} l/s", ventilationSystem.Name, supply, extract);
                    }
                }
            }

            if (mode != null && mode.IndexOf("continuity", StringComparison.OrdinalIgnoreCase) >= 0)
            {
                List<SAM.Analytical.SpaceAirMovement> movements = adjacencyCluster.GetObjects<SAM.Analytical.SpaceAirMovement>() ?? new List<SAM.Analytical.SpaceAirMovement>();
                Dictionary<string, int> outgoingCounts = new Dictionary<string, int>(StringComparer.Ordinal);
                foreach (SAM.Analytical.SpaceAirMovement movement in movements)
                {
                    outgoingCounts.TryGetValue(movement.From, out int count);
                    outgoingCounts[movement.From] = count + 1;
                }

                List<SAM.Analytical.Space> allSpaces = adjacencyCluster.GetSpaces();
                foreach (SAM.Analytical.SpaceAirMovement movement in movements)
                {
                    double flow = movement.AirFlow;
                    if (Math.Abs(flow - 0.0065) < 1e-9) flow = 0.013;
                    else if (Math.Abs(flow - 0.006) < 1e-9) flow = 0.012;
                    else if (Math.Abs(flow - 0.0075) < 1e-9) flow = outgoingCounts[movement.From] == 1 ? 0.025 : 0.015;
                    else if (Math.Abs(flow - 0.005) < 1e-9) flow = 0.010;

                    adjacencyCluster.RemoveObject(movement.GetType(), movement.Guid);
                    SAM.Analytical.SpaceAirMovement replacement = new SAM.Analytical.SpaceAirMovement(movement.Name, flow, movement.From, movement.To);
                    adjacencyCluster.AddObject(replacement);
                    SAM.Analytical.Space target = allSpaces.Find(space => new SAM.Core.ObjectReference(space).ToString() == movement.To);
                    if (target != null) adjacencyCluster.AddRelation(replacement, target);
                }

                W("diagnostic variant: transfer duties changed to exact per-zone continuity");
            }

            Shape shape = Measure(adjacencyCluster);

            W("design read back: {0} room(s), {1} unit(s), {2} supply / {3} extract / {4} transfer leg(s)",
                shape.Rooms, shape.AirHandlingUnits, shape.Supply, shape.Extract, shape.Transfer);

            // ---------------------------------------------------------------- the thermal source
            Dictionary<Guid, string> zoneReferences = new Dictionary<Guid, string>();

            foreach (string line in File.ReadAllLines(Path.Combine(directory, "acc-zones.txt")))
            {
                string[] parts = line.Split('\t');
                if (parts.Length == 2)
                {
                    zoneReferences[Guid.Parse(parts[0])] = parts[1];
                }
            }

            W("room identities read back: {0}", zoneReferences.Count);

            //Re-evidenced rather than re-simulated: the files this run stands on already exist, and
            //Prepare would delete the TSD it is about to be asked to prove.
            SAM.Core.Tas.SimulationEvidence simulationEvidence = new SAM.Core.Tas.SimulationEvidence(
                SAM.Core.Tas.SimulationOutputShape.SeparateOutputFile, path_TBD, path_TSD);

            simulationEvidence.RecordCallReturned();
            simulationEvidence.Conclude();

            SAM.Core.Tas.NoIzamThermalSource noIzamThermalSource = new SAM.Core.Tas.NoIzamThermalSource(
                path_TBD, path_TSD, true, true, simulationEvidence, zoneReferences, null, null);

            W("thermal source IsComplete: {0}", noIzamThermalSource.IsComplete);

            if (!noIzamThermalSource.IsComplete)
            {
                foreach (string refusal in noIzamThermalSource.Refusals)
                {
                    W("  refusal: {0}", refusal);
                }

                foreach (string refusal in simulationEvidence.Refusals)
                {
                    W("  evidence refusal: {0}", refusal);
                }

                Report(directory, "acceptance.log");
                return 6;
            }

            // ---------------------------------------------------------------- PR1
            W("");
            W("== PRODUCTION SAM_Systems Create.MechanicalVentilation ==");

            SAM.Core.Systems.SystemEnergyCentre systemEnergyCentre_Template =
                SAM.Analytical.Systems.Query.SystemEnergyCentre(TemplatePath());

            //const1: the frozen #111 parity operating configuration - an ordinary 8760-hour YearlySchedule
            //holding 1.0 in every hour, supplied through PR1's own generic settings exactly as the
            //Part O caller does. Without it PR1 leaves the template's own fan schedule assignment.
            SAM.Analytical.Systems.MechanicalVentilationSettings mechanicalVentilationSettings = null;

            if (mode != null && mode.IndexOf("const1", StringComparison.OrdinalIgnoreCase) >= 0)
            {
                //CONST1_NAME only renames the schedule. PR1's deterministic identities hash the schedule's
                //name, so this changes every generated guid - and the order TAS receives components in -
                //while the carrier itself stays the identical 8760 x 1.0 yearly table.
                string name_Schedule = Environment.GetEnvironmentVariable("CONST1_NAME");
                if (string.IsNullOrWhiteSpace(name_Schedule))
                {
                    name_Schedule = "Part O Continuous Operation";
                }

                SAM.Analytical.Systems.YearlySchedule yearlySchedule = new SAM.Analytical.Systems.YearlySchedule(name_Schedule);
                double[] values = new double[8760];
                for (int i = 0; i < values.Length; i++)
                {
                    values[i] = 1.0;
                }

                yearlySchedule.Values = values;

                mechanicalVentilationSettings = new SAM.Analytical.Systems.MechanicalVentilationSettings { Schedule = yearlySchedule };

                W("  PR1 settings: operating schedule \"{0}\", {1} hourly values, all 1.0", yearlySchedule.Name, yearlySchedule.Values.Length);
            }
            else
            {
                W("  PR1 settings: none - the template's own fan schedule assignment is left in place");
            }

            SAM.Analytical.Systems.MechanicalVentilationMaterialisation mechanicalVentilationMaterialisation =
                SAM.Analytical.Systems.Create.MechanicalVentilation(adjacencyCluster, systemEnergyCentre_Template, mechanicalVentilationSettings);

            foreach (string note in mechanicalVentilationMaterialisation.Notes)
            {
                W("  note: {0}", note);
            }

            foreach (string refusal in mechanicalVentilationMaterialisation.Refusals)
            {
                W("  refusal: {0}", refusal);
            }

            if (!mechanicalVentilationMaterialisation.IsMaterialised)
            {
                W("REFUSED: PR1 did not materialise.");
                Report(directory, "acceptance.log");
                return 7;
            }

            // ---------------------------------------------------------------- PR2
            W("");
            W("== PRODUCTION Create.SystemVentilationRoute ==");

            SAM.Analytical.Tas.TPD.SystemVentilationRoute systemVentilationRoute =
                SAM.Analytical.Tas.TPD.Create.SystemVentilationRoute(
                    noIzamThermalSource, mechanicalVentilationMaterialisation, path_TPD, startHour, endHour);

            foreach (string note in systemVentilationRoute.Notes)
            {
                W("  note: {0}", note);
            }

            foreach (string refusal in systemVentilationRoute.Refusals)
            {
                W("  refusal: {0}", refusal);
            }

            W("  IsComplete: {0}", systemVentilationRoute.IsComplete);

            if (systemVentilationRoute.SimulationEvidence != null)
            {
                W("  NativeDiagnostic: [{0}]  kind={1}",
                    systemVentilationRoute.SimulationEvidence.NativeDiagnostic ?? "<null>",
                    systemVentilationRoute.SimulationEvidence.NativeDiagnosticKind);
            }

            if (!systemVentilationRoute.IsComplete)
            {
                if (mode != null && mode.IndexOf("per-system", StringComparison.OrdinalIgnoreCase) >= 0)
                {
                    using (SAM.Core.Tas.SAMTPDDocument document = new SAM.Core.Tas.SAMTPDDocument(path_TPD))
                    {
                        TPD.EnergyCentre energyCentre = document.TPDDocument.EnergyCentre;
                        for (int plantIndex = 1; plantIndex <= energyCentre.GetPlantRoomCount(); plantIndex++)
                        {
                            TPD.PlantRoom plantRoom = energyCentre.GetPlantRoom(plantIndex);
                            for (int systemIndex = 1; systemIndex <= plantRoom.GetSystemCount(); systemIndex++)
                            {
                                TPD.System nativeSystem = plantRoom.GetSystem(systemIndex);
                                W("diagnostic per-system: {0} -> {1}", nativeSystem.Name, nativeSystem.Simulate(startHour + 1, endHour + 1, 0));
                            }
                        }
                    }
                }
                W("REFUSED: the route did not complete.");
                Report(directory, "acceptance.log");
                return 8;
            }

            Acceptance(systemVentilationRoute, mechanicalVentilationMaterialisation, adjacencyCluster, shape, path_TPD, startHour, endHour);

            Report(directory, "acceptance.log");
            return 0;
        }

        /// <summary>Reads the shape back off a design that was written by <see cref="Design"/>.</summary>
        private static Shape Measure(SAM.Analytical.AdjacencyCluster adjacencyCluster)
        {
            Shape result = new Shape();

            result.AirHandlingUnits = adjacencyCluster.GetObjects<SAM.Analytical.AirHandlingUnit>().Count;

            Dictionary<Guid, bool> supply = new Dictionary<Guid, bool>();
            Dictionary<Guid, bool> extract = new Dictionary<Guid, bool>();

            List<SAM.Analytical.VentilationSystem> ventilationSystems = adjacencyCluster.GetObjects<SAM.Analytical.VentilationSystem>();

            HashSet<Guid> rooms = new HashSet<Guid>();

            foreach (SAM.Analytical.VentilationSystem ventilationSystem in ventilationSystems)
            {
                foreach (SAM.Analytical.VentilationTerminal ventilationTerminal in adjacencyCluster.GetRelatedObjects<SAM.Analytical.VentilationTerminal>(ventilationSystem) ?? new List<SAM.Analytical.VentilationTerminal>())
                {
                    foreach (SAM.Analytical.Space space in adjacencyCluster.GetRelatedObjects<SAM.Analytical.Space>(ventilationTerminal) ?? new List<SAM.Analytical.Space>())
                    {
                        rooms.Add(space.Guid);

                        if (ventilationTerminal.FlowClassification == SAM.Analytical.FlowClassification.Supply)
                        {
                            supply[space.Guid] = true;
                        }
                        else
                        {
                            extract[space.Guid] = true;
                        }
                    }
                }
            }

            foreach (SAM.Analytical.SpaceAirMovement spaceAirMovement in adjacencyCluster.GetObjects<SAM.Analytical.SpaceAirMovement>())
            {
                result.Transfer++;

                foreach (SAM.Analytical.Space space in adjacencyCluster.GetRelatedObjects<SAM.Analytical.Space>(spaceAirMovement) ?? new List<SAM.Analytical.Space>())
                {
                    rooms.Add(space.Guid);
                }
            }

            //The hall is reached only through the transfers, and Measure cannot see its guid through a
            //relation the movement does not carry, so the room count comes from the movements' own
            //endpoints being materialised - which the route reports and item 5 checks against PR1.
            result.Rooms = rooms.Count;
            result.Supply = supply.Count;
            result.Extract = extract.Count;

            foreach (Guid guid in rooms)
            {
                bool hasSupply = supply.ContainsKey(guid);
                bool hasExtract = extract.ContainsKey(guid);

                if (hasSupply && hasExtract)
                {
                    result.Both++;
                }
                else if (hasSupply)
                {
                    result.SupplyOnly++;
                }
                else if (hasExtract)
                {
                    result.ExtractOnly++;
                }
                else
                {
                    result.TransferOnly++;
                }
            }

            return result;
        }

        private static void Acceptance(
            SAM.Analytical.Tas.TPD.SystemVentilationRoute systemVentilationRoute,
            SAM.Analytical.Systems.MechanicalVentilationMaterialisation mechanicalVentilationMaterialisation,
            SAM.Analytical.AdjacencyCluster adjacencyCluster,
            Shape shape,
            string path_TPD,
            int startHour,
            int endHour)
        {
            List<SAM.Analytical.Tas.TPD.SystemVentilationBinding> bindings = systemVentilationRoute.Bindings;
            List<SAM.Analytical.Tas.TPD.SystemVentilationConnectionBinding> connectionBindings = systemVentilationRoute.ConnectionBindings;

            // ---------------------------------------------------------------- what PR1 said, independently
            Dictionary<Guid, Guid> airSystem_By_AirHandlingUnit = new Dictionary<Guid, Guid>();
            Dictionary<Guid, Guid> space_By_SystemSpace = new Dictionary<Guid, Guid>();
            Dictionary<Guid, Guid> spaceAirMovement_By_Connection = new Dictionary<Guid, Guid>();

            foreach (SAM.Analytical.Systems.MechanicalVentilationBinding mechanicalVentilationBinding in mechanicalVentilationMaterialisation.Bindings)
            {
                switch (mechanicalVentilationBinding.BindingType)
                {
                    case SAM.Analytical.Systems.MechanicalVentilationBindingType.AirSystem:
                        airSystem_By_AirHandlingUnit[mechanicalVentilationBinding.Guid_Analytical] = mechanicalVentilationBinding.Guid_Systems;
                        break;
                    case SAM.Analytical.Systems.MechanicalVentilationBindingType.SystemSpace:
                        space_By_SystemSpace[mechanicalVentilationBinding.Guid_Systems] = mechanicalVentilationBinding.Guid_Analytical;
                        break;
                    case SAM.Analytical.Systems.MechanicalVentilationBindingType.TransferConnection:
                        spaceAirMovement_By_Connection[mechanicalVentilationBinding.Guid_Systems] = mechanicalVentilationBinding.Guid_Analytical;
                        break;
                }
            }

            int count_AirHandlingUnits = adjacencyCluster.GetObjects<SAM.Analytical.AirHandlingUnit>().Count;

            // ---------------------------------------------------------------- the native document
            global::TPD.TPDDoc tPDDoc = new global::TPD.TPDDoc();

            List<object> systems = new List<object>();
            Dictionary<string, object> component_By_Guid = new Dictionary<string, object>(StringComparer.OrdinalIgnoreCase);
            Dictionary<string, string> system_By_Component = new Dictionary<string, string>(StringComparer.OrdinalIgnoreCase);
            List<string> fans = new List<string>();
            List<string> zoneFlags = new List<string>();
            int count_ZoneFlagsWrong = 0;

            try
            {
                tPDDoc.Open(path_TPD);

                object energyCentre = tPDDoc.EnergyCentre;

                int count_PlantRoom = (int)Dbl(Dump.Call(energyCentre, "GetPlantRoomCount"));

                for (int p = 1; p <= count_PlantRoom; p++)
                {
                    object plantRoom = Dump.Call(energyCentre, "GetPlantRoom", p);
                    int count_System = (int)Dbl(Dump.Call(plantRoom, "GetSystemCount"));

                    for (int s = 1; s <= count_System; s++)
                    {
                        object system = Dump.Call(plantRoom, "GetSystem", s);
                        systems.Add(system);

                        string reference_System = Dump.Obj(system, "GUID") as string;

                        foreach (object component in Components(system))
                        {
                            string reference = Dump.Obj(component, "GUID") as string;
                            if (reference == null)
                            {
                                continue;
                            }

                            component_By_Guid[reference] = component;
                            system_By_Component[reference] = reference_System;

                            if (component is global::TPD.Fan fan)
                            {
                                fans.Add(string.Format(CultureInfo.InvariantCulture,
                                    "fan {0} DesignFlowType={1} value={2}",
                                    reference, fan.DesignFlowType, fan.DesignFlowRate == null ? double.NaN : fan.DesignFlowRate.Value));
                            }

                            if (component is global::TPD.SystemZone systemZone)
                            {
                                int flags = systemZone.Flags;

                                bool ok = (flags & (int)global::TPD.tpdSystemZoneFlags.tpdSystemZoneFlagModelVentFlow) == 0
                                    && (flags & (int)global::TPD.tpdSystemZoneFlags.tpdSystemZoneFlagModelInterzoneFlow) == 0;

                                if (!ok)
                                {
                                    count_ZoneFlagsWrong++;
                                }

                                zoneFlags.Add(string.Format(CultureInfo.InvariantCulture, "zone {0} Flags={1}", reference, flags));
                            }
                        }
                    }
                }

                Native(
                    systemVentilationRoute,
                    shape,
                    bindings,
                    connectionBindings,
                    space_By_SystemSpace,
                    spaceAirMovement_By_Connection,
                    airSystem_By_AirHandlingUnit,
                    count_AirHandlingUnits,
                    systems,
                    component_By_Guid,
                    system_By_Component,
                    fans,
                    zoneFlags,
                    count_ZoneFlagsWrong,
                    startHour,
                    endHour);
            }
            finally
            {
                try { tPDDoc.Close(); } catch { }
            }

            //Items 13 and 14 both work on COPIES of the document, and both need it closed first:
            //13 re-simulates each air system with a raw late-bound call so its read owes nothing to
            //the code under test, and 14 renames every zone and damper to one shared string.
            Agreement(systemVentilationRoute, path_TPD, startHour, endHour);

            //Item 14 is proved by MUTATING the document: every zone and every damper is renamed to one
            //shared string, and the results are then read again through the same bindings. If any step
            //of the route resolved by name, the second read would alias or fail.
            DuplicateNames(systemVentilationRoute, path_TPD, startHour, endHour);
        }

        private static IEnumerable<object> Components(object container)
        {
            List<object> result = new List<object>();

            Collect(container, result, 0);

            return result;
        }

        private static void Collect(object container, List<object> result, int depth)
        {
            if (container == null || depth > 8)
            {
                return;
            }

            for (int index = 1; index <= 20000; index++)
            {
                object component = Dump.Call(container, "GetComponent", index);
                if (component == null)
                {
                    break;
                }

                result.Add(component);

                if (component is global::TPD.ComponentGroup)
                {
                    Collect(component, result, depth + 1);
                }
            }
        }

        private static void Flows(
            List<SAM.Analytical.Tas.TPD.SystemVentilationBinding> bindings,

                if (!binding.DesignFlowRate_Supply_Lps.HasValue)
                {
                    continue;
                }

                if (!component_By_Guid.TryGetValue(binding.Reference_SystemZone, out object component) || !(component is global::TPD.SystemZone systemZone))
                {
                    mismatches_Supply.Add(string.Format("room {0}: no native zone", binding.Guid_Space));

                double value = systemZone.FlowRate == null ? double.NaN : systemZone.FlowRate.Value;

                    && Close(freshAir, binding.DesignFlowRate_Supply_Lps.Value)
                    && systemZone.FlowRate.Type == global::TPD.tpdSizedVariable.tpdSizedVariableValue;

                if (!ok)
                {
                    mismatches_Supply.Add(string.Format(CultureInfo.InvariantCulture,
                        "room {0}: binding {1}, native {2}/{3}, type {4}",
                        binding.Guid_Space, binding.DesignFlowRate_Supply_Lps, value, freshAir, systemZone.FlowRate.Type));
                }
            }

            int count_SupplyRooms = bindings.Count(x => x.DesignFlowRate_Supply_Lps.HasValue);

            Item(7, "supply DesignFlowRate is preserved, absolutely, on the room's own zone",
                mismatches_Supply.Count == 0 && count_SupplyRooms > 0,
                string.Format("{0} supply room(s) checked against the native zone, {1} mismatch(es). {2}",
                    count_SupplyRooms, mismatches_Supply.Count, string.Join("; ", mismatches_Supply)));

            // ---------------------------------------------------------------- items 8 and 9
            Legs(connectionBindings, component_By_Guid, SAM.Analytical.Tas.TPD.SystemVentilationConnectionType.Extract, 8, "extract");
            Legs(connectionBindings, component_By_Guid, SAM.Analytical.Tas.TPD.SystemVentilationConnectionType.Transfer, 9, "transfer");
        }

        private static void Legs(
            List<SAM.Analytical.Tas.TPD.SystemVentilationConnectionBinding> connectionBindings,

                if (connectionBinding.ConnectionType != systemVentilationConnectionType)

        private static bool Close(double value_1, double value_2)
        {
            return !double.IsNaN(value_1) && !double.IsNaN(value_2)
                && Math.Abs(value_1 - value_2) <= Math.Max(1e-6, Math.Max(Math.Abs(value_1), Math.Abs(value_2)) * 1e-5);
        }

        // ---------------------------------------------------------------- item 13

        /// <summary>
        /// Item 13. Reads every room's series a SECOND time, independently of the production code, and
        /// compares it value for value.
        /// <para>
        /// <b>Per system, and on a copy.</b> Licensed TAS replaces an air system's in-memory result
        /// surface when <c>ISystem.Simulate</c> is called for the next system in the same document, so a
        /// single pass after the loop can only ever see the last system's zones - which is what silently
        /// reduced this check to four of nine rooms before. Here each system is simulated by a raw
        /// late-bound call and its own rooms are read straight afterwards, on a copy of the document so
        /// the route's own output is left as it was produced.
        /// </para>
        /// <para>
        /// The denominator is <b>every bound room</b>. A room whose independent read failed counts as a
        /// disagreement, not as a room that was never asked about.
        /// </para>
        /// </summary>
        private static void Agreement(
            SAM.Analytical.Tas.TPD.SystemVentilationRoute systemVentilationRoute,
            string path_TPD,
            int startHour,
            int endHour)
        {
            List<SAM.Analytical.Tas.TPD.SystemVentilationBinding> bindings = systemVentilationRoute.Bindings;

            Dictionary<string, List<SAM.Analytical.Tas.TPD.SystemVentilationBinding>> bindings_By_System =
                new Dictionary<string, List<SAM.Analytical.Tas.TPD.SystemVentilationBinding>>(StringComparer.OrdinalIgnoreCase);

            foreach (SAM.Analytical.Tas.TPD.SystemVentilationBinding binding in bindings)
            {
                if (!bindings_By_System.TryGetValue(binding.Reference_System, out List<SAM.Analytical.Tas.TPD.SystemVentilationBinding> list))
                {
                    list = new List<SAM.Analytical.Tas.TPD.SystemVentilationBinding>();
                    bindings_By_System[binding.Reference_System] = list;
                }

                list.Add(binding);
            }

            int expected = endHour - startHour + 1;
            int count_Agreed = 0;
            List<string> disagreements = new List<string>();

            string path_Agree = Path.ChangeExtension(path_TPD, ".agree.tpd");

            File.Copy(path_TPD, path_Agree, true);

            global::TPD.TPDDoc tPDDoc = new global::TPD.TPDDoc();

            try
            {
                tPDDoc.Open(path_Agree);

                object energyCentre = tPDDoc.EnergyCentre;

                int count_PlantRoom = (int)Dbl(Dump.Call(energyCentre, "GetPlantRoomCount"));

                for (int p = 1; p <= count_PlantRoom; p++)
                {
                    object plantRoom = Dump.Call(energyCentre, "GetPlantRoom", p);
                    int count_System = (int)Dbl(Dump.Call(plantRoom, "GetSystemCount"));

                    for (int sIndex = 1; sIndex <= count_System; sIndex++)
                    {
                        object system = Dump.Call(plantRoom, "GetSystem", sIndex);

                        string reference_System = Dump.Obj(system, "GUID") as string;

                        if (reference_System == null || !bindings_By_System.TryGetValue(reference_System, out List<SAM.Analytical.Tas.TPD.SystemVentilationBinding> list))
                        {
                            continue;
                        }

                        //A raw late-bound simulate, so nothing about this read goes through the code
                        //under test.
                        string returned = Dump.Call(system, "Simulate", startHour + 1, endHour + 1, 0) as string;

                        Dictionary<string, object> component_By_Guid = new Dictionary<string, object>(StringComparer.OrdinalIgnoreCase);

                        foreach (object component in Components(system))
                        {
                            string reference = Dump.Obj(component, "GUID") as string;
                            if (reference != null)
                            {
                                component_By_Guid[reference] = component;
                            }
                        }

                        foreach (SAM.Analytical.Tas.TPD.SystemVentilationBinding binding in list)
                        {
                            SAM.Analytical.Tas.TPD.SystemZoneTemperatureResult result =
                                systemVentilationRoute.SystemZoneTemperatureResults.Result(binding.Guid_Space);

                            if (result == null || result.Count == 0)
                            {
                                disagreements.Add(string.Format("room {0}: the route produced no series", binding.Guid_Space));
                                continue;
                            }

                            if (!component_By_Guid.TryGetValue(binding.Reference_SystemZone, out object component))
                            {
                                disagreements.Add(string.Format("room {0}: native zone not in system {1}", binding.Guid_Space, reference_System));
                                continue;
                            }

                            object data;

                            try
                            {
                                data = component.GetType().InvokeMember(
                                    "GetResultsData",
                                    BindingFlags.InvokeMethod,
                                    null,
                                    component,
                                    new object[]
                                    {
                                        (int)global::TPD.tpdResultsPeriod.tpdResultsPeriodHourly,
                                        (int)global::TPD.tpdCombinerType.tpdCombinerTypeMax,
                                        9,
                                        startHour + 1,
                                        expected,
                                    },
                                    CultureInfo.InvariantCulture);
                            }
                            catch (Exception exception)
                            {
                                disagreements.Add(string.Format("room {0}: {1} (system answered \"{2}\")",
                                    binding.Guid_Space, (exception.InnerException ?? exception).Message.Trim(), returned));
                                continue;
                            }

                            Array array = data as Array;

                            if (array == null)
                            {
                                disagreements.Add(string.Format("room {0}: no native series (system answered \"{1}\")", binding.Guid_Space, returned));
                                continue;
                            }

                            int index = startHour;
                            bool agreed = true;
                            int count_Compared = 0;

                            foreach (object element in array)
                            {
                                double value_Native = Dbl(element);

                                if (!result.TryGetValue(index, out double value_Api) || !Close(value_Api, value_Native))
                                {
                                    agreed = false;
                                    disagreements.Add(string.Format(CultureInfo.InvariantCulture,
                                        "room {0} hour {1}: API {2}, native {3}", binding.Guid_Space, index, value_Api, value_Native));
                                    break;
                                }

                                index++;
                                count_Compared++;
                            }

                            if (agreed && count_Compared != expected)
                            {
                                agreed = false;
                                disagreements.Add(string.Format("room {0}: native series held {1} value(s), expected {2}",
                                    binding.Guid_Space, count_Compared, expected));
                            }

                            if (agreed)
                            {
                                count_Agreed++;
                            }
                        }
                    }
                }
            }
            finally
            {
                try { tPDDoc.Close(); } catch { }
            }

            Item(13, "the route's values agree with TAS's own, read independently",
                bindings.Count > 0 && count_Agreed == bindings.Count,
                string.Format("{0} of {1} room(s) agreed value for value over {2} hour(s), each read after a raw late-bound simulate of its own air system. {3}",
                    count_Agreed, bindings.Count, expected, string.Join("; ", disagreements.Take(3))));
        }

        // ---------------------------------------------------------------- item 14

        private static void DuplicateNames(
            SAM.Analytical.Tas.TPD.SystemVentilationRoute systemVentilationRoute,
            string path_TPD,
            int startHour,
            int endHour)
        {
            string path_Renamed = Path.ChangeExtension(path_TPD, ".renamed.tpd");

            File.Copy(path_TPD, path_Renamed, true);

            int count_Renamed = 0;

            global::TPD.TPDDoc tPDDoc = new global::TPD.TPDDoc();

            try
            {
                tPDDoc.Open(path_Renamed);

                object energyCentre = tPDDoc.EnergyCentre;

                int count_PlantRoom = (int)Dbl(Dump.Call(energyCentre, "GetPlantRoomCount"));

                for (int p = 1; p <= count_PlantRoom; p++)
                {
                    object plantRoom = Dump.Call(energyCentre, "GetPlantRoom", p);
                    int count_System = (int)Dbl(Dump.Call(plantRoom, "GetSystemCount"));

                    for (int s = 1; s <= count_System; s++)
                    {
                        foreach (object component in Components(Dump.Call(plantRoom, "GetSystem", s)))
                        {
                            if (!(component is global::TPD.SystemZone) && !(component is global::TPD.Damper))
                            {
                                continue;
                            }

                            try
                            {
                                component.GetType().InvokeMember(
                                    component is global::TPD.SystemZone ? "name" : "Name",
                                    BindingFlags.SetProperty,
                                    null,
                                    component,
                                    new object[] { "System Zone 1" },
                                    CultureInfo.InvariantCulture);

                                count_Renamed++;
                            }
                            catch
                            {
                            }
                        }
                    }
                }

                Dump.Call(tPDDoc, "Save");
            }
            finally
            {
                try { tPDDoc.Close(); } catch { }
            }

            SAM.Analytical.Tas.TPD.Modify.SimulateSystems(
                path_Renamed,
                systemVentilationRoute.Bindings,
                startHour,
                endHour,
                out SAM.Core.Tas.SimulationEvidence simulationEvidence,
                out SAM.Analytical.Tas.TPD.SystemZoneTemperatureResults systemZoneTemperatureResults);

            bool validated = systemZoneTemperatureResults.Validate(systemVentilationRoute.Bindings);

            //And the values must be the SAME values, not merely a complete set: a name-keyed lookup
            //that aliased two rooms would still be complete.
            int count_Same = 0;

            foreach (SAM.Analytical.Tas.TPD.SystemZoneTemperatureResult result in systemZoneTemperatureResults.Results)
            {
                SAM.Analytical.Tas.TPD.SystemZoneTemperatureResult result_Original = systemVentilationRoute.SystemZoneTemperatureResults.Result(result.Guid_Space);

                if (result_Original == null || result.Count == 0 || result_Original.Count == 0)
                {
                    continue;
                }

                bool same = true;

                for (int i = startHour; i <= endHour; i++)
                {
                    if (!result.TryGetValue(i, out double value_1) || !result_Original.TryGetValue(i, out double value_2) || !Close(value_1, value_2))
                    {
                        same = false;
                        break;
                    }
                }

                if (same)
                {
                    count_Same++;
                }
            }

            Item(14, "duplicate display names do not alter identity",
                validated && count_Same == systemVentilationRoute.Bindings.Count && count_Renamed > 0,
                string.Format("{0} native zone(s) and damper(s) renamed to one shared string; results still validate: {1}; {2} of {3} room(s) returned identical values.",
                    count_Renamed, validated, count_Same, systemVentilationRoute.Bindings.Count));
        }
    }
