// Fan heat gain and the operation (schedule) factor, measured on a produced TPD.
//
// "Continuous operation" is not a comment: it is a PlantSchedule whose operable-hour count is 8760
// and whose yearly value is 1 in every hour, together with a ProfileData whose Value is 1.0 and which
// carries no modifier that could scale it down. All of that is read here, per fan and per zone.
//
// Nothing here is production code.

using System;
using System.Collections.Generic;
using System.Globalization;
using System.IO;
using System.Text;
using SAM.Core.Tas;

namespace Inv
{
    internal static class Oper
    {
        private static string Profile(object component, string name)
        {
            object profileData = Dump.Obj(component, name);

            if (profileData == null)
            {
                return name + "=<null>";
            }

            int countModifier = Graph.Int(Dump.Call(profileData, "GetModifierCount"));
            List<string> types = new List<string>();

            for (int i = 1; i <= countModifier; i++)
            {
                types.Add(Dump.CallText(profileData, "GetModifierType", i));
            }

            return string.Format(
                CultureInfo.InvariantCulture,
                "{0}.Value={1} modifiers={2}{3}",
                name,
                Dump.Get(profileData, "Value"),
                countModifier,
                types.Count == 0 ? "" : " [" + string.Join(", ", types) + "]");
        }

        private static string Schedule(object component)
        {
            object plantSchedule = Dump.Call(component, "GetSchedule");

            if (plantSchedule == null)
            {
                return "schedule=<none attached>";
            }

            int operable = Graph.Int(Dump.Call(plantSchedule, "GetNumOperableHours"));

            int nonOne = 0;
            int zero = 0;
            int sampled = 0;

            for (int hour = 1; hour <= 8760; hour++)
            {
                object value = Dump.Call(plantSchedule, "GetYearlyValue", hour);
                if (value == null)
                {
                    break;
                }

                sampled++;
                int v = Graph.Int(value);
                if (v != 1)
                {
                    nonOne++;
                }

                if (v == 0)
                {
                    zero++;
                }
            }

            return string.Format(
                CultureInfo.InvariantCulture,
                "schedule name={0} type={1} functionType={2} operableHours={3} yearlyValues sampled={4} notEqualToOne={5} zero={6}",
                Dump.Get(plantSchedule, "Name"),
                Dump.Get(plantSchedule, "Type"),
                Dump.Get(plantSchedule, "FunctionType"),
                operable,
                sampled,
                nonOne,
                zero);
        }

        public static void Run(string path, string pathOut)
        {
            StringBuilder stringBuilder = new StringBuilder();
            stringBuilder.AppendLine("OPERATION " + path);

            using (SAMTPDDocument sAMTPDDocument = new SAMTPDDocument(path))
            {
                object energyCentre = sAMTPDDocument.TPDDocument.EnergyCentre;

                foreach (object system in Graph.Systems(energyCentre))
                {
                    stringBuilder.AppendLine();
                    stringBuilder.AppendLine("== SYSTEM " + Dump.Get(system, "Name") + " " + Graph.Guid_Of(system));

                    foreach (object component in Graph.TopLevel(system))
                    {
                        if (component is global::TPD.Fan)
                        {
                            stringBuilder.AppendLine(string.Format(
                                CultureInfo.InvariantCulture,
                                "  FAN {0} {1}",
                                Graph.Guid_Of(component),
                                Dump.Get(component, "Name")));

                            stringBuilder.AppendLine("        HeatGainFactor      = " + Dump.Get(component, "HeatGainFactor"));
                            stringBuilder.AppendLine("        ControlType         = " + Dump.Get(component, "ControlType"));
                            stringBuilder.AppendLine("        Pressure            = " + Dump.Get(component, "Pressure"));
                            stringBuilder.AppendLine("        MinimumFlowFraction = " + Dump.Get(component, "MinimumFlowFraction"));
                            stringBuilder.AppendLine("        DesignFlowType      = " + Dump.Get(component, "DesignFlowType"));
                            stringBuilder.AppendLine("        " + Profile(component, "PartLoad"));
                            stringBuilder.AppendLine("        " + Profile(component, "OverallEfficiency"));
                            stringBuilder.AppendLine("        " + Schedule(component));
                        }
                        else if (component is global::TPD.SystemZone)
                        {
                            stringBuilder.AppendLine(string.Format(
                                CultureInfo.InvariantCulture,
                                "  ZONE {0} {1}",
                                Graph.Guid_Of(component),
                                Dump.Get(component, "Name")));

                            stringBuilder.AppendLine("        Flags               = " + Dump.Get(component, "Flags")
                                + " (1=DisplacementVent, 2=ModelInterzoneFlow, 4=ModelVentFlow)");
                            stringBuilder.AppendLine("        MinimumFlowFraction = " + Dump.Get(component, "MinimumFlowFraction"));
                            stringBuilder.AppendLine("        " + Profile(component, "TemperatureSetpoint"));
                            stringBuilder.AppendLine("        " + Schedule(component));
                        }
                        else if (component is global::TPD.Damper)
                        {
                            stringBuilder.AppendLine(string.Format(
                                CultureInfo.InvariantCulture,
                                "  DAMPER {0} {1}  DesignFlowType={2} MinimumFlowType={3} MinimumFlowFraction={4}",
                                Graph.Guid_Of(component),
                                Dump.Get(component, "Name"),
                                Dump.Get(component, "DesignFlowType"),
                                Dump.Get(component, "MinimumFlowType"),
                                Dump.Get(component, "MinimumFlowFraction")));

                            stringBuilder.AppendLine("        " + Schedule(component));
                        }
                    }
                }
            }

            string text = stringBuilder.ToString();
            Console.WriteLine(text);

            if (!string.IsNullOrWhiteSpace(pathOut))
            {
                File.WriteAllText(pathOut, text);
                Console.WriteLine("written " + pathOut);
            }
        }
    }
}
