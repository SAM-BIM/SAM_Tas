using System;
using System.Collections;
using System.Collections.Generic;
using System.Globalization;
using System.IO;
using System.Linq;
using TPD;

namespace B0P
{
    public static partial class Steps
    {
        public static List<PlantRoom> PlantRooms(TPDDoc doc)
        {
            List<PlantRoom> result = new List<PlantRoom>();
            EnergyCentre ec = doc.EnergyCentre;
            int i = 1;
            PlantRoom pr = ec.GetPlantRoom(i);
            while (pr != null) { result.Add(pr); i++; pr = ec.GetPlantRoom(i); }
            return result;
        }

        public static List<SystemComponent> Components(global::TPD.System sys)
        {
            List<SystemComponent> result = new List<SystemComponent>();
            int count = sys.GetComponentCount();
            for (int i = 1; i <= count; i++)
            {
                SystemComponent sc = sys.GetComponent(i);
                if (sc == null) continue;
                result.Add(sc);
                if (sc is ComponentGroup) result.AddRange(Components((ComponentGroup)sc));
            }
            return result;
        }

        public static List<SystemComponent> Components(ComponentGroup cg)
        {
            List<SystemComponent> result = new List<SystemComponent>();
            int count = cg.GetComponentCount();
            for (int i = 1; i <= count; i++)
            {
                SystemComponent sc = cg.GetComponent(i);
                if (sc == null) continue;
                result.Add(sc);
                if (sc is ComponentGroup) result.AddRange(Components((ComponentGroup)sc));
            }
            return result;
        }

        public static string LastDiag;

        static string Compact(string s) { return s == null ? null : System.Text.RegularExpressions.Regex.Replace(s, @"\s+", " "); }

        public static double[] Results(object component, int variable, int start, int end)
        {
            object o;
            LastDiag = null;
            try { o = ((dynamic)component).GetResultsData(tpdResultsPeriod.tpdResultsPeriodHourly, tpdCombinerType.tpdCombinerTypeMax, variable, start, end - start + 1); }
            catch (Exception ex) { LastDiag = ex.GetType().Name + ": " + Compact(ex.Message); return null; }
            if (o == null) { LastDiag = "null"; return null; }
            IEnumerable e = o as IEnumerable;
            if (e == null) return null;
            List<double> values = new List<double>();
            foreach (float f in e) values.Add(f);
            return values.ToArray();
        }

        // tpdzt <file.tpd> <outdir>
        public static void TpdZoneTemperature(string[] args)
        {
            string path = args[1];
            string outDir = args[2];
            Directory.CreateDirectory(outDir);
            TPDDoc doc = new TPDDoc();
            if (args.Length > 3 && args[3] == "rw") doc.Open(path); else doc.OpenReadOnly(path);
            try
            {
                EnergyCentre ec = doc.EnergyCentre;
                W("TPD {0}", path);
                try { W("TSDPATH|{0}", ((dynamic)ec.GetTSDData(1)).TSDPath); } catch (Exception ex) { W("TSDPATH|ERR|{0}", Compact(ex.Message)); }
                int start = 1, end = 8760;
                var series = new List<Tuple<string, double[]>>();
                foreach (PlantRoom pr in PlantRooms(doc))
                {
                    W("PLANTROOM|{0}", ((dynamic)pr).Name);
                    int sysCount = pr.GetSystemCount();
                    for (int s = 1; s <= sysCount; s++)
                    {
                        global::TPD.System sys = pr.GetSystem(s);
                        if (sys == null) continue;
                        W("SYSTEM|{0}|{1}", ((dynamic)sys).Name, ((dynamic)sys).GUID);
                        foreach (SystemComponent sc in Components(sys))
                        {
                            string cname = null, cguid = null, ctype = sc.GetType().Name;
                            try { cname = ((dynamic)sc).Name; } catch { }
                            try { cguid = ((dynamic)sc).GUID; } catch { }
                            bool isZone = sc is SystemZone;
                            W("COMP|{0}|{1}|{2}|{3}", ctype, cname, cguid, isZone);
                            if (!isZone) continue;
                            SystemZone sz = (SystemZone)sc;
                            dynamic zl = null;
                            try { zl = sz.GetSystemZoneZoneLoad(); } catch { }
                            string loadName = null; double area = 0, vol = 0;
                            if (zl != null) { try { loadName = zl.Name; area = zl.FloorArea; vol = zl.Volume; } catch { } }
                            W("SYSZONE|{0}|{1}|load='{2}'|area={3}|vol={4}", cname, cguid, loadName, area, vol);
                            for (int v = 1; v <= 11; v++)
                            {
                                double[] r = Results(sc, v, start, end);
                                if (r == null) { W("   var{0,-3} NULL {1}", v, LastDiag); continue; }
                                W("   var{0,-3} n={1} min={2:F4} max={3:F4} mean={4:F4}", v, r.Length, r.Min(), r.Max(), r.Average());
                                series.Add(Tuple.Create(string.Format("{0}|{1}|var{2}", cguid, loadName, v), r));
                            }
                        }
                    }
                }
                string outPath = Path.Combine(outDir, "tpd-zone-series.csv");
                using (StreamWriter sw = new StreamWriter(outPath, false))
                {
                    sw.WriteLine("hour," + string.Join(",", series.Select(x => x.Item1)));
                    int n = series.Count == 0 ? 0 : series.Min(x => x.Item2.Length);
                    for (int h = 0; h < n; h++)
                    {
                        sw.Write(h);
                        foreach (var ser in series) { sw.Write(','); sw.Write(ser.Item2[h].ToString("R", CultureInfo.InvariantCulture)); }
                        sw.WriteLine();
                    }
                }
                W("WROTE {0} ({1} series)", outPath, series.Count);
            }
            finally { doc.Close(); }
        }
    }
