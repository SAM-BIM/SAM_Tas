// PHASE C - ONE fresh no-IZAM thermal source, generated from the exact new fixture.
//
// The model that goes in is the model Phase A prepared, unaltered. Nothing here writes a ventilation
// design; every step is the production call:
//
//   SAM.Analytical.gbXML.Convert.TogbXML             the geometry export
//   SAM.Analytical.Tas.Create.NoIzamThermalSource    the thermal source, both cleanups forced on
//
// The produced TBD is then read back INDEPENDENTLY through TBD's own 0-based accessors for the no-IZAM
// evidence, and both files are hashed.

using System;
using System.Collections.Generic;
using System.Globalization;
using System.IO;
using System.Security.Cryptography;
using System.Text;

namespace Inv
{
    internal static class Src
    {
        private static readonly StringBuilder log = new StringBuilder();
        private static readonly List<string> verdicts = new List<string>();

        private static void W(string format, params object[] args)
        {
            string line = args == null || args.Length == 0 ? format : string.Format(CultureInfo.InvariantCulture, format, args);
            Console.WriteLine(line);
            log.AppendLine(line);
        }

        private static void Item(int number, string title, bool pass, string evidence)
        {
            string line = string.Format(CultureInfo.InvariantCulture, "ITEM {0,2}  {1,-4}  {2}\n            {3}", number, pass ? "PASS" : "FAIL", title, evidence);
            verdicts.Add(line);
            W("");
            W(line);
        }

        internal static string Hash(string path)
        {
            if (!File.Exists(path))
            {
                return "<absent>";
            }

            using (SHA256 sHA256 = SHA256.Create())
            using (FileStream fileStream = File.OpenRead(path))
            {
                return BitConverter.ToString(sHA256.ComputeHash(fileStream)).Replace("-", string.Empty);
            }
        }

        public static int Main(string path_SAM, string directory, string path_TWD, string mode)
        {
            log.Clear();
            verdicts.Clear();

            bool full = mode != null && mode.StartsWith("full", StringComparison.OrdinalIgnoreCase);

            Directory.CreateDirectory(directory);

            string path_gbXML = Path.Combine(directory, "src.xml");
            string path_TBD = Path.Combine(directory, "src.tbd");
            string path_TSD = Path.Combine(directory, "src.tsd");

            W("PHASE C - the no-IZAM thermal source");
            W("  model    {0}  sha256 {1}", path_SAM, Hash(path_SAM));
            W("  weather  {0}  sha256 {1}", path_TWD, Hash(path_TWD));
            W("  period   {0}", full ? "days 1..365 (full year)" : "day 1 (sizing check)");
            W("");

            SAM.Analytical.AnalyticalModel analyticalModel = SAM.Core.Convert.ToSAM<SAM.Analytical.AnalyticalModel>(path_SAM)[0];

            gbXMLSerializer.gbXML gbXML = SAM.Analytical.gbXML.Convert.TogbXML(analyticalModel, SAM.Core.Tolerance.MacroDistance, SAM.Core.Tolerance.Distance);
            SAM.Core.gbXML.Create.gbXML(gbXML, path_gbXML);

            W("gbXML written: {0} bytes", new FileInfo(path_gbXML).Length);

            List<SAM.Weather.WeatherData> weatherDatas = SAM.Weather.Tas.Convert.ToSAM_WeatherDatas(path_TWD);

            W("weather years read: {0}", weatherDatas == null ? 0 : weatherDatas.Count);

            if (weatherDatas == null || weatherDatas.Count == 0)
            {
                W("REFUSED: no weather data was read, so nothing simulatable can be produced.");
                File.WriteAllText(Path.Combine(directory, "phase-c.log"), log.ToString());
                return 6;
            }

            W("weather used: {0}", weatherDatas[0].Name);

            SAM.Analytical.Tas.WorkflowSettings workflowSettings = new SAM.Analytical.Tas.WorkflowSettings
            {
                Path_gbXML = path_gbXML,
                Path_TBD = path_TBD,
                WeatherData = weatherDatas[0],
                Sizing = true,
                Simulate = true,
                SimulateFrom = 1,
                SimulateTo = full ? 365 : 1,
                UpdateZones = true,
                UnmetHours = false,
                RemoveExistingTBD = true,
            };

            W("");
            W("== PRODUCTION Create.NoIzamThermalSource ==");

            SAM.Core.Tas.NoIzamThermalSource noIzamThermalSource = SAM.Analytical.Tas.Create.NoIzamThermalSource(
                analyticalModel, workflowSettings, out SAM.Analytical.AnalyticalModel analyticalModel_Result);

            foreach (string note in noIzamThermalSource.Notes)
            {
                W("  note: {0}", note);
            }

            foreach (string refusal in noIzamThermalSource.Refusals)
            {
                W("  REFUSAL: {0}", refusal);
            }

            if (noIzamThermalSource.SimulationEvidence != null)
            {
                foreach (string note in noIzamThermalSource.SimulationEvidence.Notes)
                {
                    W("  evidence: {0}", note);
                }

                foreach (string refusal in noIzamThermalSource.SimulationEvidence.Refusals)
                {
                    W("  evidence REFUSAL: {0}", refusal);
                }
            }

            W("  TBD  {0}", noIzamThermalSource.Path_TBD);
            W("  TSD  {0}", noIzamThermalSource.Path_TSD);
            W("  room identities: {0}", noIzamThermalSource.Count_ZoneReferences);
            W("  IsComplete: {0}", noIzamThermalSource.IsComplete);

            //The zone map, so every later process rebuilds the SAME source object without re-simulating.
            List<string> lines = new List<string>();
            foreach (KeyValuePair<Guid, string> keyValuePair in noIzamThermalSource.ZoneReferences)
            {
                lines.Add(string.Concat(keyValuePair.Key.ToString("D"), "\t", keyValuePair.Value));
            }

            lines.Sort(StringComparer.Ordinal);

            File.WriteAllLines(Path.Combine(directory, "src-zones.txt"), lines);

            W("");
            W("== provenance ==");
            W("  src.tbd  sha256 {0}  {1} bytes", Hash(path_TBD), File.Exists(path_TBD) ? new FileInfo(path_TBD).Length : 0);
            W("  src.tsd  sha256 {0}  {1} bytes", Hash(path_TSD), File.Exists(path_TSD) ? new FileInfo(path_TSD).Length : 0);

            File.WriteAllLines(Path.Combine(directory, "provenance.txt"), new string[]
            {
                string.Concat("fixture.sam\t", Hash(path_SAM)),
                string.Concat("weather.twd\t", Hash(path_TWD)),
                string.Concat("src.tbd\t", Hash(path_TBD)),
                string.Concat("src.tsd\t", Hash(path_TSD)),
            });

            if (noIzamThermalSource.IsComplete)
            {
                NoIzamEvidence(path_TBD);
            }

            W("");
            W("================================ PHASE C SUMMARY ================================");
            foreach (string verdict in verdicts)
            {
                W(verdict);
            }

            File.WriteAllText(Path.Combine(directory, "phase-c.log"), log.ToString());

            return noIzamThermalSource.IsComplete ? 0 : 6;
        }

        // ---------------------------------------------------------------- the no-IZAM evidence, items 1-3

        /// <summary>
        /// Reads the produced TBD back through TBD's OWN 0-based null-terminated accessors, so the
        /// evidence is independent of the code that wrote the file.
        /// </summary>
        private static void NoIzamEvidence(string path_TBD)
        {
            int count_IZAM = 0;
            int count_InternalCondition = 0;
            int count_TicVNonZero = 0;
            int count_TicIPresent = 0;
            int count_TicINonZero = 0;

            global::TBD.TBDDocument tBDDocument = new global::TBD.TBDDocument();

            try
            {
                tBDDocument.open(path_TBD);

                global::TBD.Building building = tBDDocument.Building;

                while (building.GetIZAM(count_IZAM) != null && count_IZAM < 100000)
                {
                    count_IZAM++;
                }

                int index = 0;
                global::TBD.InternalCondition internalCondition = building.GetIC(index);

                while (internalCondition != null && index < 100000)
                {
                    count_InternalCondition++;

                    global::TBD.InternalGain internalGain = internalCondition.GetInternalGain();

                    global::TBD.profile profile_V = internalGain?.GetProfile((int)global::TBD.Profiles.ticV);
                    if (profile_V != null && (profile_V.value != 0 || profile_V.factor != 0 || profile_V.setbackValue != 0 || profile_V.schedule != null))
                    {
                        count_TicVNonZero++;
                    }

                    global::TBD.profile profile_I = internalGain?.GetProfile((int)global::TBD.Profiles.ticI);
                    if (profile_I != null)
                    {
                        count_TicIPresent++;

                        if (profile_I.value != 0 || profile_I.setbackValue != 0)
                        {
                            count_TicINonZero++;
                        }
                    }

                    index++;
                    internalCondition = building.GetIC(index);
                }
            }
            finally
            {
                try { tBDDocument.close(); } catch { }
            }

            Item(1, "the thermal source carries no IZAM",
                count_IZAM == 0,
                string.Format("IZAM count {0} on {1}.", count_IZAM, path_TBD));

            Item(2, "the mechanical ventilation gain (ticV) is zero on every internal condition",
                count_TicVNonZero == 0,
                string.Format("{0} internal condition(s), {1} with a non-zero ticV.", count_InternalCondition, count_TicVNonZero));

            Item(3, "the infiltration term (ticI) is untouched and non-zero where the model states one",
                count_TicIPresent == count_InternalCondition && count_InternalCondition > 0 && count_TicINonZero > 0,
                string.Format("ticI present on {0} of {1} internal condition(s), non-zero on {2}.", count_TicIPresent, count_InternalCondition, count_TicINonZero));
        }
    }
}
