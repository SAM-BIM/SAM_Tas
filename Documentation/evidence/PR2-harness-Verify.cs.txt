// Proof that the PRODUCTION correction works: SAM.Analytical.Tas.TPD.Modify.SimulateSystems on a
// production-converted TPD, and then the ZoneTemperature series that come back from it.
//
// This is the regression the native experiment earned: the document-level Simulate answers
// "Sizing Flow Failed" on the same file, and the system-level one answers "Done" with results.

using System;
using System.Collections.Generic;
using System.Globalization;
using System.IO;
using System.Reflection;
using System.Text;

namespace Inv
{
    internal static class Verify
    {
        private static readonly StringBuilder log = new StringBuilder();

        private static void W(string format, params object[] args)
        {
            string line = args == null || args.Length == 0 ? format : string.Format(CultureInfo.InvariantCulture, format, args);
            Console.WriteLine(line);
            log.AppendLine(line);
        }

        private static double Dbl(object value)
        {
            try
            {
                return value == null ? double.NaN : Convert.ToDouble(value, CultureInfo.InvariantCulture);
            }
            catch
            {
                return double.NaN;
            }
        }

        public static int Run(string path_TPD, int startHour, int endHour)
        {
            log.Clear();
            W("verify {0}  hours {1}..{2}", path_TPD, startHour, endHour);

            if (!File.Exists(path_TPD))
            {
                W("REFUSED: TPD missing");
                Flush(path_TPD);
                return 5;
            }

            // ---------------------------------------------------------------- the OLD path
            W("");
            W("== PRODUCTION Modify.Simulate (document level - simulates the PLANT too) ==");

            bool ok_Document = SAM.Analytical.Tas.TPD.Modify.Simulate(
                path_TPD, startHour, endHour, out SAM.Core.Tas.SimulationEvidence evidence_Document);

            W("returned: {0}", ok_Document);
            W("NativeDiagnostic: [{0}]  kind={1}", evidence_Document.NativeDiagnostic ?? "<null>", evidence_Document.NativeDiagnosticKind);
            W("Completed: {0}", evidence_Document.Completed);
            foreach (string refusal in evidence_Document.Refusals)
            {
                W("  refusal: {0}", refusal);
            }

            // ---------------------------------------------------------------- the NEW path
            W("");
            W("== PRODUCTION Modify.SimulateSystems (air systems only) ==");

            bool ok_Systems = SAM.Analytical.Tas.TPD.Modify.SimulateSystems(
                path_TPD, startHour, endHour, out SAM.Core.Tas.SimulationEvidence evidence_Systems);

            W("returned: {0}", ok_Systems);
            W("NativeDiagnostic: [{0}]  kind={1}", evidence_Systems.NativeDiagnostic ?? "<null>", evidence_Systems.NativeDiagnosticKind);
            W("Completed (before reconciliation): {0}", evidence_Systems.Completed);
            foreach (string note in evidence_Systems.Notes)
            {
                W("  note: {0}", note);
            }
            foreach (string refusal in evidence_Systems.Refusals)
            {
                W("  refusal: {0}", refusal);
            }

            // ---------------------------------------------------------------- the results
            W("");
            W("== ZoneTemperature, read back from the same document ==");

            global::TPD.TPDDoc tPDDoc = new global::TPD.TPDDoc();
            int count_Complete = 0;
            int count_Zone = 0;

            try
            {
                tPDDoc.Open(path_TPD);
                object energyCentre = tPDDoc.EnergyCentre;

                int expected = endHour - startHour + 1;
                int count_PlantRoom = (int)Dbl(Dump.Call(energyCentre, "GetPlantRoomCount"));

                for (int p = 1; p <= count_PlantRoom; p++)
                {
                    object plantRoom = Dump.Call(energyCentre, "GetPlantRoom", p);
                    int count_System = (int)Dbl(Dump.Call(plantRoom, "GetSystemCount"));

                    for (int s = 1; s <= count_System; s++)
                    {
                        List<object> zones = new List<object>();
                        Collect(Dump.Call(plantRoom, "GetSystem", s), zones, 0);

                        foreach (object zone in zones)
                        {
                            count_Zone++;

                            object zoneLoad = Dump.Call(zone, "GetSystemZoneZoneLoad");

                            object data = null;
                            try
                            {
                                data = zone.GetType().InvokeMember(
                                    "GetResultsData",
                                    BindingFlags.InvokeMethod,
                                    null,
                                    zone,
                                    new object[]
                                    {
                                        (int)global::TPD.tpdResultsPeriod.tpdResultsPeriodHourly,
                                        (int)global::TPD.tpdCombinerType.tpdCombinerTypeMax,
                                        9,
                                        startHour + 1,
                                        expected,
                                    },
                                    CultureInfo.InvariantCulture);
                            }
                            catch (Exception exception)
                            {
                                Exception inner = exception.InnerException ?? exception;
                                W("  zone {0}: GetResultsData THREW {1}: {2}",
                                    Dump.Get(zone, "GUID"), inner.GetType().Name, inner.Message.Trim());
                                continue;
                            }

                            Array array = data as Array;
                            if (array == null)
                            {
                                W("  zone {0}: no series", Dump.Get(zone, "GUID"));
                                continue;
                            }

                            int finite = 0;
                            int total = 0;
                            double min = double.MaxValue;
                            double max = double.MinValue;

                            foreach (object element in array)
                            {
                                double value = Dbl(element);
                                total++;
                                if (!double.IsNaN(value) && !double.IsInfinity(value))
                                {
                                    finite++;
                                    if (value < min) { min = value; }
                                    if (value > max) { max = value; }
                                }
                            }

                            bool complete = total == expected && finite == expected;
                            if (complete)
                            {
                                count_Complete++;
                            }

                            W("  zone {0} load={1} guid={2}: {3}/{4} values, {5} finite, min={6} max={7}  {8}",
                                Dump.Get(zone, "GUID"),
                                Dump.Get(zoneLoad, "Name"),
                                Dump.Get(zoneLoad, "GUID"),
                                total,
                                expected,
                                finite,
                                min == double.MaxValue ? double.NaN : min,
                                max == double.MinValue ? double.NaN : max,
                                complete ? "COMPLETE" : "INCOMPLETE");
                        }
                    }
                }
            }
            finally
            {
                try { tPDDoc.Close(); } catch { }
            }

            W("");
            W("VERDICT: {0} of {1} zones returned a complete finite ZoneTemperature series for hours {2}..{3}",
                count_Complete, count_Zone, startHour, endHour);

            if (count_Zone > 0 && count_Complete == count_Zone)
            {
                evidence_Systems.RecordResultsReconciled(count_Zone, startHour, endHour);
                W("SimulationEvidence.Completed after reconciliation: {0}", evidence_Systems.Completed);
            }

            Flush(path_TPD);
            return 0;
        }

        private static void Collect(object container, List<object> zones, int depth)
        {
            if (container == null || depth > 8)
            {
                return;
            }

            for (int index = 1; index <= 4000; index++)
            {
                object systemComponent = Dump.Call(container, "GetComponent", index);
                if (systemComponent == null)
                {
                    break;
                }

                if (systemComponent is global::TPD.SystemZone)
                {
                    zones.Add(systemComponent);
                }

                if (systemComponent is global::TPD.ComponentGroup)
                {
                    Collect(systemComponent, zones, depth + 1);
                }
            }
        }

        private static void Flush(string path_TPD)
        {
            try
            {
                File.WriteAllText(Path.ChangeExtension(path_TPD, ".verify.log"), log.ToString());
            }
            catch
            {
            }
        }
    }
}
