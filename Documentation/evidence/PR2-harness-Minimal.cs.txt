// Step 4: a genuinely successful Systems simulation from a graph REPRESENTATIVE OF THE PR2 ROUTE.
//
// The MV.json template drags in a chiller, a DHW circuit, radiators and fancoil units - heating and
// cooling kit that a Part O Iteration 3 ventilation-only route does not have and does not want. Every
// converted zone carries ZoneComponentCount=2, and those zone components are themselves a sizing
// burden. So rather than keep repairing the template, this builds the minimum the brief specifies:
//
//   one real ZoneLoad, one SystemZone, explicit absolute supply flow, a fan, valid plant/system
//   topology, the load TSD, a simulation period, and the System Modelled flags.
//
// Both ITPD.Simulate and IEnergyCentre.SimulateExx are called, and both return strings are recorded,
// because SimulateExx exposes resetSizes/simSystem and may answer differently.

using System;
using System.Collections.Generic;
using System.Globalization;
using System.IO;
using System.Reflection;
using System.Text;

namespace Inv
{
    internal static class Minimal
    {
        private static readonly StringBuilder log = new StringBuilder();

        private static void W(string format, params object[] args)
        {
            string line = args == null || args.Length == 0 ? format : string.Format(CultureInfo.InvariantCulture, format, args);
            Console.WriteLine(line);
            log.AppendLine(line);
        }

        private static void SetProperty(object target, string name, object value)
        {
            target.GetType().InvokeMember(
                name, BindingFlags.SetProperty, null, target, new object[] { value }, CultureInfo.InvariantCulture);
        }

        private static double Dbl(object value)
        {
            try
            {
                return value == null ? double.NaN : Convert.ToDouble(value, CultureInfo.InvariantCulture);
            }
            catch
            {
                return double.NaN;
            }
        }

        private static void Try(string what, Action action)
        {
            try
            {
                action();
            }
            catch (Exception exception)
            {
                Exception inner = exception.InnerException ?? exception;
                W("  !! {0} FAILED: {1}: {2}", what, inner.GetType().Name, inner.Message.Replace(Environment.NewLine, " ").Trim());
            }
        }

        private static void Duct(object system, string label, object up, int portUp, object down, int portDown)
        {
            try
            {
                system.GetType().InvokeMember(
                    "AddDuct", BindingFlags.InvokeMethod, null, system,
                    new object[] { up, portUp, down, portDown }, CultureInfo.InvariantCulture);
                W("  duct {0}: ok", label);
            }
            catch (Exception exception)
            {
                Exception inner = exception.InnerException ?? exception;
                W("  duct {0}: THREW {1}: {2}", label, inner.GetType().Name, inner.Message.Replace(Environment.NewLine, " ").Trim());
            }
        }

        private static void Absolute(string label, object sizedFlowVariable, double value_Lps)
        {
            if (sizedFlowVariable == null)
            {
                W("  {0}: no carrier", label);
                return;
            }

            Try(label, delegate
            {
                SetProperty(sizedFlowVariable, "Value", value_Lps);
                SetProperty(sizedFlowVariable, "Type", (int)global::TPD.tpdSizedVariable.tpdSizedVariableValue);
            });

            W("  {0} -> Value={1} Type={2}", label, Dump.Get(sizedFlowVariable, "Value"), Dump.Get(sizedFlowVariable, "Type"));
        }

        /// <param name="plantKind">
        /// none = no plant component at all; electrical = one portless ElectricalGroup (nothing to
        /// connect, so it cannot be an invalid circuit).
        /// </param>
        public static int Run(string path_TPD, string path_TSD, string plantKind, int startHour, int endHour)
        {
            log.Clear();
            W("minimal {0}  tsd={1}  plant={2}  hours {3}..{4}", path_TPD, path_TSD, plantKind, startHour, endHour);

            if (!File.Exists(path_TSD))
            {
                W("REFUSED: TSD missing");
                Flush(path_TPD);
                return 5;
            }

            if (File.Exists(path_TPD))
            {
                File.Delete(path_TPD);
            }

            global::TPD.TPDDoc tPDDoc = new global::TPD.TPDDoc();
            try
            {
                tPDDoc.Create(path_TPD);

                object energyCentre = tPDDoc.EnergyCentre;
                if (energyCentre == null)
                {
                    W("REFUSED: no energy centre");
                    return 3;
                }

                Try("AddTSDData", delegate
                {
                    energyCentre.GetType().InvokeMember(
                        "AddTSDData", BindingFlags.InvokeMethod, null, energyCentre,
                        new object[] { path_TSD, 0 }, CultureInfo.InvariantCulture);
                });

                object tSDData = Dump.Call(energyCentre, "GetTSDData", 1);
                int count_ZoneLoad = (int)Dbl(Dump.Call(tSDData, "GetZoneLoadCount"));
                W("TSD attached: zone loads = {0}", count_ZoneLoad);

                if (count_ZoneLoad < 1)
                {
                    W("REFUSED: no zone loads");
                    return 6;
                }

                Try("energy centre hours", delegate
                {
                    SetProperty(energyCentre, "StartHour", startHour + 1);
                    SetProperty(energyCentre, "EndHour", endHour + 1);
                });
                W("hours: {0}..{1}", Dump.CallText(energyCentre, "GetStartHour"), Dump.CallText(energyCentre, "GetEndHour"));

                object plantRoom = Dump.Call(energyCentre, "AddPlantRoom");
                Try("plant room name", delegate { SetProperty(plantRoom, "Name", "PR"); });

                if (string.Equals(plantKind, "electrical", StringComparison.OrdinalIgnoreCase))
                {
                    // An ElectricalGroup declares no ports, so it satisfies "the plant room has
                    // components" without introducing a circuit that could be invalid.
                    object electricalGroup = Dump.Call(plantRoom, "AddElectricalGroup");
                    W("plant: AddElectricalGroup -> {0}", electricalGroup == null ? "NULL" : "ok");
                }
                else
                {
                    W("plant: none");
                }

                W("plantComponents = {0}", Dump.CallText(plantRoom, "GetComponentCount"));

                object system = Dump.Call(plantRoom, "AddSystem");
                Try("system name", delegate { SetProperty(system, "Name", "MVHR"); });

                W("");
                W("== one zone, bound to a real load, with an absolute supply duty ==");

                object zoneLoad = Dump.Call(tSDData, "GetZoneLoad", 1);
                object zone = Dump.Call(system, "AddSystemZone");

                Try("AddZoneLoad", delegate
                {
                    zone.GetType().InvokeMember(
                        "AddZoneLoad", BindingFlags.InvokeMethod, null, zone,
                        new object[] { zoneLoad }, CultureInfo.InvariantCulture);
                });

                W("  zone {0} -> load {1} volume {2} floorArea {3}",
                    Dump.Get(zone, "GUID"),
                    Dump.Get(Dump.Call(zone, "GetSystemZoneZoneLoad"), "Name"),
                    Dump.Get(zoneLoad, "Volume"),
                    Dump.Get(zoneLoad, "FloorArea"));

                W("  ZoneComponentCount = {0}  (a ventilation-only zone carries none)",
                    Dump.CallText(zone, "GetZoneComponentCount"));

                Absolute("zone FlowRate", Dump.Obj(zone, "FlowRate"), 17.0);
                Absolute("zone FreshAir", Dump.Obj(zone, "FreshAir"), 17.0);

                Try("zone flags", delegate
                {
                    int flags = Convert.ToInt32(Dump.Obj(zone, "Flags"), CultureInfo.InvariantCulture);
                    SetProperty(
                        zone,
                        "Flags",
                        flags
                            | (int)global::TPD.tpdSystemZoneFlags.tpdSystemZoneFlagModelVentFlow
                            | (int)global::TPD.tpdSystemZoneFlags.tpdSystemZoneFlagModelInterzoneFlow);
                });
                W("  zone flags = {0}", Dump.Get(zone, "Flags"));

                W("");
                W("== fans and the intake/exhaust junctions ==");

                object junction_Intake = Dump.Call(system, "AddJunction");
                object junction_Exhaust = Dump.Call(system, "AddJunction");
                object fan_Supply = Dump.Call(system, "AddFan");
                object fan_Extract = Dump.Call(system, "AddFan");

                foreach (object fan in new[] { fan_Supply, fan_Extract })
                {
                    Try("fan DesignFlowType -> AllAttachedZonesFlowRate", delegate
                    {
                        SetProperty(fan, "DesignFlowType", (int)global::TPD.tpdFlowRateType.tpdFlowRateAllAttachedZonesFlowRate);
                    });
                    Try("fan HeatGainFactor 0", delegate { SetProperty(fan, "HeatGainFactor", 0.0); });
                }

                Duct(system, "intake -> supply fan", junction_Intake, 1, fan_Supply, 1);
                Duct(system, "supply fan -> zone", fan_Supply, 1, zone, 1);
                Duct(system, "zone -> extract fan", zone, 1, fan_Extract, 1);
                Duct(system, "extract fan -> exhaust", fan_Extract, 1, junction_Exhaust, 1);

                tPDDoc.Save();
                W("saved {0} bytes", new FileInfo(path_TPD).Length);

                // ------------------------------------------------------------------ simulate

                W("");
                W("== ITPD.Simulate ==");
                string returned = null;
                try
                {
                    returned = tPDDoc.Simulate(startHour + 1, endHour + 1, 0);
                    W("RESULT Simulate: isNull={0} length={1} value=[{2}]",
                        returned == null, returned == null ? -1 : returned.Length, returned ?? "<null>");
                }
                catch (Exception exception)
                {
                    Exception inner = exception.InnerException ?? exception;
                    W("RESULT Simulate: THREW {0}: {1}", inner.GetType().Name, inner.Message.Replace(Environment.NewLine, " ").Trim());
                }

                W("");
                W("== IEnergyCentre.SimulateExx (resetSizes=1, simSystem=1) ==");
                try
                {
                    object returnedExx = energyCentre.GetType().InvokeMember(
                        "SimulateExx",
                        BindingFlags.InvokeMethod,
                        null,
                        energyCentre,
                        new object[] { startHour + 1, endHour + 1, 0, 1, 1, 0, null, 1, 1, 0 },
                        CultureInfo.InvariantCulture);

                    W("RESULT SimulateExx: isNull={0} value=[{1}]",
                        returnedExx == null, returnedExx == null ? "<null>" : returnedExx.ToString());
                }
                catch (Exception exception)
                {
                    Exception inner = exception.InnerException ?? exception;
                    W("RESULT SimulateExx: THREW {0}: {1}", inner.GetType().Name, inner.Message.Replace(Environment.NewLine, " ").Trim());
                }

                tPDDoc.Save();

                // ------------------------------------------------------------------ results

                W("");
                W("== ZoneTemperature probe ==");
                W("  zone FlowRate now Value={0} Type={1}",
                    Dump.Get(Dump.Obj(zone, "FlowRate"), "Value"), Dump.Get(Dump.Obj(zone, "FlowRate"), "Type"));

                foreach (object fan in new[] { fan_Supply, fan_Extract })
                {
                    W("  fan DesignFlowRate Value={0} Type={1}",
                        Dump.Get(Dump.Obj(fan, "DesignFlowRate"), "Value"),
                        Dump.Get(Dump.Obj(fan, "DesignFlowRate"), "Type"));
                }

                Probe(zone, startHour, endHour);
            }
            finally
            {
                try { tPDDoc.Close(); } catch { }
            }

            Flush(path_TPD);
            return 0;
        }

        private static void Probe(object zone, int startHour, int endHour)
        {
            object data = null;
            try
            {
                data = zone.GetType().InvokeMember(
                    "GetResultsData",
                    BindingFlags.InvokeMethod,
                    null,
                    zone,
                    new object[]
                    {
                        (int)global::TPD.tpdResultsPeriod.tpdResultsPeriodHourly,
                        (int)global::TPD.tpdCombinerType.tpdCombinerTypeMax,
                        9,
                        startHour + 1,
                        endHour - startHour + 1,
                    },
                    CultureInfo.InvariantCulture);
            }
            catch (Exception exception)
            {
                Exception inner = exception.InnerException ?? exception;
                W("  GetResultsData THREW {0}: {1}", inner.GetType().Name, inner.Message.Replace(Environment.NewLine, " ").Trim());
                return;
            }

            if (data == null)
            {
                W("  GetResultsData -> <null>  (NO RESULTS)");
                return;
            }

            Array array = data as Array;
            if (array == null)
            {
                W("  GetResultsData -> {0} (not an array)", data.GetType().Name);
                return;
            }

            int finite = 0;
            double min = double.MaxValue;
            double max = double.MinValue;
            StringBuilder sample = new StringBuilder();

            for (int i = 0; i < array.Length; i++)
            {
                double value = Dbl(array.GetValue(i));
                if (!double.IsNaN(value) && !double.IsInfinity(value))
                {
                    finite++;
                    if (value < min) { min = value; }
                    if (value > max) { max = value; }
                }

                if (i < 6)
                {
                    sample.Append(value.ToString("F3", CultureInfo.InvariantCulture)).Append(' ');
                }
            }

            W("  ZoneTemperature: {0} values, {1} FINITE, min={2} max={3}",
                array.Length,
                finite,
                min == double.MaxValue ? double.NaN : min,
                max == double.MinValue ? double.NaN : max);
            W("  first values: {0}", sample.ToString().Trim());
        }

        private static void Flush(string path_TPD)
        {
            try
            {
                File.WriteAllText(Path.ChangeExtension(path_TPD, ".min.log"), log.ToString());
            }
            catch
            {
            }
        }
    }
}
