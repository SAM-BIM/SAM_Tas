import pandas as pd, numpy as np
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt
W='C:/TasOut/b0w/'
D=pd.read_csv(W+'out/hourly-dataset.csv')
A=pd.read_csv(W+'A.csv')
SH=pd.read_csv(W+'out/shift-analysis.csv')
DU=pd.read_csv('C:/TasOut/b0r4/ducts.csv')
BOUND=['Bathroom_2','Studio 1_0','Bedroom 2_3','Ensuite_5','Kitchen_4','Bedroom 2_6','Ensuite_8','Kitchen_7']
MON=['Jan','Feb','Mar','Apr','May','Jun','Jul','Aug','Sep','Oct','Nov','Dec']

# 1 RMSE vs shift
f,ax=plt.subplots(1,2,figsize=(8,3),constrained_layout=True)
for i,(pair,t) in enumerate([('A_air vs B_zone(achieved air)','A air vs B0 ZoneTemperature'),('A_res vs B_res','A vs B0 ResultantTemperature')]):
    s=SH[(SH.pair==pair)&(SH.scope=='pooled')].sort_values('shift')
    ax[i].plot(s['shift'],s.rmse,'o-',color='#1f77b4',label='RMSE')
    ax[i].plot(s['shift'],s.mae,'s--',color='#ff7f0e',label='MAE')
    ax[i].axvline(0,color='k',lw=0.6)
    ax[i].set_title(t+'\nminimum at k=0 (no residual time shift)',fontsize=8)
    ax[i].set_xlabel('shift k applied to B, comparing A(t) with B(t+k)  [hours]'); ax[i].set_ylabel('K'); ax[i].legend()
f.savefig(W+'out/fig1-rmse-vs-shift.png'); plt.close(f)

# 2 monthly
pool=np.concatenate([D[D.room==r].d_res.values for r in BOUND]).reshape(len(BOUND),8760)
bias=[pool[:,mon==i].mean() for i in range(12)]; rmse=[np.sqrt((pool[:,mon==i]**2).mean()) for i in range(12)]
f,ax=plt.subplots(figsize=(7,3),constrained_layout=True)
x=np.arange(12)
ax.bar(x-0.2,bias,0.4,label='bias B-A',color='#1f77b4')
ax.bar(x+0.2,rmse,0.4,label='RMSE',color='#aec7e8')
ax2=ax.twinx(); ax2.plot(x,[odb[mon==i].mean() for i in range(12)],'k.-',lw=1,label='mean ODB'); ax2.set_ylabel('mean outdoor dry bulb (C)'); ax2.grid(False)
ax.axhline(0,color='k',lw=0.6); ax.set_xticks(x); ax.set_xticklabels(MON); ax.set_ylabel('K')
ax.set_title('Monthly A/B ResultantTemperature difference, 8 bound rooms\nthe +0.373 K annual bias is a heating-season effect; warm months are slightly negative',fontsize=8)
ax.legend(loc='upper center'); ax2.legend(loc='upper right')
f.savefig(W+'out/fig2-monthly.png'); plt.close(f)

# 3 critical windows
def dcol(sub,var=3):
    m=[c for c in DU.columns if sub in c and c.endswith('|var%d'%var)]
    return DU[m[0]].values if len(m)==1 else None
f,ax=plt.subplots(2,1,figsize=(8,5),constrained_layout=True)
for k,(h0,h1,title) in enumerate([(2900,3040,'hour 2968 - the annual maximum of the pooled |B-A|, early May'),
                                  (5270,5520,'hour 5491 / 5299 region - August evenings, wet rooms')]):
    sl=slice(h0,h1)
    for r,c in zip(['Bedroom 2_6','Kitchen_7','Ensuite_8'],['#1f77b4','#2ca02c','#d62728']):
        d=D[D.room==r].reset_index(drop=True)
        ax[k].plot(range(h0,h1),d.A_res.values[sl],c=c,lw=1.0,label='A '+r)
        ax[k].plot(range(h0,h1),d.B_res.values[sl],c=c,lw=1.0,ls='--',label='B0 '+r)
    ax[k].plot(range(h0,h1),odb[sl],color='0.5',lw=0.9,label='outdoor')
    ax[k].plot(range(h0,h1),A['{D301D74D-D9CD-4080-88C5-D374A8CD5A02}|dryBulbTemp'].values[sl],color='0.2',lw=0.9,ls=':',label='A MVHR-03 plenum (= A supply air)')
    for hh in ([2968] if k==0 else [5299,5491]): ax[k].axvline(hh,color='k',lw=0.6,alpha=0.6)
    ax[k].set_title(title,fontsize=8); ax[k].set_ylabel('C'); ax[k].set_xlabel('hour of year (0-based)')
    ax[k].legend(ncol=4,fontsize=6)
f.savefig(W+'out/fig3-critical-windows.png'); plt.close(f)
