        private static readonly StringBuilder log = new StringBuilder();

        private static void W(string format, params object[] args)
        {
            string line = args == null || args.Length == 0 ? format : string.Format(CultureInfo.InvariantCulture, format, args);
            Console.WriteLine(line);
            log.AppendLine(line);
        }

        private static void Usage()
        {
            W("modes:");
            W("  make-empty-tpd <out.tpd>              a TPD with an energy centre but nothing in it");
            W("  simulate-tpd   <in.tpd> <start> <end> call the PRODUCTION Modify.Simulate, print what it returned");
            W("  probe-tpd      <out.tpd>              build a minimal system, measure identity/units/flags");
            W("  dump-tpd       <in.tpd>               walk an existing TPD and report what it holds");
            W("  togbxml        <in.sam> <out.xml>");
            W("  noizam         <in.sam> <in.xml> <out.tbd> <weather.twd> <sizing|full>");
            W("  dump-tbd       <in.tbd>               IZAM / ticV / ticI inventory");
            W("  dump-graph     <in.tpd> <out.txt>     exhaustive late-bound native graph dump");
            W("  build-ref      <out.tpd> [s eo bs be t1 t2]   reference system with asymmetric flows");
        }

        private static int Main(string[] args)
        {
            int exitCode;
            try
            {
                exitCode = Run(args);
            }
            catch (Exception exception)
            {
                W("EXCEPTION {0}: {1}", exception.GetType().Name, exception.Message);
                W(exception.StackTrace ?? string.Empty);
                exitCode = 2;
            }

            try
            {
                File.WriteAllText(Path.Combine(@"C:\TasOut\inv", "last-run.log"), log.ToString());
            }
            catch
            {
            }

            return exitCode;
        }

        private static int Run(string[] args)
        {
            if (args == null || args.Length == 0)
            {
                Usage();
                return 1;
            }

            switch (args[0].ToLowerInvariant())
            {
                case "make-empty-tpd":
                    return MakeEmptyTpd(args[1]);
                case "simulate-tpd":
                    return SimulateTpd(args[1], int.Parse(args[2]), int.Parse(args[3]));
                case "probe-tpd":
                    return ProbeTpd(args[1]);
                case "dump-tpd":
                    return DumpTpd(args[1]);
                case "togbxml":
                    return ToGbXml(args[1], args[2]);
                case "noizam":
                    return NoIzam(args[1], args[2], args[3], args[4], args[5], args.Length > 6 && string.Equals(args[6], "control", StringComparison.OrdinalIgnoreCase));
                case "dump-tbd":
                    return DumpTbd(args[1]);
                case "dump-graph":
                    return DumpGraph(args[1], args[2]);
                case "verify":
                    return Verify.Run(args[1], int.Parse(args[2]), int.Parse(args[3]));
                case "route-source":
                    return Route.Source(args[1], args[2], args[3], args.Length > 4 ? args[4] : "sizing");
                case "route-acceptance":
                    return Route.Run(args[1], args[2], args[3], args.Length > 4 ? args[4] : "sizing");
                case "fanout":
                    return FanOut.Run(args[1], int.Parse(args[2]));
                case "zoneload-identity":
                    return Ident.Run(args[1], args[2], args[3]);
                case "minimal":
                    return Minimal.Run(args[1], args[2], args[3], int.Parse(args[4]), int.Parse(args[5]));
                case "exp":
                    return Experiment.Run(args[1], args[2], int.Parse(args[3]), int.Parse(args[4]));
                case "convert-template":
                    return ConvertRef.Build(args[1], args[2], args[3]);
                case "prove":
                    return ConvertRef.Prove(args[1], int.Parse(args[2]), int.Parse(args[3]), int.Parse(args[4]));
                case "seed":
                    return Seed.SeedTbd(args[1], args[2], args.Length > 3 ? int.Parse(args[3]) : 2, args.Length > 4 ? double.Parse(args[4], System.Globalization.CultureInfo.InvariantCulture) : 5.0);
                case "clean":
                    return Seed.CleanTbd(args[1]);
                case "resize":
                    return Resize.Run(args[1], args[2], int.Parse(args[3]));
                case "attach-tsd":
                    return Attach.Run(args[1], args[2], args.Length > 3 ? int.Parse(args[3]) : 8);
                case "build-ref2":
                    return BuildRef2.Build(args[1], Ref.Supply_Only_Lps, Ref.Extract_Only_Lps, Ref.Both_Supply_Lps, Ref.Both_Extract_Lps, Ref.Transfer_Lps, Ref.Transfer_Second_Lps);
                case "build-ref":
                    return BuildRef.Build(
                        args[1],
                        args.Length > 2 ? double.Parse(args[2], System.Globalization.CultureInfo.InvariantCulture) : Ref.Supply_Only_Lps,
                        args.Length > 3 ? double.Parse(args[3], System.Globalization.CultureInfo.InvariantCulture) : Ref.Extract_Only_Lps,
                        args.Length > 4 ? double.Parse(args[4], System.Globalization.CultureInfo.InvariantCulture) : Ref.Both_Supply_Lps,
                        args.Length > 5 ? double.Parse(args[5], System.Globalization.CultureInfo.InvariantCulture) : Ref.Both_Extract_Lps,
                        args.Length > 6 ? double.Parse(args[6], System.Globalization.CultureInfo.InvariantCulture) : Ref.Transfer_Lps,
                        args.Length > 7 ? double.Parse(args[7], System.Globalization.CultureInfo.InvariantCulture) : Ref.Transfer_Second_Lps);
                default:
                    Usage();
                    return 1;
            }
        }

        // ---------------------------------------------------------------- measurement helpers

        /// <summary>Runs one measurement, reporting a COM refusal instead of aborting the whole probe.</summary>
        private static void Try(string what, Action action)
        {
            try
            {
                action();
            }
            catch (Exception exception)
            {
                W("  !! {0} FAILED: {1}: {2}", what, exception.GetType().Name, exception.Message);
            }
        }

        private static void TryW(string what, Func<string> func)
        {
            try
            {
                W("  {0}: {1}", what, func());
            }
            catch (Exception exception)
            {
                W("  !! {0} FAILED: {1}: {2}", what, exception.GetType().Name, exception.Message);
            }
        }

        /// <summary>Reads a value that may be an unset COM variant, reporting what happened rather than throwing.</summary>
        private static string Describe(Func<string> func)
        {
            try
            {
                string value = func();
                return value == null ? "<null>" : string.Concat("[", value, "] len=", value.Length);
            }
            catch (Exception exception)
            {
                return string.Concat("<THREW ", exception.GetType().Name, ": ", exception.Message, ">");
            }
        }

        /// <summary>Exactly what production does: a late-bound GUID read off a dynamic reference.</summary>
        private static string DescribeDynamic(object systemComponent)
        {
            try
            {
                dynamic @dynamic = systemComponent;
                object value = @dynamic.GUID;
                return value == null
                    ? "<null>"
                    : string.Concat("[", value, "] runtimeType=", value.GetType().FullName);
            }
            catch (Exception exception)
            {
                return string.Concat("<THREW ", exception.GetType().Name, ": ", exception.Message, ">");
            }
        }

        private static List<TPD.System> Systems(TPD.PlantRoom plantRoom)
        {
            List<TPD.System> result = new List<TPD.System>();
            int count = plantRoom.GetSystemCount();
            for (int i = 1; i <= count; i++)
            {
                TPD.System system = plantRoom.GetSystem(i);
                if (system != null)
                {
                    result.Add(system);
                }
            }

            return result;
        }

        private static List<TPD.SystemComponent> Components(TPD.System system)
        {
            List<TPD.SystemComponent> result = new List<TPD.SystemComponent>();
            for (int index = 1; index <= 5000; index++)
            {
                TPD.SystemComponent systemComponent = null;
                try
                {
                    systemComponent = system.GetComponent(index) as TPD.SystemComponent;
                }
                catch
                {
                    break;
                }

                if (systemComponent == null)
                {
                    break;
                }

                result.Add(systemComponent);
            }

            return result;
        }

        private static int MakeEmptyTpd(string path_TPD)
        {
            W("make-empty-tpd {0}", path_TPD);

            if (File.Exists(path_TPD))
            {
                File.Delete(path_TPD);
            }

            TPD.TPDDoc tPDDoc = new TPD.TPDDoc();
            try
            {
                tPDDoc.Create(path_TPD);
                W("EnergyCentre after Create: {0}", tPDDoc.EnergyCentre == null ? "NULL" : "present");
                if (tPDDoc.EnergyCentre != null)
                {
                    W("PlantRoomCount: {0}", tPDDoc.EnergyCentre.GetPlantRoomCount());
                }

                tPDDoc.Save();
            }
            finally
            {
                try { tPDDoc.Close(); } catch { }
            }

            W("exists after save: {0}, length {1}", File.Exists(path_TPD), new FileInfo(path_TPD).Length);
            return 0;
        }

        /// <summary>
        /// Calls the PRODUCTION SAM.Analytical.Tas.TPD.Modify.Simulate and reports what it returned alongside
        /// what the document actually holds. This is the E1 reproduction.
        /// </summary>
        private static int SimulateTpd(string path_TPD, int startHour, int endHour)
        {
            W("simulate-tpd {0} {1}..{2}", path_TPD, startHour, endHour);
            W("exists before: {0}", File.Exists(path_TPD));

            if (File.Exists(path_TPD))
            {
                FileInfo fileInfo = new FileInfo(path_TPD);
                W("before: {0} bytes, written {1:o}", fileInfo.Length, fileInfo.LastWriteTimeUtc);
            }

            DateTime utcStarted = DateTime.UtcNow;
            W("UtcStarted {0:o}", utcStarted);

            bool result = SAM.Analytical.Tas.TPD.Modify.Simulate(path_TPD, startHour, endHour);

            W("PRODUCTION Modify.Simulate RETURNED: {0}", result);

            if (File.Exists(path_TPD))
            {
                FileInfo fileInfo = new FileInfo(path_TPD);
                W("after: {0} bytes, written {1:o}", fileInfo.Length, fileInfo.LastWriteTimeUtc);
            }

            string path_ErrorLog = Path.Combine(
                Path.GetDirectoryName(path_TPD) ?? string.Empty,
                Path.GetFileNameWithoutExtension(path_TPD) + "_error_log.txt");

            W("error log {0}: {1}", path_ErrorLog, File.Exists(path_ErrorLog) ? "PRESENT" : "absent");
            if (File.Exists(path_ErrorLog))
            {
                W("error log text: {0}", File.ReadAllText(path_ErrorLog).Trim());
            }

            return 0;
        }

        /// <summary>
        /// Builds the smallest possible TPD system by hand and measures the facts PR2 must not guess: which
        /// GUID a SystemZone answers, whether it equals its zone load's, what the flow carriers accept, what
        /// the zone flags read back as, and what a duct exposes.
        /// </summary>
        private static int ProbeTpd(string path_TPD)
        {
            W("probe-tpd {0}", path_TPD);

            if (File.Exists(path_TPD))
            {
                File.Delete(path_TPD);
            }

            TPD.TPDDoc tPDDoc = new TPD.TPDDoc();
            try
            {
                tPDDoc.Create(path_TPD);

                TPD.EnergyCentre energyCentre = tPDDoc.EnergyCentre;
                W("EnergyCentre: {0}", energyCentre == null ? "NULL" : "present");
                if (energyCentre == null)
                {
                    return 3;
                }

                TPD.PlantRoom plantRoom = energyCentre.AddPlantRoom();
                Try("PlantRoom.Name=PR1", delegate { plantRoom.Name = "PR1"; });

                TPD.System system = plantRoom.AddSystem();
                Try("System.Name=AHU1", delegate { system.Name = "AHU1"; });
                TryW("System.GUID", delegate { return Describe(delegate { return system.GUID; }); });

                TPD.SystemZone systemZone = system.AddSystemZone() as TPD.SystemZone;
                W("AddSystemZone -> {0}", systemZone == null ? "NULL" : "SystemZone");
                if (systemZone == null)
                {
                    return 4;
                }

                TPD.SystemComponent zoneAsComponent = systemZone as TPD.SystemComponent;

                W("---- IDENTITY: what does a SystemZone answer for GUID? ----");
                TryW("zone GetComponentType()", delegate { return zoneAsComponent.GetComponentType().ToString(CultureInfo.InvariantCulture); });
                TryW("zone .Name (typed)", delegate { return Describe(delegate { return zoneAsComponent.Name; }); });
                TryW("zone .GUID (typed string)", delegate { return Describe(delegate { return zoneAsComponent.GUID; }); });
                TryW("zone .GUID (dynamic - what production uses)", delegate { return DescribeDynamic(zoneAsComponent); });
                TryW("zone GetZoneLoadCount()", delegate { return zoneAsComponent.GetZoneLoadCount().ToString(CultureInfo.InvariantCulture); });
                TryW("zone GetSystemZoneZoneLoad()", delegate { return systemZone.GetSystemZoneZoneLoad() == null ? "NULL" : "present"; });

                Try("zone load identity", delegate
                {
                    TPD.ZoneLoad zoneLoad = systemZone.GetSystemZoneZoneLoad();
                    if (zoneLoad == null)
                    {
                        W("  (no zone load: none attached until a TSD supplies one)");
                        return;
                    }

                    W("  ZoneLoad.GUID: {0}", Describe(delegate { return zoneLoad.GUID; }));
                    W("  ZoneLoad.Name: {0}", Describe(delegate { return zoneLoad.Name; }));
                });

                W("---- FLOW CARRIERS: the only two a zone has ----");
                Try("SystemZone.FlowRate", delegate
                {
                    TPD.SizedFlowVariable variable = systemZone.FlowRate;
                    W("  FlowRate object: {0}", variable == null ? "NULL" : "present");
                    if (variable == null)
                    {
                        return;
                    }

                    W("  Value before: {0}", Describe(delegate { return variable.Value.ToString(CultureInfo.InvariantCulture); }));
                    Try("  set FlowRate.Value=42", delegate { variable.Value = 42.0; });
                    W("  Value after set 42: {0}", Describe(delegate { return variable.Value.ToString(CultureInfo.InvariantCulture); }));
                    W("  Method: {0}", Describe(delegate { return variable.Method.ToString(); }));
                    W("  Type: {0}", Describe(delegate { return variable.Type.ToString(); }));
                });

                Try("SystemZone.FreshAir", delegate
                {
                    TPD.SizedFlowVariable variable = systemZone.FreshAir;
                    W("  FreshAir object: {0}", variable == null ? "NULL" : "present");
                    if (variable == null)
                    {
                        return;
                    }

                    Try("  set FreshAir.Value=17", delegate { variable.Value = 17.0; });
                    W("  Value after set 17: {0}", Describe(delegate { return variable.Value.ToString(CultureInfo.InvariantCulture); }));
                });

                W("---- FLAGS ----");
                foreach (TPD.tpdSystemZoneFlags flag in Enum.GetValues(typeof(TPD.tpdSystemZoneFlags)))
                {
                    W("  enum tpdSystemZoneFlags.{0} = {1}", flag, (int)flag);
                }

                TryW("Flags before", delegate { return systemZone.Flags.ToString(CultureInfo.InvariantCulture); });
                Try("set ModelVentFlow|ModelInterzoneFlow", delegate
                {
                    systemZone.Flags = systemZone.Flags
                        | (int)TPD.tpdSystemZoneFlags.tpdSystemZoneFlagModelVentFlow
                        | (int)TPD.tpdSystemZoneFlags.tpdSystemZoneFlagModelInterzoneFlow;
                });
                TryW("Flags after", delegate { return systemZone.Flags.ToString(CultureInfo.InvariantCulture); });

                W("---- DUCT SURFACE: any design-flow carrier on a leg? ----");
                TryW("GetSystemZoneInputDuct()", delegate { return systemZone.GetSystemZoneInputDuct() == null ? "NULL" : "present"; });
                TryW("GetSystemZoneOutputDuct()", delegate { return systemZone.GetSystemZoneOutputDuct() == null ? "NULL" : "present"; });
                TryW("GetZoneComponentCount()", delegate { return systemZone.GetZoneComponentCount().ToString(CultureInfo.InvariantCulture); });

                W("---- FAN ----");
                TPD.Fan fan = null;
                Try("AddFan", delegate { fan = system.AddFan() as TPD.Fan; });

                if (fan != null)
                {
                    TPD.SystemComponent fanAsComponent = fan as TPD.SystemComponent;
                    Try("Fan.Name=SupplyFan", delegate { fanAsComponent.Name = "SupplyFan"; });
                    TryW("Fan .GUID (typed)", delegate { return Describe(delegate { return fanAsComponent.GUID; }); });
                    TryW("Fan .GUID (dynamic)", delegate { return DescribeDynamic(fanAsComponent); });
                    TryW("Fan.DesignFlowType default", delegate { return fan.DesignFlowType.ToString(); });

                    foreach (TPD.tpdFlowRateType flowRateType in Enum.GetValues(typeof(TPD.tpdFlowRateType)))
                    {
                        W("  enum tpdFlowRateType.{0} = {1}", flowRateType, (int)flowRateType);
                    }

                    Try("Fan.DesignFlowRate", delegate
                    {
                        TPD.SizedFlowVariable variable = fan.DesignFlowRate;
                        W("  DesignFlowRate object: {0}", variable == null ? "NULL" : "present");
                        if (variable == null)
                        {
                            return;
                        }

                        Try("  set DesignFlowRate.Value=123", delegate { variable.Value = 123.0; });
                        W("  Value after set 123: {0}", Describe(delegate { return variable.Value.ToString(CultureInfo.InvariantCulture); }));
                    });

                    W("---- DUCT between fan and zone ----");
                    Try("AddDuct(fan,1 -> zone,1)", delegate
                    {
                        TPD.Duct duct = system.AddDuct(fanAsComponent, 1, zoneAsComponent, 1);
                        W("  duct: {0}", duct == null ? "NULL" : "present");
                        if (duct == null)
                        {
                            return;
                        }

                        W("  upstream: {0}", Describe(delegate { return duct.GetUpstreamComponent().Name; }));
                        W("  duct.GetFlowRate(1): {0}", Describe(delegate { return duct.GetFlowRate(1).ToString(CultureInfo.InvariantCulture); }));
                    });
                }

                W("---- ROUND TRIP BY GUID ----");
                Try("GetComponentByGUID", delegate
                {
                    string guid = null;
                    try
                    {
                        guid = zoneAsComponent.GUID;
                    }
                    catch
                    {
                    }

                    if (string.IsNullOrEmpty(guid))
                    {
                        W("  zone GUID unreadable or empty - cannot round trip the zone");
                        return;
                    }

                    TPD.SystemComponent back = system.GetComponentByGUID(guid) as TPD.SystemComponent;
                    W("  GetComponentByGUID([{0}]) -> {1}", guid, back == null ? "NULL" : "resolved");
                });

                W("---- COMPONENT WALK ----");
                foreach (TPD.SystemComponent systemComponent in Components(system))
                {
                    W(
                        "  component type={0} name={1} guid={2}",
                        Describe(delegate { return systemComponent.GetComponentType().ToString(CultureInfo.InvariantCulture); }),
                        Describe(delegate { return systemComponent.Name; }),
                        Describe(delegate { return systemComponent.GUID; }));
                }

                W("---- SAVE ----");
                tPDDoc.Save();
                W("saved; length {0}", new FileInfo(path_TPD).Length);
            }
            finally
            {
                try { tPDDoc.Close(); } catch { }
            }

            return 0;
        }

        private static SAM.Analytical.AnalyticalModel Read(string path_SAM)
        {
            SAM.Analytical.AnalyticalModel result = SAM.Core.Convert.ToSAM<SAM.Analytical.AnalyticalModel>(path_SAM)[0];
            W("model: {0} spaces", result.AdjacencyCluster.GetSpaces().Count);
            return result;
        }

        private static int ToGbXml(string path_SAM, string path_gbXML)
        {
            W("togbxml {0} -> {1}", path_SAM, path_gbXML);
            SAM.Analytical.AnalyticalModel analyticalModel = Read(path_SAM);
            gbXMLSerializer.gbXML gbXML = SAM.Analytical.gbXML.Convert.TogbXML(analyticalModel, SAM.Core.Tolerance.MacroDistance, SAM.Core.Tolerance.Distance);
            SAM.Core.gbXML.Create.gbXML(gbXML, path_gbXML);
            W("gbXML written: {0} bytes", new FileInfo(path_gbXML).Length);
            return 0;
        }

        /// <summary>
        /// The Part O Iteration 3 thermal source: the normal workflow with IZAM creation OFF, the IZAM sweep
        /// ON and the mechanical ventilation gain zeroed.
        /// </summary>
        private static int NoIzam(string path_SAM, string path_gbXML, string path_TBD, string path_TWD, string mode, bool control)
        {
            W("noizam {0} mode={1} control={2}", path_TBD, mode, control);

            SAM.Analytical.AnalyticalModel analyticalModel = Read(path_SAM);

            List<SAM.Weather.WeatherData> weatherDatas = SAM.Weather.Tas.Convert.ToSAM_WeatherDatas(path_TWD);
            W("weather years: {0}", weatherDatas == null ? 0 : weatherDatas.Count);
            if (weatherDatas == null || weatherDatas.Count == 0)
            {
                W("REFUSED: no weather data read from {0}", path_TWD);
                return 5;
            }

            bool full = string.Equals(mode, "full", StringComparison.OrdinalIgnoreCase);

            SAM.Analytical.Tas.WorkflowSettings workflowSettings = new SAM.Analytical.Tas.WorkflowSettings
            {
                Path_gbXML = path_gbXML,
                Path_TBD = path_TBD,
                WeatherData = weatherDatas[0],
                Sizing = true,
                Simulate = full,
                SimulateFrom = 1,
                SimulateTo = full ? 365 : 1,
                UpdateZones = true,
                UnmetHours = false,
                RemoveExistingTBD = true,

                // control = the pre-Iteration-3 behaviour, so the no-IZAM numbers can be compared to something.
                AddIZAMs = control,
                RemoveIZAMs = !control,
                RemoveMechanicalVentilationGains = !control,
            };

            SAM.Analytical.Tas.WorkflowCalculator workflowCalculator = new SAM.Analytical.Tas.WorkflowCalculator(workflowSettings);
            SAM.Analytical.AnalyticalModel result = workflowCalculator.Calculate(analyticalModel);

            W("workflow returned: {0}", result == null ? "NULL" : "model");

            if (workflowCalculator.Notes != null)
            {
                foreach (string note in workflowCalculator.Notes)
                {
                    W("  note: {0}", note);
                }
            }

            W("TBD exists: {0}", File.Exists(path_TBD));
            if (File.Exists(path_TBD))
            {
                W("TBD length: {0}", new FileInfo(path_TBD).Length);
            }

            string path_TSD = Path.Combine(
                Path.GetDirectoryName(path_TBD) ?? string.Empty,
                Path.GetFileNameWithoutExtension(path_TBD) + ".tsd");
            W("TSD {0}: {1}", path_TSD, File.Exists(path_TSD) ? new FileInfo(path_TSD).Length + " bytes" : "absent");

            return 0;
        }

        /// <summary>
        /// Reads a TBD back INDEPENDENTLY of the code under test, walking TAS's own accessors: are there any
        /// IZAMs left, is every ticV zero, and is ticI still there?
        /// </summary>
        private static int DumpTbd(string path_TBD)
        {
            W("dump-tbd {0}", path_TBD);

            TBD.TBDDocument tBDDocument = new TBD.TBDDocument();
            try
            {
                tBDDocument.open(path_TBD);
                TBD.Building building = tBDDocument.Building;
                W("Building: {0}", building == null ? "NULL" : "present");
                if (building == null)
                {
                    return 0;
                }

                int count_IZAM = 0;
                TBD.IZAM iZAM = building.GetIZAM(0);
                while (iZAM != null)
                {
                    W("  IZAM[{0}] name={1}", count_IZAM, iZAM.name);
                    count_IZAM++;
                    iZAM = building.GetIZAM(count_IZAM);
                    if (count_IZAM > 10000) { break; }
                }
                W("IZAM COUNT: {0}", count_IZAM);

                int index = 0;
                int count_IC = 0;
                int count_TicVNonZero = 0;
                int count_TicIPresent = 0;
                TBD.InternalCondition internalCondition = building.GetIC(index);
                while (internalCondition != null)
                {
                    count_IC++;
                    TBD.InternalGain internalGain = internalCondition.GetInternalGain();
                    if (internalGain != null)
                    {
                        TBD.profile profile_TicV = internalGain.GetProfile((int)TBD.Profiles.ticV);
                        TBD.profile profile_TicI = internalGain.GetProfile((int)TBD.Profiles.ticI);

                        double ticV_value = profile_TicV == null ? 0 : profile_TicV.value;
                        double ticV_factor = profile_TicV == null ? 0 : profile_TicV.factor;
                        double ticV_setback = profile_TicV == null ? 0 : profile_TicV.setbackValue;
                        bool ticV_schedule = profile_TicV != null && profile_TicV.schedule != null;

                        double ticI_value = profile_TicI == null ? 0 : profile_TicI.value;
                        double ticI_factor = profile_TicI == null ? 0 : profile_TicI.factor;

                        bool ticV_nonZero = ticV_value != 0 || ticV_factor != 0 || ticV_setback != 0 || ticV_schedule;
                        if (ticV_nonZero) { count_TicVNonZero++; }
                        if (ticI_value != 0 || ticI_factor != 0) { count_TicIPresent++; }

                        W(
                            "  IC[{0}] name={1} ticV(v={2} f={3} sb={4} sched={5}) ticI(v={6} f={7})",
                            index,
                            internalCondition.name,
                            ticV_value, ticV_factor, ticV_setback, ticV_schedule,
                            ticI_value, ticI_factor);
                    }

                    index++;
                    internalCondition = building.GetIC(index);
                    if (index > 10000) { break; }
                }

                W("IC COUNT: {0}", count_IC);
                W("ticV NON-ZERO COUNT: {0}", count_TicVNonZero);
                W("ticI PRESENT COUNT: {0}", count_TicIPresent);

                int count_Zone = 0;
                TBD.zone zone = building.GetZone(0);
                while (zone != null)
                {
                    count_Zone++;
                    zone = building.GetZone(count_Zone);
                    if (count_Zone > 100000) { break; }
                }
                W("ZONE COUNT: {0}", count_Zone);
            }
            finally
            {
                try { tBDDocument.close(); } catch { }
            }

            return 0;
        }


        private static int DumpGraph(string path_TPD, string path_Out)
        {
            W("dump-graph {0} -> {1}", path_TPD, path_Out);
            string text = Ref.Graph(path_TPD);
            File.WriteAllText(path_Out, text);
            W("written: {0} bytes, {1} lines", text.Length, text.Split('\n').Length);
            return 0;
        }

        private static int DumpTpd(string path_TPD)
        {
            W("dump-tpd {0}", path_TPD);

            TPD.TPDDoc tPDDoc = new TPD.TPDDoc();
            try
            {
                tPDDoc.Open(path_TPD);

                TPD.EnergyCentre energyCentre = tPDDoc.EnergyCentre;
                W("EnergyCentre: {0}", energyCentre == null ? "NULL" : "present");
                if (energyCentre == null)
                {
                    return 0;
                }

                TryW("StartHour", delegate { return energyCentre.GetStartHour().ToString(CultureInfo.InvariantCulture); });
                TryW("EndHour", delegate { return energyCentre.GetEndHour().ToString(CultureInfo.InvariantCulture); });
                W("PlantRoomCount {0}", energyCentre.GetPlantRoomCount());

                for (int i = 1; i <= energyCentre.GetPlantRoomCount(); i++)
                {
                    TPD.PlantRoom plantRoom = energyCentre.GetPlantRoom(i);
                    if (plantRoom == null)
                    {
                        continue;
                    }

                    W("PlantRoom[{0}] name={1} systems={2}", i, Describe(delegate { return plantRoom.Name; }), plantRoom.GetSystemCount());

                    foreach (TPD.System system in Systems(plantRoom))
                    {
                        W("  System name={0} guid={1}", Describe(delegate { return system.Name; }), Describe(delegate { return system.GUID; }));

                        foreach (TPD.SystemComponent systemComponent in Components(system))
                        {
                            TPD.SystemZone systemZone = systemComponent as TPD.SystemZone;

                            W(
                                "    component type={0} name={1} guid={2} zoneLoads={3}{4}",
                                Describe(delegate { return systemComponent.GetComponentType().ToString(CultureInfo.InvariantCulture); }),
                                Describe(delegate { return systemComponent.Name; }),
                                Describe(delegate { return systemComponent.GUID; }),
                                Describe(delegate { return systemComponent.GetZoneLoadCount().ToString(CultureInfo.InvariantCulture); }),
                                systemZone == null ? string.Empty : " [SystemZone]");

                            int count_ZoneLoad = 0;
                            try
                            {
                                count_ZoneLoad = systemComponent.GetZoneLoadCount();
                            }
                            catch
                            {
                            }

                            for (int z = 1; z <= count_ZoneLoad; z++)
                            {
                                TPD.ZoneLoad zoneLoad = null;
                                try
                                {
                                    zoneLoad = systemComponent.GetZoneLoad(z);
                                }
                                catch
                                {
                                }

                                if (zoneLoad != null)
                                {
                                    W(
                                        "        zoneLoad[{0}] name={1} guid={2}",
                                        z,
                                        Describe(delegate { return zoneLoad.Name; }),
                                        Describe(delegate { return zoneLoad.GUID; }));
                                }
                            }
                        }
                    }
                }
            }
            finally
            {
                try { tPDDoc.Close(); } catch { }
            }

            return 0;
        }
