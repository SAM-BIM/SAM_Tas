// Diagnostic only, and NEVER part of the acceptance evidence: drives WorkflowCalculator directly with
// the same settings Create.NoIzamThermalSource forces on, purely so a failing step can be NAMED.
// Create.NoIzamThermalSource owns its calculator, so its Timings are not reachable from outside.

using System;
using System.Collections.Generic;
using System.Globalization;
using System.IO;
using System.Text;

namespace Inv
{
    internal static class Diag
    {
        public static int Main(string path_SAM, string directory, string path_TWD, string mode)
        {
            StringBuilder log = new StringBuilder();

            Action<string> W = line =>
            {
                Console.WriteLine(line);
                log.AppendLine(line);
            };

            bool full = mode != null && mode.StartsWith("full", StringComparison.OrdinalIgnoreCase);

            Directory.CreateDirectory(directory);

            string path_gbXML = Path.Combine(directory, "src.xml");
            string path_TBD = Path.Combine(directory, "src.tbd");

            SAM.Analytical.AnalyticalModel analyticalModel = SAM.Core.Convert.ToSAM<SAM.Analytical.AnalyticalModel>(path_SAM)[0];

            gbXMLSerializer.gbXML gbXML = SAM.Analytical.gbXML.Convert.TogbXML(analyticalModel, SAM.Core.Tolerance.MacroDistance, SAM.Core.Tolerance.Distance);
            SAM.Core.gbXML.Create.gbXML(gbXML, path_gbXML);

            List<SAM.Weather.WeatherData> weatherDatas = SAM.Weather.Tas.Convert.ToSAM_WeatherDatas(path_TWD);

            W(string.Format("weather years {0}, using '{1}'", weatherDatas.Count, weatherDatas[0].Name));

            SAM.Analytical.Tas.WorkflowSettings workflowSettings = new SAM.Analytical.Tas.WorkflowSettings
            {
                Path_gbXML = path_gbXML,
                Path_TBD = path_TBD,
                WeatherData = weatherDatas[0],
                Sizing = true,
                Simulate = true,
                SimulateFrom = 1,
                SimulateTo = full ? 365 : 1,
                UpdateZones = true,
                UnmetHours = false,
                RemoveExistingTBD = true,
                AddIZAMs = false,
                RemoveIZAMs = true,
                RemoveMechanicalVentilationGains = true,
            };

            SAM.Analytical.Tas.WorkflowCalculator workflowCalculator = new SAM.Analytical.Tas.WorkflowCalculator(workflowSettings);

            string failure = null;

            try
            {
                SAM.Analytical.AnalyticalModel analyticalModel_Result = workflowCalculator.Calculate(analyticalModel);

                W(string.Format("Calculate returned {0}", analyticalModel_Result == null ? "<null>" : "a model"));
            }
            catch (Exception exception)
            {
                failure = string.Format("{0}: {1}", exception.GetType().Name, exception.Message);

                W(string.Concat("EXCEPTION ", failure));
                W(exception.StackTrace ?? string.Empty);
            }

            W("");
            W("== steps, in order, with the last one being where it got to ==");

            foreach (KeyValuePair<string, double> keyValuePair in workflowCalculator.Timings ?? new List<KeyValuePair<string, double>>())
            {
                W(string.Format(CultureInfo.InvariantCulture, "  {0,10:0.0} ms  {1}", keyValuePair.Value, keyValuePair.Key));
            }

            W("");
            W("== notes ==");

            foreach (string note in workflowCalculator.Notes ?? new List<string>())
            {
                W(string.Concat("  ", note));
            }

            File.WriteAllText(Path.Combine(directory, "diag.log"), log.ToString());

            return failure == null ? 0 : 6;
        }
    }
}
