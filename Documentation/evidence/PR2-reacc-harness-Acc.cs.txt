        internal static void W(string format, params object[] args)
        {

        {

            W("");

        {
            try
            {
                File.WriteAllText(path, log.ToString());

        {
            try
            {
                return value == null ? double.NaN : Convert.ToDouble(value, CultureInfo.InvariantCulture);
            }
            catch
            {
                return double.NaN;
            }
        }

        private static void NoIzamEvidence(string path_TBD)
        {
            int count_IZAM = 0;
            int count_InternalCondition = 0;
            int count_TicVNonZero = 0;
            int count_TicIPresent = 0;

            global::TBD.TBDDocument tBDDocument = new global::TBD.TBDDocument();

            try
            {
                tBDDocument.open(path_TBD);

                global::TBD.Building building = tBDDocument.Building;

                while (building.GetIZAM(count_IZAM) != null && count_IZAM < 100000)
                {
                    count_IZAM++;
                }

                int index = 0;
                global::TBD.InternalCondition internalCondition = building.GetIC(index);

                while (internalCondition != null && index < 100000)
                {
                    count_InternalCondition++;

                    global::TBD.InternalGain internalGain = internalCondition.GetInternalGain();

                    global::TBD.profile profile_V = internalGain?.GetProfile((int)global::TBD.Profiles.ticV);
                    if (profile_V != null && (profile_V.value != 0 || profile_V.factor != 0 || profile_V.setbackValue != 0 || profile_V.schedule != null))
                    {
                        count_TicVNonZero++;
                    }

                    global::TBD.profile profile_I = internalGain?.GetProfile((int)global::TBD.Profiles.ticI);
                    if (profile_I != null)
                    {
                        count_TicIPresent++;
                    }

                    index++;
                    internalCondition = building.GetIC(index);
                }
            }
            finally
            {
                try { tBDDocument.close(); } catch { }
            }

            Item(1, "the thermal source carries no IZAM",
                count_IZAM == 0,
                string.Format("IZAM count {0} on {1}.", count_IZAM, path_TBD));

            Item(2, "the mechanical ventilation gain is zero on every internal condition",
                count_TicVNonZero == 0,
                string.Format("{0} internal condition(s), {1} with a non-zero ticV.", count_InternalCondition, count_TicVNonZero));

            Item(3, "the infiltration term is untouched",
                count_TicIPresent == count_InternalCondition && count_InternalCondition > 0,
                string.Format("ticI present on {0} of {1} internal condition(s).", count_TicIPresent, count_InternalCondition));
        }

        internal static void Acceptance(
            SAM.Analytical.Tas.TPD.SystemVentilationRoute systemVentilationRoute,
            SAM.Analytical.Systems.MechanicalVentilationMaterialisation mechanicalVentilationMaterialisation,
            SAM.Analytical.AdjacencyCluster adjacencyCluster,
            Shape shape,
            string path_TPD,
            int startHour,
            int endHour)
        {
            List<SAM.Analytical.Tas.TPD.SystemVentilationBinding> bindings = systemVentilationRoute.Bindings;
            List<SAM.Analytical.Tas.TPD.SystemVentilationConnectionBinding> connectionBindings = systemVentilationRoute.ConnectionBindings;

            // ---------------------------------------------------------------- what PR1 said, independently
            Dictionary<Guid, Guid> airSystem_By_AirHandlingUnit = new Dictionary<Guid, Guid>();
            Dictionary<Guid, Guid> space_By_SystemSpace = new Dictionary<Guid, Guid>();
            Dictionary<Guid, Guid> spaceAirMovement_By_Connection = new Dictionary<Guid, Guid>();

            foreach (SAM.Analytical.Systems.MechanicalVentilationBinding mechanicalVentilationBinding in mechanicalVentilationMaterialisation.Bindings)
            {
                switch (mechanicalVentilationBinding.BindingType)
                {
                    case SAM.Analytical.Systems.MechanicalVentilationBindingType.AirSystem:
                        airSystem_By_AirHandlingUnit[mechanicalVentilationBinding.Guid_Analytical] = mechanicalVentilationBinding.Guid_Systems;
                        break;
                    case SAM.Analytical.Systems.MechanicalVentilationBindingType.SystemSpace:
                        space_By_SystemSpace[mechanicalVentilationBinding.Guid_Systems] = mechanicalVentilationBinding.Guid_Analytical;
                        break;
                    case SAM.Analytical.Systems.MechanicalVentilationBindingType.TransferConnection:
                        spaceAirMovement_By_Connection[mechanicalVentilationBinding.Guid_Systems] = mechanicalVentilationBinding.Guid_Analytical;
                        break;
                }
            }

            //The units PR1 MATERIALISED, read off its own bindings - not every AirHandlingUnit the model
            //carries. A real model carries units that no mechanical ventilation system serves from (this
            //fixture carries 'AHU1', named only by a legacy MV system with no design terminal), and
            //counting those would compare native systems against plant the design never asked for.
            int count_AirHandlingUnits = airSystem_By_AirHandlingUnit.Count;

            W("");
            W("air handling units: {0} materialised by PR1, {1} present on the model.",
                airSystem_By_AirHandlingUnit.Count,
                adjacencyCluster.GetObjects<SAM.Analytical.AirHandlingUnit>().Count);

            // ---------------------------------------------------------------- the native document
            global::TPD.TPDDoc tPDDoc = new global::TPD.TPDDoc();

            List<object> systems = new List<object>();
            Dictionary<string, object> component_By_Guid = new Dictionary<string, object>(StringComparer.OrdinalIgnoreCase);
            Dictionary<string, string> system_By_Component = new Dictionary<string, string>(StringComparer.OrdinalIgnoreCase);
            List<string> fans = new List<string>();
            List<string> zoneFlags = new List<string>();
            int count_ZoneFlagsWrong = 0;

            try
            {
                tPDDoc.Open(path_TPD);

                object energyCentre = tPDDoc.EnergyCentre;

                int count_PlantRoom = (int)Dbl(Dump.Call(energyCentre, "GetPlantRoomCount"));

                for (int p = 1; p <= count_PlantRoom; p++)
                {
                    object plantRoom = Dump.Call(energyCentre, "GetPlantRoom", p);
                    int count_System = (int)Dbl(Dump.Call(plantRoom, "GetSystemCount"));

                    for (int s = 1; s <= count_System; s++)
                    {
                        object system = Dump.Call(plantRoom, "GetSystem", s);
                        systems.Add(system);

                        string reference_System = Dump.Obj(system, "GUID") as string;

                        foreach (object component in Components(system))
                        {
                            string reference = Dump.Obj(component, "GUID") as string;
                            if (reference == null)
                            {
                                continue;
                            }

                            component_By_Guid[reference] = component;
                            system_By_Component[reference] = reference_System;

                            if (component is global::TPD.Fan fan)
                            {
                                fans.Add(string.Format(CultureInfo.InvariantCulture,
                                    "fan {0} DesignFlowType={1} value={2}",
                                    reference, fan.DesignFlowType, fan.DesignFlowRate == null ? double.NaN : fan.DesignFlowRate.Value));
                            }

                            if (component is global::TPD.SystemZone systemZone)
                            {
                                int flags = systemZone.Flags;

                                bool ok = (flags & (int)global::TPD.tpdSystemZoneFlags.tpdSystemZoneFlagModelVentFlow) == 0
                                    && (flags & (int)global::TPD.tpdSystemZoneFlags.tpdSystemZoneFlagModelInterzoneFlow) == 0;

                                if (!ok)
                                {
                                    count_ZoneFlagsWrong++;
                                }

                                zoneFlags.Add(string.Format(CultureInfo.InvariantCulture, "zone {0} Flags={1}", reference, flags));
                            }
                        }
                    }
                }

                Native(
                    systemVentilationRoute,
                    shape,
                    bindings,
                    connectionBindings,
                    space_By_SystemSpace,
                    spaceAirMovement_By_Connection,
                    airSystem_By_AirHandlingUnit,
                    count_AirHandlingUnits,
                    systems,
                    component_By_Guid,
                    system_By_Component,
                    fans,
                    zoneFlags,
                    count_ZoneFlagsWrong,
                    startHour,
                    endHour);
            }
            finally
            {
                try { tPDDoc.Close(); } catch { }
            }

            //Items 13 and 14 both work on COPIES of the document, and both need it closed first:
            //13 re-simulates each air system with a raw late-bound call so its read owes nothing to
            //the code under test, and 14 renames every zone and damper to one shared string.
            Agreement(systemVentilationRoute, path_TPD, startHour, endHour);

            //Item 14 is proved by MUTATING the document: every zone and every damper is renamed to one
            //shared string, and the results are then read again through the same bindings. If any step
            //of the route resolved by name, the second read would alias or fail.
            DuplicateNames(systemVentilationRoute, path_TPD, startHour, endHour);
        }

        private static IEnumerable<object> Components(object container)
        {
            List<object> result = new List<object>();

            Collect(container, result, 0);

            return result;
        }

        private static void Collect(object container, List<object> result, int depth)
        {
            if (container == null || depth > 8)
            {
                return;
            }

            for (int index = 1; index <= 20000; index++)
            {
                object component = Dump.Call(container, "GetComponent", index);
                if (component == null)
                {
                    break;
                }

                result.Add(component);

                if (component is global::TPD.ComponentGroup)
                {
                    Collect(component, result, depth + 1);
                }
            }
        }

        private static void Native(
            SAM.Analytical.Tas.TPD.SystemVentilationRoute systemVentilationRoute,
            Shape shape,
            List<SAM.Analytical.Tas.TPD.SystemVentilationBinding> bindings,
            List<SAM.Analytical.Tas.TPD.SystemVentilationConnectionBinding> connectionBindings,
            Dictionary<Guid, Guid> space_By_SystemSpace,
            Dictionary<Guid, Guid> spaceAirMovement_By_Connection,
            Dictionary<Guid, Guid> airSystem_By_AirHandlingUnit,
            int count_AirHandlingUnits,
            List<object> systems,
            Dictionary<string, object> component_By_Guid,
            Dictionary<string, string> system_By_Component,
            List<string> fans,
            List<string> zoneFlags,
            int count_ZoneFlagsWrong,
            int startHour,
            int endHour)
        {
            // ---------------------------------------------------------------- item 4
            HashSet<string> references_System = new HashSet<string>(bindings.Select(x => x.Reference_System), StringComparer.OrdinalIgnoreCase);

            Item(4, "separate analytical air handling units remain separate TAS systems",
                systems.Count == count_AirHandlingUnits && references_System.Count == count_AirHandlingUnits && count_AirHandlingUnits > 1,
                string.Format("{0} analytical unit(s), {1} native TAS system(s), {2} distinct system(s) named by the room bindings.",
                    count_AirHandlingUnits, systems.Count, references_System.Count));

            // ---------------------------------------------------------------- item 5
            int count_WrongSystem = 0;
            int count_Missing = 0;

            foreach (SAM.Analytical.Tas.TPD.SystemVentilationBinding binding in bindings)
            {
                if (!component_By_Guid.ContainsKey(binding.Reference_SystemZone))
                {
                    count_Missing++;
                    continue;
                }

                if (!system_By_Component.TryGetValue(binding.Reference_SystemZone, out string reference_System)
                    || !string.Equals(reference_System, binding.Reference_System, StringComparison.OrdinalIgnoreCase))
                {
                    count_WrongSystem++;
                }
            }

            Item(5, "every intended room belongs to the correct TAS system",
                count_Missing == 0 && count_WrongSystem == 0 && bindings.Count == space_By_SystemSpace.Count,
                string.Format("{0} room(s) bound, {1} materialised by PR1; {2} native zone(s) missing, {3} in the wrong system.",
                    bindings.Count, space_By_SystemSpace.Count, count_Missing, count_WrongSystem));

            // ---------------------------------------------------------------- item 6
            int count_Supply = connectionBindings.Count(x => x.ConnectionType == SAM.Analytical.Tas.TPD.SystemVentilationConnectionType.Supply);
            int count_Extract = connectionBindings.Count(x => x.ConnectionType == SAM.Analytical.Tas.TPD.SystemVentilationConnectionType.Extract);
            int count_Transfer = connectionBindings.Count(x => x.ConnectionType == SAM.Analytical.Tas.TPD.SystemVentilationConnectionType.Transfer);

            int count_Both = bindings.Count(x => x.DesignFlowRate_Supply_Lps.HasValue && x.DesignFlowRate_Extract_Lps.HasValue);
            int count_SupplyOnly = bindings.Count(x => x.DesignFlowRate_Supply_Lps.HasValue && !x.DesignFlowRate_Extract_Lps.HasValue);
            int count_ExtractOnly = bindings.Count(x => !x.DesignFlowRate_Supply_Lps.HasValue && x.DesignFlowRate_Extract_Lps.HasValue);
            int count_TransferOnly = bindings.Count(x => !x.DesignFlowRate_Supply_Lps.HasValue && !x.DesignFlowRate_Extract_Lps.HasValue);

            //Asserted against the design that was written, not against a literal - so the same check
            //holds whichever model the acceptance is run on.
            Item(6, "supply / extract / both / transfer-only topology is correct",
                count_Supply == shape.Supply && count_Extract == shape.Extract && count_Transfer == shape.Transfer
                    && count_SupplyOnly == shape.SupplyOnly && count_ExtractOnly == shape.ExtractOnly
                    && count_Both == shape.Both && count_TransferOnly == shape.TransferOnly,
                string.Format("found/designed - legs: {0}/{7} supply, {1}/{8} extract, {2}/{9} transfer. rooms: {3}/{10} supply-only, {4}/{11} extract-only, {5}/{12} both, {6}/{13} transfer-only.",
                    count_Supply, count_Extract, count_Transfer, count_SupplyOnly, count_ExtractOnly, count_Both, count_TransferOnly,
                    shape.Supply, shape.Extract, shape.Transfer, shape.SupplyOnly, shape.ExtractOnly, shape.Both, shape.TransferOnly));

            // ---------------------------------------------------------------- items 7, 8, 9
            Flows(bindings, connectionBindings, component_By_Guid);

            // ---------------------------------------------------------------- item 10
            //The two bits that matter are ModelVentFlow (4) and ModelInterzoneFlow (2): with the
            //ventilation modelled explicitly in TAS Systems, leaving either on would deliver the same
            //air twice. DisplacementVent (1) is a different property - it is what the template's
            //prototype room states about how air is delivered, not a second ventilation model - so it
            //is reported rather than required to be clear.
            Item(10, "no duplicated ventilation: the building's own vent and interzone flows are off",
                count_ZoneFlagsWrong == 0 && zoneFlags.Count == bindings.Count,
                string.Format("{0} native zone(s), {1} still modelling their own ventilation (bit 4) or interzone flow (bit 2). Flags seen: {2}.",
                    zoneFlags.Count,
                    count_ZoneFlagsWrong,
                    string.Join(", ", zoneFlags.Select(x => x.Substring(x.IndexOf("Flags=", StringComparison.Ordinal))).Distinct())));

            // ---------------------------------------------------------------- item 11
            bool fansDerived = fans.Count != 0 && fans.TrueForAll(x => !x.Contains("tpdFlowRateValue"));

            Item(11, "fan duty is derived from the attached zones, never authored",
                fansDerived,
                fans.Count == 0 ? "no fan found." : string.Join("; ", fans));

            // ---------------------------------------------------------------- item 12
            SAM.Analytical.Tas.TPD.SystemZoneTemperatureResults systemZoneTemperatureResults = systemVentilationRoute.SystemZoneTemperatureResults;

            int expected = endHour - startHour + 1;
            int count_Complete = systemZoneTemperatureResults.Results.Count(x => x.IsComplete);

            Item(12, "every room returns a complete finite ZoneTemperature series for the requested period",
                systemZoneTemperatureResults.IsComplete && count_Complete == bindings.Count,
                string.Format("{0} of {1} room(s) complete over hours {2}..{3} ({4} values each).",
                    count_Complete, bindings.Count, startHour, endHour, expected));

            // ---------------------------------------------------------------- item 15
            int count_RoundTrip = 0;
            int count_LoadResolved = 0;

            foreach (SAM.Analytical.Tas.TPD.SystemVentilationBinding binding in bindings)
            {
                if (!component_By_Guid.TryGetValue(binding.Reference_SystemZone, out object component))
                {
                    continue;
                }

                if (string.Equals(Dump.Obj(component, "GUID") as string, binding.Reference_SystemZone, StringComparison.OrdinalIgnoreCase))
                {
                    count_RoundTrip++;
                }

                object zoneLoad = Dump.Call(component, "GetSystemZoneZoneLoad");
                if (string.Equals(Dump.Obj(zoneLoad, "GUID") as string, binding.Reference_ZoneLoad, StringComparison.OrdinalIgnoreCase))
                {
                    count_LoadResolved++;
                }
            }

            bool distinct = bindings.Select(x => x.Reference_SystemZone).Distinct(StringComparer.OrdinalIgnoreCase).Count() == bindings.Count
                && bindings.Select(x => x.Reference_ZoneLoad).Distinct(StringComparer.OrdinalIgnoreCase).Count() == bindings.Count;

            Item(15, "the native SystemZone / ZoneLoad identity chain holds end to end",
                count_RoundTrip == bindings.Count && count_LoadResolved == bindings.Count && distinct,
                string.Format("{0}/{1} zone guids round-trip, {2}/{1} zone loads resolve to the bound load, all identities distinct: {3}.",
                    count_RoundTrip, bindings.Count, count_LoadResolved, distinct));
        }

        private static void Flows(
            List<SAM.Analytical.Tas.TPD.SystemVentilationBinding> bindings,
            List<SAM.Analytical.Tas.TPD.SystemVentilationConnectionBinding> connectionBindings,
            Dictionary<string, object> component_By_Guid)
        {
            // ---------------------------------------------------------------- item 7, supply
            List<string> mismatches_Supply = new List<string>();

            foreach (SAM.Analytical.Tas.TPD.SystemVentilationBinding binding in bindings)
            {
                if (!binding.DesignFlowRate_Supply_Lps.HasValue)
                {
                    continue;
                }

                if (!component_By_Guid.TryGetValue(binding.Reference_SystemZone, out object component) || !(component is global::TPD.SystemZone systemZone))
                {
                    mismatches_Supply.Add(string.Format("room {0}: no native zone", binding.Guid_Space));
                    continue;
                }

                double value = systemZone.FlowRate == null ? double.NaN : systemZone.FlowRate.Value;
                double freshAir = systemZone.FreshAir == null ? double.NaN : systemZone.FreshAir.Value;

                bool ok = Close(value, binding.DesignFlowRate_Supply_Lps.Value)
                    && Close(freshAir, binding.DesignFlowRate_Supply_Lps.Value)
                    && systemZone.FlowRate.Type == global::TPD.tpdSizedVariable.tpdSizedVariableValue;

                if (!ok)
                {
                    mismatches_Supply.Add(string.Format(CultureInfo.InvariantCulture,
                        "room {0}: binding {1}, native {2}/{3}, type {4}",
                        binding.Guid_Space, binding.DesignFlowRate_Supply_Lps, value, freshAir, systemZone.FlowRate.Type));
                }
            }

            int count_SupplyRooms = bindings.Count(x => x.DesignFlowRate_Supply_Lps.HasValue);

            Item(7, "supply DesignFlowRate is preserved, absolutely, on the room's own zone",
                mismatches_Supply.Count == 0 && count_SupplyRooms > 0,
                string.Format("{0} supply room(s) checked against the native zone, {1} mismatch(es). {2}",
                    count_SupplyRooms, mismatches_Supply.Count, string.Join("; ", mismatches_Supply)));

            // ---------------------------------------------------------------- items 8 and 9
            Legs(connectionBindings, component_By_Guid, SAM.Analytical.Tas.TPD.SystemVentilationConnectionType.Extract, 8, "extract");
            Legs(connectionBindings, component_By_Guid, SAM.Analytical.Tas.TPD.SystemVentilationConnectionType.Transfer, 9, "transfer");
        }

        private static void Legs(
            List<SAM.Analytical.Tas.TPD.SystemVentilationConnectionBinding> connectionBindings,
            Dictionary<string, object> component_By_Guid,
            SAM.Analytical.Tas.TPD.SystemVentilationConnectionType systemVentilationConnectionType,
            int number,
            string what)
        {
            List<string> mismatches = new List<string>();
            int count = 0;

            foreach (SAM.Analytical.Tas.TPD.SystemVentilationConnectionBinding connectionBinding in connectionBindings)
            {
                if (connectionBinding.ConnectionType != systemVentilationConnectionType)
                {
                    continue;
                }

                count++;

                if (connectionBinding.Reference_FlowController == null
                    || !component_By_Guid.TryGetValue(connectionBinding.Reference_FlowController, out object component)
                    || !(component is global::TPD.Damper damper))
                {
                    mismatches.Add(string.Format("leg {0}: no native damper", connectionBinding.Guid_SystemConnection));
                    continue;
                }

                double value = damper.DesignFlowRate == null ? double.NaN : damper.DesignFlowRate.Value;

                bool ok = Close(value, connectionBinding.DesignFlowRate_Lps)
                    && damper.DesignFlowType == global::TPD.tpdFlowRateType.tpdFlowRateValue
                    && damper.DesignFlowRate.Type == global::TPD.tpdSizedVariable.tpdSizedVariableValue;

                if (!ok)
                {
                    mismatches.Add(string.Format(CultureInfo.InvariantCulture,
                        "leg {0}: binding {1}, native {2}, flowType {3}, sizedType {4}",
                        connectionBinding.Guid_SystemConnection, connectionBinding.DesignFlowRate_Lps, value,
                        damper.DesignFlowType, damper.DesignFlowRate == null ? "<none>" : damper.DesignFlowRate.Type.ToString()));
                }
            }

            Item(number, string.Format("{0} DesignFlowRate is preserved on its own in-line damper", what),
                mismatches.Count == 0 && count > 0,
                string.Format("{0} {1} leg(s) checked against the native damper, {2} mismatch(es). {3}",
                    count, what, mismatches.Count, string.Join("; ", mismatches)));
        }

        private static bool Close(double value_1, double value_2)
        {
            return !double.IsNaN(value_1) && !double.IsNaN(value_2)
                && Math.Abs(value_1 - value_2) <= Math.Max(1e-6, Math.Max(Math.Abs(value_1), Math.Abs(value_2)) * 1e-5);
        }

        // ---------------------------------------------------------------- item 13

        /// <summary>
        /// Item 13. Reads every room's series a SECOND time, independently of the production code, and
        /// compares it value for value.
        /// <para>
        /// <b>Per system, and on a copy.</b> Licensed TAS replaces an air system's in-memory result
        /// surface when <c>ISystem.Simulate</c> is called for the next system in the same document, so a
        /// single pass after the loop can only ever see the last system's zones - which is what silently
        /// reduced this check to four of nine rooms before. Here each system is simulated by a raw
        /// late-bound call and its own rooms are read straight afterwards, on a copy of the document so
        /// the route's own output is left as it was produced.
        /// </para>
        /// <para>
        /// The denominator is <b>every bound room</b>. A room whose independent read failed counts as a
        /// disagreement, not as a room that was never asked about.
        /// </para>
        /// </summary>
        private static void Agreement(
            SAM.Analytical.Tas.TPD.SystemVentilationRoute systemVentilationRoute,
            string path_TPD,
            int startHour,
            int endHour)
        {
            List<SAM.Analytical.Tas.TPD.SystemVentilationBinding> bindings = systemVentilationRoute.Bindings;

            Dictionary<string, List<SAM.Analytical.Tas.TPD.SystemVentilationBinding>> bindings_By_System =
                new Dictionary<string, List<SAM.Analytical.Tas.TPD.SystemVentilationBinding>>(StringComparer.OrdinalIgnoreCase);

            foreach (SAM.Analytical.Tas.TPD.SystemVentilationBinding binding in bindings)
            {
                if (!bindings_By_System.TryGetValue(binding.Reference_System, out List<SAM.Analytical.Tas.TPD.SystemVentilationBinding> list))
                {
                    list = new List<SAM.Analytical.Tas.TPD.SystemVentilationBinding>();
                    bindings_By_System[binding.Reference_System] = list;
                }

                list.Add(binding);
            }

            int expected = endHour - startHour + 1;
            int count_Agreed = 0;
            List<string> disagreements = new List<string>();

            string path_Agree = Path.ChangeExtension(path_TPD, ".agree.tpd");

            File.Copy(path_TPD, path_Agree, true);

            global::TPD.TPDDoc tPDDoc = new global::TPD.TPDDoc();

            try
            {
                tPDDoc.Open(path_Agree);

                object energyCentre = tPDDoc.EnergyCentre;

                int count_PlantRoom = (int)Dbl(Dump.Call(energyCentre, "GetPlantRoomCount"));

                for (int p = 1; p <= count_PlantRoom; p++)
                {
                    object plantRoom = Dump.Call(energyCentre, "GetPlantRoom", p);
                    int count_System = (int)Dbl(Dump.Call(plantRoom, "GetSystemCount"));

                    for (int sIndex = 1; sIndex <= count_System; sIndex++)
                    {
                        object system = Dump.Call(plantRoom, "GetSystem", sIndex);

                        string reference_System = Dump.Obj(system, "GUID") as string;

                        if (reference_System == null || !bindings_By_System.TryGetValue(reference_System, out List<SAM.Analytical.Tas.TPD.SystemVentilationBinding> list))
                        {
                            continue;
                        }

                        //A raw late-bound simulate, so nothing about this read goes through the code
                        //under test.
                        string returned = Dump.Call(system, "Simulate", startHour + 1, endHour + 1, 0) as string;

                        Dictionary<string, object> component_By_Guid = new Dictionary<string, object>(StringComparer.OrdinalIgnoreCase);

                        foreach (object component in Components(system))
                        {
                            string reference = Dump.Obj(component, "GUID") as string;
                            if (reference != null)
                            {
                                component_By_Guid[reference] = component;
                            }
                        }

                        foreach (SAM.Analytical.Tas.TPD.SystemVentilationBinding binding in list)
                        {
                            SAM.Analytical.Tas.TPD.SystemZoneTemperatureResult result =
                                systemVentilationRoute.SystemZoneTemperatureResults.Result(binding.Guid_Space);

                            if (result == null || result.Count == 0)
                            {
                                disagreements.Add(string.Format("room {0}: the route produced no series", binding.Guid_Space));
                                continue;
                            }

                            if (!component_By_Guid.TryGetValue(binding.Reference_SystemZone, out object component))
                            {
                                disagreements.Add(string.Format("room {0}: native zone not in system {1}", binding.Guid_Space, reference_System));
                                continue;
                            }

                            object data;

                            try
                            {
                                data = component.GetType().InvokeMember(
                                    "GetResultsData",
                                    BindingFlags.InvokeMethod,
                                    null,
                                    component,
                                    new object[]
                                    {
                                        (int)global::TPD.tpdResultsPeriod.tpdResultsPeriodHourly,
                                        (int)global::TPD.tpdCombinerType.tpdCombinerTypeMax,
                                        9,
                                        startHour + 1,
                                        expected,
                                    },
                                    CultureInfo.InvariantCulture);
                            }
                            catch (Exception exception)
                            {
                                disagreements.Add(string.Format("room {0}: {1} (system answered \"{2}\")",
                                    binding.Guid_Space, (exception.InnerException ?? exception).Message.Trim(), returned));
                                continue;
                            }

                            Array array = data as Array;

                            if (array == null)
                            {
                                disagreements.Add(string.Format("room {0}: no native series (system answered \"{1}\")", binding.Guid_Space, returned));
                                continue;
                            }

                            int index = startHour;
                            bool agreed = true;
                            int count_Compared = 0;

                            foreach (object element in array)
                            {
                                double value_Native = Dbl(element);

                                if (!result.TryGetValue(index, out double value_Api) || !Close(value_Api, value_Native))
                                {
                                    agreed = false;
                                    disagreements.Add(string.Format(CultureInfo.InvariantCulture,
                                        "room {0} hour {1}: API {2}, native {3}", binding.Guid_Space, index, value_Api, value_Native));
                                    break;
                                }

                                index++;
                                count_Compared++;
                            }

                            if (agreed && count_Compared != expected)
                            {
                                agreed = false;
                                disagreements.Add(string.Format("room {0}: native series held {1} value(s), expected {2}",
                                    binding.Guid_Space, count_Compared, expected));
                            }

                            if (agreed)
                            {
                                count_Agreed++;
                            }
                        }
                    }
                }
            }
            finally
            {
                try { tPDDoc.Close(); } catch { }
            }

            Item(13, "the route's values agree with TAS's own, read independently",
                bindings.Count > 0 && count_Agreed == bindings.Count,
                string.Format("{0} of {1} room(s) agreed value for value over {2} hour(s), each read after a raw late-bound simulate of its own air system. {3}",
                    count_Agreed, bindings.Count, expected, string.Join("; ", disagreements.Take(3))));
        }

        private static void DuplicateNames(
            SAM.Analytical.Tas.TPD.SystemVentilationRoute systemVentilationRoute,
            string path_TPD,
            int startHour,
            int endHour)
        {
            string path_Renamed = Path.ChangeExtension(path_TPD, ".renamed.tpd");

            File.Copy(path_TPD, path_Renamed, true);

            int count_Renamed = 0;

            global::TPD.TPDDoc tPDDoc = new global::TPD.TPDDoc();

            try
            {
                tPDDoc.Open(path_Renamed);

                object energyCentre = tPDDoc.EnergyCentre;

                int count_PlantRoom = (int)Dbl(Dump.Call(energyCentre, "GetPlantRoomCount"));

                for (int p = 1; p <= count_PlantRoom; p++)
                {
                    object plantRoom = Dump.Call(energyCentre, "GetPlantRoom", p);
                    int count_System = (int)Dbl(Dump.Call(plantRoom, "GetSystemCount"));

                    for (int s = 1; s <= count_System; s++)
                    {
                        foreach (object component in Components(Dump.Call(plantRoom, "GetSystem", s)))
                        {
                            if (!(component is global::TPD.SystemZone) && !(component is global::TPD.Damper))
                            {
                                continue;
                            }

                            try
                            {
                                component.GetType().InvokeMember(
                                    component is global::TPD.SystemZone ? "name" : "Name",
                                    BindingFlags.SetProperty,
                                    null,
                                    component,
                                    new object[] { "System Zone 1" },
                                    CultureInfo.InvariantCulture);

                                count_Renamed++;
                            }
                            catch
                            {
                            }
                        }
                    }
                }

                Dump.Call(tPDDoc, "Save");
            }
            finally
            {
                try { tPDDoc.Close(); } catch { }
            }

            SAM.Analytical.Tas.TPD.Modify.SimulateSystems(
                path_Renamed,
                systemVentilationRoute.Bindings,
                startHour,
                endHour,
                out SAM.Core.Tas.SimulationEvidence simulationEvidence,
                out SAM.Analytical.Tas.TPD.SystemZoneTemperatureResults systemZoneTemperatureResults);

            bool validated = systemZoneTemperatureResults.Validate(systemVentilationRoute.Bindings);

            //And the values must be the SAME values, not merely a complete set: a name-keyed lookup
            //that aliased two rooms would still be complete.
            int count_Same = 0;

            foreach (SAM.Analytical.Tas.TPD.SystemZoneTemperatureResult result in systemZoneTemperatureResults.Results)
            {
                SAM.Analytical.Tas.TPD.SystemZoneTemperatureResult result_Original = systemVentilationRoute.SystemZoneTemperatureResults.Result(result.Guid_Space);

                if (result_Original == null || result.Count == 0 || result_Original.Count == 0)
                {
                    continue;
                }

                bool same = true;

                for (int i = startHour; i <= endHour; i++)
                {
                    if (!result.TryGetValue(i, out double value_1) || !result_Original.TryGetValue(i, out double value_2) || !Close(value_1, value_2))
                    {
                        same = false;
                        break;
                    }
                }

                if (same)
                {
                    count_Same++;
                }
            }

            Item(14, "duplicate display names do not alter identity",
                validated && count_Same == systemVentilationRoute.Bindings.Count && count_Renamed > 0,
                string.Format("{0} native zone(s) and damper(s) renamed to one shared string; results still validate: {1}; {2} of {3} room(s) returned identical values.",
                    count_Renamed, validated, count_Same, systemVentilationRoute.Bindings.Count));
        }
    }
