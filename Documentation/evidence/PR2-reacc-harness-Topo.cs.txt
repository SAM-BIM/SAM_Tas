// DIRECTED TOPOLOGY ACCEPTANCE.
//
// The previous acceptance failed here, so nothing in this file infers a direction from anything but
// TAS's own connector ports: every edge comes from IDuct.GetUpstreamComponent /
// GetDownstreamComponent. No screen coordinate, no display name, no creation order and no drawing is
// read.
//
// Three independent statements are compared, per leg:
//
//   1. the ANALYTICAL INTENT      intent.tsv, written by the production route before any conversion
//   2. the ROUTE'S CLAIM          claim-legs.tsv / claim-rooms.tsv, written by PR2 before the read-back
//   3. the NATIVE DIRECTED GRAPH  read here, off the document, through TAS's own accessors
//
// A leg passes only when all three agree on the two endpoints, the direction between them, and the
// duty - and when the native carrier the duty sits on is the one the frozen rules name.

using System;
using System.Collections.Generic;
using System.Globalization;
using System.IO;
using System.Text;
using SAM.Core.Tas;

namespace Inv
{
    internal static class Topo
    {
        private static readonly StringBuilder log = new StringBuilder();

        private static void W(string format, params object[] args)
        {
            string line = args == null || args.Length == 0 ? format : string.Format(CultureInfo.InvariantCulture, format, args);
            Console.WriteLine(line);
            log.AppendLine(line);
        }

        private sealed class Node
        {
            public string Guid;
            public string Kind;
            public string Name;
            public bool Transparent;
            public double Duty = double.NaN;
            public string DutyType = string.Empty;
            public List<string> Out = new List<string>();
            public List<string> In = new List<string>();
        }

        private static bool IsTransparent(string kind)
        {
            return kind == "Junction" || kind == "GroupJunction" || kind == "ComponentGroup";
        }

        private static string KindOf(object component)
        {
            string kind = Graph.Kind(component);
            int index = kind.IndexOf('=');
            if (index >= 0)
            {
                kind = kind.Substring(index + 1).Trim('"');
            }

            return kind;
        }

        private static double Dbl(object value)
        {
            try
            {
                return value == null ? double.NaN : System.Convert.ToDouble(value, CultureInfo.InvariantCulture);
            }
            catch
            {
                return double.NaN;
            }
        }

        private static void Collect(object container, Dictionary<string, Node> nodes, List<object> groups, int depth)
        {
            if (depth > 6)
            {
                return;
            }

            foreach (object component in Graph.TopLevel(container))
            {
                string guid = Graph.Guid_Of(component);
                string kind = KindOf(component);

                Node node = new Node
                {
                    Guid = guid,
                    Kind = kind,
                    Name = (Dump.Get(component, "Name") ?? string.Empty).Trim('"'),
                    Transparent = IsTransparent(kind),
                };

                if (component is global::TPD.SystemZone systemZone)
                {
                    node.Duty = systemZone.FlowRate == null ? double.NaN : systemZone.FlowRate.Value;
                    node.DutyType = systemZone.FlowRate == null ? string.Empty : systemZone.FlowRate.Type.ToString();
                }
                else if (component is global::TPD.Damper damper)
                {
                    node.Duty = damper.DesignFlowRate == null ? double.NaN : damper.DesignFlowRate.Value;
                    node.DutyType = damper.DesignFlowType.ToString();
                }
                else if (component is global::TPD.Fan fan)
                {
                    node.Duty = fan.DesignFlowRate == null ? double.NaN : fan.DesignFlowRate.Value;
                    node.DutyType = fan.DesignFlowType.ToString();
                }

                nodes[guid] = node;

                if (component is global::TPD.ComponentGroup)
                {
                    groups.Add(component);
                    Collect(component, nodes, groups, depth + 1);
                }
            }
        }

        /// <summary>
        /// Every edge, taken from the duct's own upstream/downstream accessors and recorded in BOTH
        /// directions so an upstream question and a downstream question are answered off the same ports.
        /// </summary>
        private static void Edges(object container, Dictionary<string, Node> nodes)
        {
            foreach (object duct in Graph.Ducts(container))
            {
                string up = Graph.Guid_Of(Dump.Call(duct, "GetUpstreamComponent"));
                string down = Graph.Guid_Of(Dump.Call(duct, "GetDownstreamComponent"));

                Node node_Up;
                Node node_Down;

                if (nodes.TryGetValue(up, out node_Up))
                {
                    node_Up.Out.Add(down);
                }

                if (nodes.TryGetValue(down, out node_Down))
                {
                    node_Down.In.Add(up);
                }
            }
        }

        /// <summary>Every duty-carrying component reachable from a node through transparent nodes only.</summary>
        private static List<string> Reach(string guid, Dictionary<string, Node> nodes, bool downstream)
        {
            List<string> result = new List<string>();
            HashSet<string> seen = new HashSet<string>(StringComparer.Ordinal);
            Queue<string> queue = new Queue<string>();

            Node start;
            if (nodes.TryGetValue(guid, out start))
            {
                foreach (string next in downstream ? start.Out : start.In)
                {
                    queue.Enqueue(next);
                }
            }

            while (queue.Count > 0)
            {
                string current = queue.Dequeue();

                if (!seen.Add(current))
                {
                    continue;
                }

                Node node;
                if (!nodes.TryGetValue(current, out node))
                {
                    continue;
                }

                if (!node.Transparent)
                {
                    result.Add(current);
                    continue;
                }

                foreach (string next in downstream ? node.Out : node.In)
                {
                    queue.Enqueue(next);
                }
            }

            result.Sort(StringComparer.Ordinal);

            return result;
        }

        /// <summary>
        /// True where a damper carries no duty of its own - <c>tpdFlowRateAllAttachedZonesFlowRate</c>,
        /// the template's trunk damper, which DERIVES its flow from the zones attached downstream.
        /// <para>
        /// It is a duty-carrying component to <c>Reach</c> but not a leg of the design, so a design leg's
        /// direction question has to see through it exactly as it sees through a junction. The distinction
        /// is the native <c>DesignFlowType</c> and nothing else: an authored duty is
        /// <c>tpdFlowRateValue</c>, which is the frozen carrier rule for every extract and transfer leg.
        /// </para>
        /// </summary>
        private static bool IsDerivedDamper(Node node)
        {
            return node != null && node.Kind == "Damper" && node.DutyType != "tpdFlowRateValue";
        }

        /// <summary>
        /// The first component upstream of a node that carries a duty the DESIGN states, seeing through
        /// junctions, groups and the unit's own derived-duty trunk damper.
        /// </summary>
        private static List<string> ReachDesign(string guid, Dictionary<string, Node> nodes, bool downstream)
        {
            List<string> result = new List<string>();
            HashSet<string> seen = new HashSet<string>(StringComparer.OrdinalIgnoreCase);
            Queue<string> queue = new Queue<string>();

            queue.Enqueue(guid);

            while (queue.Count > 0)
            {
                string current = queue.Dequeue();

                foreach (string next in Reach(current, nodes, downstream))
                {
                    if (!seen.Add(next))
                    {
                        continue;
                    }

                    Node node;
                    nodes.TryGetValue(next, out node);

                    if (IsDerivedDamper(node))
                    {
                        queue.Enqueue(next);
                        continue;
                    }

                    result.Add(next);
                }
            }

            result.Sort(StringComparer.Ordinal);

            return result;
        }

        private static bool Close(double value_1, double value_2)
        {
            if (double.IsNaN(value_1) || double.IsNaN(value_2))
            {
                return false;
            }

            return Math.Abs(value_1 - value_2) <= 1e-6 + 1e-6 * Math.Max(Math.Abs(value_1), Math.Abs(value_2));
        }

        private static List<string[]> Tsv(string path)
        {
            List<string[]> result = new List<string[]>();

            string[] lines = File.ReadAllLines(path);

            for (int index = 1; index < lines.Length; index++)
            {
                if (!string.IsNullOrWhiteSpace(lines[index]))
                {
                    result.Add(lines[index].Split('\t'));
                }
            }

            return result;
        }

        private static string Describe(Dictionary<string, Node> nodes, List<string> guids)
        {
            if (guids.Count == 0)
            {
                return "<none>";
            }

            List<string> parts = new List<string>();

            foreach (string guid in guids)
            {
                Node node;
                parts.Add(nodes.TryGetValue(guid, out node)
                    ? string.Format(CultureInfo.InvariantCulture, "{0} \"{1}\"", node.Kind, node.Name)
                    : "?");
            }

            return string.Join(" + ", parts);
        }

        private static string Trim(string value, int length)
        {
            if (value == null)
            {
                return string.Empty;
            }

            return value.Length <= length ? value : value.Substring(0, length);
        }

        public static int Run(string path_TPD, string directory, string path_Out)
        {
            log.Clear();

            W("DIRECTED TOPOLOGY ACCEPTANCE");
            W("  document {0}  sha256 {1}", path_TPD, Src.Hash(path_TPD));
            W("  intent   {0}", Path.Combine(directory, "intent.tsv"));
            W("  claim    {0} / {1}", "claim-rooms.tsv", "claim-legs.tsv");
            W("");
            W("Every edge below comes from IDuct.GetUpstreamComponent / GetDownstreamComponent. No screen");
            W("coordinate, display name or creation order is read.");
            W("");

            // ---- 1. the analytical intent ----------------------------------------------------------

            Dictionary<Guid, string> name_By_Space = new Dictionary<Guid, string>();
            Dictionary<Guid, double> supply_By_Space = new Dictionary<Guid, double>();
            Dictionary<Guid, double> extract_By_Space = new Dictionary<Guid, double>();
            Dictionary<Guid, string> system_By_Space = new Dictionary<Guid, string>();
            List<Tuple<Guid, Guid, double>> intent_Transfer = new List<Tuple<Guid, Guid, double>>();

            foreach (string[] parts in Tsv(Path.Combine(directory, "intent.tsv")))
            {
                if (parts[0] == "room")
                {
                    Guid guid = Guid.Parse(parts[3]);
                    system_By_Space[guid] = parts[1];
                    name_By_Space[guid] = parts[4];
                    supply_By_Space[guid] = double.Parse(parts[5], CultureInfo.InvariantCulture);
                    extract_By_Space[guid] = double.Parse(parts[6], CultureInfo.InvariantCulture);
                }
                else if (parts[0] == "transfer")
                {
                    intent_Transfer.Add(Tuple.Create(Guid.Parse(parts[7]), Guid.Parse(parts[8]), double.Parse(parts[9], CultureInfo.InvariantCulture)));
                }
            }

            // ---- 2. the route's claim --------------------------------------------------------------

            Dictionary<Guid, string> zone_By_Space = new Dictionary<Guid, string>();
            Dictionary<Guid, string> system_Reference_By_Space = new Dictionary<Guid, string>();

            foreach (string[] parts in Tsv(Path.Combine(directory, "claim-rooms.tsv")))
            {
                Guid guid = Guid.Parse(parts[0]);
                system_Reference_By_Space[guid] = parts[3];
                zone_By_Space[guid] = parts[4];
            }

            //One claimed leg per (type, from, to).
            Dictionary<string, Tuple<double, string>> claim_By_Leg = new Dictionary<string, Tuple<double, string>>(StringComparer.Ordinal);

            foreach (string[] parts in Tsv(Path.Combine(directory, "claim-legs.tsv")))
            {
                claim_By_Leg[string.Concat(parts[0], "|", parts[1], "|", parts[3])] = Tuple.Create(
                    double.Parse(parts[5], CultureInfo.InvariantCulture),
                    parts.Length > 6 ? parts[6] : string.Empty);
            }

            // ---- 3. the native directed graph -------------------------------------------------------

            Dictionary<string, Node> nodes = new Dictionary<string, Node>(StringComparer.OrdinalIgnoreCase);
            Dictionary<string, string> system_By_Node = new Dictionary<string, string>(StringComparer.OrdinalIgnoreCase);
            Dictionary<string, string> systemName_By_Reference = new Dictionary<string, string>(StringComparer.OrdinalIgnoreCase);

            using (SAMTPDDocument sAMTPDDocument = new SAMTPDDocument(path_TPD, true))
            {
                object energyCentre = sAMTPDDocument.TPDDocument.EnergyCentre;

                foreach (object system in Graph.Systems(energyCentre))
                {
                    string reference_System = Graph.Guid_Of(system);

                    systemName_By_Reference[reference_System] = (Dump.Get(system, "Name") ?? string.Empty).Trim('"');

                    Dictionary<string, Node> nodes_System = new Dictionary<string, Node>(StringComparer.OrdinalIgnoreCase);
                    List<object> groups = new List<object>();

                    Collect(system, nodes_System, groups, 0);
                    Edges(system, nodes_System);

                    foreach (object group in groups)
                    {
                        Edges(group, nodes_System);
                    }

                    //Portal edges across each group boundary, so the transparent closure runs straight
                    //through a group rather than stopping at it.
                    foreach (object group in groups)
                    {
                        string guid_Group = Graph.Guid_Of(group);

                        HashSet<string> hasIncoming = new HashSet<string>(StringComparer.OrdinalIgnoreCase);
                        HashSet<string> hasOutgoing = new HashSet<string>(StringComparer.OrdinalIgnoreCase);

                        foreach (object duct in Graph.Ducts(group))
                        {
                            hasOutgoing.Add(Graph.Guid_Of(Dump.Call(duct, "GetUpstreamComponent")));
                            hasIncoming.Add(Graph.Guid_Of(Dump.Call(duct, "GetDownstreamComponent")));
                        }

                        foreach (object member in Graph.TopLevel(group))
                        {
                            if (KindOf(member) != "GroupJunction")
                            {
                                continue;
                            }

                            string guid_Member = Graph.Guid_Of(member);

                            if (!hasIncoming.Contains(guid_Member))
                            {
                                nodes_System[guid_Group].Out.Add(guid_Member);
                                nodes_System[guid_Member].In.Add(guid_Group);
                            }

                            if (!hasOutgoing.Contains(guid_Member))
                            {
                                nodes_System[guid_Member].Out.Add(guid_Group);
                                nodes_System[guid_Group].In.Add(guid_Member);
                            }
                        }
                    }

                    foreach (KeyValuePair<string, Node> keyValuePair in nodes_System)
                    {
                        nodes[keyValuePair.Key] = keyValuePair.Value;
                        system_By_Node[keyValuePair.Key] = reference_System;
                    }
                }
            }

            W("native graph: {0} component(s) across {1} air system(s)", nodes.Count, systemName_By_Reference.Count);

            // =================================================================== the per-leg comparison

            int count_Pass = 0;
            int count_Fail = 0;
            List<string> failures = new List<string>();

            W("");
            W("{0,-11} | {1,-13} | {2,-13} | {3,-8} | {4,-30} | {5,-30} | {6,-9} | {7}",
                "leg", "intended from", "intended to", "duty l/s", "native upstream", "native downstream", "native duty", "verdict");
            W(new string('-', 170));

            // ---- the transfer legs, which is where the old acceptance reversed ---------------------

            foreach (Tuple<Guid, Guid, double> leg in intent_Transfer)
            {
                string name_From = name_By_Space.ContainsKey(leg.Item1) ? name_By_Space[leg.Item1] : leg.Item1.ToString("D");
                string name_To = name_By_Space.ContainsKey(leg.Item2) ? name_By_Space[leg.Item2] : leg.Item2.ToString("D");

                string key = string.Concat("Transfer|", leg.Item1.ToString("D"), "|", leg.Item2.ToString("D"));

                Tuple<double, string> claim;
                if (!claim_By_Leg.TryGetValue(key, out claim))
                {
                    count_Fail++;
                    failures.Add(string.Format("transfer {0} -> {1}: the route claims no leg in this direction.", name_From, name_To));
                    W("{0,-11} | {1,-13} | {2,-13} | {3,-8} | {4,-30} | {5,-30} | {6,-9} | FAIL - no claimed leg",
                        "transfer", Trim(name_From, 13), Trim(name_To, 13), leg.Item3.ToString("0.###", CultureInfo.InvariantCulture), "-", "-", "-");
                    continue;
                }

                //The native damper the route says carries this duty. Its ports are then read, and both
                //ends must be the zones the INTENT names - not the ones the claim names.
                Node node_Damper;
                if (string.IsNullOrWhiteSpace(claim.Item2) || !nodes.TryGetValue(claim.Item2, out node_Damper))
                {
                    count_Fail++;
                    failures.Add(string.Format("transfer {0} -> {1}: the claimed flow controller is not in the native document.", name_From, name_To));
                    continue;
                }

                List<string> upstream = Reach(node_Damper.Guid, nodes, false);
                List<string> downstream = Reach(node_Damper.Guid, nodes, true);

                string reference_Zone_From = zone_By_Space.ContainsKey(leg.Item1) ? zone_By_Space[leg.Item1] : null;
                string reference_Zone_To = zone_By_Space.ContainsKey(leg.Item2) ? zone_By_Space[leg.Item2] : null;

                bool ok_Upstream = reference_Zone_From != null && upstream.Count == 1 && string.Equals(upstream[0], reference_Zone_From, StringComparison.OrdinalIgnoreCase);
                bool ok_Downstream = reference_Zone_To != null && downstream.Count == 1 && string.Equals(downstream[0], reference_Zone_To, StringComparison.OrdinalIgnoreCase);
                bool ok_Duty = Close(node_Damper.Duty, leg.Item3) && Close(claim.Item1, leg.Item3);

                //The frozen rule: an extract or transfer duty rides on a TYPED Damper.DesignFlowRate,
                //never on a junction and never on an authored fan value.
                bool ok_Carrier = node_Damper.Kind == "Damper" && node_Damper.DutyType == "tpdFlowRateValue";

                //And it must not cross a dwelling system boundary.
                bool ok_System = system_By_Node.ContainsKey(node_Damper.Guid)
                    && reference_Zone_From != null
                    && system_By_Node.ContainsKey(reference_Zone_From)
                    && string.Equals(system_By_Node[node_Damper.Guid], system_By_Node[reference_Zone_From], StringComparison.OrdinalIgnoreCase)
                    && reference_Zone_To != null
                    && system_By_Node.ContainsKey(reference_Zone_To)
                    && string.Equals(system_By_Node[node_Damper.Guid], system_By_Node[reference_Zone_To], StringComparison.OrdinalIgnoreCase);

                bool pass = ok_Upstream && ok_Downstream && ok_Duty && ok_Carrier && ok_System;

                if (pass)
                {
                    count_Pass++;
                }
                else
                {
                    count_Fail++;

                    //Named specifically, because a reversal is the failure this whole document exists for.
                    bool reversed = reference_Zone_To != null && upstream.Count == 1 && string.Equals(upstream[0], reference_Zone_To, StringComparison.OrdinalIgnoreCase);

                    failures.Add(string.Format(
                        "transfer {0} -> {1}: upstream {2}, downstream {3}, duty {4}, carrier {5}, system {6}{7}",
                        name_From, name_To,
                        ok_Upstream ? "ok" : "WRONG",
                        ok_Downstream ? "ok" : "WRONG",
                        ok_Duty ? "ok" : "WRONG",
                        ok_Carrier ? "ok" : "WRONG",
                        ok_System ? "ok" : "CROSSES A SYSTEM",
                        reversed ? "  *** THE NATIVE LEG IS REVERSED ***" : string.Empty));
                }

                W("{0,-11} | {1,-13} | {2,-13} | {3,-8} | {4,-30} | {5,-30} | {6,-9} | {7}",
                    "transfer",
                    Trim(name_From, 13),
                    Trim(name_To, 13),
                    leg.Item3.ToString("0.###", CultureInfo.InvariantCulture),
                    Trim(Describe(nodes, upstream), 30),
                    Trim(Describe(nodes, downstream), 30),
                    node_Damper.Duty.ToString("0.###", CultureInfo.InvariantCulture),
                    pass ? "PASS" : "FAIL");
            }

            // ---- the supply legs: the fan is upstream of the room's own zone -----------------------

            foreach (KeyValuePair<Guid, double> keyValuePair in supply_By_Space)
            {
                if (keyValuePair.Value <= 0)
                {
                    continue;
                }

                Guid guid_Space = keyValuePair.Key;
                string name = name_By_Space[guid_Space];

                string reference_Zone = zone_By_Space.ContainsKey(guid_Space) ? zone_By_Space[guid_Space] : null;

                Node node_Zone;
                if (reference_Zone == null || !nodes.TryGetValue(reference_Zone, out node_Zone))
                {
                    count_Fail++;
                    failures.Add(string.Format("supply {0}: no native zone.", name));
                    continue;
                }

                //ReachDesign, not Reach: the template puts its own trunk damper between the supply fan
                //and the branch junction, and that damper derives its flow from the attached zones
                //(tpdFlowRateAllAttachedZonesFlowRate) rather than carrying a duty of its own. It is
                //therefore transparent to the question "what supplies this room", exactly as a junction is.
                List<string> upstream = ReachDesign(reference_Zone, nodes, false);

                bool ok_Upstream = upstream.Count == 1 && nodes[upstream[0]].Kind == "Fan";

                //The frozen rule: a supply duty rides on the room's own SystemZone FlowRate, absolutely.
                bool ok_Duty = Close(node_Zone.Duty, keyValuePair.Value) && node_Zone.DutyType == "tpdSizedVariableValue";

                bool pass = ok_Upstream && ok_Duty;

                if (pass)
                {
                    count_Pass++;
                }
                else
                {
                    count_Fail++;
                    failures.Add(string.Format("supply {0}: upstream {1}, duty {2}.", name, ok_Upstream ? "ok" : "WRONG", ok_Duty ? "ok" : "WRONG"));
                }

                W("{0,-11} | {1,-13} | {2,-13} | {3,-8} | {4,-30} | {5,-30} | {6,-9} | {7}",
                    "supply",
                    "<the unit>",
                    Trim(name, 13),
                    keyValuePair.Value.ToString("0.###", CultureInfo.InvariantCulture),
                    Trim(Describe(nodes, upstream), 30),
                    Trim(string.Format("SystemZone \"{0}\"", node_Zone.Name), 30),
                    node_Zone.Duty.ToString("0.###", CultureInfo.InvariantCulture),
                    pass ? "PASS" : "FAIL");
            }

            // ---- the extract legs: the room's own zone is upstream of its extract damper -----------

            foreach (KeyValuePair<Guid, double> keyValuePair in extract_By_Space)
            {
                if (keyValuePair.Value <= 0)
                {
                    continue;
                }

                Guid guid_Space = keyValuePair.Key;
                string name = name_By_Space[guid_Space];

                //The extract leg is claimed room -> the unit, so the claim's "to" is the unit and resolves
                //to Guid.Empty in claim-legs.tsv.
                string key = string.Concat("Extract|", guid_Space.ToString("D"), "|", Guid.Empty.ToString("D"));

                Tuple<double, string> claim;
                if (!claim_By_Leg.TryGetValue(key, out claim))
                {
                    count_Fail++;
                    failures.Add(string.Format("extract {0}: the route claims no extract leg out of this room.", name));
                    continue;
                }

                Node node_Damper;
                if (string.IsNullOrWhiteSpace(claim.Item2) || !nodes.TryGetValue(claim.Item2, out node_Damper))
                {
                    count_Fail++;
                    failures.Add(string.Format("extract {0}: the claimed flow controller is not in the native document.", name));
                    continue;
                }

                List<string> upstream = Reach(node_Damper.Guid, nodes, false);
                List<string> downstream = Reach(node_Damper.Guid, nodes, true);

                string reference_Zone = zone_By_Space.ContainsKey(guid_Space) ? zone_By_Space[guid_Space] : null;

                bool ok_Upstream = reference_Zone != null && upstream.Count == 1 && string.Equals(upstream[0], reference_Zone, StringComparison.OrdinalIgnoreCase);
                bool ok_Downstream = downstream.Count == 1 && nodes[downstream[0]].Kind == "Fan";
                bool ok_Duty = Close(node_Damper.Duty, keyValuePair.Value) && Close(claim.Item1, keyValuePair.Value);
                bool ok_Carrier = node_Damper.Kind == "Damper" && node_Damper.DutyType == "tpdFlowRateValue";

                bool pass = ok_Upstream && ok_Downstream && ok_Duty && ok_Carrier;

                if (pass)
                {
                    count_Pass++;
                }
                else
                {
                    count_Fail++;
                    failures.Add(string.Format("extract {0}: upstream {1}, downstream {2}, duty {3}, carrier {4}.",
                        name,
                        ok_Upstream ? "ok" : "WRONG",
                        ok_Downstream ? "ok" : "WRONG",
                        ok_Duty ? "ok" : "WRONG",
                        ok_Carrier ? "ok" : "WRONG"));
                }

                W("{0,-11} | {1,-13} | {2,-13} | {3,-8} | {4,-30} | {5,-30} | {6,-9} | {7}",
                    "extract",
                    Trim(name, 13),
                    "<the unit>",
                    keyValuePair.Value.ToString("0.###", CultureInfo.InvariantCulture),
                    Trim(Describe(nodes, upstream), 30),
                    Trim(Describe(nodes, downstream), 30),
                    node_Damper.Duty.ToString("0.###", CultureInfo.InvariantCulture),
                    pass ? "PASS" : "FAIL");
            }

            // ---- what must NOT be there --------------------------------------------------------------

            //An extra native leg the design does not state. Every duty-carrying damper must be one of the
            //legs above, and every zone-to-zone adjacency must be a transfer the design states.
            int count_Damper = 0;
            int count_Unaccounted = 0;

            HashSet<string> accounted = new HashSet<string>(StringComparer.OrdinalIgnoreCase);

            foreach (KeyValuePair<string, Tuple<double, string>> keyValuePair in claim_By_Leg)
            {
                if (!string.IsNullOrWhiteSpace(keyValuePair.Value.Item2))
                {
                    accounted.Add(keyValuePair.Value.Item2);
                }
            }

            int count_Trunk = 0;

            foreach (KeyValuePair<string, Node> keyValuePair in nodes)
            {
                if (keyValuePair.Value.Kind != "Damper")
                {
                    continue;
                }

                count_Damper++;

                //A derived-duty damper is the unit's own trunk damper. It states no design duty, so it
                //cannot be an extra leg - it is reported, with the flow TAS sized it to, and checked
                //against the system's own supply total further down.
                if (IsDerivedDamper(keyValuePair.Value))
                {
                    count_Trunk++;

                    W("  trunk damper {0} \"{1}\" in system \"{2}\": {3} = {4} l/s, derived from the attached zones",
                        keyValuePair.Key,
                        keyValuePair.Value.Name,
                        systemName_By_Reference.ContainsKey(system_By_Node[keyValuePair.Key]) ? systemName_By_Reference[system_By_Node[keyValuePair.Key]] : "?",
                        keyValuePair.Value.DutyType,
                        keyValuePair.Value.Duty.ToString("0.###", CultureInfo.InvariantCulture));

                    continue;
                }

                if (!accounted.Contains(keyValuePair.Key))
                {
                    count_Unaccounted++;
                    failures.Add(string.Format("native damper {0} \"{1}\" carries an authored duty of {2} l/s and belongs to no leg the design states.",
                        keyValuePair.Key, keyValuePair.Value.Name, keyValuePair.Value.Duty.ToString("0.###", CultureInfo.InvariantCulture)));
                }
            }

            //One trunk damper per air system, and no more: a second would be a duplicated supply path.
            if (count_Trunk != systemName_By_Reference.Count)
            {
                failures.Add(string.Format("{0} derived-duty trunk damper(s) across {1} air system(s); one per system is the template's shape.",
                    count_Trunk, systemName_By_Reference.Count));
            }

            //And no zone may be reachable from a zone in another air system.
            int count_CrossSystem = 0;

            foreach (KeyValuePair<string, Node> keyValuePair in nodes)
            {
                if (keyValuePair.Value.Transparent)
                {
                    continue;
                }

                string reference_System = system_By_Node[keyValuePair.Key];

                foreach (string guid in Reach(keyValuePair.Key, nodes, true))
                {
                    if (system_By_Node.ContainsKey(guid) && !string.Equals(system_By_Node[guid], reference_System, StringComparison.OrdinalIgnoreCase))
                    {
                        count_CrossSystem++;
                        failures.Add(string.Format("native edge {0} -> {1} crosses an air system boundary.", keyValuePair.Key, guid));
                    }
                }
            }

            W("");
            W("");
            W("native dampers: {0} total, {1} derived-duty trunk damper(s), {2} authored-duty damper(s) belonging to no stated leg",
                count_Damper, count_Trunk, count_Unaccounted);
            W("native edges crossing an air system boundary: {0}", count_CrossSystem);

            W("");
            W("DIRECTED TOPOLOGY: {0} leg(s) PASS, {1} FAIL", count_Pass, count_Fail + count_Unaccounted + count_CrossSystem);

            if (failures.Count != 0)
            {
                W("");
                W("PR2 CORRECTNESS BLOCKER - every mismatch, in full:");

                foreach (string failure in failures)
                {
                    W("  {0}", failure);
                }
            }

            if (!string.IsNullOrWhiteSpace(path_Out))
            {
                File.WriteAllText(path_Out, log.ToString());
            }

            return failures.Count == 0 ? 0 : 10;
        }
    }
