// Hourly delivered flow on every duct of a simulated air system.
//
// The fan's operation on this route is NOT a yearly or hourly schedule table: the schedule attached
// to it reads tpdScheduleType.tpdScheduleFunction with
// tpdScheduleFunctionType.tpdScheduleFunctionAllZonesLoad, so TAS derives the fan's operation from
// the attached zones' loads and GetNumOperableHours()/GetYearlyValue() - which describe a yearly or
// hourly table - answer 0 for it. Reading those and concluding "the fan never runs" would be wrong.
//
// What settles operation is the flow TAS actually delivered: IDuct.GetFlowRate(hour) over the whole
// simulated year, compared against the leg's design duty. A duct that carries its design duty in
// every one of 8760 hours is continuous operation at factor 1.0, measured rather than asserted.
//
// Nothing here is production code.

using System;
using System.Collections.Generic;
using System.Globalization;
using System.IO;
using System.Text;
using SAM.Core.Tas;

namespace Inv
{
    internal static class Flow
    {
        public static void Run(string path, string pathOut, int hours)
        {
            StringBuilder stringBuilder = new StringBuilder();
            stringBuilder.AppendLine("DUCT FLOW " + path + " hours=" + hours);

            using (SAMTPDDocument sAMTPDDocument = new SAMTPDDocument(path))
            {
                object energyCentre = sAMTPDDocument.TPDDocument.EnergyCentre;

                foreach (object system in Graph.Systems(energyCentre))
                {
                    string returned = Dump.Call(system, "Simulate", 1, hours, 0) as string;

                    stringBuilder.AppendLine();
                    stringBuilder.AppendLine(string.Format(
                        CultureInfo.InvariantCulture,
                        "== SYSTEM {0} {1} simulate answered \"{2}\"",
                        Graph.Guid_Of(system),
                        Dump.Get(system, "Name"),
                        returned));

                    List<object> ducts = Graph.Ducts(system);
                    int d = 0;

                    foreach (object duct in ducts)
                    {
                        d++;

                        object upstream = Dump.Call(duct, "GetUpstreamComponent");
                        object downstream = Dump.Call(duct, "GetDownstreamComponent");

                        double min = double.MaxValue;
                        double max = double.MinValue;
                        double sum = 0;
                        int read = 0;
                        int zero = 0;
                        int nonFinite = 0;
                        string threw = null;

                        for (int hour = 0; hour < hours; hour++)
                        {
                            object value = null;

                            try
                            {
                                value = duct.GetType().InvokeMember(
                                    "GetFlowRate",
                                    System.Reflection.BindingFlags.InvokeMethod,
                                    null,
                                    duct,
                                    new object[] { hour },
                                    CultureInfo.InvariantCulture);
                            }
                            catch (Exception exception)
                            {
                                threw = (exception.InnerException ?? exception).Message.Trim();
                                break;
                            }

                            if (value == null)
                            {
                                break;
                            }

                            double v = System.Convert.ToDouble(value, CultureInfo.InvariantCulture);

                            if (double.IsNaN(v) || double.IsInfinity(v))
                            {
                                nonFinite++;
                                continue;
                            }

                            read++;
                            sum += v;
                            if (v < min) { min = v; }
                            if (v > max) { max = v; }
                            if (v == 0) { zero++; }
                        }

                        stringBuilder.AppendLine(string.Format(
                            CultureInfo.InvariantCulture,
                            "  duct {0,-3} [{1} {2}] -> [{3} {4}]",
                            d,
                            Trim(Graph.Kind(upstream)),
                            Dump.Get(upstream, "Name"),
                            Trim(Graph.Kind(downstream)),
                            Dump.Get(downstream, "Name")));

                        if (threw != null)
                        {
                            stringBuilder.AppendLine("        GetFlowRate threw: " + threw);
                            continue;
                        }

                        stringBuilder.AppendLine(string.Format(
                            CultureInfo.InvariantCulture,
                            "        hours={0} min={1:R} max={2:R} mean={3:R} hoursAtZero={4} nonFinite={5} constant={6}",
                            read,
                            read == 0 ? double.NaN : min,
                            read == 0 ? double.NaN : max,
                            read == 0 ? double.NaN : sum / read,
                            zero,
                            nonFinite,
                            read > 0 && min == max));
                    }
                }
            }

            string text = stringBuilder.ToString();
            Console.WriteLine(text);

            if (!string.IsNullOrWhiteSpace(pathOut))
            {
                File.WriteAllText(pathOut, text);
                Console.WriteLine("written " + pathOut);
            }
        }

        private static string Trim(string kind)
        {
            int index = kind.IndexOf('=');
            return index < 0 ? kind : kind.Substring(index + 1).Trim('"');
        }
    }
}
