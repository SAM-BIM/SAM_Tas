// Checkpoint 1B, part A continued: can an INDIVIDUAL leg carry its own absolute design airflow?
//
// The zone has only FlowRate/FreshAir and a duct has no design-flow property at all, so if extract and
// transfer are representable it must be through an in-line flow-controlling component. Damper is the
// candidate: IDamper declares DesignFlowRate (SizedFlowVariable) and DesignFlowType (tpdFlowRateType),
// the same pair a Fan uses.
//
// This builds the full asymmetric reference with a damper on every leg whose duty PR1 authors, writes
// deliberately distinct non-round values, saves, and the caller then re-opens and dumps it. A value TAS
// refuses to keep shows up as a mismatch on the way back in, not as an assumption here.

using System;
using System.Globalization;
using System.IO;
using System.Reflection;
using System.Text;

namespace Inv
{
    internal static class BuildRef2
    {
        private static readonly StringBuilder log = new StringBuilder();

        private static void W(string format, params object[] args)
        {
            string line = args == null || args.Length == 0 ? format : string.Format(CultureInfo.InvariantCulture, format, args);
            Console.WriteLine(line);
            log.AppendLine(line);
        }

        private static void SetProperty(object target, string name, object value)
        {
            target.GetType().InvokeMember(
                name,
                BindingFlags.SetProperty,
                null,
                target,
                new object[] { value },
                CultureInfo.InvariantCulture);
        }

        /// <summary>Writes an absolute duty and reads it straight back, so a refusal is visible.</summary>
        private static bool SetAbsoluteFlow(string label, object owner, string propertyName, double value_Lps)
        {
            object sizedFlowVariable = Dump.Obj(owner, propertyName);

            if (sizedFlowVariable == null)
            {
                W("  {0}.{1}: NO CARRIER", label, propertyName);
                return false;
            }

            try
            {
                SetProperty(sizedFlowVariable, "Type", (int)global::TPD.tpdSizedVariable.tpdSizedVariableValue);
                SetProperty(sizedFlowVariable, "Value", value_Lps);
            }
            catch (Exception exception)
            {
                Exception inner = exception.InnerException ?? exception;
                W("  {0}.{1}: THREW {2}: {3}", label, propertyName, inner.GetType().Name, inner.Message.Trim());
                return false;
            }

            string readBack = Dump.Get(sizedFlowVariable, "Value");
            W("  {0}.{1} <- {2}   reads back Value={3} Type={4}",
                label, propertyName, value_Lps.ToString("R", CultureInfo.InvariantCulture),
                readBack, Dump.Get(sizedFlowVariable, "Type"));

            return true;
        }

        private static object AddDuct(object system, string label, object up, int portUp, object down, int portDown)
        {
            try
            {
                object duct = system.GetType().InvokeMember(
                    "AddDuct",
                    BindingFlags.InvokeMethod,
                    null,
                    system,
                    new object[] { up, portUp, down, portDown },
                    CultureInfo.InvariantCulture);

                W("  duct {0}: {1}", label, duct == null ? "NULL" : "ok");
                return duct;
            }
            catch (Exception exception)
            {
                Exception inner = exception.InnerException ?? exception;
                W("  duct {0}: THREW {1}: {2}", label, inner.GetType().Name, inner.Message.Replace("\r", " ").Replace("\n", " ").Trim());
                return null;
            }
        }

        /// <summary>A damper carrying an explicit absolute duty on one leg.</summary>
        private static object AddDamper(object system, string label, double value_Lps)
        {
            object damper = Dump.Call(system, "AddDamper");

            if (damper == null)
            {
                W("  damper {0}: AddDamper returned NULL", label);
                return null;
            }

            try
            {
                SetProperty(damper, "DesignFlowType", (int)global::TPD.tpdFlowRateType.tpdFlowRateValue);
            }
            catch (Exception exception)
            {
                W("  damper {0}: DesignFlowType threw {1}", label, exception.GetType().Name);
            }

            SetAbsoluteFlow(string.Concat("damper ", label), damper, "DesignFlowRate", value_Lps);

            W("  damper {0}: guid={1} DesignFlowType={2}", label, Dump.Get(damper, "GUID"), Dump.Get(damper, "DesignFlowType"));

            return damper;
        }

        private static object AddZone(object system, string label, double flowRate_Lps, double freshAir_Lps)
        {
            object zone = Dump.Call(system, "AddSystemZone");
            W("  zone {0}: guid={1}", label, Dump.Get(zone, "GUID"));
            SetAbsoluteFlow(string.Concat("zone ", label), zone, "FlowRate", flowRate_Lps);
            SetAbsoluteFlow(string.Concat("zone ", label), zone, "FreshAir", freshAir_Lps);

            try
            {
                int flags = Convert.ToInt32(Dump.Obj(zone, "Flags"), CultureInfo.InvariantCulture);
                SetProperty(
                    zone,
                    "Flags",
                    flags
                        | (int)global::TPD.tpdSystemZoneFlags.tpdSystemZoneFlagModelVentFlow
                        | (int)global::TPD.tpdSystemZoneFlags.tpdSystemZoneFlagModelInterzoneFlow);
                W("  zone {0}: flags {1} -> {2}", label, flags, Dump.Get(zone, "Flags"));
            }
            catch (Exception exception)
            {
                W("  zone {0}: flags threw {1}", label, exception.GetType().Name);
            }

            return zone;
        }

        public static int Build(string path_TPD, double supplyOnly, double extractOnly, double bothSupply, double bothExtract, double transfer1, double transfer2)
        {
            log.Clear();
            W("build-ref2 {0}", path_TPD);
            W("supplyOnly={0} extractOnly={1} bothSupply={2} bothExtract={3} transfer1={4} transfer2={5}",
                supplyOnly, extractOnly, bothSupply, bothExtract, transfer1, transfer2);

            if (File.Exists(path_TPD))
            {
                File.Delete(path_TPD);
            }

            global::TPD.TPDDoc tPDDoc = new global::TPD.TPDDoc();
            try
            {
                tPDDoc.Create(path_TPD);

                object energyCentre = tPDDoc.EnergyCentre;
                object plantRoom = Dump.Call(energyCentre, "AddPlantRoom");
                object system = Dump.Call(plantRoom, "AddSystem");

                try
                {
                    SetProperty(plantRoom, "Name", "PR");
                    SetProperty(system, "Name", "AHU");
                }
                catch
                {
                }

                W("system guid={0}", Dump.Get(system, "GUID"));

                W("");
                W("== zones ==");
                object zone_SupplyOnly = AddZone(system, "SupplyOnly", supplyOnly, supplyOnly);
                object zone_ExtractOnly = AddZone(system, "ExtractOnly", extractOnly, 0.0);
                object zone_Both = AddZone(system, "Both", bothSupply, bothSupply);
                object zone_T1 = AddZone(system, "TransferTo1", transfer1, 0.0);
                object zone_T2 = AddZone(system, "TransferTo2", transfer2, 0.0);

                W("");
                W("== fans ==");
                object fan_Supply = Dump.Call(system, "AddFan");
                object fan_Extract = Dump.Call(system, "AddFan");
                W("  supply fan guid={0}", Dump.Get(fan_Supply, "GUID"));
                W("  extract fan guid={0}", Dump.Get(fan_Extract, "GUID"));

                // A fan derives its duty from the attached zones - measured on the TAS-authored reference,
                // where DesignFlowType=3 gave DesignFlowRate exactly equal to the zone's FreshAir.
                try
                {
                    SetProperty(fan_Supply, "DesignFlowType", (int)global::TPD.tpdFlowRateType.tpdFlowRateAllAttachedZonesFreshAir);
                    SetProperty(fan_Extract, "DesignFlowType", (int)global::TPD.tpdFlowRateType.tpdFlowRateAllAttachedZonesFreshAir);
                    W("  both fans DesignFlowType <- tpdFlowRateAllAttachedZonesFreshAir");
                }
                catch (Exception exception)
                {
                    W("  fan DesignFlowType threw {0}", exception.GetType().Name);
                }

                W("");
                W("== CASE 1: supply-only leg, damper {0} l/s ==", supplyOnly);
                object damper_Supply1 = AddDamper(system, "supply->SupplyOnly", supplyOnly);
                AddDuct(system, "fan_Supply -> damper_Supply1", fan_Supply, 1, damper_Supply1, 1);
                AddDuct(system, "damper_Supply1 -> zone_SupplyOnly", damper_Supply1, 1, zone_SupplyOnly, 1);

                W("");
                W("== CASE 2: extract-only leg, damper {0} l/s ==", extractOnly);
                object damper_Extract1 = AddDamper(system, "ExtractOnly->extract", extractOnly);
                AddDuct(system, "zone_ExtractOnly -> damper_Extract1", zone_ExtractOnly, 1, damper_Extract1, 1);
                AddDuct(system, "damper_Extract1 -> fan_Extract", damper_Extract1, 1, fan_Extract, 1);

                W("");
                W("== CASE 3: supply {0} and extract {1} on the SAME room, independently ==", bothSupply, bothExtract);
                object damper_Supply2 = AddDamper(system, "supply->Both", bothSupply);
                object damper_Extract2 = AddDamper(system, "Both->extract", bothExtract);
                AddDuct(system, "fan_Supply -> damper_Supply2", fan_Supply, 1, damper_Supply2, 1);
                AddDuct(system, "damper_Supply2 -> zone_Both", damper_Supply2, 1, zone_Both, 1);
                AddDuct(system, "zone_Both -> damper_Extract2", zone_Both, 1, damper_Extract2, 1);
                AddDuct(system, "damper_Extract2 -> fan_Extract", damper_Extract2, 1, fan_Extract, 1);

                W("");
                W("== CASE 4/5: branching transfer from zone_Both, {0} and {1} l/s ==", transfer1, transfer2);
                object junction = Dump.Call(system, "AddJunction");
                W("  junction guid={0}", Dump.Get(junction, "GUID"));
                object damper_T1 = AddDamper(system, "transfer leg 1", transfer1);
                object damper_T2 = AddDamper(system, "transfer leg 2", transfer2);
                AddDuct(system, "zone_Both -> junction", zone_Both, 1, junction, 1);
                AddDuct(system, "junction -> damper_T1", junction, 1, damper_T1, 1);
                AddDuct(system, "damper_T1 -> zone_T1", damper_T1, 1, zone_T1, 1);
                AddDuct(system, "junction -> damper_T2", junction, 1, damper_T2, 1);
                AddDuct(system, "damper_T2 -> zone_T2", damper_T2, 1, zone_T2, 1);

                W("");
                W("== SAVE ==");
                tPDDoc.Save();
                W("saved {0} bytes", new FileInfo(path_TPD).Length);
            }
            finally
            {
                try { tPDDoc.Close(); } catch { }
            }

            try
            {
                File.WriteAllText(Path.ChangeExtension(path_TPD, ".build.log"), log.ToString());
            }
            catch
            {
            }

            return 0;
        }
    }
}
