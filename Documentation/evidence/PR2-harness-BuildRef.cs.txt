// Checkpoint 1B, part A: build a minimal reference system with deliberately asymmetric, non-round
// flows and observe what TAS actually stores.
//
// The cases, per the brief:
//   supply-only      17 l/s
//   extract-only     23 l/s
//   supply+extract   supply 31 / extract 19   (deliberately DIFFERENT)
//   transfer         11 l/s
//   branching        a second transfer leg of 7 l/s off the same source
//
// Nothing here assumes a carrier. Each authoring attempt is made and then read back, so a value TAS
// silently refuses to store shows up as a mismatch rather than as an assumption that it worked.

using System;
using System.Globalization;
using System.IO;
using System.Text;

namespace Inv
{
    internal static class BuildRef
    {
        private static readonly StringBuilder log = new StringBuilder();

        private static void W(string format, params object[] args)
        {
            string line = args == null || args.Length == 0 ? format : string.Format(CultureInfo.InvariantCulture, format, args);
            Console.WriteLine(line);
            log.AppendLine(line);
        }

        /// <summary>
        /// Writes an absolute duty onto a SizedFlowVariable the way PR1's values must arrive: an explicit
        /// value, with Type forced to tpdSizedVariableValue so TAS uses the number rather than sizing.
        /// Reads it straight back so a refusal is visible.
        /// </summary>
        private static void SetFlow(string label, object owner, string propertyName, double value_Lps)
        {
            object sizedFlowVariable = Dump.Obj(owner, propertyName);

            if (sizedFlowVariable == null)
            {
                W("  {0}.{1}: NO CARRIER (property answered null)", label, propertyName);
                return;
            }

            try
            {
                sizedFlowVariable.GetType().InvokeMember(
                    "Type",
                    System.Reflection.BindingFlags.SetProperty,
                    null,
                    sizedFlowVariable,
                    new object[] { (int)global::TPD.tpdSizedVariable.tpdSizedVariableValue },
                    CultureInfo.InvariantCulture);
            }
            catch (Exception exception)
            {
                W("  {0}.{1}: setting Type threw {2}", label, propertyName, exception.GetType().Name);
            }

            try
            {
                sizedFlowVariable.GetType().InvokeMember(
                    "Value",
                    System.Reflection.BindingFlags.SetProperty,
                    null,
                    sizedFlowVariable,
                    new object[] { value_Lps },
                    CultureInfo.InvariantCulture);
            }
            catch (Exception exception)
            {
                W("  {0}.{1}: setting Value threw {2}", label, propertyName, exception.GetType().Name);
                return;
            }

            W("  {0}.{1} <- {2}  reads back Value={3} Type={4} Method={5}",
                label,
                propertyName,
                value_Lps.ToString("R", CultureInfo.InvariantCulture),
                Dump.Get(sizedFlowVariable, "Value"),
                Dump.Get(sizedFlowVariable, "Type"),
                Dump.Get(sizedFlowVariable, "Method"));
        }

        private static object AddZone(object system, string label)
        {
            object systemZone = Dump.Call(system, "AddSystemZone");
            W("{0}: AddSystemZone -> {1} guid={2}", label, systemZone == null ? "NULL" : "ok", Dump.Get(systemZone, "GUID"));
            return systemZone;
        }

        private static object AddFan(object system, string label)
        {
            object fan = Dump.Call(system, "AddFan");
            W("{0}: AddFan -> {1} guid={2}", label, fan == null ? "NULL" : "ok", Dump.Get(fan, "GUID"));
            return fan;
        }

        private static object AddDuct(object system, string label, object upstream, int portUp, object downstream, int portDown)
        {
            object duct = null;
            try
            {
                duct = system.GetType().InvokeMember(
                    "AddDuct",
                    System.Reflection.BindingFlags.InvokeMethod,
                    null,
                    system,
                    new object[] { upstream, portUp, downstream, portDown },
                    CultureInfo.InvariantCulture);
            }
            catch (Exception exception)
            {
                Exception inner = exception.InnerException ?? exception;
                W("{0}: AddDuct THREW {1}: {2}", label, inner.GetType().Name, inner.Message.Replace("\r", " ").Replace("\n", " ").Trim());
                return null;
            }

            W("{0}: AddDuct -> {1}", label, duct == null ? "NULL" : "ok");
            return duct;
        }

        public static int Build(string path_TPD, double supplyOnly, double extractOnly, double bothSupply, double bothExtract, double transfer1, double transfer2)
        {
            log.Clear();
            W("build-ref {0}", path_TPD);
            W("values: supplyOnly={0} extractOnly={1} bothSupply={2} bothExtract={3} transfer1={4} transfer2={5}",
                supplyOnly, extractOnly, bothSupply, bothExtract, transfer1, transfer2);

            if (File.Exists(path_TPD))
            {
                File.Delete(path_TPD);
            }

            global::TPD.TPDDoc tPDDoc = new global::TPD.TPDDoc();
            try
            {
                tPDDoc.Create(path_TPD);

                object energyCentre = tPDDoc.EnergyCentre;
                if (energyCentre == null)
                {
                    W("REFUSED: no energy centre");
                    return 3;
                }

                object plantRoom = Dump.Call(energyCentre, "AddPlantRoom");
                object system = Dump.Call(plantRoom, "AddSystem");
                W("system guid={0}", Dump.Get(system, "GUID"));

                // ---------------------------------------------------------------- supply-only room
                W("");
                W("== CASE 1: supply-only room, {0} l/s ==", supplyOnly);
                object zone_SupplyOnly = AddZone(system, "zone_SupplyOnly");
                SetFlow("zone_SupplyOnly", zone_SupplyOnly, "FlowRate", supplyOnly);
                SetFlow("zone_SupplyOnly", zone_SupplyOnly, "FreshAir", supplyOnly);

                object fan_Supply = AddFan(system, "fan_Supply");
                AddDuct(system, "supply fan -> supply-only zone", fan_Supply, 1, zone_SupplyOnly, 1);

                // ---------------------------------------------------------------- extract-only room
                W("");
                W("== CASE 2: extract-only room, {0} l/s ==", extractOnly);
                W("A zone has exactly two flow carriers, FlowRate and FreshAir, and both describe what");
                W("ENTERS. The question is whether an extract magnitude can be stated independently.");
                object zone_ExtractOnly = AddZone(system, "zone_ExtractOnly");
                SetFlow("zone_ExtractOnly", zone_ExtractOnly, "FlowRate", extractOnly);
                SetFlow("zone_ExtractOnly", zone_ExtractOnly, "FreshAir", 0.0);

                object fan_Extract = AddFan(system, "fan_Extract");
                AddDuct(system, "extract-only zone -> extract fan", zone_ExtractOnly, 1, fan_Extract, 1);

                // Is there any second flow carrier on the zone at all? Enumerate every declared property.
                W("  -- every declared SystemZone property, to prove no extract carrier is being missed --");
                Dump.Properties(new StringBuilder(), string.Empty, zone_ExtractOnly, typeof(global::TPD.ISystemZone));
                StringBuilder properties = new StringBuilder();
                Dump.Properties(properties, "     ", zone_ExtractOnly, typeof(global::TPD.ISystemZone));
                Dump.Properties(properties, "     ", zone_ExtractOnly, typeof(global::TPD.ISystemComponent));
                foreach (string line in properties.ToString().Split('\n'))
                {
                    if (!string.IsNullOrWhiteSpace(line))
                    {
                        W(line.TrimEnd());
                    }
                }

                // ---------------------------------------------------------------- both, asymmetric
                W("");
                W("== CASE 3: supply {0} / extract {1}, DIFFERENT on purpose ==", bothSupply, bothExtract);
                object zone_Both = AddZone(system, "zone_Both");
                SetFlow("zone_Both", zone_Both, "FlowRate", bothSupply);
                SetFlow("zone_Both", zone_Both, "FreshAir", bothSupply);
                AddDuct(system, "supply fan -> both zone", fan_Supply, 1, zone_Both, 1);
                AddDuct(system, "both zone -> extract fan", zone_Both, 1, fan_Extract, 1);
                W("  There is nowhere on the zone to state {0} independently of {1}.", bothExtract, bothSupply);

                // ---------------------------------------------------------------- transfer
                W("");
                W("== CASE 4: room-to-room transfer, {0} l/s ==", transfer1);
                object zone_TransferTo = AddZone(system, "zone_TransferTo");
                SetFlow("zone_TransferTo", zone_TransferTo, "FlowRate", transfer1);
                SetFlow("zone_TransferTo", zone_TransferTo, "FreshAir", 0.0);

                W("  Can a duct run DIRECTLY from one zone's outlet to another zone's inlet?");
                object duct_Transfer = AddDuct(system, "zone_Both -> zone_TransferTo (direct)", zone_Both, 1, zone_TransferTo, 1);

                // ---------------------------------------------------------------- branching transfer
                W("");
                W("== CASE 5: branching transfer, second leg {0} l/s ==", transfer2);
                object zone_TransferTo2 = AddZone(system, "zone_TransferTo2");
                SetFlow("zone_TransferTo2", zone_TransferTo2, "FlowRate", transfer2);
                SetFlow("zone_TransferTo2", zone_TransferTo2, "FreshAir", 0.0);

                W("  A second leg from the SAME source zone outlet:");
                AddDuct(system, "zone_Both -> zone_TransferTo2 (second leg)", zone_Both, 1, zone_TransferTo2, 1);

                W("  And through a junction, which is how a real system branches:");
                object junction = Dump.Call(system, "AddJunction");
                W("  AddJunction -> {0} guid={1}", junction == null ? "NULL" : "ok", Dump.Get(junction, "GUID"));
                if (junction != null)
                {
                    AddDuct(system, "zone_Both -> junction", zone_Both, 1, junction, 1);
                    AddDuct(system, "junction -> zone_TransferTo", junction, 1, zone_TransferTo, 1);
                    AddDuct(system, "junction -> zone_TransferTo2", junction, 1, zone_TransferTo2, 1);
                }

                // ---------------------------------------------------------------- flags
                W("");
                W("== Flags: System Modelled ventilation and interzone flow ==");
                foreach (object zone in new[] { zone_SupplyOnly, zone_ExtractOnly, zone_Both, zone_TransferTo, zone_TransferTo2 })
                {
                    if (zone == null)
                    {
                        continue;
                    }

                    try
                    {
                        int flags = Convert.ToInt32(Dump.Obj(zone, "Flags"), CultureInfo.InvariantCulture);
                        int wanted = flags
                            | (int)global::TPD.tpdSystemZoneFlags.tpdSystemZoneFlagModelVentFlow
                            | (int)global::TPD.tpdSystemZoneFlags.tpdSystemZoneFlagModelInterzoneFlow;

                        zone.GetType().InvokeMember(
                            "Flags",
                            System.Reflection.BindingFlags.SetProperty,
                            null,
                            zone,
                            new object[] { wanted },
                            CultureInfo.InvariantCulture);

                        W("  zone {0}: flags {1} -> {2}", Dump.Get(zone, "GUID"), flags, Dump.Get(zone, "Flags"));
                    }
                    catch (Exception exception)
                    {
                        W("  zone flags threw {0}", exception.GetType().Name);
                    }
                }

                W("");
                W("== SAVE ==");
                tPDDoc.Save();
                W("saved {0} bytes", new FileInfo(path_TPD).Length);
            }
            finally
            {
                try { tPDDoc.Close(); } catch { }
            }

            try
            {
                File.WriteAllText(Path.ChangeExtension(path_TPD, ".build.log"), log.ToString());
            }
            catch
            {
            }

            return 0;
        }
    }
}
