using System;
using System.Globalization;
using System.Linq;
using System.Reflection;
using System.Collections;
using System.Collections.Generic;

namespace B0P
{
    public static class Program
    {
        [STAThread]
        public static int Main(string[] args)
        {
            try
            {
                switch (args.Length == 0 ? "help" : args[0])
                {
                    case "map": Steps.MapModel(args); return 0;
                    case "tsdzones": Steps.TsdZones(args); return 0;
                    case "extract": Steps.Extract(args); return 0;
                    case "tpdzt": Steps.TpdZoneTemperature(args); return 0;
                    case "refl": Steps.Refl(args); return 0;
                    case "tpdinfo": Steps.TpdInfo(args); return 0;
                    case "tbdizam": Steps.TbdIzam(args); return 0;
                    case "topo": Steps.Topo(args); return 0;
                    case "simsys": Steps.SimSys(args); return 0;
                    case "ducts": Steps.Ducts(args); return 0;
                    case "zonecomp": Steps.ZoneComp(args); return 0;
                }
                Console.WriteLine("verbs: map tsdzones extract tpdzt tbdstat tpddump");
                return 2;
            }
            catch (Exception ex) { Console.WriteLine("FATAL {0}: {1}\n{2}", ex.GetType().Name, ex.Message, ex.StackTrace); return 1; }
        }
    }
}
