// Can one TAS air-system component port feed SEVERAL ducts?
//
// It decides the shape of the explicit (non-replicated) Part O ventilation conversion. PR1 hangs every
// room of one air handling unit off the SAME template attachment - a Damper on the supply side and a
// Junction on the extract side - so without ComponentGroup replication those single ports must fan out
// to N rooms. If a Damper refuses and a Junction accepts, the conversion has to say so rather than
// discover it on a customer's model.

using System;
using System.Collections.Generic;
using System.Globalization;
using System.IO;
using System.Reflection;
using System.Text;

namespace Inv
{
    internal static class FanOut
    {
        private static readonly StringBuilder log = new StringBuilder();

        private static void W(string format, params object[] args)
        {
            string line = args == null || args.Length == 0 ? format : string.Format(CultureInfo.InvariantCulture, format, args);
            Console.WriteLine(line);
            log.AppendLine(line);
        }

        private static void Flush(string path)
        {
            try
            {
                File.WriteAllText(path, log.ToString());
            }
            catch
            {
            }
        }

        private static bool Duct(object system, string label, object up, int portUp, object down, int portDown)
        {
            try
            {
                object duct = system.GetType().InvokeMember(
                    "AddDuct", BindingFlags.InvokeMethod, null, system,
                    new object[] { up, portUp, down, portDown }, CultureInfo.InvariantCulture);

                W("  duct {0}: {1}", label, duct == null ? "returned NULL" : "ok");
                return duct != null;
            }
            catch (Exception exception)
            {
                Exception inner = exception.InnerException ?? exception;
                W("  duct {0}: THREW {1}: {2}", label, inner.GetType().Name, inner.Message.Replace(Environment.NewLine, " ").Trim());
                return false;
            }
        }

        public static int Run(string path_TPD, int rooms)
        {
            log.Clear();
            W("fanout {0}  rooms={1}", path_TPD, rooms);

            if (File.Exists(path_TPD))
            {
                File.Delete(path_TPD);
            }

            global::TPD.TPDDoc tPDDoc = new global::TPD.TPDDoc();
            try
            {
                tPDDoc.Create(path_TPD);

                object energyCentre = tPDDoc.EnergyCentre;
                object plantRoom = Dump.Call(energyCentre, "AddPlantRoom");
                object system = Dump.Call(plantRoom, "AddSystem");

                object junction_Supply = Dump.Call(system, "AddJunction");
                object damper_Supply = Dump.Call(system, "AddDamper");
                object junction_Extract = Dump.Call(system, "AddJunction");

                W("");
                W("== one JUNCTION port -> one DAMPER ==");
                Duct(system, "junctionSupply[1] -> damperSupply[1]", junction_Supply, 1, damper_Supply, 1);

                List<object> zones = new List<object>();
                for (int i = 0; i < rooms; i++)
                {
                    zones.Add(Dump.Call(system, "AddSystemZone"));
                }

                W("");
                W("== one DAMPER output port -> N zones (the supply-side fan-out) ==");
                int ok_Damper = 0;
                for (int i = 0; i < rooms; i++)
                {
                    if (Duct(system, string.Format("damperSupply[1] -> zone{0}[1]", i + 1), damper_Supply, 1, zones[i], 1))
                    {
                        ok_Damper++;
                    }
                }

                W("");
                W("== N zones -> one JUNCTION input port (the extract-side fan-in) ==");
                int ok_Junction = 0;
                for (int i = 0; i < rooms; i++)
                {
                    if (Duct(system, string.Format("zone{0}[1] -> junctionExtract[1]", i + 1), zones[i], 1, junction_Extract, 1))
                    {
                        ok_Junction++;
                    }
                }

                W("");
                W("== a DAMPER inserted in one room's extract leg, and a room-to-room transfer damper ==");
                object damper_Extract = Dump.Call(system, "AddDamper");
                Duct(system, "zone1[1] -> damperExtract[1]", zones[0], 1, damper_Extract, 1);
                Duct(system, "damperExtract[1] -> junctionExtract[1]", damper_Extract, 1, junction_Extract, 1);

                if (rooms >= 2)
                {
                    object damper_Transfer = Dump.Call(system, "AddDamper");
                    Duct(system, "zone1[1] -> damperTransfer[1]", zones[0], 1, damper_Transfer, 1);
                    Duct(system, "damperTransfer[1] -> zone2[1]", damper_Transfer, 1, zones[1], 1);
                }

                W("");
                W("== output duct counts actually recorded ==");
                W("  damperSupply GetOutputDuctCount(1)  = {0}", Dump.CallText(damper_Supply, "GetOutputDuctCount", 1));
                W("  junctionSupply GetOutputDuctCount(1)= {0}", Dump.CallText(junction_Supply, "GetOutputDuctCount", 1));
                W("  junctionExtract GetInputDuctCount(1)= {0}", Dump.CallText(junction_Extract, "GetInputDuctCount", 1));
                for (int i = 0; i < rooms; i++)
                {
                    W("  zone{0} inputDucts(1)={1} outputDucts(1)={2}",
                        i + 1,
                        Dump.CallText(zones[i], "GetInputDuctCount", 1),
                        Dump.CallText(zones[i], "GetOutputDuctCount", 1));
                }

                Dump.Call(tPDDoc, "Save");

                W("");
                W("VERDICT: damper output fan-out {0}/{1}, junction input fan-in {2}/{3}",
                    ok_Damper, rooms, ok_Junction, rooms);
            }
            finally
            {
                try { tPDDoc.Close(); } catch { }
            }

            Flush(Path.ChangeExtension(path_TPD, ".fanout.log"));
            return 0;
        }
    }
}
