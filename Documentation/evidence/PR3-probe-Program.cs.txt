// PR3 licensed convention probes. One operation per process. Read-only on the input unless the mode
// says it writes, and then only on the copy it is given.
using System;
using System.Collections.Generic;
using System.Globalization;
using System.IO;
using System.Text;

namespace Probe
{
    internal static class Program
    {
        private static readonly StringBuilder log = new StringBuilder();

        private static void W(string format, params object[] args)
        {
            string line = args.Length == 0 ? format : string.Format(CultureInfo.InvariantCulture, format, args);
            Console.WriteLine(line);
            log.AppendLine(line);
        }

        private static int Main(string[] args)
        {
            try
            {
                switch (args[0])
                {
                    case "dump": Dump(args[1]); break;
                    case "yearly": Yearly(args[1]); break;
                    case "selfimpose": SelfImpose(args[1], args[2], args[3], args.Length > 4 ? int.Parse(args[4]) : 0); break;
                    case "tsd": Tsd(args[1], args[2], args[3]); break;
                    default: W("unknown mode"); return 1;
                }
            }
            catch (Exception exception)
            {
                W("EXCEPTION {0}: {1}\n{2}", exception.GetType().Name, exception.Message, exception.StackTrace);
                return 2;
            }
            finally
            {
                if (args.Length > 0)
                {
                    File.WriteAllText(args[args.Length - 1] + ".probe.log", log.ToString());
                }
            }

            return 0;
        }

        // ------------------------------------------------------------------------------------------ dump
        private static void Dump(string path_TBD)
        {
            TBD.TBDDocument tBDDocument = new TBD.TBDDocument();
            tBDDocument.openReadOnly(path_TBD);
            try
            {
                TBD.Building building = tBDDocument.Building;
                W("building year {0}", building.year);

                int count_IZAM = 0;
                while (building.GetIZAM(count_IZAM) != null) count_IZAM++;
                W("IZAM count {0}", count_IZAM);

                int count_IC = 0;
                while (building.GetIC(count_IC) != null) count_IC++;
                W("building IC count {0}", count_IC);

                for (int i = 0; i < count_IC; i++)
                {
                    TBD.InternalCondition ic = building.GetIC(i);
                    List<string> zones = new List<string>();
                    int j = 0;
                    TBD.zone z;
                    while ((z = ic.GetZone(j)) != null) { zones.Add(z.GUID); j++; }
                    List<string> dayTypes = new List<string>();
                    int k = 0;
                    TBD.dayType d;
                    while ((d = ic.GetDayType(k)) != null) { dayTypes.Add(d.name); k++; }
                    W("IC[{0}] '{1}' zones={2} [{3}] dayTypes=[{4}]", i, ic.name, zones.Count, string.Join(",", zones), string.Join(",", dayTypes));
                }

                int index = 0;
                TBD.zone zone;
                while ((zone = building.GetZone(index)) != null)
                {
                    W("ZONE[{0}] guid={1} name='{2}' sizeH={3} sizeC={4} maxH={5} maxC={6}", index, zone.GUID, zone.name, zone.sizeHeating, zone.sizeCooling, zone.maxHeatingLoad, zone.maxCoolingLoad);
                    int n = 0;
                    TBD.InternalCondition ic;
                    while ((ic = zone.GetIC(n)) != null)
                    {
                        TBD.Thermostat t = ic.GetThermostat();
                        W("   IC#{0} '{1}' thermostat: controlRange={2} proportional={3} radiantProportion={4} minDeadBand={5}", n, ic.name, t.controlRange, t.proportionalControl, t.radiantProportion, t.GetMinimumDeadBand());
                        foreach (TBD.Profiles slot in new[] { TBD.Profiles.ticUL, TBD.Profiles.ticLL })
                        {
                            TBD.profile p = t.GetProfile((int)slot);
                            W("      {0}: type={1} factor={2} value={3} name='{4}' extreme(max)={5} extreme(min)={6}", slot, p?.type, p?.factor, p?.value, p?.name, p?.GetExtremeValue(true), p?.GetExtremeValue(false));
                        }
                        TBD.Emitter eh = ic.GetHeatingEmitter();
                        TBD.Emitter ec = ic.GetCoolingEmitter();
                        W("      heating emitter '{0}' radiant={1} type={2} | cooling emitter '{3}' radiant={4} type={5}", eh?.name, eh?.radiantProportion, eh?.emitterType, ec?.name, ec?.radiantProportion, ec?.emitterType);
                        n++;
                    }
                    index++;
                }

                TBD.Calendar calendar = building.GetCalendar();
                int dt = 0;
                TBD.dayType dayType;
                List<string> names = new List<string>();
                while ((dayType = calendar.dayTypes(dt)) != null) { names.Add(dayType.name); dt++; }
                W("calendar dayTypes [{0}]", string.Join(",", names));
                List<string> days = new List<string>();
                for (int day = 1; day <= 366; day++)
                {
                    TBD.dayType dd = null;
                    try { dd = calendar.days(day)?.dayType; } catch (Exception e) { days.Add(day + ":EX " + e.Message); break; }
                    if (day <= 3 || day >= 364) days.Add(day + ":" + (dd == null ? "<null>" : dd.name));
                }
                W("calendar days {0}", string.Join(" ", days));
            }
            finally
            {
                tBDDocument.close();
            }
        }

        // ---------------------------------------------------------------------------------------- yearly
        // Writes onto the COPY it is given and never saves: the question is only how TAS indexes a yearly
        // profile across SetYearlyValues / yearlyValues[i] / GetYearlyValues.
        private static void Yearly(string path_TBD_Copy)
        {
            TBD.TBDDocument tBDDocument = new TBD.TBDDocument();
            tBDDocument.open(path_TBD_Copy);
            try
            {
                TBD.Building building = tBDDocument.Building;
                TBD.profile p = building.GetZone(0).GetIC(0).GetThermostat().GetProfile((int)TBD.Profiles.ticUL);
                p.type = TBD.ProfileTypes.ticYearlyProfile;
                p.factor = 1;

                foreach (int length in new[] { 8760, 8761 })
                {
                    float[] values = new float[length];
                    for (int i = 0; i < length; i++) values[i] = 1000 + i;   // element i carries 1000+i
                    p.SetYearlyValues(values);

                    W("SetYearlyValues(float[{0}]), element i = 1000+i:", length);
                    foreach (int h in new[] { 0, 1, 2, 8759, 8760, 8761 })
                    {
                        try { W("   yearlyValues[{0}] = {1}", h, p.get_yearlyValues(h)); }
                        catch (Exception e) { W("   yearlyValues[{0}] threw {1}", h, e.Message); }
                    }
                    object o = p.GetYearlyValues();
                    Array a = o as Array;
                    W("   GetYearlyValues -> {0} rank={1} lb={2} ub={3} first={4} last={5}", o?.GetType().Name, a?.Rank, a?.GetLowerBound(0), a?.GetUpperBound(0), a?.GetValue(a.GetLowerBound(0)), a?.GetValue(a.GetUpperBound(0)));
                }

                // Per-element write, 1-based, then read both ways.
                for (int h = 1; h <= 8760; h++) p.set_yearlyValues(h, 5000 + h);
                W("set_yearlyValues(h, 5000+h) for h=1..8760:");
                foreach (int h in new[] { 1, 2, 8760 }) W("   yearlyValues[{0}] = {1}", h, p.get_yearlyValues(h));
                Array b = p.GetYearlyValues() as Array;
                W("   GetYearlyValues lb={0} ub={1} first={2} second={3} last={4}", b.GetLowerBound(0), b.GetUpperBound(0), b.GetValue(b.GetLowerBound(0)), b.GetValue(b.GetLowerBound(0) + 1), b.GetValue(b.GetUpperBound(0)));
                W("   extreme(max)={0} extreme(min)={1}", p.GetExtremeValue(true), p.GetExtremeValue(false));
            }
            finally
            {
                tBDDocument.close();
            }
        }

        // ------------------------------------------------------------------------------------ selfimpose
        // The scientific control for the bridge mechanism, independent of TPD: impose on a COPY of a
        // simulated TBD its OWN free-running dry bulb series (from its TSD) as both thermostat limits,
        // simulate, and see whether the copy's air temperature follows the imposed series. `shift` offsets
        // the imposed series by whole hours so a misalignment shows up as a larger deviation.
        private static void SelfImpose(string path_TSD_Source, string path_TBD_Copy, string path_TSD_Out, int shift)
        {
            Dictionary<string, float[]> dryBulb = new Dictionary<string, float[]>(StringComparer.OrdinalIgnoreCase);
            TSD.TSDDocument tSDDocument = new TSD.TSDDocument();
            tSDDocument.openReadOnly(path_TSD_Source);
            try
            {
                TSD.BuildingData buildingData = tSDDocument.SimulationData.GetBuildingData();
                int i = 1;
                TSD.ZoneData zoneData;
                while ((zoneData = buildingData.GetZoneData(i)) != null)
                {
                    object[] raw = ToObjects(zoneData.GetAnnualZoneResult((int)TSD.tsdZoneArray.dryBulbTemp));
                    float[] values = new float[raw.Length];
                    for (int k = 0; k < raw.Length; k++) values[k] = Convert.ToSingle(raw[k]);
                    dryBulb[zoneData.zoneGUID] = values;
                    W("source TSD zone {0} {1} '{2}': {3} dry bulb values", i, zoneData.zoneGUID, zoneData.name, values.Length);
                    i++;
                }
            }
            finally
            {
                tSDDocument.close();
            }

            TBD.TBDDocument tBDDocument = new TBD.TBDDocument();
            tBDDocument.open(path_TBD_Copy);
            try
            {
                TBD.Building building = tBDDocument.Building;
                int index = 0;
                TBD.zone zone;
                while ((zone = building.GetZone(index)) != null)
                {
                    index++;
                    if (!dryBulb.TryGetValue(zone.GUID, out float[] series)) { W("zone {0} has no source series", zone.GUID); continue; }
                    int n = 0;
                    TBD.InternalCondition ic;
                    while ((ic = zone.GetIC(n)) != null)
                    {
                        n++;
                        TBD.Thermostat t = ic.GetThermostat();
                        foreach (TBD.Profiles slot in new[] { TBD.Profiles.ticUL, TBD.Profiles.ticLL })
                        {
                            TBD.profile p = t.GetProfile((int)slot);
                            p.type = TBD.ProfileTypes.ticYearlyProfile;
                            p.factor = 1;
                            for (int h = 1; h <= 8760; h++)
                            {
                                int s = h - 1 + shift;
                                s = ((s % 8760) + 8760) % 8760;
                                p.set_yearlyValues(h, series[s]);
                            }
                        }
                    }
                }

                tBDDocument.save();
                if (File.Exists(path_TSD_Out)) File.Delete(path_TSD_Out);
                DateTime start = DateTime.UtcNow;
                tBDDocument.simulate(1, 365, 0, 1, 0, 0, path_TSD_Out, 1, 0);
                W("simulate returned; output {0} bytes, written {1:o} (start {2:o})", File.Exists(path_TSD_Out) ? new FileInfo(path_TSD_Out).Length : -1, File.Exists(path_TSD_Out) ? File.GetLastWriteTimeUtc(path_TSD_Out) : DateTime.MinValue, start);
                tBDDocument.save();
            }
            finally
            {
                tBDDocument.close();
            }
        }

        // ------------------------------------------------------------------------------------------- tsd
        // Compares a bridge TSD against the source TSD per zone guid: dry bulb follow, resultant, loads.
        private static void Tsd(string path_TSD_Source, string path_TSD_Bridge, string path_Out)
        {
            Dictionary<string, float[][]> a = Read(path_TSD_Source);
            Dictionary<string, float[][]> b = Read(path_TSD_Bridge);
            foreach (KeyValuePair<string, float[][]> kv in a)
            {
                if (!b.TryGetValue(kv.Key, out float[][] bb)) { W("zone {0} missing in bridge", kv.Key); continue; }
                W("zone {0} n={1}/{2}", kv.Key, kv.Value[0].Length, bb[0].Length);
                foreach (int shift in new[] { -1, 0, 1 })
                {
                    double max = 0, sum = 0; int cnt = 0;
                    for (int h = 0; h < 8760; h++)
                    {
                        int s = h + shift;
                        if (s < 0 || s >= 8760) continue;
                        double d = Math.Abs(bb[0][h] - kv.Value[0][s]);
                        max = Math.Max(max, d); sum += d; cnt++;
                    }
                    W("   dryBulb bridge[h] vs source[h{0:+0;-0;+0}]: max|d|={1:0.0000} mean|d|={2:0.0000}", shift, max, sum / cnt);
                }
                double maxR = 0, maxHL = 0, maxCL = 0;
                for (int h = 0; h < 8760; h++)
                {
                    maxR = Math.Max(maxR, Math.Abs(bb[1][h] - kv.Value[1][h]));
                    maxHL = Math.Max(maxHL, bb[2][h]);
                    maxCL = Math.Max(maxCL, bb[3][h]);
                }
                W("   resultant bridge vs source max|d|={0:0.0000}; bridge max heating load={1:0.0} W, max cooling load={2:0.0} W", maxR, maxHL, maxCL);
            }
            File.WriteAllText(path_Out, log.ToString());
        }

        private static Dictionary<string, float[][]> Read(string path_TSD)
        {
            Dictionary<string, float[][]> result = new Dictionary<string, float[][]>(StringComparer.OrdinalIgnoreCase);
            TSD.TSDDocument tSDDocument = new TSD.TSDDocument();
            tSDDocument.openReadOnly(path_TSD);
            try
            {
                TSD.BuildingData buildingData = tSDDocument.SimulationData.GetBuildingData();
                int i = 1;
                TSD.ZoneData zoneData;
                while ((zoneData = buildingData.GetZoneData(i)) != null)
                {
                    float[][] series = new float[4][];
                    int k = 0;
                    foreach (TSD.tsdZoneArray q in new[] { TSD.tsdZoneArray.dryBulbTemp, TSD.tsdZoneArray.resultantTemp, TSD.tsdZoneArray.heatingLoad, TSD.tsdZoneArray.coolingLoad })
                    {
                        object[] raw = ToObjects(zoneData.GetAnnualZoneResult((int)q));
                        series[k] = new float[raw.Length];
                        for (int m = 0; m < raw.Length; m++) series[k][m] = Convert.ToSingle(raw[m]);
                        k++;
                    }
                    result[zoneData.zoneGUID] = series;
                    i++;
                }
            }
            finally
            {
                tSDDocument.close();
            }
            return result;
        }

        private static object[] ToObjects(object value)
        {
            List<object> result = new List<object>();
            foreach (object o in (System.Collections.IEnumerable)value) result.Add(o);
            return result.ToArray();
        }
    }
