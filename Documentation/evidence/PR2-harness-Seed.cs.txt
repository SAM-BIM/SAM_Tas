// Checkpoint 1B, part D: a STRONG control for the no-IZAM acceptance.
//
// The earlier fixtures were weak controls and were reported as such: neither SAM model carries
// ventilation systems, so even the control TBD had IZAM count 0, and its ticV read v=0 f=1 - product
// zero - so nothing non-zero was ever removed.
//
// This builds the missing control by CONTROLLED MUTATION of a real TBD:
//
//   seed  - inject genuine inherited IZAMs and a genuinely non-zero mechanical ticV, then record the
//           full before state including infiltration and the natural-ventilation aperture types
//   clean - run the PRODUCTION cleanup (Modify.RemoveIZAMs + Modify.RemoveVentilationGains) on it
//
// The caller then dumps before and after and compares. No production architecture is changed to make
// the fixture: the mutation is applied to a file, not to the code under test.

using System;
using System.Collections.Generic;
using System.Globalization;
using System.IO;
using System.Text;

namespace Inv
{
    internal static class Seed
    {
        private static readonly StringBuilder log = new StringBuilder();

        private static void W(string format, params object[] args)
        {
            string line = args == null || args.Length == 0 ? format : string.Format(CultureInfo.InvariantCulture, format, args);
            Console.WriteLine(line);
            log.AppendLine(line);
        }

        private static void Flush(string path)
        {
            try
            {
                File.WriteAllText(path, log.ToString());
            }
            catch
            {
            }
        }

        /// <summary>Everything the acceptance compares, read by walking TAS's own accessors.</summary>
        public static void Inventory(string label, TBD.Building building)
        {
            W("---- {0} ----", label);

            int count_IZAM = 0;
            TBD.IZAM iZAM = building.GetIZAM(0);
            while (iZAM != null && count_IZAM < 10000)
            {
                W("  IZAM[{0}] name={1} fromOutside={2}", count_IZAM, iZAM.name, iZAM.fromOutside);
                count_IZAM++;
                iZAM = building.GetIZAM(count_IZAM);
            }

            W("  IZAM COUNT = {0}", count_IZAM);

            int index = 0;
            int count_TicVEffective = 0;
            double sum_TicVEffective = 0.0;
            TBD.InternalCondition internalCondition = building.GetIC(index);

            while (internalCondition != null && index < 10000)
            {
                TBD.InternalGain internalGain = internalCondition.GetInternalGain();
                if (internalGain != null)
                {
                    TBD.profile profile_TicV = internalGain.GetProfile((int)TBD.Profiles.ticV);
                    TBD.profile profile_TicI = internalGain.GetProfile((int)TBD.Profiles.ticI);

                    // A TBD profile is factor x value, so the EFFECTIVE contribution is the product.
                    // The earlier weak control had value 0 and factor 1 - product zero, nothing to remove.
                    double ticV_value = profile_TicV == null ? 0 : profile_TicV.value;
                    double ticV_factor = profile_TicV == null ? 0 : profile_TicV.factor;
                    double ticV_setback = profile_TicV == null ? 0 : profile_TicV.setbackValue;
                    double ticV_effective = ticV_value * ticV_factor;
                    bool ticV_schedule = profile_TicV != null && profile_TicV.schedule != null;

                    double ticI_value = profile_TicI == null ? 0 : profile_TicI.value;
                    double ticI_factor = profile_TicI == null ? 0 : profile_TicI.factor;

                    if (ticV_effective != 0.0)
                    {
                        count_TicVEffective++;
                        sum_TicVEffective += ticV_effective;
                    }

                    W("  IC[{0}] name={1} ticV(v={2} f={3} sb={4} sched={5} EFFECTIVE={6}) ticI(v={7} f={8})",
                        index,
                        internalCondition.name,
                        ticV_value, ticV_factor, ticV_setback, ticV_schedule, ticV_effective,
                        ticI_value, ticI_factor);
                }

                index++;
                internalCondition = building.GetIC(index);
            }

            W("  IC COUNT = {0}", index);
            W("  ticV EFFECTIVE NON-ZERO COUNT = {0}", count_TicVEffective);
            W("  ticV EFFECTIVE SUM = {0}", sum_TicVEffective.ToString("R", CultureInfo.InvariantCulture));

            // Natural ventilation: aperture types and their opening schedules, which must not move.
            int count_ApertureType = 0;
            TBD.ApertureType apertureType = building.GetApertureType(0);
            while (apertureType != null && count_ApertureType < 10000)
            {
                TBD.profile profile = null;
                try
                {
                    profile = apertureType.GetProfile();
                }
                catch
                {
                }

                W("  APERTURETYPE[{0}] name={1} Cd={2} sheltered={3} profile(v={4} f={5} sched={6})",
                    count_ApertureType,
                    apertureType.name,
                    apertureType.dischargeCoefficient,
                    apertureType.sheltered,
                    profile == null ? 0 : profile.value,
                    profile == null ? 0 : profile.factor,
                    profile == null || profile.schedule == null ? "<none>" : profile.schedule.name);

                count_ApertureType++;
                apertureType = building.GetApertureType(count_ApertureType);
            }

            W("  APERTURETYPE COUNT = {0}", count_ApertureType);
        }

        /// <summary>Injects genuine inherited IZAMs and a genuinely non-zero mechanical ticV.</summary>
        public static int SeedTbd(string path_Source, string path_TBD, int izamCount, double ticV_Value)
        {
            log.Clear();
            W("seed {0} -> {1}  (inject {2} IZAMs, ticV value {3})", path_Source, path_TBD, izamCount, ticV_Value);

            if (!File.Exists(path_Source))
            {
                W("REFUSED: source TBD does not exist");
                Flush(Path.ChangeExtension(path_TBD, ".seed.log"));
                return 5;
            }

            File.Copy(path_Source, path_TBD, true);

            TBD.TBDDocument tBDDocument = new TBD.TBDDocument();
            try
            {
                tBDDocument.open(path_TBD);
                TBD.Building building = tBDDocument.Building;

                if (building == null)
                {
                    W("REFUSED: no building");
                    return 3;
                }

                W("");
                W("== INJECTING inherited IZAMs ==");
                for (int i = 0; i < izamCount; i++)
                {
                    try
                    {
                        TBD.IZAM iZAM = building.AddIZAM(null);
                        if (iZAM == null)
                        {
                            W("  AddIZAM({0}) returned NULL", i);
                            continue;
                        }

                        iZAM.name = string.Concat("INHERITED_IZAM_", i + 1);
                        iZAM.description = "Injected to give the no-IZAM sweep something real to remove.";
                        W("  added IZAM '{0}'", iZAM.name);
                    }
                    catch (Exception exception)
                    {
                        Exception inner = exception.InnerException ?? exception;
                        W("  AddIZAM({0}) THREW {1}: {2}", i, inner.GetType().Name, inner.Message.Trim());
                    }
                }

                W("");
                W("== INJECTING a genuinely non-zero mechanical ticV ==");
                int index = 0;
                int count_Set = 0;
                TBD.InternalCondition internalCondition = building.GetIC(index);

                while (internalCondition != null && index < 10000)
                {
                    TBD.InternalGain internalGain = internalCondition.GetInternalGain();
                    TBD.profile profile = internalGain?.GetProfile((int)TBD.Profiles.ticV);

                    if (profile != null)
                    {
                        try
                        {
                            profile.value = (float)ticV_Value;
                            profile.factor = (float)1.0;
                            profile.setbackValue = (float)(ticV_Value / 2.0);
                            count_Set++;
                        }
                        catch (Exception exception)
                        {
                            W("  IC[{0}] ticV write THREW {1}", index, exception.GetType().Name);
                        }
                    }

                    index++;
                    internalCondition = building.GetIC(index);
                }

                W("  set a non-zero ticV on {0} internal condition(s)", count_Set);

                W("");
                Inventory("BEFORE (seeded, pre-cleanup)", building);

                tBDDocument.save();
                W("");
                W("saved {0} bytes", new FileInfo(path_TBD).Length);
            }
            finally
            {
                try { tBDDocument.close(); } catch { }
            }

            Flush(Path.ChangeExtension(path_TBD, ".seed.log"));
            return 0;
        }

        /// <summary>Runs the PRODUCTION cleanup on the seeded TBD - the code under test, unmodified.</summary>
        public static int CleanTbd(string path_TBD)
        {
            log.Clear();
            W("clean {0}", path_TBD);

            if (!File.Exists(path_TBD))
            {
                W("REFUSED: TBD does not exist");
                Flush(Path.ChangeExtension(path_TBD, ".clean.log"));
                return 5;
            }

            TBD.TBDDocument tBDDocument = new TBD.TBDDocument();
            try
            {
                tBDDocument.open(path_TBD);
                TBD.Building building = tBDDocument.Building;

                W("");
                Inventory("BEFORE (as read by this process)", building);

                W("");
                W("== PRODUCTION CLEANUP ==");

                bool removed = SAM.Analytical.Tas.Modify.RemoveIZAMs(building);
                W("  Modify.RemoveIZAMs returned {0}", removed);

                // RemoveVentilationGains only requires a non-null cluster; it reads the TBD, not the model.
                List<string> names = SAM.Analytical.Tas.Modify.RemoveVentilationGains(
                    tBDDocument, new SAM.Analytical.AdjacencyCluster());

                W("  Modify.RemoveVentilationGains touched {0} internal condition(s)", names == null ? 0 : names.Count);

                tBDDocument.save();

                W("");
                Inventory("AFTER (post-cleanup)", building);

                W("");
                W("saved {0} bytes", new FileInfo(path_TBD).Length);
            }
            finally
            {
                try { tBDDocument.close(); } catch { }
            }

            Flush(Path.ChangeExtension(path_TBD, ".clean.log"));
            return 0;
        }
    }
}
