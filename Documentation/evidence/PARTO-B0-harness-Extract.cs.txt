using System;
using System.Collections;
using System.Collections.Generic;
using System.Globalization;
using System.IO;
using System.Linq;

namespace B0P
{
    public static partial class Steps
    {
        public static double[] Series(TSD.ZoneData zd, TSD.tsdZoneArray arr)
        {
            double[] result = new double[8760];
            for (int d = 1; d <= 365; d++)
            {
                float[] day = (zd.GetDailyZoneResult(d, (short)arr) as IEnumerable).Cast<float>().ToArray();
                int start = (d - 1) * 24;
                for (int h = 0; h < 24; h++) result[start + h] = day[h];
            }
            return result;
        }

        public static double[] BuildingSeries(TSD.BuildingData bd, TSD.tsdBuildingArray arr)
        {
            double[] result = new double[8760];
            for (int d = 1; d <= 365; d++)
            {
                float[] day = (bd.GetDailyBuildingResult(d, (short)arr) as IEnumerable).Cast<float>().ToArray();
                int start = (d - 1) * 24;
                for (int h = 0; h < 24; h++) result[start + h] = day[h];
            }
            return result;
        }

        // extract <tsd> <zoneGuidCsv> <arrayCsv> <out.csv>
        public static void Extract(string[] args)
        {
            string path = args[1];
            string[] guids = args[2].Split(',');
            string[] arrays = args[3].Split(',');
            string outPath = args[4];

            TSD.TSDDocument doc = new TSD.TSDDocument();
            doc.openReadOnly(path);
            try
            {
                TSD.SimulationData sd = doc.SimulationData;
                TSD.BuildingData bd = sd.GetBuildingData();
                Dictionary<string, TSD.ZoneData> byGuid = new Dictionary<string, TSD.ZoneData>(StringComparer.OrdinalIgnoreCase);
                int i = 1;
                TSD.ZoneData zd = bd.GetZoneData(i);
                while (zd != null) { byGuid[zd.zoneGUID] = zd; i++; zd = bd.GetZoneData(i); }

                using (StreamWriter sw = new StreamWriter(outPath, false))
                {
                    List<string> header = new List<string> { "hour" };
                    List<double[]> cols = new List<double[]>();
                    foreach (string a in arrays)
                    {
                        if (a.StartsWith("BLD:"))
                        {
                            TSD.tsdBuildingArray ba = (TSD.tsdBuildingArray)Enum.Parse(typeof(TSD.tsdBuildingArray), a.Substring(4));
                            header.Add(a);
                            cols.Add(BuildingSeries(bd, ba));
                        }
                    }
                    foreach (string g in guids)
                    {
                        if (!byGuid.ContainsKey(g)) { Console.WriteLine("MISSING ZONE {0}", g); continue; }
                        foreach (string a in arrays)
                        {
                            if (a.StartsWith("BLD:")) continue;
                            TSD.tsdZoneArray za = (TSD.tsdZoneArray)Enum.Parse(typeof(TSD.tsdZoneArray), a);
                            header.Add(g + "|" + a);
                            cols.Add(Series(byGuid[g], za));
                        }
                    }
                    sw.WriteLine(string.Join(",", header));
                    for (int h = 0; h < 8760; h++)
                    {
                        sw.Write(h);
                        for (int c = 0; c < cols.Count; c++) { sw.Write(','); sw.Write(cols[c][h].ToString("R", CultureInfo.InvariantCulture)); }
                        sw.WriteLine();
                    }
                }
                Console.WriteLine("WROTE {0}", outPath);
            }
            finally { doc.close(); }
        }
    }
}
