// Is a component's position presentation only? Two modes, each one document cycle:
//
//   spread <in.tpd> <out.tpd>   copy the document and move EVERY top-level component of every system to
//                               its own cell of a wide grid (and set every direction LeftRight). Nothing
//                               else is touched - no duct, duty, guid or schedule.
//   zt     <in.tpd> <out.txt>   on a copy, simulate each air system with a raw late-bound call and write
//                               every zone's 8760 hourly ZoneTemperature (variable 9), keyed by zone guid.
//
// Comparing zt of the original against zt of the spread copy answers the question natively.
//
// Nothing here is production code.

using System;
using System.Collections.Generic;
using System.Globalization;
using System.IO;
using System.Reflection;
using System.Text;

namespace Inv
{
    internal static class Spread
    {
        private static int I(object value)
        {
            try { return value == null ? 0 : Convert.ToInt32(value, CultureInfo.InvariantCulture); } catch { return 0; }
        }

        private static IEnumerable<object> Systems(object energyCentre)
        {
            int count_PlantRoom = I(Dump.Call(energyCentre, "GetPlantRoomCount"));
            for (int p = 1; p <= count_PlantRoom; p++)
            {
                object plantRoom = Dump.Call(energyCentre, "GetPlantRoom", p);
                int count_System = I(Dump.Call(plantRoom, "GetSystemCount"));
                for (int s = 1; s <= count_System; s++)
                {
                    yield return Dump.Call(plantRoom, "GetSystem", s);
                }
            }
        }

        private static List<object> TopLevel(object system)
        {
            List<object> result = new List<object>();
            for (int index = 1; index <= 20000; index++)
            {
                object component = Dump.Call(system, "GetComponent", index);
                if (component == null)
                {
                    break;
                }

                result.Add(component);
            }

            return result;
        }

        public static int Move(string path_In, string path_Out)
        {
            File.Copy(path_In, path_Out, true);
            int count = 0;

            global::TPD.TPDDoc tPDDoc = new global::TPD.TPDDoc();
            try
            {
                tPDDoc.Open(path_Out);
                foreach (object system in Systems(tPDDoc.EnergyCentre))
                {
                    int cell = 0;
                    foreach (object component in TopLevel(system))
                    {
                        ((dynamic)component).SetPosition(2000 + (cell % 6) * 200, 2000 + (cell / 6) * 200);
                        ((dynamic)component).SetDirection(global::TPD.tpdDirection.tpdLeftRight);
                        cell++;
                        count++;
                    }
                }

                Dump.Call(tPDDoc, "Save");
            }
            finally
            {
                try { tPDDoc.Close(); } catch { }
            }

            Console.WriteLine("moved " + count + " component(s) into " + path_Out);
            return 0;
        }

        public static int ZoneTemperatures(string path_TPD, string pathOut)
        {
            string path_Copy = Path.ChangeExtension(path_TPD, ".zt.tpd");
            File.Copy(path_TPD, path_Copy, true);

            StringBuilder stringBuilder = new StringBuilder();

            global::TPD.TPDDoc tPDDoc = new global::TPD.TPDDoc();
            try
            {
                tPDDoc.Open(path_Copy);
                foreach (object system in Systems(tPDDoc.EnergyCentre))
                {
                    string answered = Dump.CallText(system, "Simulate", 1, 8760, 0);
                    stringBuilder.AppendLine("#SYSTEM " + Dump.Get(system, "Name") + " answered " + answered);

                    foreach (object component in TopLevel(system))
                    {
                        if (!(component is global::TPD.SystemZone))
                        {
                            continue;
                        }

                        object data = null;
                        try
                        {
                            data = component.GetType().InvokeMember("GetResultsData", BindingFlags.InvokeMethod, null, component,
                                new object[] { (int)global::TPD.tpdResultsPeriod.tpdResultsPeriodHourly, (int)global::TPD.tpdCombinerType.tpdCombinerTypeMax, 9, 1, 8760 },
                                CultureInfo.InvariantCulture);
                        }
                        catch
                        {
                        }

                        List<string> values = new List<string>();
                        if (data is Array array)
                        {
                            foreach (object element in array)
                            {
                                values.Add(Convert.ToDouble(element, CultureInfo.InvariantCulture).ToString("R", CultureInfo.InvariantCulture));
                            }
                        }

                        //Keyed by the zone's paired ZoneLoad guid - the TBD zone guid, which is stable across
                        //documents built from the same TBD. A native SystemZone guid is not.
                        object zoneLoad = Dump.Call(component, "GetSystemZoneZoneLoad");
                        string key = (Dump.Obj(zoneLoad, "GUID") as string) ?? (Dump.Obj(component, "GUID") as string);

                        stringBuilder.AppendLine(key + "\t" + values.Count + "\t" + string.Join(",", values));
                    }
                }
            }
            finally
            {
                try { tPDDoc.Close(); } catch { }
            }

            File.WriteAllText(pathOut, stringBuilder.ToString());
            Console.WriteLine("written " + pathOut);
            return 0;
        }
    }
}
