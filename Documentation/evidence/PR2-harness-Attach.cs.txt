// Checkpoint 1B, part B: attach a REAL no-IZAM TSD to an energy centre, bind zones to actual zone
// loads, and settle two things that cannot be measured on a zone with no TSD:
//
//   1. IDENTITY - is the SystemZone component guid the same as its ZoneLoad guid, and does
//      TSDData.GetZoneLoadForGuid resolve the load back?
//
//   2. UNITS - what absolute unit does SizedFlowVariable.Value carry? Measured, not inferred: a zone
//      is sized by TAS's own ACH rule against a zone load whose volume is known, and the sized answer
//      is compared to volume * ACH in each candidate unit. Only one candidate can match.

using System;
using System.Collections.Generic;
using System.Globalization;
using System.IO;
using System.Reflection;
using System.Text;

namespace Inv
{
    internal static class Attach
    {
        private static readonly StringBuilder log = new StringBuilder();

        private static void W(string format, params object[] args)
        {
            string line = args == null || args.Length == 0 ? format : string.Format(CultureInfo.InvariantCulture, format, args);
            Console.WriteLine(line);
            log.AppendLine(line);
        }

        private static void SetProperty(object target, string name, object value)
        {
            target.GetType().InvokeMember(
                name, BindingFlags.SetProperty, null, target, new object[] { value }, CultureInfo.InvariantCulture);
        }

        private static double Dbl(object value)
        {
            try
            {
                return value == null ? double.NaN : Convert.ToDouble(value, CultureInfo.InvariantCulture);
            }
            catch
            {
                return double.NaN;
            }
        }


        private static void Duct(object system, string label, object up, int portUp, object down, int portDown)
        {
            try
            {
                system.GetType().InvokeMember(
                    "AddDuct", BindingFlags.InvokeMethod, null, system,
                    new object[] { up, portUp, down, portDown }, CultureInfo.InvariantCulture);
                W("  duct {0}: ok", label);
            }
            catch (Exception exception)
            {
                Exception inner = exception.InnerException ?? exception;
                W("  duct {0}: THREW {1}: {2}", label, inner.GetType().Name, inner.Message.Replace(Environment.NewLine, " ").Trim());
            }
        }

        public static int Run(string path_TPD, string path_TSD, int achForSizing)
        {
            log.Clear();
            W("attach-tsd {0} <- {1}  (sizing at {2} ACH)", path_TPD, path_TSD, achForSizing);

            if (!File.Exists(path_TSD))
            {
                W("REFUSED: TSD does not exist: {0}", path_TSD);
                return 5;
            }

            if (File.Exists(path_TPD))
            {
                File.Delete(path_TPD);
            }

            global::TPD.TPDDoc tPDDoc = new global::TPD.TPDDoc();
            try
            {
                tPDDoc.Create(path_TPD);

                object energyCentre = tPDDoc.EnergyCentre;
                if (energyCentre == null)
                {
                    W("REFUSED: no energy centre");
                    return 3;
                }

                W("");
                W("== attaching the no-IZAM TSD ==");
                bool attached = false;
                foreach (int hWnd in new[] { 0, 1 })
                {
                    try
                    {
                        energyCentre.GetType().InvokeMember(
                            "AddTSDData", BindingFlags.InvokeMethod, null, energyCentre,
                            new object[] { path_TSD, hWnd }, CultureInfo.InvariantCulture);
                        W("AddTSDData(hWnd={0}) returned without throwing", hWnd);
                    }
                    catch (Exception exception)
                    {
                        Exception inner = exception.InnerException ?? exception;
                        W("AddTSDData(hWnd={0}) THREW {1}: {2}", hWnd, inner.GetType().Name, inner.Message.Replace(Environment.NewLine, " ").Trim());
                    }

                    if ((int)Dbl(Dump.Call(energyCentre, "GetTSDDataCount")) > 0)
                    {
                        attached = true;
                        W("attached with hWnd={0}", hWnd);
                        break;
                    }
                }

                if (!attached)
                {
                    W("neither hWnd value attached the TSD");
                }

                int count_TSDData = (int)Dbl(Dump.Call(energyCentre, "GetTSDDataCount"));
                W("TSDDataCount = {0}", count_TSDData);

                if (count_TSDData < 1)
                {
                    W("REFUSED: the TSD did not attach");
                    return 6;
                }

                object tSDData = Dump.Call(energyCentre, "GetTSDData", 1);
                W("TBDPath = {0}", Dump.Get(tSDData, "TBDPath"));
                W("TSDPath = {0}", Dump.Get(tSDData, "TSDPath"));
                W("StartDay = {0}  EndDay = {1}", Dump.Get(tSDData, "StartDay"), Dump.Get(tSDData, "EndDay"));

                int count_ZoneLoad = (int)Dbl(Dump.Call(tSDData, "GetZoneLoadCount"));
                W("ZoneLoadCount = {0}", count_ZoneLoad);

                if (count_ZoneLoad < 1)
                {
                    W("REFUSED: the attached TSD carries no zone loads");
                    return 7;
                }

                object plantRoom = Dump.Call(energyCentre, "AddPlantRoom");
                object system = Dump.Call(plantRoom, "AddSystem");

                W("");
                W("== binding each zone load to its own SystemZone ==");
                W("(SystemComponent.AddZoneLoad is the explicit pairing point - no name is involved)");

                List<object> zones = new List<object>();
                List<object> zoneLoads = new List<object>();

                for (int i = 1; i <= count_ZoneLoad; i++)
                {
                    object zoneLoad = Dump.Call(tSDData, "GetZoneLoad", i);
                    if (zoneLoad == null)
                    {
                        continue;
                    }

                    object zone = Dump.Call(system, "AddSystemZone");
                    if (zone == null)
                    {
                        W("  zone {0}: AddSystemZone returned NULL", i);
                        continue;
                    }

                    try
                    {
                        zone.GetType().InvokeMember(
                            "AddZoneLoad", BindingFlags.InvokeMethod, null, zone,
                            new object[] { zoneLoad }, CultureInfo.InvariantCulture);
                    }
                    catch (Exception exception)
                    {
                        Exception inner = exception.InnerException ?? exception;
                        W("  zone {0}: AddZoneLoad THREW {1}: {2}", i, inner.GetType().Name, inner.Message.Trim());
                        continue;
                    }

                    zones.Add(zone);
                    zoneLoads.Add(zoneLoad);

                    string guid_Zone = Dump.Get(zone, "GUID");
                    string guid_Load = Dump.Get(zoneLoad, "GUID");

                    W("  [{0}] load Name={1}", i, Dump.Get(zoneLoad, "Name"));
                    W("      ZoneLoad.GUID       = {0}", guid_Load);
                    W("      SystemZone.GUID     = {0}", guid_Zone);
                    W("      EQUAL?              = {0}", string.Equals(guid_Zone, guid_Load, StringComparison.OrdinalIgnoreCase));
                    W("      FloorArea={0} Volume={1}", Dump.Get(zoneLoad, "FloorArea"), Dump.Get(zoneLoad, "Volume"));

                    // Does the zone hand its load back explicitly?
                    object back = Dump.Call(zone, "GetSystemZoneZoneLoad");
                    W("      GetSystemZoneZoneLoad -> {0}", back == null ? "<null>" : Dump.Get(back, "GUID"));

                    // Does the load-side bridge resolve it?
                    object resolved = Dump.Call(tSDData, "GetZoneLoadForGuid", guid_Load.Trim('"'));
                    W("      GetZoneLoadForGuid    -> {0}", resolved == null ? "<null>" : Dump.Get(resolved, "Name"));

                    // And does the component round trip through the system?
                    object roundTripped = Dump.Call(system, "GetComponentByGUID", guid_Zone.Trim('"'));
                    W("      GetComponentByGUID    -> {0}", roundTripped == null ? "<null>" : Dump.Get(roundTripped, "GUID"));

                    // Size this zone by TAS's own ACH rule so the unit can be read off the answer.
                    object flowRate = Dump.Obj(zone, "FlowRate");
                    if (flowRate != null)
                    {
                        try
                        {
                            SetProperty(flowRate, "Type", (int)global::TPD.tpdSizedVariable.tpdSizedVariableSize);
                            SetProperty(flowRate, "Method", (int)global::TPD.tpdSizeFlowMethod.tpdSizeFlowACH);
                            SetProperty(flowRate, "SizeValue1", (double)achForSizing);
                            SetProperty(flowRate, "SizeValue2", (double)achForSizing);
                            SetProperty(flowRate, "SizeFraction", 1.0);
                        }
                        catch (Exception exception)
                        {
                            W("      sizing setup threw {0}", exception.GetType().Name);
                        }
                    }
                }

                W("");
                W("== a topologically COMPLETE system, shaped like the TAS-authored reference ==");
                W("(intake junction -> supply fan -> zones -> extract fan -> exhaust junction)");

                object junction_Intake = Dump.Call(system, "AddJunction");
                object junction_Exhaust = Dump.Call(system, "AddJunction");
                object fan = Dump.Call(system, "AddFan");
                object fan_Extract = Dump.Call(system, "AddFan");

                try
                {
                    SetProperty(fan, "DesignFlowType", (int)global::TPD.tpdFlowRateType.tpdFlowRateAllAttachedZonesFlowRate);
                    SetProperty(fan_Extract, "DesignFlowType", (int)global::TPD.tpdFlowRateType.tpdFlowRateAllAttachedZonesFlowRate);
                }
                catch
                {
                }

                Duct(system, "intake -> supply fan", junction_Intake, 1, fan, 1);

                for (int i = 0; i < zones.Count; i++)
                {
                    Duct(system, string.Concat("supply fan -> zone ", i + 1), fan, 1, zones[i], 1);
                    Duct(system, string.Concat("zone ", i + 1, " -> extract fan"), zones[i], 1, fan_Extract, 1);
                }

                Duct(system, "extract fan -> exhaust", fan_Extract, 1, junction_Exhaust, 1);

                W("");
                W("== does TAS need a PLANT-side component to simulate at all? ==");
                W("  plantRoom.GetComponentCount before = {0}", Dump.CallText(plantRoom, "GetComponentCount"));
                try
                {
                    object boiler = Dump.Call(plantRoom, "AddBoiler");
                    W("  AddBoiler -> {0}", boiler == null ? "NULL" : "ok");
                }
                catch (Exception exception)
                {
                    W("  AddBoiler threw {0}", exception.GetType().Name);
                }
                W("  plantRoom.GetComponentCount after  = {0}", Dump.CallText(plantRoom, "GetComponentCount"));

                W("");
                W("== design condition, so a sizing pass has something to size against ==");
                try
                {
                    object designCondition = Dump.Call(energyCentre, "AddDesignCondition");
                    W("  AddDesignCondition -> {0} Name={1}", designCondition == null ? "NULL" : "ok", Dump.Get(designCondition, "Name"));

                    if (designCondition != null)
                    {
                        for (int i = 0; i < zones.Count; i++)
                        {
                            object flowRate = Dump.Obj(zones[i], "FlowRate");
                            if (flowRate == null)
                            {
                                continue;
                            }

                            try
                            {
                                flowRate.GetType().InvokeMember(
                                    "AddDesignCondition", BindingFlags.InvokeMethod, null, flowRate,
                                    new object[] { designCondition }, CultureInfo.InvariantCulture);
                            }
                            catch (Exception exception)
                            {
                                W("  zone[{0}] AddDesignCondition threw {1}", i + 1, exception.GetType().Name);
                            }
                        }

                        W("  design condition attached to {0} zone flow variables", zones.Count);
                    }
                }
                catch (Exception exception)
                {
                    W("  design condition setup threw {0}", exception.GetType().Name);
                }

                tPDDoc.Save();
                W("saved {0} bytes", new FileInfo(path_TPD).Length);

                W("");
                W("== SIZING: let TAS compute the ACH-sized flow, then read the unit off the answer ==");

                // ITPD.Simulate returns a STRING. Production discards it; capture it here, because a
                // status line from TAS itself is far better evidence than a literal bool.
                try
                {
                    string returned = tPDDoc.Simulate(1, 24, 0);
                    W("Simulate(1,24,0) RETURNED STRING: [{0}] (length {1})", returned ?? "<null>", returned == null ? 0 : returned.Length);
                }
                catch (Exception exception)
                {
                    Exception inner = exception.InnerException ?? exception;
                    W("Simulate threw {0}: {1}", inner.GetType().Name, inner.Message.Replace(Environment.NewLine, " ").Trim());
                }

                tPDDoc.Save();

                W("");
                W("== UNIT VERDICT ==");
                W("For each zone: TAS was asked for {0} ACH against a zone load of known volume.", achForSizing);
                W("volume[m3] * ACH / 3600 = m3/s ; * 1000 = l/s ; volume * ACH = m3/h");
                W("");

                for (int i = 0; i < zones.Count; i++)
                {
                    object flowRate = Dump.Obj(zones[i], "FlowRate");
                    double sized = Dbl(Dump.Obj(flowRate, "Value"));
                    double volume = Dbl(Dump.Obj(zoneLoads[i], "Volume"));

                    double expected_m3s = volume * achForSizing / 3600.0;
                    double expected_lps = expected_m3s * 1000.0;
                    double expected_m3h = volume * achForSizing;

                    W("  zone[{0}] volume={1} sizedValue={2} Type={3}",
                        i + 1,
                        volume.ToString("R", CultureInfo.InvariantCulture),
                        sized.ToString("R", CultureInfo.InvariantCulture),
                        Dump.Get(flowRate, "Type"));
                    W("      if m3/s -> {0}   if l/s -> {1}   if m3/h -> {2}",
                        expected_m3s.ToString("F4", CultureInfo.InvariantCulture),
                        expected_lps.ToString("F4", CultureInfo.InvariantCulture),
                        expected_m3h.ToString("F4", CultureInfo.InvariantCulture));

                    string verdict = "no match";
                    if (Math.Abs(sized - expected_lps) < Math.Max(0.01, expected_lps * 0.001)) { verdict = "MATCHES l/s"; }
                    else if (Math.Abs(sized - expected_m3s) < Math.Max(0.0001, expected_m3s * 0.001)) { verdict = "MATCHES m3/s"; }
                    else if (Math.Abs(sized - expected_m3h) < Math.Max(0.01, expected_m3h * 0.001)) { verdict = "MATCHES m3/h"; }
                    W("      -> {0}", verdict);
                }
            }
            finally
            {
                try { tPDDoc.Close(); } catch { }
            }

            try
            {
                File.WriteAllText(Path.ChangeExtension(path_TPD, ".attach.log"), log.ToString());
            }
            catch
            {
            }

            return 0;
        }
    }
}
