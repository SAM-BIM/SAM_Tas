// PR2 corrective re-acceptance harness. One operation per process; every path absolute and short.

using System;
using System.IO;

namespace Inv
{
    internal static class Program
    {
        private static void Usage()
        {
            Console.WriteLine("modes:");
            Console.WriteLine("  prep <in.sam> <outdir> [zoneCategory|-] [sizepartf|nosize]");
            Console.WriteLine("  vs   <in.sam> <out.txt>");
            Console.WriteLine("  pr1  <prepared.sam> <outdir> [none|nounit|noterminal]");
            Console.WriteLine("  source <prepared.sam> <outdir> <weather.twd> [sizing|full]");
            Console.WriteLine("  run    <designDir> <sourceDir> <outdir> [sizing|full][ noconst]");
            Console.WriteLine("  source-check <in.tbd> <out.txt>");
            Console.WriteLine("  topo   <in.tpd> <dir with intent.tsv/claim-*.tsv> [out.txt]");
            Console.WriteLine("  graph  <in.tpd> <out.txt>");
            Console.WriteLine("  net    <in.tpd> <out.txt>");
            Console.WriteLine("  layout <in.tpd> [out.txt]");
            Console.WriteLine("  zt     <in.tpd> <out.txt>");
            Console.WriteLine("  fanprobe <in.tpd> <startHour> <endHour> [out.txt]");
        }

        private static int Main(string[] args)
        {
            try
            {
                if (args == null || args.Length == 0)
                {
                    Usage();
                    return 1;
                }

                switch (args[0].ToLowerInvariant())
                {
                    case "prep":
                        return Prep.Main(
                            args[1],
                            args[2],
                            args.Length > 3 && args[3] != "-" ? args[3] : null,
                            args.Length <= 4 || !string.Equals(args[4], "nosize", StringComparison.OrdinalIgnoreCase));

                    case "vs":
                        return Vs.Main(args[1], args[2]);

                    case "pr1":
                        return Pr1.Main(args[1], args[2], args.Length > 3 ? args[3] : "none");

                    case "source":
                        return Src.Main(args[1], args[2], args[3], args.Length > 4 ? args[4] : "sizing");

                    case "diag":
                        return Diag.Main(args[1], args[2], args[3], args.Length > 4 ? args[4] : "sizing");

                    case "run":
                        return Run.Main(args[1], args[2], args[3], args.Length > 4 ? args[4] : "sizing");

                    case "source-check":
                        return Check.Run(args[1], args.Length > 2 ? args[2] : null);

                    case "graph":
                        Graph.Walk(args[1], args[2]);
                        return 0;

                    case "net":
                        Net.Run(args[1], args[2]);
                        return 0;

                    case "layout":
                        return Layout.Run(args[1], args.Length > 2 ? args[2] : null);

                    case "zt":
                        return Spread.ZoneTemperatures(args[1], args[2]);

                    case "fanprobe":
                        return FanProbe.Run(args[1], int.Parse(args[2]), int.Parse(args[3]), args.Length > 4 ? args[4] : null);

                    case "topo":
                        return Topo.Run(args[1], args[2], args.Length > 3 ? args[3] : null);

                    case "spread":
                        return Spread.Move(args[1], args[2]);

                    default:
                        Usage();
                        return 1;
                }
            }
            catch (Exception exception)
            {
                Console.WriteLine("EXCEPTION {0}: {1}", exception.GetType().Name, exception.Message);
                Console.WriteLine(exception.StackTrace ?? string.Empty);
                return 2;
            }
        }
    }
}
