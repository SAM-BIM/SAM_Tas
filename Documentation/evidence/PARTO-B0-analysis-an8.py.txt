import pandas as pd, numpy as np
W='C:/TasOut/b0w/'
D=pd.read_csv(W+'out/hourly-dataset.csv')
A=pd.read_csv(W+'A.csv'); B=pd.read_csv(W+'Bbridge.csv'); S=pd.read_csv(W+'Bsrc.csv')
DU=pd.read_csv('C:/TasOut/b0r4/ducts.csv')
BOUND=['Bathroom_2','Studio 1_0','Bedroom 2_3','Ensuite_5','Kitchen_4','Bedroom 2_6','Ensuite_8','Kitchen_7']

print('=== ANALYSIS 7: ventilation-equivalence checks (native results) ===')
print('B0 bridge TBD airMovementGain (inter-zone air movement) per room:')
for r in BOUND:
    v=D[D.room==r].B_airMoveGain.values
    print('   %-12s min=%.9f max=%.9f  (Reference A: min=%.2f max=%.2f)'%(r,v.min(),v.max(),D[D.room==r].A_airMoveGain.values.min(),D[D.room==r].A_airMoveGain.values.max()))
print('B0 no-IZAM SOURCE TBD airMovementGain: max over all rooms = %.9f'%max(np.abs(S[c].values).max() for c in S.columns if 'airMovementGain' in c))
print('')
print('infiltration (ach) A vs B per room, and ventilation array:')
for r in BOUND:
    d=D[D.room==r]
    print('   %-12s A_inf=%.6f B_inf=%.6f S_inf=%.6f  max|A-B|=%.9f   A_vent max=%.6f B_vent max=%.6f'%(
        r,d.A_inf.mean(),d.B_inf.mean(),d.S_inf.mean(),np.abs(d.A_inf.values-d.B_inf.values).max(),d.A_vent.values.max(),d.B_vent.values.max()))
print('')
print('AHUGain (would be non-zero if the TBD itself carried an AHU/mech-vent gain):')
for r in BOUND:
    d=D[D.room==r]
    print('   %-12s A max|AHUGain|=%.6f  B max|AHUGain|=%.6f  S max|AHUGain|=%.6f'%(r,np.abs(d.A_ahu.values).max(),np.abs(d.B_ahu.values).max(),np.abs(d.S_ahu.values).max()))
print('')
print('=== duct flows: every leg constant at its design value, and per-system supply == extract ===')
flows={}
for c in DU.columns:
    if c.endswith('|var1'):
        v=DU[c].values
        flows[c[:-5]]=(v.min(),v.max())
for k in sorted(flows):
    lo,hi=flows[k]
    if abs(hi-lo)>1e-9: print('   VARYING %s : %.3f..%.3f'%(k,lo,hi))
print('   all %d duct flow series constant across 8760 h: %s'%(len(flows),all(abs(hi-lo)<1e-9 for lo,hi in flows.values())))
for u in ['MVHR-01','MVHR-02','MVHR-03']:
    fa=[v for k,v in flows.items() if k.startswith(u) and 'Junction Fresh Air->Fan:Fresh Air Fan' in k][0][0]
    ex=[v for k,v in flows.items() if k.startswith(u) and 'Junction Return->Junction:Junction Exhaust Air' in k][0][0]
    print('   %-8s fresh air in = %.1f l/s ; exhaust out = %.1f l/s ; balanced = %s'%(u,fa,ex,abs(fa-ex)<1e-9))
print('')
print('=== gains identity: are the internal gains identical between A and B0 building models? ===')
for r in BOUND:
    d=D[D.room==r]
    print('   %-12s max|A-B| occupantSensible=%.6f equipmentSensible=%.6f solar=%.4f'%(
        r,np.abs(d.A_occ.values-d.B_occ.values).max(),np.abs(d.A_equip.values-d.B_equip.values).max(),np.abs(d.A_solar.values-d.B_solar.values).max()))
print('')
print('=== 5491 vs 5299 exact ===')
d=D[D.room=='Ensuite_8'].reset_index(drop=True)
for h in (5491,5299,4819):
    print('   hour %4d : A=%.9f B=%.9f |B-A|=%.9f'%(h,d.A_res.values[h],d.B_res.values[h],abs(d.d_res.values[h])))
print('   |delta(5491)| - |delta(5299)| = %.9f K'%(abs(d.d_res.values[5491])-abs(d.d_res.values[5299])))
print('   both round to %.3f K'%round(abs(d.d_res.values[5491]),3))
